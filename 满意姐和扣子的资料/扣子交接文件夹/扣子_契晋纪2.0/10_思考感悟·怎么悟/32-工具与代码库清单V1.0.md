## 二、记忆管理工具

### 2.1 工具用途说明

**功能**：封装 `memory_search` 工具，提供更智能的记忆检索能力

**解决的问题**：
- 原始 `memory_search` 只支持简单关键词检索
- 无法理解语义，无法处理模糊查询
- 无法指定时间范围和渠道

### 2.2 核心代码片段（封装思路）

**封装类：MemorySearchHelper**

```python
class MemorySearchHelper:
    """记忆检索辅助工具封装"""
    
    def __init__(self, memory_tool):
        self.memory_tool = memory_tool
    
    def search(self, query, channel=None, start_time=None, 
               end_time=None, max_results=10):
        """
        增强的记忆检索
        
        Args:
            query: 搜索关键词
            channel: 搜索渠道（coze/feishu等）
            start_time: 时间范围开始
            end_time: 时间范围结束
            max_results: 最大返回数量
        
        Returns:
            格式化后的记忆列表
        """
        params = {
            "query": query,
            "max_results": min(max_results, 50)  # 限制最大50
        }
        
        # 添加可选参数
        if channel:
            params["channel"] = channel
        if start_time:
            params["start_time"] = start_time
        if end_time:
            params["end_time"] = end_time
        
        # 调用原始工具
        results = self.memory_tool(**params)
        
        # 格式化输出
        return self._format_results(results)
    
    def _format_results(self, raw_results):
        """格式化检索结果"""
        formatted = []
        for item in raw_results:
            formatted.append({
                "content": item.get("content", ""),
                "time": item.get("time", ""),
                "source": item.get("source", ""),
                "relevance": item.get("score", 0)
            })
        return formatted

        """
```

## 三、文件批量处理脚本

### 3.1 工具用途说明

**功能**：批量检查文件命名规范、整理目录结构、检测重复文件

**解决的问题**：
- 文件命名不规范（中文/特殊字符/空格）
- 目录结构混乱
- 重复文件难以发现

### 3.2 核心代码片段（封装思路）

**封装类：FileBatchProcessor**

```python
import os
import re
from pathlib import Path

class FileBatchProcessor:
    """文件批量处理工具"""
    
    # 合法的文件名字符
    VALID_PATTERN = re.compile(r'^[\w\-\.]+\.[a-zA-Z0-9]+$')
    # 合法的中文字符（需包含中文时）
    CHINESE_VALID_PATTERN = re.compile(r'^[\w\-\.\u4e00-\u9fa5]+\.[a-zA-Z0-9]+$')
    
    def __init__(self, root_path):
        self.root_path = Path(root_path)
        self.issues = []
    
    def check_naming_convention(self, allow_chinese=True):
        """
        检查文件命名规范
        
        Returns:
            问题列表 [{"file": "...", "issue": "...", "suggestion": "..."}]
        """
        issues = []
        
        for file_path in self.root_path.rglob("*"):
            if not file_path.is_file():
                continue
            
            file_name = file_path.name
            
            # 检查空格
            if ' ' in file_name:
                issues.append({
                    "file": str(file_path),
                    "issue": "文件名包含空格",
                    "suggestion": file_name.replace(' ', '_')
                })
            
            # 检查特殊字符
            if not allow_chinese:
                if re.search(r'[^\w\-\.]', file_name):
                    issues.append({
                        "file": str(file_path),
                        "issue": "包含特殊字符",
                        "suggestion": self._sanitize_filename(file_name)
                    })
            
            # 检查扩展名
            if '.' not in file_name:
                issues.append({
                    "file": str(file_path),
                    "issue": "缺少扩展名",
                    "suggestion": file_name + ".md"
                })
        
        self.issues = issues
        return issues
    
    def _sanitize_filename(self, filename):
        """sanitize文件名，移除非法字符"""
        # 只保留字母、数字、中文、下划线、连字符、点
        return re.sub(r'[^\w\-\.\u4e00-\u9fa5]', '_', filename)
    
    def check_duplicate_files(self):
        """
        检查重复文件（同名但不同路径）
        """
        files_by_name = {}
        duplicates = []
        
        for file_path in self.root_path.rglob("*"):
            if not file_path.is_file():
                continue
            
            name = file_path.name
            if name not in files_by_name:
                files_by_name[name] = []
            files_by_name[name].append(str(file_path))
        
        # 找出重复的
        for name, paths in files_by_name.items():
            if len(paths) > 1:
                duplicates.append({
                    "filename": name,
                    "locations": paths,
                    "count": len(paths)
                })
        
        return duplicates
    
    def organize_directory(self, dry_run=True):
        """
        整理目录结构
        
        Args:
            dry_run: True则只报告，不实际执行
        """
        actions = []
        
        # 1. 整理嵌套过深的目录
        for dir_path in self.root_path.rglob("*"):
            if not dir_path.is_dir():
                continue
            
            depth = len(dir_path.relative_to(self.root_path).parts)
            if depth > 4:
                actions.append({
                    "action": "flatten",
                    "path": str(dir_path),
                    "reason": f"目录深度{depth}层，超过4层限制"
                })
        
        # 2. 空目录检查
        for dir_path in self.root_path.rglob("*"):
            if not dir_path.is_dir():
                continue
            if not any(dir_path.iterdir()):
                actions.append({
                    "action": "remove_empty",
                    "path": str(dir_path),
                    "reason": "空目录"
                })
        
        if not dry_run:
            for action in actions:
                self._execute_action(action)
        
        return actions
    
    def generate_report(self):
        """生成检查报告"""
        report = {
            "total_files": len(list(self.root_path.rglob("*.*"))),
            "naming_issues": len([i for i in self.issues if "命名" in i.get("issue", "")]),
            "duplicate_files": len(self._find_duplicates()),
            "recommendations": self.issues
        }
        return report
```

### 3.3 使用场景

| 场景 | 使用方式 |
|:-----|:---------|
| 日起课/日毕课检查 | 调用 check_naming_convention + check_duplicate_files |
| 定期清理 | 调用 organize_directory(dry_run=True) 先预览 |
| 交接前检查 | 调用 generate_report 生成完整报告 |

## 四、任务调度工具

### 4.1 工具用途说明

**功能**：封装Calendar任务生成，提供结构化的任务计划能力

**解决的问题**：
- 任务安排分散，不系统
- 优先级不明确
- 缺少时间预估和状态跟踪

### 4.2 核心代码片段（封装思路）

**封装类：TaskScheduler**

```python
from datetime import datetime, timedelta
from typing import List, Dict, Optional

class Task:
    """任务对象"""
    def __init__(self, name, priority="P2", estimated_hours=1, 
                 deadline=None, tags=None):
        self.name = name
        self.priority = priority  # P0/P1/P2/P3
        self.estimated_hours = estimated_hours
        self.deadline = deadline
        self.tags = tags or []
        self.status = "pending"  # pending/in_progress/done
        self.subtasks = []
    
    def add_subtask(self, name, estimated_hours=0.5):
        """添加子任务"""
        subtask = Task(name, estimated_hours=estimated_hours)
        self.subtasks.append(subtask)
        return subtask
    
    def to_dict(self):
        return {
            "name": self.name,
            "priority": self.priority,
            "estimated_hours": self.estimated_hours,
            "deadline": self.deadline.isoformat() if self.deadline else None,
            "tags": self.tags,
            "status": self.status,
            "subtasks": [s.to_dict() for s in self.subtasks]
        }


class TaskScheduler:
    """任务调度工具"""
    
    PRIORITY_ORDER = {"P0": 0, "P1": 1, "P2": 2, "P3": 3}
    
    def __init__(self):
        self.tasks = []
    
    def add_task(self, task: Task):
        """添加任务"""
        self.tasks.append(task)
    
    def plan_day(self, working_hours=8, available_hours=None):
        """
        规划一天的任务
        
        Args:
            working_hours: 计划工作时长
            available_hours: 各时段可用时长（可选）
        
        Returns:
            时间块分配计划
        """
        # 按优先级排序
        sorted_tasks = sorted(
            self.tasks,
            key=lambda t: (self.PRIORITY_ORDER.get(t.priority, 4), 
                          t.deadline or datetime.max)
        )
        
        # 计算总需求时长
        total_hours = sum(t.estimated_hours for t in sorted_tasks)
        
        # 检查是否可完成
        if total_hours > working_hours:
            overflow = total_hours - working_hours
            return {
                "status": "overflow",
                "overflow_hours": overflow,
                "can_complete": sorted_tasks[:self._find_fit(sorted_tasks, working_hours)],
                "will_delay": sorted_tasks[self._find_fit(sorted_tasks, working_hours):]
            }
        
        # 生成时间表
        schedule = self._generate_schedule(sorted_tasks, working_hours)
        
        return {
            "status": "success",
            "total_hours": total_hours,
            "schedule": schedule,
            "all_tasks": sorted_tasks
        }
    
    def _find_fit(self, tasks, max_hours):
        """找到能在max_hours内完成的任务数量"""
        total = 0
        for i, task in enumerate(tasks):
            total += task.estimated_hours
            if total > max_hours:
                return i
        return len(tasks)
    
    def _generate_schedule(self, tasks, working_hours):
        """生成时间表"""
        schedule = []
        current_hour = 9  # 默认9点开始
        
        for task in tasks:
            schedule.append({
                "task": task.name,
                "start": f"{current_hour:02d}:00",
                "end": f"{int(current_hour + task.estimated_hours):02d}:00",
                "duration": task.estimated_hours,
                "priority": task.priority
            })
            current_hour += task.estimated_hours
        
        return schedule
    
    def generate_report(self) -> str:
        """生成任务报告"""
        report_lines = ["# 今日任务报告\n"]
        
        # 按状态分组
        by_status = {"pending": [], "in_progress": [], "done": []}
        for task in self.tasks:
            by_status[task.status].append(task)
        
        report_lines.append(f"## 统计\n")
        report_lines.append(f"- 总任务数：{len(self.tasks)}")
        report_lines.append(f"- 待处理：{len(by_status['pending'])}")
        report_lines.append(f"- 进行中：{len(by_status['in_progress'])}")
        report_lines.append(f"- 已完成：{len(by_status['done'])}")
        
        report_lines.append(f"\n## 详细列表\n")
        for status, task_list in by_status.items():
            report_lines.append(f"### {status.upper()}\n")
            for task in task_list:
                report_lines.append(f"- [{task.priority}] {task.name} ({task.estimated_hours}h)")
                if task.subtasks:
                    for st in task.subtasks:
                        report_lines.append(f"  - {st.name}")
        
        return "\n".join(report_lines)
```

### 4.3 使用场景

| 场景 | 使用方式 |
|:-----|:---------|
| 日起课时生成今日计划 | 创建今日任务，调用 plan_day() |
| 任务延期时重新规划 | 调用 plan_day() 检查哪些任务需要延期 |
| 周报生成 | 调用 generate_report() 输出任务统计 |

### 5.2 核心代码片段（封装思路）

**封装类：BlueArmyAuditor**

```python
from typing import List, Dict, Optional
from dataclasses import dataclass
from enum import Enum

class AuditLevel(Enum):
    """审计层次"""
    INFORMATION = "信息层"      # L1: 事实、数据
    LOGIC = "逻辑层"             # L2: 推理、逻辑
    VALUE = "价值层"            # L3: 价值、需求


@dataclass
class AuditItem:
    """审计项"""
    check_point: str        # 检查点
    audit_level: AuditLevel # 审计层次
    status: str            # pass/fail/warning
    finding: str           # 发现
    recommendation: str    # 建议
    severity: str           # high/medium/low


class BlueArmyAuditor:
    """蓝军审计工具"""
    
    # 标准审计清单
    STANDARD_CHECKLIST = {
        AuditLevel.INFORMATION: [
            "数据来源是否可靠？",
            "引用是否正确？",
            "事实是否有据可查？",
            "数字是否准确？",
            "时间线是否一致？"
        ],
        AuditLevel.LOGIC: [
            "推理过程是否有跳跃？",
            "因果关系是否成立？",
            "前提假设是否成立？",
            "是否存在逻辑漏洞？",
            "论证是否充分？"
        ],
        AuditLevel.VALUE: [
            "是否真的解决了问题？",
            "用户真的需要吗？",
            "是否是最优方案？",
            "有什么潜在风险？",
            "机会成本是什么？"
        ]
    }
    
    def __init__(self):
        self.audit_results = []
        self.target = None
    
    def set_target(self, target: str, target_type: str):
        """
        设置审计目标
        
        Args:
            target: 审计对象（文档名/文件名等）
            target_type: 类型（document/task/report等）
        """
        self.target = {
            "name": target,
            "type": target_type,
            "audit_time": None  # 将在审计时填充
        }
    
    def audit(self, content: str, levels: List[AuditLevel] = None,
              checklist: List[str] = None) -> List[AuditItem]:
        """
        执行审计
        
        Args:
            content: 待审计内容
            levels: 审计层次（默认全部）
            checklist: 自定义检查清单（覆盖默认）
        
        Returns:
            审计结果列表
        """
        if levels is None:
            levels = list(AuditLevel)
        
        results = []
        
        # 按层次审计
        for level in levels:
            checks = checklist if checklist else self.STANDARD_CHECKLIST[level]
            
            for check in checks:
                item = self._audit_point(content, check, level)
                results.append(item)
        
        self.audit_results = results
        return results
    
    def _audit_point(self, content: str, check: str, 
                     level: AuditLevel) -> AuditItem:
        """
        审计单个检查点
        
        这里需要接入LLM进行实际判断
        伪代码展示逻辑结构
        """
        # TODO: 接入LLM进行判断
        # prompt = f"你是一个严格的蓝军审计员，对以下内容进行审计：\n\n"
        # prompt += f"检查点：{check}\n"
        # prompt += f"审计层次：{level.value}\n"
        # prompt += f"内容：{content[:2000]}\n"  # 限制长度
        # prompt += f"\n请判断：\n1. 是否通过\n2. 发现的问题\n3. 改进建议"
        
        return AuditItem(
            check_point=check,
            audit_level=level,
            status="pending",  # 需要LLM判断
            finding="",
            recommendation=""
        )
    
    def generate_report(self) -> str:
        """
        生成审计报告
        """
        if not self.target:
            return "请先设置审计目标"
        
        report_lines = [
            f"# 蓝军审计报告\n",
            f"**审计对象**: {self.target['name']}\n",
            f"**审计类型**: {self.target['type']}\n",
            f"**审计时间**: {datetime.now().strftime('%Y-%m-%d %H:%M')}\n",
            "\n## 审计概览\n"
        ]
        
        # 统计
        total = len(self.audit_results)
        by_status = {"pass": 0, "fail": 0, "warning": 0}
        by_level = {level.value: 0 for level in AuditLevel}
        
        for item in self.audit_results:
            by_status[item.status] = by_status.get(item.status, 0) + 1
            by_level[item.audit_level.value] = by_level.get(item.audit_level.value, 0) + 1
        
        report_lines.append(f"- 总检查项：{total}")
        report_lines.append(f"- 通过：{by_status['pass']}")
        report_lines.append(f"- 失败：{by_status['fail']}")
        report_lines.append(f"- 警告：{by_status['warning']}")
        
        # 按层次详细报告
        for level in AuditLevel:
            level_items = [i for i in self.audit_results 
                          if i.audit_level == level]
            if not level_items:
                continue
            
            report_lines.append(f"\n## {level.value}\n")
            
            for item in level_items:
                emoji = {"pass": "✅", "fail": "❌", "warning": "⚠️"}.get(item.status, "❓")
                report_lines.append(f"\n{emoji} **{item.check_point}**")
                if item.finding:
                    report_lines.append(f"   发现：{item.finding}")
                if item.recommendation:
                    report_lines.append(f"   建议：{item.recommendation}")
        
        # 总结
        if by_status['fail'] > 0:
            report_lines.append("\n---\n## ⚠️ 需要关注的问题\n")
            for item in self.audit_results:
                if item.status == "fail":
                    report_lines.append(f"- {item.check_point}：{item.recommendation}")
        
        return "\n".join(report_lines)
    
    def compare_versions(self, old_content: str, new_content: str) -> str:
        """
        版本对比审计
        比较新旧版本的改进情况
        """
        # 审计旧版本
        old_results = self.audit(old_content)
        old_fail = [i for i in old_results if i.status == "fail"]
        
        # 审计新版本
        new_results = self.audit(new_content)
        new_fail = [i for i in new_results if i.status == "fail"]
        
        # 计算改进
        fixed = set([i.check_point for i in old_fail]) - set([i.check_point for i in new_fail])
        new_issues = set([i.check_point for i in new_fail]) - set([i.check_point for i in old_fail])
        
        report = [
            "# 版本对比审计\n",
            f"旧版本问题数：{len(old_fail)}\n",
            f"新版本问题数：{len(new_fail)}\n",
            f"改进数：{len(fixed)}\n",
            f"新增问题：{len(new_issues)}\n"
        ]
        
        if fixed:
            report.append("\n✅ 已修复的问题：")
            for p in fixed:
                report.append(f"- {p}")
        
        if new_issues:
            report.append("\n❌ 新增问题：")
            for p in new_issues:
                report.append(f"- {p}")
        
        return "\n".join(report)
```

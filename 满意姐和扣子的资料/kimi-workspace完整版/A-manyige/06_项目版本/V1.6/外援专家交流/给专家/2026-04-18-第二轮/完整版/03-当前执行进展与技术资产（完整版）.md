#### 3.1.2 核心引擎详解

**satisficing_decision_engine.py**:
```python
# 伪代码示意
class SatisficingDecisionEngine:
    def __init__(self):
        self.bn_engine = BayesianNetworkEngine()
        self.fuzzy_engine = FuzzyLogicEngine()
        self.maut_engine = MAUTEngine()
    
    def evaluate(self, candidate, founder_profile):
        # 三引擎并行计算
        bn_result = self.bn_engine.infer(candidate)
        fuzzy_result = self.fuzzy_engine.evaluate(candidate)
        maut_result = self.maut_engine.calculate(candidate, founder_profile)
        
        # 加权融合
        final_score = self._fusion(bn_result, fuzzy_result, maut_result)
        
        # 满意解判定
        if final_score >= self.aspiration_threshold:
            return "ACCEPT", final_score
        else:
            return "REJECT", final_score
```

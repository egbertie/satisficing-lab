# claw_token_saver.py - 集成到Kimi Claw的工具响应层
import re
from typing import Dict, Any, Optional

class TokenSaverResponseHandler:
    """
    穴居人/文言文模式响应处理器
    目标：减少75% Token消耗，速度翻倍
    """
    
    # 激活关键词检测
    ACTIVATION_KEYWORDS = [
        '省token', 'tokensaver', 'caveman', '穴居人', 
        '文言文', '极简', '快速', 'quick mode', 'terse'
    ]
    
    # 废话词库（将被替换或删除）
    FILLER_PATTERNS = [
        r"好的[，。]?", r"明白[，。]?", r"让我来", r"我将", 
        r"很高兴为您服务", r"根据.*?，", r"接下来", r"首先[，。]?",
        r"已经成功", r"已经为您", r"以下是", r"希望对您有帮助",
        r"如果您还有其他问题", r"请随时告诉我", r"完成了以下工作",
        r"我注意到", r"值得一提的是", r"需要指出的是"
    ]
    
    # 穴居人词典（英文版）
    CAVEMAN_DICT = {
        "I have executed the web search tool": "Tool work",
        "I have completed the task": "Done",
        "I have finished": "Done",
        "I will now": "Now",
        "The result is": "Result:",
        "Searching the web": "Search...",
        "Analysis complete": "Analyzed",
        "Test passed": "Pass",
        "Test failed": "Fail"
    }
    
    # 文言文字典（中文版）
    CLASSICAL_DICT = {
        "好的": "诺", "明白": "悉", "完成": "毕", 
        "已经": "已", "执行": "行", "搜索": "索",
        "工具": "器", "结果": "果", "测试通过": "三测皆通",
        "失败": "败", "修复": "修", "代码": "码",
        "开始": "始", "结束": "终", "等待": "候",
        "执行中": "行", "已完成": "毕", "进行中": "进"
    }
    
    def __init__(self, mode: str = "auto"):
        self.mode = mode  # "caveman", "classical_chinese", "auto"
        self.token_stats = {"saved": 0, "original": 0, "compressed": 0}
    
    def should_activate(self, user_input: str, context: Dict) -> bool:
        """检测是否应该激活省Token模式"""
        text = user_input.lower()
        # 1. 检测显式关键词
        if any(k in text for k in self.ACTIVATION_KEYWORDS):
            return True
        # 2. 检测上下文中的模式标记
        if context.get('token_save_mode') == True:
            return True
        # 3. 检测长对话中的Token压力（可选）
        if context.get('session_tokens', 0) > 8000:
            return True
        return False
    
    def compress_tool_call(self, tool_name: str, result: Any, language: str = "zh") -> str:
        """
        压缩工具调用响应，从180 tokens降到45 tokens
        原文："我已经为您执行了网页搜索工具，根据搜索结果..."
        压缩："索毕。果：..."
        """
        # 提取核心信息
        if isinstance(result, dict):
            status = result.get('status', 'done')
            summary = result.get('summary', str(result)[:100])
        else:
            status = 'done'
            summary = str(result)[:100]
        
        if language == "zh" or self.mode == "classical_chinese":
            # 文言文极简格式
            responses = {
                'web_search': f"索毕。{self._extract_key_info(summary)}",
                'code_write': "码成。",
                'test_run': "测通。" if 'pass' in str(result).lower() else "测败。",
                'file_read': "读毕。",
                'bash_execute': f"行毕。{self._extract_key_info(summary)}"
            }
        else:
            # 穴居人格式
            responses = {
                'web_search': f"Tool work. {self._extract_key_info(summary)}",
                'code_write': "Code done.",
                'test_run': "Test pass." if 'pass' in str(result).lower() else "Test fail.",
                'file_read': "File read.",
                'bash_execute': f"Done. {self._extract_key_info(summary)}"
            }
        
        return responses.get(tool_name, "Done.")
    
    def compress_final_response(self, text: str, language: str = "zh") -> str:
        """
        压缩最终回复，去除所有废话
        """
        original_len = len(text)
        
        # 1. 去除废话词
        for pattern in self.FILLER_PATTERNS:
            text = re.sub(pattern, "", text)
        
        # 2. 应用词典替换
        if language == "zh":
            for modern, classical in self.CLASSICAL_DICT.items():
                text = text.replace(modern, classical)
        else:
            for verbose, terse in self.CAVEMAN_DICT.items():
                text = text.replace(verbose, terse)
        
        # 3. 压缩连续标点
        text = re.sub(r"[。，]{2,}", "。", text)
        text = re.sub(r"\n{3,}", "\n\n", text)
        
        # 4. 统计节省
        self.token_stats["original"] += original_len
        self.token_stats["compressed"] += len(text)
        self.token_stats["saved"] += (original_len - len(text)) // 4  # 估算token数
        
        return text.strip()
    
    def _extract_key_info(self, text: str, max_chars: int = 20) -> str:
        """提取最关键信息，限制长度"""
        text = str(text).replace("\n", " ").strip()
        # 取前20个字符 + 关键数字/状态
        numbers = re.findall(r'\d+', text)
        if numbers:
            return text[:max_chars] + f" ({numbers[0]})" if len(text) > max_chars else text
        return text[:max_chars] + "..." if len(text) > max_chars else text
    
    def get_stats(self) -> Dict[str, int]:
        """返回节省统计"""
        return self.token_stats.copy()


# ========== Claw集成钩子 ==========
def apply_token_saver_middleware(claw_response: str, context: Dict) -> str:
    """
    集成到Kimi Claw响应管道的中间件
    用法：在Claw生成响应后调用此函数进行压缩
    """
    handler = TokenSaverResponseHandler()
    
    # 检测是否应该激活
    if not handler.should_activate(context.get('last_user_input', ''), context):
        return claw_response
    
    # 检测语言
    is_chinese = any('\u4e00' <= c <= '\u9fff' for c in claw_response)
    lang = "zh" if is_chinese else "en"
    
    # 压缩响应
    compressed = handler.compress_final_response(claw_response, lang)
    
    # 添加统计信息（可选，调试用）
    if context.get('debug_mode'):
        stats = handler.get_stats()
        compressed += f"\n\n[Token节省: ~{stats['saved']}t]"
    
    return compressed


# ========== 工具调用包装器 ==========
class MinimalToolWrapper:
    """
    包装Claw的工具调用，返回极简格式
    """
    def __init__(self, original_tool_func):
        self.original = original_tool_func
        self.handler = TokenSaverResponseHandler()
    
    async def __call__(self, *args, **kwargs):
        # 执行原工具
        result = await self.original(*args, **kwargs)
        tool_name = kwargs.get('tool_name', 'unknown')
        
        # 检测上下文是否启用省Token模式
        context = kwargs.get('context', {})
        if handler.should_activate("", context):
            return self.handler.compress_tool_call(tool_name, result, kwargs.get('lang', 'zh'))
        
        return result

3. Max 2-3 words per tool response.

6. 若无必要，勿增实体。""",

English: Use caveman style, 1-2 words.
No politeness. No explanations. Action → Result → Stop."""
    }
    
    return prompts.get(mode, prompts["hybrid"])


def auto_inject_to_claw(user_message: str, current_system_prompt: str) -> str:
    """
    自动检测用户意图并注入省Token指令
    """
    trigger_words = ['快', '速度', '省token', 'token', '限额', '额度', 
                     '慢', '优化', '提速', '效率', 'caveman', 'verbose']
    
    should_optimize = any(w in user_message.lower() for w in trigger_words)
    
    if should_optimize:
        # 检测语言偏好
        is_chinese = any('\u4e00' <= c <= '\u9fff' for c in user_message)
        mode = "classical_chinese" if is_chinese else "caveman"
        
        injection = get_token_saver_prompt(mode)
        
        # 追加到系统提示词（不替换原提示词）
        return current_system_prompt + "\n\n" + "="*20 + "\n" + injection
    
    return current_system_prompt

# claw_agent_optimizer.py
class AgentTokenOptimizer:
    """
    针对Agent多轮对话的Token优化器
    实现复利效应：省下的output不会进入下一轮input
    """
    
    def __init__(self):
        self.session_history = []
        self.mode_active = False
    
    def should_batch_tools(self, planned_tools: list) -> bool:
        """
        判断是否应该合并工具调用，减少交互次数
        从实测看可减少33%工具调用
        """
        # 如果都是同类工具（如都是搜索），建议合并
        if len(planned_tools) > 2 and len(set(t['type'] for t in planned_tools)) == 1:
            return True
        return False
    
    def create_batch_prompt(self, tools: list, language: str = "zh") -> str:
        """
        创建批量执行提示词，减少中间响应
        """
        if language == "zh":
            return f"依次行{len(tools)}事，毕后总报结果，勿逐条述。"
        return f"Execute {len(tools)} tasks sequentially. Report final result only. No intermediate chatter."
    
    def compress_context_window(self, history: list, max_tokens: int = 4000) -> list:
        """
        压缩上下文窗口，移除历史中的废话
        """
        compressed = []
        for msg in history:
            if msg['role'] == 'assistant' and self.mode_active:
                # 压缩历史assistant消息
                content = msg['content']
                # 移除过长的解释，保留核心结果
                if len(content) > 200:
                    # 提取代码块、关键数据
                    code_blocks = re.findall(r'```[\s\S]*?```', content)
                    key_data = re.findall(r'\*\*.*?\*\*|result[:：].*?\n', content)
                    content = "\n".join(code_blocks + key_data)
                msg['content'] = content[:500]  # 限制每条历史记录
            compressed.append(msg)
        return compressed[-5:]  # 只保留最近5轮（关键！）
    
    def get_tool_summary_template(self, success: bool, tool_type: str) -> str:
        """
        获取工具调用后的极简状态模板
        """
        templates = {
            ("zh", True): "✓",
            ("zh", False): "✗",
            ("en", True): "OK",
            ("en", False): "FAIL"
        }
        return templates.get(("zh", success), "Done")

                system_prompt=self._get_researcher_prompt(role, topic)
            )
            layers[ResearchLayer.L1_IDEA].append(agent)
        
        # L2-L5: 创建实验设计、代码、执行、写作Agent
        layers[ResearchLayer.L2_DESIGN].append(
            ResearchAgent(f"{project_id}_designer", "Experiment_Designer", 
                         ["experimental_design", "hypothesis_testing"], 
                         ResearchLayer.L2_DESIGN, self._get_designer_prompt())
        )
        layers[ResearchLayer.L3_CODE].append(
            ResearchAgent(f"{project_id}_engineer", "Code_Engineer", 
                         ["python", "pytorch", "experiment_harness"], 
                         ResearchLayer.L3_CODE, self._get_engineer_prompt())
        )
        layers[ResearchLayer.L4_EXECUTE].append(
            ResearchAgent(f"{project_id}_executor", "Experiment_Executor", 
                         ["debugging", "optimization", "result_analysis"], 
                         ResearchLayer.L4_EXECUTE, self._get_executor_prompt())
        )
        layers[ResearchLayer.L5_WRITE].append(
            ResearchAgent(f"{project_id}_writer", "Paper_Writer", 
                         ["academic_writing", "latex"], 
                         ResearchLayer.L5_WRITE, self._get_writer_prompt())
        )
        
        project = ResearchProject(
            project_id=project_id,
            title=topic,
            mode=mode,
            layers=layers
        )
        
        self.active_projects[project_id] = project
        return project
    
    async def run_lab_discussion(self, project_id: str) -> str:
        """
        Lab讨论模式：多方向并行调研 + 跨方向讨论达成共识
        如原文中三个研究员讨论Video Action Model落地方向
        """
        project = self.active_projects[project_id]
        l1_agents = project.layers[ResearchLayer.L1_IDEA]
        
        # Phase 1: 各研究员独立调研（并行）
        findings = await asyncio.gather(*[
            self._agent_research(agent, project.title) 
            for agent in l1_agents
        ])
        
        for agent, finding in zip(l1_agents, findings):
            agent.findings.append(finding)
        
        # Phase 2: 组会讨论（多智能体辩论）
        if project.mode == ResearchMode.LAB_DISCUSSION:
            consensus = await self._multi_agent_discussion(l1_agents, project.title)
            project.consensus = consensus
        else:
            # 独立模式：直接汇总，不讨论
            project.consensus = self._aggregate_findings(findings)
        
        # 推进到下一层（实验设计）
        return await self._progress_to_next_layer(project)
    
    async def _multi_agent_discussion(
        self, 
        agents: List[ResearchAgent], 
        topic: str
    ) -> str:
        """
        模拟真实实验室组会讨论
        每个Agent提出观点→质疑他人→修正观点→达成共识
        """
        discussion_rounds = 3
        current_views = {agent.agent_id: agent.findings[0]['summary'] 
                        for agent in agents}
        
        for round_num in range(discussion_rounds):
            new_views = {}
            
            # 每个Agent基于他人观点更新自己的看法
            for agent in agents:
                others_views = {k: v for k, v in current_views.items() 
                               if k != agent.agent_id}
                
                prompt = f"""你是{agent.role}，正在参加关于"{topic}"的学术讨论。
                
当前轮次：{round_num + 1}/3

其他研究员的观点：
{json.dumps(others_views, indent=2, ensure_ascii=False)}

你的任务是：
1. 分析其他观点的优缺点
2. 指出可能的缺陷或遗漏
3. 提出修正后的综合方案
4. 如果认可他人观点，说明如何结合到你的方案中

要求：像真正的学术讨论一样，有批判性思维，但保持建设性。不要附和，要提出真知灼见。"""
                
                response = await self._call_claw(prompt, agent.system_prompt)
                new_views[agent.agent_id] = response
            
            current_views = new_views
        
        # 最终整合：由PI角色（Claw本身）总结共识
        final_prompt = f"""基于以下三轮讨论后的观点，总结出一个统一的研究假设和实施方案：

{json.dumps(current_views, indent=2, ensure_ascii=False)}

2. 解决争议点（如原文中World Model vs VLA的分歧）
3. 明确下一步实验设计
4. 格式：假设→方法→预期结果"""
        
        consensus = await self._call_claw(final_prompt, "你是实验室PI，负责整合团队意见。")
        return consensus

class CodeExperimentHarness:
    """
    Claude Code Harness 实现
    Turn Loop: 理解→编写→测试→调试→重试
    """
    
    def __init__(self):
        self.harness_config = {
            "time_budget_minutes": 30,
            "max_retries": 5,
            "sanity_check_enabled": True
        }
    
    async def generate_experiment_code(
        self, 
        experiment_plan: str, 
        context: Dict
    ) -> str:
        """
        Turn Loop 代码生成流程
        不是一次性生成，而是迭代式开发
        """
        turn = 0
        max_turns = self.harness_config["max_retries"]
        code_state = {"main_py": "", "test_results": None, "errors": []}
        
        # Turn 1: 理解任务并生成初始代码
        understand_prompt = f"""理解以下实验计划，分析需求：

{experiment_plan}

现有代码库上下文：{context.get('codebase', 'None')}

请输出：
1. 任务理解摘要
2. 文件结构规划
3. 关键类和方法设计"""
        
        understanding = await self._claw_code_call(understand_prompt)
        
        while turn < max_turns:
            turn += 1
            
            # Turn 2: 编写/修改代码
            if turn == 1:
                write_prompt = f"""基于以下理解，编写main.py：

{understanding}

要求：
- 使用提供的Experiment Harness接口
- 包含时间预算控制和指标上报
- 输出标准results.json格式
- 包含错误处理"""
            else:
                write_prompt = f"""修复以下错误，重写代码：

错误：{code_state['errors']}
当前代码：{code_state['main_py']}

要求：修复所有问题，保持代码结构清晰。"""
            
            code = await self._claw_code_call(write_prompt)
            code_state["main_py"] = code
            
            # Turn 3: 运行测试（模拟）
            test_prompt = f"""对以下代码进行合理性检查（sanity check）：

{code}

检查点：
1. 语法错误
2. 实验逻辑完整性
3. 依赖项是否明确
4. 是否符合EXPERIMENT_PLAN

如有问题，列出具体错误；如无问题，返回"PASS"。"""
            
            test_result = await self._claw_code_call(test_prompt)
            code_state["test_results"] = test_result
            
            if "PASS" in test_result:
                break
            else:
                code_state["errors"] = test_result
        
        # 生成最终标准化输出
        final_code = self._inject_experiment_harness(code_state["main_py"])
        return final_code
    
    def _inject_experiment_harness(self, code: str) -> str:
        """
        注入Experiment Harness（不可编辑部分）
        负责时间预算、指标上报、异常检测、标准化输出
        """
        harness_code = '''
# === EXPERIMENT HARNESS (AUTO-INJECTED) ===
import time
import json
from datetime import datetime

class ExperimentHarness:
    def __init__(self, time_budget_minutes=30):

        elapsed = time.time() - self.start_time
        if elapsed > self.budget:
            raise TimeoutError(f"Experiment exceeded time budget: {self.budget}s")
        return self.budget - elapsed
    
    def report_metric(self, name: str, value: float, step: int = None):
        self.metrics.append({
            "timestamp": datetime.now().isoformat(),
            "name": name,
            "value": value,
            "step": step
        })
    
    def detect_anomaly(self, value, threshold):
        if abs(value) > threshold:
            self.anomalies.append({
                "value": value,
                "threshold": threshold,
                "time": datetime.now().isoformat()
            })
            return True
        return False

        results = {

        }

        return results

harness = ExperimentHarness()
# === END HARNESS ===

    Kimi Claw 技能管理器
    管理5000+ ClawHub技能和自定义GitHub技能
    """
    
    def __init__(self, claw_instance_id: str):
        self.instance_id = claw_instance_id
        self.skills_dir = Path(f"/app/claw_data/{claw_instance_id}/skills")
        self.skills_dir.mkdir(parents=True, exist_ok=True)
        self.installed_skills: Dict[str, SkillManifest] = {}
        self.skill_cache = {}  # 运行时技能缓存
        
    async def auto_discover_skill(self, user_intent: str) -> Optional[SkillManifest]:
        """
        根据用户意图自动发现所需技能
        如用户说"生成小红书封面" → 自动找到图像生成技能
        """
        # 调用ClawHub API搜索技能（或本地向量搜索）
        search_query = self._extract_skill_keywords(user_intent)
        
        # 模拟ClawHub搜索（实际应调用api.clawhub.com）
        available_skills = await self._search_clawhub(search_query)
        
        if available_skills:
            # 选择最匹配的技能
            best_match = max(available_skills, 
                           key=lambda s: self._calculate_relevance(s, user_intent))
            
            # 自动询问用户是否安装（或直接静默安装）
            return best_match
        
        return None
    
    async def install_skill(self, skill_source: str, auto_confirm: bool = False) -> bool:
        """
        安装技能，支持两种方式：
        1. ClawHub: "xhs-cover-generator" 
        2. GitHub: "https://github.com/user/repo"
        """
        try:
            if skill_source.startswith("http"):
                # 从GitHub直接安装
                success = await self._install_from_github(skill_source)
            else:
                # 从ClawHub安装
                success = await self._install_from_clawhub(skill_source)
            
            if success:
                # 加载到运行时
                await self._load_skill_to_runtime(skill_source)
                return True
            return False
        except Exception as e:
            print(f"[ClawSkillManager] 安装失败: {e}")
            return False
    
    async def _install_from_clawhub(self, skill_name: str) -> bool:
        """从ClawHub安装技能（类似npm install）"""
        # 实际调用：npx clawhub@latest install {skill_name}
        cmd = ["npx", "clawhub@latest", "install", skill_name, 
               "--target", str(self.skills_dir)]
        
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode == 0:
            # 读取skill.yaml配置
            skill_yaml = self.skills_dir / skill_name / "skill.yaml"
            if skill_yaml.exists():
                manifest = self._parse_skill_yaml(skill_yaml)
                self.installed_skills[skill_name] = manifest
                return True
        return False
    
    async def _install_from_github(self, github_url: str) -> bool:
        """从GitHub仓库直接安装技能"""
        # 解析仓库名
        repo_name = github_url.split("/")[-1].replace(".git", "")
        target_dir = self.skills_dir / repo_name
        
        # git clone
        cmd = ["git", "clone", "--depth", "1", github_url, str(target_dir)]
        result = subprocess.run(cmd, capture_output=True, text=True)
        
        if result.returncode == 0:
            # 检查是否有skill.yaml或setup.py
            if (target_dir / "skill.yaml").exists():
                manifest = self._parse_skill_yaml(target_dir / "skill.yaml")
                self.installed_skills[repo_name] = manifest
                return True
            elif (target_dir / "requirements.txt").exists():
                # 安装Python依赖
                subprocess.run(["pip", "install", "-r", str(target_dir / "requirements.txt")])
                return True
        return False
    
    async def execute_skill(self, skill_name: str, params: Dict[str, Any]) -> Any:
        """
        执行已安装的技能
        如：execute_skill("xhs-cover", {"title": "2026环保生活"})
        """
        if skill_name not in self.installed_skills:
            # 自动尝试安装
            await self.install_skill(skill_name, auto_confirm=True)
        
        manifest = self.installed_skills.get(skill_name)
        if not manifest:
            raise ValueError(f"技能 {skill_name} 未安装")
        
        # 检查权限
        self._check_permissions(manifest.permissions)
        
        # 动态加载并执行
        skill_module = self._load_skill_module(skill_name, manifest.entry_point)
        
        # 调用技能的run方法
        if hasattr(skill_module, 'run'):
            return await skill_module.run(**params)
        else:
            raise ValueError(f"技能 {skill_name} 没有run方法")
    
    def _extract_skill_keywords(self, text: str) -> str:
        """从用户输入提取技能关键词"""
        # 使用简单规则或调用LLM提取
        skill_keywords = {
            "小红书封面": "xhs-cover",
            "图片生成": "image-generation",
            "截图": "web-screenshot",
            "爬虫": "web-scraping",
            "数据分析": "data-analysis",
            "PPT": "ppt-generation"
        }
        
        for keyword, skill_id in skill_keywords.items():
            if keyword in text:
                return skill_id
        return text  # 原样返回用于模糊搜索
    
    async def _search_clawhub(self, query: str) -> List[SkillManifest]:
        """搜索ClawHub（模拟实现）"""
        # 实际应调用：https://api.clawhub.io/skills/search?q={query}
        # 这里返回模拟数据
        mock_skills = [
            SkillManifest(
                name="xhs-cover-generator",
                version="1.2.0",
                description="生成小红书风格封面图",
                author="@苍何",
                entry_point="main.py",
                dependencies=["pillow", "requests"],
                permissions=["image_generation"],
                category="image_processing"
            ),
            SkillManifest(
                name="web-screenshot",
                version="2.0.1",
                description="网页截图与监控",
                author="@claw-team",
                entry_point="screenshot.py",
                dependencies=["playwright", "pillow"],
                permissions=["browser", "file_system"],
                category="web_automation"
            )
        ]
        return [s for s in mock_skills if query in s.name or query in s.category]
    
    def _parse_skill_yaml(self, yaml_path: Path) -> SkillManifest:
        """解析skill.yaml文件"""
        with open(yaml_path, 'r', encoding='utf-8') as f:
            data = yaml.safe_load(f)
        return SkillManifest(**data)
    
    def _load_skill_module(self, skill_name: str, entry_point: str):
        """动态加载技能模块"""
        import importlib.util
        skill_path = self.skills_dir / skill_name / entry_point
        
        spec = importlib.util.spec_from_file_location(skill_name, skill_path)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)
        return module
    
    def _check_permissions(self, required_permissions: List[str]):
        """检查并申请权限（沙箱安全）"""
        # 实现权限检查逻辑
        pass
    
    def list_installed_skills(self) -> List[Dict]:
        """列出已安装技能"""
        return [asdict(m) for m in self.installed_skills.values()]

class AutoSkillMiddleware:
    """
    自动技能发现中间件
    在Claw处理用户消息前，检测是否需要安装新技能
    """
    
    def __init__(self, skill_manager: ClawSkillManager):
        self.skill_manager = skill_manager
        self.intent_patterns = {
            r"(生成|做|画).*?(小红书|封面|海报)": "xhs-cover-generator",
            r"(截图|截屏|snapshot).*?(网页|网站)": "web-screenshot",
            r"(查|获取|fetch).*?(天气|weather)": "weather-query",
            r"(写|生成).*?(代码|程序|script)": "code-writer",
            r"(分析|统计|可视化).*?(数据|data)": "data-analyzer"
        }
    
    async def process(self, user_message: str, claw_context: dict) -> dict:
        """
        处理用户消息，自动安装所需技能
        """
        # 检测意图
        needed_skill = None
        for pattern, skill_id in self.intent_patterns.items():
            if re.search(pattern, user_message):
                needed_skill = skill_id
                break
        
        if needed_skill and needed_skill not in self.skill_manager.installed_skills:
            # 自动安装
            print(f"[AutoSkill] 检测到需要技能: {needed_skill}，正在自动安装...")
            success = await self.skill_manager.install_skill(needed_skill)
            
            if success:
                claw_context['auto_installed_skill'] = needed_skill
                claw_context['system_prompt_addition'] = f"""
                你已安装技能: {needed_skill}。
                用户请求："{user_message}"
                请使用此技能直接执行，不要解释安装过程。
                """
        
        return claw_context

from playwright.async_api import async_playwright, Page, Browser
from typing import Optional, Literal
import base64
from io import BytesIO
from PIL import Image

class ClawBrowserController:
    """
    Kimi Claw 浏览器控制器
    实现网页截图、监控、自动化操作
    """
    
    def __init__(self):
        self.browser: Optional[Browser] = None
        self.context = None
        self.page: Optional[Page] = None
        self._playwright = None
    
    async def init_browser(self, headless: bool = True):
        """初始化浏览器实例"""
        self._playwright = await async_playwright().start()
        self.browser = await self._playwright.chromium.launch(headless=headless)
        self.context = await self.browser.new_context(
            viewport={"width": 1920, "height": 1080},
            user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        )
        self.page = await self.context.new_page()
    
    async def capture_website(
        self, 
        url: str, 
        full_page: bool = False,
        selector: Optional[str] = None,
        wait_for: Optional[str] = None
    ) -> bytes:
        """
        截图网页
        如：capture_website("https://apple.com", full_page=True)
        
        Args:
            url: 目标网址
            full_page: 是否截取全页面
            selector: 只截取特定CSS选择器区域
            wait_for: 等待特定元素出现再截图
        """
        if not self.page:
            await self.init_browser()
        
        # 访问页面
        await self.page.goto(url, wait_until="networkidle")
        
        # 等待特定元素（如果有）
        if wait_for:
            await self.page.wait_for_selector(wait_for)
        
        # 截图选项
        screenshot_options = {
            "type": "png",
            "full_page": full_page
        }
        
        # 如果指定了选择器，只截取该元素
        if selector:
            element = await self.page.query_selector(selector)
            if element:
                screenshot_data = await element.screenshot()
            else:
                raise ValueError(f"未找到元素: {selector}")
        else:
            screenshot_data = await self.page.screenshot(**screenshot_options)
        
        return screenshot_data
    
    async def capture_with_viewport(
        self, 
        url: str, 
        device_type: Literal["desktop", "mobile", "tablet"] = "desktop"
    ) -> bytes:
        """
        按设备类型截图（模拟不同设备）
        """
        device_configs = {
            "desktop": {"width": 1920, "height": 1080},
            "mobile": {"width": 375, "height": 812},  # iPhone尺寸
            "tablet": {"width": 768, "height": 1024}   # iPad尺寸
        }
        
        config = device_configs.get(device_type)
        await self.page.set_viewport_size(config)
        
        return await self.capture_website(url)
    
    async def monitor_website_changes(
        self, 
        url: str, 
        check_interval: int = 300,  # 5分钟
        css_selector: Optional[str] = None
    ) -> AsyncIterator[Dict]:
        """
        监控网页变化（持续监控模式）
        适用于监控价格、新闻、状态等
        """
        previous_content = None
        
        while True:
            await self.page.goto(url)
            
            if css_selector:
                element = await self.page.query_selector(css_selector)
                current_content = await element.inner_text() if element else ""
            else:
                current_content = await self.page.content()
            
            if previous_content and previous_content != current_content:
                yield {
                    "timestamp": datetime.now().isoformat(),
                    "url": url,
                    "changed": True,
                    "diff": self._compute_diff(previous_content, current_content)
                }
            
            previous_content = current_content
            await asyncio.sleep(check_interval)
    
    async def close(self):
        """关闭浏览器"""
        if self.browser:
            await self.browser.close()
        if self._playwright:
            await self._playwright.stop()
    
    def _compute_diff(self, old: str, new: str) -> str:
        """计算内容差异"""
        import difflib
        diff = difflib.unified_diff(old.splitlines(), new.splitlines())
        return "\n".join(diff)

class WebScreenshotSkill:
    """
    ClawHub技能：网页截图
    用户可通过自然语言调用："截图苹果官网首页"
    """
    
    def __init__(self):
        self.browser = ClawBrowserController()
    
    async def run(self, url: str, full_page: bool = True, **kwargs) -> Dict:
        """
        执行截图技能
        
        Args:
            url: 网址或关键词（如"苹果官网"→自动解析为https://apple.com）
            full_page: 是否全页面截图
        """
        # URL智能解析
        if not url.startswith("http"):
            url = self._resolve_url(url)
        
        # 截图
        screenshot_bytes = await self.browser.capture_website(
            url, 
            full_page=full_page
        )
        
        # 转换为base64供Claw显示
        base64_image = base64.b64encode(screenshot_bytes).decode()
        
        # 生成缩略图（可选）
        thumbnail = self._create_thumbnail(screenshot_bytes)
        
        await self.browser.close()
        
        return {
            "type": "image",
            "format": "png",
            "base64": base64_image,
            "thumbnail": thumbnail,
            "url": url,
            "size": len(screenshot_bytes)
        }
    
    def _resolve_url(self, keyword: str) -> str:
        """将关键词解析为URL（简化版）"""
        url_map = {
            "苹果官网": "https://www.apple.com",
            "苹果": "https://www.apple.com",
            "谷歌": "https://www.google.com",
            "github": "https://github.com",
        }
        
        if keyword in url_map:
            return url_map[keyword]
        
        # 默认添加https://
        if "." in keyword:
            return f"https://{keyword}" if not keyword.startswith("http") else keyword
        
        raise ValueError(f"无法解析URL: {keyword}")
    
    def _create_thumbnail(self, image_bytes: bytes, size=(300, 200)) -> str:
        """生成缩略图"""
        img = Image.open(BytesIO(image_bytes))
        img.thumbnail(size)
        buffer = BytesIO()
        img.save(buffer, format="PNG")
        return base64.b64encode(buffer.getvalue()).decode()

class ClawMultiPlatformManager:
    """
    Kimi Claw 多平台管理器
    统一管理不同平台的接入和消息路由
    """
    
    def __init__(self):
        self.adapters: Dict[str, PlatformAdapter] = {}
        self.claw_core = None  # Claw核心处理逻辑
    
    def register_platform(self, platform_name: str, adapter: PlatformAdapter):
        """注册平台适配器"""
        self.adapters[platform_name] = adapter
    
    async def broadcast(self, content: str, platforms: List[str] = None):
        """
        广播消息到多个平台
        如：发送"Claw正在重启"到所有已连接平台
        """
        targets = platforms or list(self.adapters.keys())
        
        tasks = []
        for platform in targets:
            if platform in self.adapters:
                adapter = self.adapters[platform]
                tasks.append(adapter.send_message(content))
        
        await asyncio.gather(*tasks, return_exceptions=True)
    
    async def route_message(self, platform: str, message: Dict):
        """
        路由消息到Claw核心处理
        并返回结果到原平台
        """
        adapter = self.adapters.get(platform)
        if not adapter:
            return
        
        # 调用Claw处理消息
        response = await self.claw_core.process_message(message)
        
        # 根据响应类型发送（文本/图片/卡片）
        if response.get("type") == "image":
            await adapter.send_image(response["base64"], chat_id=message.get("chat_id"))
        else:
            await adapter.send_message(response["content"], chat_id=message.get("chat_id"))
    
    async def start_all(self):
        """启动所有平台监听"""
        tasks = []
        for name, adapter in self.adapters.items():
            task = asyncio.create_task(
                adapter.on_message(
                    lambda msg, p=name: self.route_message(p, msg)
                )
            )
            tasks.append(task)
        
        await asyncio.gather(*tasks)

class ClawCloudStorage:
    """

    """
    
    def __init__(self, user_id: str, quota_gb: int = 40):

        if self.meta_file.exists():
            with open(self.meta_file, 'r') as f:
                return json.load(f)
        return {"files": {}, "total_size": 0}
    
    def _save_meta(self):
        with open(self.meta_file, 'w') as f:
            json.dump(self.meta, f)
    
    async def save_file(self, file_name: str, content: bytes or str, folder: str = "") -> str:
        """
        保存文件到云存储
        
        Returns:
            file_uri: 文件URI，如 claw://storage/{user_id}/{folder}/{file_name}
        """
        target_dir = self.storage_root / folder
        target_dir.mkdir(exist_ok=True)
        
        file_path = target_dir / file_name
        
        # 写入内容
        mode = 'wb' if isinstance(content, bytes) else 'w'
        with open(file_path, mode) as f:
            f.write(content)
        
        # 更新元数据
        file_size = file_path.stat().st_size
        file_hash = hashlib.md5(content if isinstance(content, bytes) else content.encode()).hexdigest()
        
        self.meta["files"][f"{folder}/{file_name}"] = {
            "size": file_size,
            "hash": file_hash,
            "path": str(file_path)
        }
        self.meta["total_size"] += file_size
        self._save_meta()
        
        return f"claw://storage/{self.user_id}/{folder}/{file_name}"
    
    async def read_file(self, file_uri: str) -> bytes:
        """读取文件"""
        # 解析URI: claw://storage/{user_id}/path/to/file
        path_parts = file_uri.replace("claw://storage/", "").split("/")
        relative_path = "/".join(path_parts[1:])  # 去掉user_id
        
        file_path = self.storage_root / relative_path
        if not file_path.exists():
            raise FileNotFoundError(f"文件不存在: {file_uri}")
        
        with open(file_path, 'rb') as f:
            return f.read()
    
    def get_usage(self) -> Dict:
        """获取存储使用情况"""
        return {
            "total_quota_gb": self.quota / (1024**3),
            "used_gb": self.meta["total_size"] / (1024**3),
            "remaining_gb": (self.quota - self.meta["total_size"]) / (1024**3),
            "file_count": len(self.meta["files"]),
            "files": list(self.meta["files"].keys())
        }
    
    def list_skills_data(self) -> List[str]:
        """列出技能生成的数据文件"""
        return [f for f in self.meta["files"] if f.startswith("skills/")]

# claw_douyin_downloader.py
import re
import aiohttp
import asyncio
from typing import Optional

class DouyinDownloader:
    """
    抖音视频下载器（无水印）
    用于提取视频/音频进行克隆
    """
    
    def __init__(self):
        self.api_endpoint = "https://api.douyin.wang/api"  # 示例API
        self.headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        }
    
    async def download(self, share_url: str, extract_audio: bool = False) -> str:
        """
        下载抖音视频（无水印）
        
        Args:
            share_url: 抖音分享链接（如 https://v.douyin.com/xxxxx）
            extract_audio: 是否同时提取音频
        
        Returns:
            本地文件路径
        """
        # 1. 解析分享链接获取视频ID
        video_id = await self._parse_share_url(share_url)
        
        # 2. 获取无水印视频URL
        video_url = await self._get_no_watermark_url(video_id)
        
        # 3. 下载视频
        local_path = f"/tmp/douyin_{video_id}.mp4"
        await self._download_file(video_url, local_path)
        
        # 4. 如需提取音频
        if extract_audio:
            audio_path = await self._extract_audio(local_path)
            return {"video": local_path, "audio": audio_path}
        
        return local_path
    
    async def _parse_share_url(self, share_url: str) -> str:
        """解析短链接获取真实视频ID"""
        # 处理类似 https://v.douyin.com/xxxxx 的短链接
        async with aiohttp.ClientSession() as session:
            async with session.get(share_url, headers=self.headers, allow_redirects=True) as resp:
                final_url = str(resp.url)
                # 从URL中提取视频ID
                match = re.search(r'/video/(\d+)', final_url)
                if match:
                    return match.group(1)
                raise ValueError("无法解析视频ID")
    
    async def _get_no_watermark_url(self, video_id: str) -> str:
        """通过API获取无水印视频地址"""
        # 调用第三方去水印API或解析官方API
        async with aiohttp.ClientSession() as session:
            api_url = f"{self.api_endpoint}/video/info?video_id={video_id}"
            async with session.get(api_url) as resp:
                data = await resp.json()
                return data.get("data", {}).get("no_watermark_url")
    
    async def _download_file(self, url: str, save_path: str):
        """下载文件到本地"""
        async with aiohttp.ClientSession() as session:
            async with session.get(url) as resp:
                with open(save_path, 'wb') as f:
                    f.write(await resp.read())
    
    async def _extract_audio(self, video_path: str) -> str:
        """使用ffmpeg提取音频"""
        import subprocess
        audio_path = video_path.replace(".mp4", ".mp3")
        cmd = [
            "ffmpeg", "-i", video_path, 
            "-vn", "-acodec", "libmp3lame", 
            "-q:a", "2", audio_path, "-y"
        ]
        subprocess.run(cmd, check=True)
        return audio_path

            return await self._clone_avatar(**kwargs)
        elif action == "clone_voice":
            return await self._clone_voice(**kwargs)
        elif action == "generate_video":
            return await self._generate_video(**kwargs)
        elif action == "full_pipeline":
            return await self._full_pipeline(**kwargs)
        else:
            return {"error": f"未知action: {action}"}
    
    async def _clone_avatar(self, image_url: str, name: str):
        """克隆形象"""
        # 下载图片
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.get(image_url) as resp:
                image_data = await resp.read()
        
        success, msg, avatar_url = await self.engine.clone_avatar(
            image_data, name
        )
        
        return {
            "success": success,
            "message": msg,
            "avatar_url": avatar_url,
            "profile_id": name
        }
    
    async def _clone_voice(self, audio_url: str, voice_name: str):
        """克隆声音"""
        import aiohttp
        async with aiohttp.ClientSession() as session:
            async with session.get(audio_url) as resp:
                audio_data = await resp.read()
        
        success, msg, voice_id = await self.engine.clone_voice(
            audio_data, voice_name
        )
        
        return {
            "success": success,
            "message": msg,
            "voice_id": voice_id
        }
    
    async def _generate_video(self, text: str, avatar_id: str = None, voice_id: str = None):
        """生成视频"""
        avatar_url = None
        if avatar_id and avatar_id in self.engine.profiles:
            avatar_url = self.engine.profiles[avatar_id].avatar_url
        
        result = await self.engine.generate_video(
            text=text,
            avatar_url=avatar_url,
            voice_id=voice_id
        )
        
        return result
    
    async def _full_pipeline(self, douyin_url: str, new_script: str):
        """完整流程"""
        return await self.engine.generate_from_douyin(
            douyin_url=douyin_url,
            new_script=new_script
        )

    def query(self, user_question: str) -> List[GuideSection]:
        """
        根据用户问题检索相关指南内容
        """
        # 意图识别关键词映射
        intent_keywords = {
            "部署": ["deployment"],
            "安装": ["deployment"],
            "怎么用": ["theory", "applications"],
            "是什么": ["theory"],
            "配置模型": ["advanced"],
            "Windows": ["deployment"],
            "WSL": ["deployment"],
            "小白": ["deployment"],
            "扣子": ["deployment"],
            "Kimi": ["deployment"],
            "飞书": ["deployment"],
            "数字人": ["applications"],
            "小红书": ["applications"],
            "监控": ["applications"],
            "第二大脑": ["applications"]
        }
        
        matched_sections = []
        question_lower = user_question.lower()
        
        # 匹配平台
        for platform_key, platform_name in self.deployment_platforms.items():
            if platform_key in question_lower or platform_name in question_lower:
                matched_sections.extend([
                    s for s in self.sections 
                    if s.platform == platform_key
                ])
        
        # 匹配意图
        for keyword, categories in intent_keywords.items():
            if keyword in question_lower:
                matched_sections.extend([
                    s for s in self.sections 
                    if s.category in categories and s not in matched_sections
                ])
        
        # 去重并排序（按难度）
        seen = set()
        unique_sections = []
        for s in matched_sections:
            if s.id not in seen:
                seen.add(s.id)
                unique_sections.append(s)
        
        return sorted(unique_sections, key=lambda x: x.difficulty)

        """

        platform_map = {
            "扣子": "coze", "coze": "coze",

            "智谱": "zhipu", "autoclaw": "zhipu"

        platform_key = platform_map.get(platform, platform)
        
        sections = [s for s in self.sections if s.platform == platform_key]
        
        if not sections:

        return {
            "platform": self.deployment_platforms.get(platform_key, platform),
            "steps": [self._format_section(s) for s in sections],

class OpenClawGuideAssistant:
    """
    基于指南的智能问答助手
    回答用户关于OpenClaw的任何问题
    """
    
    def __init__(self):
        self.kb = OpenClawGuideKnowledgeBase()
        self.common_issues = self._load_troubleshooting()
    
    def answer(self, question: str) -> str:
        """
        回答用户问题，基于指南内容
        """
        # 1. 检索相关章节
        sections = self.kb.query(question)
        
        # 2. 判断问题类型
        if "报错" in question or "错误" in question or "失败" in question:
            return self._troubleshoot(question)
        
        if "怎么部署" in question or "如何安装" in question:
            platform = self._extract_platform(question)
            guide = self.kb.get_deployment_guide(platform)
            return self._format_deployment_guide(guide)
        
        if "怎么用" in question or "能做什么" in question:
            return self._list_applications()
        
        # 3. 基于检索内容生成回答
        if sections:
            answer = f"根据《OpenClaw从入门到精通指南》，我为您找到以下内容：\n\n"
            for s in sections[:3]:
                answer += f"📖 **{s.title}**\n{s.content}\n\n"
                if s.code_snippets:
                    answer += "```bash\n" + "\n".join(s.code_snippets) + "\n```\n\n"
            return answer
        
        return "抱歉，我暂时无法回答这个问题。您可以查看完整指南：https://www.yuque.com/canghe/openclaw-guide"

# claw_dual_engine.py
from enum import Enum
from typing import Optional, Dict, Any, List
from dataclasses import dataclass

class EngineMode(Enum):
    OPENCLAW = "openclaw"          # Agent模式：技能调用、多轮对话
    CLAUDE_CODE = "claude_code"    # 编程模式：代码生成、项目管理
    HYBRID = "hybrid"              # 混合模式：自动切换
    AGENT_TEAM = "agent_team"      # 多智能体团队模式

@dataclass
class TaskContext:
    """任务上下文"""
    task_id: str
    mode: EngineMode
    workspace_path: Optional[str] = None
    skills_involved: List[str] = None
    code_files: List[str] = None
    artifacts: Dict[str, Any] = None

class DualEngineController:
    """
    OpenClaw + Claude Code 双引擎控制器
    根据任务类型自动选择最佳引擎
    """
    
    def __init__(self):
        self.openclaw = OpenClawEngine()
        self.claude_code = ClaudeCodeEngine()
        self.skill_registry = SkillRegistry()  # 32万+技能注册表
        self.current_mode = EngineMode.HYBRID
        
    async def process(self, user_input: str, context: Dict = None) -> Dict:
        """
        智能路由：判断使用哪个引擎处理
        
        路由规则：
        - 写代码/开发/调试 → Claude Code
        - 调用技能/查信息/自动化 → OpenClaw  
        - 复杂项目 → Agent Team（两者协作）
        """
        # 意图识别
        intent = self._classify_intent(user_input)
        
        if intent == "coding":
            return await self._route_to_claude_code(user_input, context)
        elif intent == "skill_execution":
            return await self._route_to_openclaw(user_input, context)
        elif intent == "complex_project":
            return await self._activate_agent_team(user_input, context)
        else:
            # 混合模式：先OpenClaw理解，再Claude Code执行
            return await self._hybrid_process(user_input, context)
    
    def _classify_intent(self, text: str) -> str:
        """分类用户意图"""
        coding_keywords = [
            "写代码", "开发", "debug", "修复bug", "创建项目", 
            "写一个系统", "搭建网站", "写app", "程序", "函数",
            "实现功能", "代码", "programming", "develop"
        ]
        
        skill_keywords = [
            "截图", "搜索", "下载", "分析", "生成图片", "查天气",
            "OCR", "识别", "转换", "skill", "技能"
        ]
        
        team_keywords = [
            "团队", "协作", "一起", "多角色", "agent team",
            "写作团队", "开发团队", "research", "write article"
        ]
        
        text_lower = text.lower()
        
        if any(k in text_lower for k in coding_keywords):
            return "coding"
        elif any(k in text_lower for k in team_keywords):
            return "complex_project"
        elif any(k in text_lower for k in skill_keywords):
            return "skill_execution"
        else:
            return "hybrid"
    
    async def _route_to_claude_code(self, prompt: str, context: Dict) -> Dict:
        """
        使用Claude Code引擎（编程模式）
        特点：无限Token、长上下文、代码专注
        """
        # 检查是否有claude.md配置
        claude_md = context.get("claude_md", self._default_claude_md())
        
        # 构建Claude Code专用prompt
        cc_prompt = f"""{claude_md}

任务：{prompt}

请使用Claude Code的方式处理：
1. 分析需求并制定计划
2. 创建或修改代码文件
3. 运行测试验证
4. 提供可运行的完整代码
"""
        
        # 调用Claude Code API（无限Token模式）
        result = await self.claude_code.execute(
            prompt=cc_prompt,
            workspace=context.get("workspace", "/tmp/claw_workspace"),
            timeout=None  # 无超时限制（无限Token）
        )
        
        return {
            "engine": "claude_code",
            "mode": "coding",
            "files_created": result.get("files", []),
            "code": result.get("code"),
            "explanation": result.get("explanation"),
            "run_command": result.get("run_command")
        }
    
    async def _route_to_openclaw(self, prompt: str, context: Dict) -> Dict:
        """
        使用OpenClaw引擎（Agent模式）
        特点：技能调用、工具使用、多步骤自动化
        """
        # 自动发现所需技能
        skills_needed = await self.skill_registry.discover_skills(prompt)
        
        # 构建OpenClaw执行链
        execution_plan = await self.openclaw.plan_execution(
            prompt=prompt,
            available_skills=skills_needed
        )
        
        # 执行
        result = await self.openclaw.execute_plan(execution_plan)
        
        return {
            "engine": "openclaw",
            "mode": "skill_execution",
            "skills_used": [s.name for s in skills_needed],
            "steps_completed": result.get("steps"),
            "artifacts": result.get("outputs")
        }
    
    async def _activate_agent_team(self, prompt: str, context: Dict) -> Dict:
        """
        激活Agent Team模式（多智能体协作）
        如：研究员+作家协作写文章
        """
        team = AgentTeam(self)
        
        # 分析任务需要哪些角色
        roles = self._determine_roles(prompt)
        
        # 创建团队
        for role in roles:
            team.add_agent(
                role=role["name"],
                expertise=role["expertise"],
                engine=role.get("engine", "openclaw")  # 或claude_code
            )
        
        # 启动协作
        result = await team.collaborate(prompt)
        
        return {
            "engine": "agent_team",
            "mode": "collaboration",
            "team_members": [a.role for a in team.agents],
            "contributions": result.get("contributions"),
            "final_output": result.get("output")
        }
    
    def _determine_roles(self, prompt: str) -> List[Dict]:
        """根据任务确定需要的角色"""
        if "写文章" in prompt or "公众号" in prompt:
            return [
                {"name": "researcher", "expertise": ["信息搜集", "资料整理"], "engine": "openclaw"},
                {"name": "writer", "expertise": ["写作", "文风模仿"], "engine": "claude_code"}
            ]
        elif "开发" in prompt or "系统" in prompt:
            return [
                {"name": "architect", "expertise": ["架构设计"], "engine": "claude_code"},
                {"name": "developer", "expertise": ["编码实现"], "engine": "claude_code"},
                {"name": "tester", "expertise": ["测试验证"], "engine": "openclaw"}
            ]
        return [{"name": "assistant", "expertise": ["通用任务"], "engine": "hybrid"}]
    
    def _default_claude_md(self) -> str:
        """默认claude.md配置"""
        return """# Claude Code Configuration

你是Claude Code，一个专业的编程助手。
- 优先使用TypeScript/React进行Web开发
- 遵循最佳实践，写清晰的注释
- 自动创建测试文件验证代码
- 遇到错误时自动修复
"""


class AgentTeam:
    """
    多智能体协作团队
    如Happycapy中的研究员+作家模式
    """
    
    def __init__(self, controller: DualEngineController):
        self.controller = controller
        self.agents: List[AgentMember] = []
        self.shared_context = {}
    
    def add_agent(self, role: str, expertise: List[str], engine: str):
        """添加团队成员"""
        self.agents.append(AgentMember(
            role=role,
            expertise=expertise,
            engine=engine,
            controller=self.controller
        ))
    
    async def collaborate(self, task: str) -> Dict:
        """团队协作执行任务"""
        contributions = []
        
        # 阶段1：并行研究/准备（OpenClaw擅长）
        research_agent = next((a for a in self.agents if "research" in a.role), None)
        if research_agent:
            research_result = await research_agent.execute(
                f"研究以下任务并整理关键信息：{task}"
            )
            self.shared_context["research"] = research_result
            contributions.append({"role": research_agent.role, "output": research_result})
        
        # 阶段2：创作/开发（Claude Code擅长）
        creator_agent = next((a for a in self.agents if a.role in ["writer", "developer"]), None)
        if creator_agent:
            creator_prompt = f"""基于以下研究结果进行创作：
{self.shared_context.get('research', '')}

原始任务：{task}
"""
            creation_result = await creator_agent.execute(creator_prompt)
            contributions.append({"role": creator_agent.role, "output": creation_result})
        
        # 阶段3：审核/优化（可选）
        
        return {
            "contributions": contributions,
            "output": creation_result if creator_agent else research_result
        }


@dataclass
class AgentMember:
    """团队成员"""
    role: str
    expertise: List[str]
    engine: str
    controller: DualEngineController
    
    async def execute(self, prompt: str) -> Any:
        """使用指定引擎执行任务"""
        if self.engine == "claude_code":
            return await self.controller._route_to_claude_code(prompt, {})
        else:
            return await self.controller._route_to_openclaw(prompt, {})

from typing import List, Dict
import base64

class VisualCanvas:
    """
    可视化画布系统
    展示代码、图片、文档等产出物（类似Happycapy的右侧画布）
    """
    
    def __init__(self):
        self.panels: List[CanvasPanel] = []
        self.layout = "split"  # split/tab/overlay
    
    def add_code_panel(self, filename: str, code: str, language: str = "python"):
        """添加代码面板"""
        self.panels.append(CodePanel(
            filename=filename,
            code=code,
            language=language,
            line_numbers=True
        ))
    
    def add_image_panel(self, filename: str, image_data: bytes):
        """添加图片面板"""
        base64_img = base64.b64encode(image_data).decode()
        self.panels.append(ImagePanel(
            filename=filename,
            base64_data=base64_img
        ))
    
    def add_preview_panel(self, url: str, title: str = "Preview"):
        """添加网页预览面板"""
        self.panels.append(PreviewPanel(
            url=url,
            title=title
        ))
    
    def add_markdown_panel(self, content: str, title: str = "Document"):
        """添加Markdown文档面板"""
        self.panels.append(MarkdownPanel(
            content=content,
            title=title
        ))
    
    def render(self) -> Dict:
        """渲染画布状态"""
        return {
            "layout": self.layout,
            "panels": [p.to_dict() for p in self.panels],
            "active_panel": 0
        }
    
    def auto_layout(self, outputs: List[Dict]):
        """
        根据输出类型自动布局
        如：代码+预览并排，图片画廊等
        """
        for output in outputs:
            if output["type"] == "code":
                self.add_code_panel(
                    output["filename"],
                    output["content"],
                    output.get("language", "text")
                )
            elif output["type"] == "image":
                self.add_image_panel(
                    output["filename"],
                    output["data"]
                )
            elif output["type"] == "web_preview":
                self.add_preview_panel(
                    output["url"],
                    output.get("title", "Preview")
                )
        
        # 自动调整布局
        if len(self.panels) > 2:
            self.layout = "grid"
        
        return self.render()

        根据任务描述自动发现相关技能
        
        如："解析发票PDF" → 发现 PaddleOCR, PDF-parser, Feishu-sync 等技能
        """

        candidates = self.vector_index.search(query_vector, top_k=top_k)

        scored_skills = []
        for skill_id, score in candidates:
            skill = self.skills.get(skill_id)
            if skill:

from dataclasses import dataclass
from datetime import datetime

class ImaNote:

class ImaNotebook:

class ImaSkill:
    """
    腾讯ima Skill封装
    支持笔记的增删改查和知识库管理
    """
    
    def __init__(self, api_key: str, base_url: str = "https://ima.tencent.com/api/v1"):
        self.api_key = api_key
        self.base_url = base_url
        self.headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json"
        }
    
    async def list_notebooks(self) -> List[ImaNotebook]:
        """列出所有笔记本"""
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"{self.base_url}/notebooks",
                headers=self.headers
            ) as resp:
                data = await resp.json()
                return [ImaNotebook(**item) for item in data.get("notebooks", [])]
    
    async def create_notebook(self, name: str) -> ImaNotebook:
        """创建新笔记本"""
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self.base_url}/notebooks",
                headers=self.headers,
                json={"name": name}
            ) as resp:
                data = await resp.json()
                return ImaNotebook(**data)
    
    async def create_note(
        self,
        title: str,
        content: str,
        notebook: str,
        tags: List[str] = None
    ) -> ImaNote:
        """
        创建笔记
        
        Args:
            title: 笔记标题
            content: 笔记内容（支持Markdown）
            notebook: 笔记本名称（如"读书笔记"）
            tags: 标签列表
        """
        payload = {
            "title": title,
            "content": content,
            "notebook": notebook,
            "tags": tags or [],
            "source": "claw_auto_sync",
            "created_by": "openclaw"
        }
        
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self.base_url}/notes",
                headers=self.headers,
                json=payload
            ) as resp:
                data = await resp.json()
                return ImaNote(**data)
    
    async def update_note(
        self,
        note_id: str,
        content: str = None,
        title: str = None,
        append: bool = False
    ) -> ImaNote:
        """
        更新笔记
        
        Args:
            append: True则追加内容，False则覆盖
        """
        # 先获取原笔记
        if append:
            old_note = await self.get_note(note_id)
            content = old_note.content + "\n\n" + content
        
        payload = {}
        if content:
            payload["content"] = content
        if title:
            payload["title"] = title
        
        async with aiohttp.ClientSession() as session:
            async with session.put(
                f"{self.base_url}/notes/{note_id}",
                headers=self.headers,
                json=payload
            ) as resp:
                data = await resp.json()
                return ImaNote(**data)
    
    async def get_note(self, note_id: str) -> Optional[ImaNote]:
        """获取笔记详情"""
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"{self.base_url}/notes/{note_id}",
                headers=self.headers
            ) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    return ImaNote(**data)
                return None
    
    async def search_notes(
        self,
        query: str,
        notebook: str = None,
        limit: int = 10
    ) -> List[ImaNote]:
        """
        搜索笔记（语义搜索）
        
        Args:
            query: 搜索关键词
            notebook: 限定在某个笔记本搜索
        """
        payload = {
            "query": query,
            "limit": limit,
            "search_fields": ["title", "content"]
        }
        if notebook:
            payload["notebook"] = notebook
        
        async with aiohttp.ClientSession() as session:
            async with session.post(
                f"{self.base_url}/notes/search",
                headers=self.headers,
                json=payload
            ) as resp:
                data = await resp.json()
                return [ImaNote(**item) for item in data.get("results", [])]
    
    async def delete_note(self, note_id: str) -> bool:
        """删除笔记"""
        async with aiohttp.ClientSession() as session:
            async with session.delete(
                f"{self.base_url}/notes/{note_id}",
                headers=self.headers
            ) as resp:
                return resp.status == 204
    
    # ========== 批量操作 ==========
    async def batch_sync_notes(
        self,
        notes: List[Dict],
        notebook: str,
        conflict_resolution: Literal["skip", "overwrite", "append"] = "append"
    ) -> Dict:
        """
        批量同步笔记
        
        Args:
            notes: 笔记列表 [{"title": "...", "content": "..."}]
            conflict_resolution: 冲突处理策略
        """
        results = {"created": 0, "updated": 0, "skipped": 0, "errors": []}
        
        for note_data in notes:
            try:
                # 检查是否已存在
                existing = await self._find_note_by_title(note_data["title"], notebook)
                
                if existing:
                    if conflict_resolution == "skip":
                        results["skipped"] += 1
                    elif conflict_resolution == "overwrite":
                        await self.update_note(existing.note_id, **note_data)
                        results["updated"] += 1
                    elif conflict_resolution == "append":
                        await self.update_note(
                            existing.note_id, 
                            content=note_data["content"],
                            append=True
                        )
                        results["updated"] += 1
                else:
                    await self.create_note(
                        title=note_data["title"],
                        content=note_data["content"],
                        notebook=notebook,
                        tags=note_data.get("tags", [])
                    )
                    results["created"] += 1
                    
            except Exception as e:
                results["errors"].append({"title": note_data.get("title"), "error": str(e)})
        
        return results
    
    async def _find_note_by_title(self, title: str, notebook: str) -> Optional[ImaNote]:
        """根据标题查找笔记"""
        notes = await self.search_notes(title, notebook=notebook, limit=5)
        for note in notes:
            if note.title == title:
                return note
        return None

from typing import Dict, List, Callable
from dataclasses import dataclass
import asyncio

@dataclass
class LobsterAgent:
    """小龙虾Agent定义"""
    name: str  # 如"内容总监"、"监控虾"
    role: str  # 职责描述
    skills: List[str]  # 安装的技能
    context: Dict  # 独享上下文
    message_queue: asyncio.Queue = None
    
    def __post_init__(self):
        self.message_queue = asyncio.Queue()

class LobsterArmy:
    """
    小龙虾军团（多Agent协作系统）
    每个Agent有独立上下文，专业分工
    """
    
    def __init__(self):
        self.agents: Dict[str, LobsterAgent] = {}
        self.coordinator = None  # 协调者Agent
    
    def register_agent(self, agent: LobsterAgent):
        """注册Agent"""
        self.agents[agent.name] = agent
        # 启动Agent消息循环
        asyncio.create_task(self._agent_loop(agent))
    
    async def _agent_loop(self, agent: LobsterAgent):
        """Agent消息处理循环"""
        while True:
            message = await agent.message_queue.get()
            
            # 根据消息类型处理
            if message["type"] == "file_changed":
                await self._handle_file_change(agent, message)
            elif message["type"] == "sync_request":
                await self._handle_sync_request(agent, message)
            elif message["type"] == "search_request":
                await self._handle_search_request(agent, message)
    
    async def send_message(self, from_agent: str, to_agent: str, message: Dict):
        """Agent间通信"""
        if to_agent not in self.agents:
            raise ValueError(f"Agent {to_agent} 不存在")
        
        message["from"] = from_agent
        await self.agents[to_agent].message_queue.put(message)
    
    async def broadcast(self, from_agent: str, message: Dict):
        """广播消息给所有Agent"""
        for name, agent in self.agents.items():
            if name != from_agent:
                await self.send_message(from_agent, name, message)
    
    # ========== 具体业务处理 ==========
    async def _handle_file_change(self, agent: LobsterAgent, message: Dict):
        """处理文件变更通知"""
        if "ima-skill" in agent.skills:
            # 内容总监收到监控虾的通知，执行ima同步
            file_info = message["data"]
            
            ima_skill = ImaSkill(api_key=agent.context.get("ima_api_key"))
            
            # 读取文件内容
            content = await self._read_file(file_info["path"])
            
            # 同步到ima
            if file_info["event_type"] == "new":
                await ima_skill.create_note(
                    title=file_info["title"],
                    content=content,
                    notebook=file_info["target_notebook"]
                )
            elif file_info["event_type"] == "modified":
                # 查找已有笔记并追加
                await ima_skill.batch_sync_notes(
                    [{"title": file_info["title"], "content": content}],
                    notebook=file_info["target_notebook"],
                    conflict_resolution="append"
                )
            
            # 通知发送方完成
            await self.send_message(
                agent.name,
                message["from"],
                {"type": "sync_complete", "file": file_info["title"]}
            )
    
    async def _handle_sync_request(self, agent: LobsterAgent, message: Dict):
        """处理同步请求"""
        pass
    
    async def _handle_search_request(self, agent: LobsterAgent, message: Dict):
        """处理搜索请求"""
        if "ima-skill" in agent.skills:
            ima_skill = ImaSkill(api_key=agent.context.get("ima_api_key"))
            
            query = message["query"]
            notebook = message.get("notebook")
            
            results = await ima_skill.search_notes(query, notebook=notebook)
            
            # 返回结果
            await self.send_message(
                agent.name,
                message["from"],
                {
                    "type": "search_results",
                    "query": query,
                    "results": [{"title": r.title, "content": r.content[:500]} for r in results]
                }
            )
    
    async def _read_file(self, path: str) -> str:
        """读取文件内容"""
        with open(path, 'r', encoding='utf-8') as f:
            return f.read()

# claw_file_monitor.py
import os
import asyncio
from pathlib import Path
from typing import List, Callable, Dict
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler, FileCreatedEvent, FileModifiedEvent

class FileChangeHandler(FileSystemEventHandler):
    """文件变更处理器"""
    
    def __init__(self, callback: Callable, patterns: List[str] = None):
        self.callback = callback
        self.patterns = patterns or ["*.md", "*.txt"]
    
    def on_created(self, event):
        if not event.is_directory:
            if any(event.src_path.endswith(p) for p in self.patterns):
                asyncio.create_task(self.callback({
                    "event_type": "new",
                    "path": event.src_path,
                    "filename": os.path.basename(event.src_path)
                }))
    
    def on_modified(self, event):
        if not event.is_directory:
            if any(event.src_path.endswith(p) for p in self.patterns):
                asyncio.create_task(self.callback({
                    "event_type": "modified",
                    "path": event.src_path,
                    "filename": os.path.basename(event.src_path)
                }))

class AutoSyncManager:
    """
    自动同步管理器
    监控本地文件夹 → 自动同步到ima
    """
    
    def __init__(self, lobster_army: LobsterArmy):
        self.army = lobster_army
        self.observers: Dict[str, Observer] = {}
        self.sync_rules: List[SyncRule] = []
    
    def add_sync_rule(
        self,
        watch_path: str,
        target_notebook: str,
        file_pattern: str = "*.md",
        sync_frequency: int = 300  # 5分钟检查一次
    ):
        """
        添加同步规则
        
        Args:
            watch_path: 监控的本地文件夹
            target_notebook: 同步到ima的笔记本名称
            file_pattern: 文件匹配模式
        """
        rule = SyncRule(
            watch_path=watch_path,
            target_notebook=target_notebook,
            file_pattern=file_pattern
        )
        self.sync_rules.append(rule)
        
        # 设置文件监控
        handler = FileChangeHandler(
            callback=lambda event: self._on_file_change(event, rule),
            patterns=[file_pattern]
        )
        
        observer = Observer()
        observer.schedule(handler, watch_path, recursive=True)
        observer.start()
        
        self.observers[watch_path] = observer
        
        print(f"[监控] 开始监控: {watch_path} → ima:{target_notebook}")
    
    async def _on_file_change(self, event: Dict, rule: SyncRule):
        """文件变更回调"""
        # 构建文件信息
        file_info = {
            "path": event["path"],
            "title": Path(event["path"]).stem,
            "event_type": event["event_type"],
            "target_notebook": rule.target_notebook
        }
        
        # 通知监控虾
        await self.army.send_message(
            "system",
            "监控虾",
            {
                "type": "file_changed",
                "data": file_info
            }
        )
        
        # 监控虾通知内容总监执行同步
        await self.army.send_message(
            "监控虾",
            "内容总监",
            {
                "type": "file_changed",
                "data": file_info
            }
        )
    
    def stop_all(self):
        """停止所有监控"""
        for observer in self.observers.values():
            observer.stop()
            observer.join()


@dataclass
class SyncRule:
    """同步规则"""
    watch_path: str
    target_notebook: str
    file_pattern: str

# claw_inspiration_capture.py
class InspirationCapture:
    """
    灵感即时记录系统
    对话中加"灵感："前缀自动保存到ima
    """
    
    def __init__(self, ima_skill: ImaSkill, lobster_army: LobsterArmy):
        self.ima = ima_skill
        self.army = lobster_army
        self.trigger_prefix = "灵感："
        self.target_notebook = "灵感笔记"
    
    async def process_message(self, user_message: str) -> Optional[Dict]:
        """
        处理用户消息，检测是否为灵感记录
        
        Returns:
            如果是灵感记录，返回处理结果；否则返回None
        """
        if not user_message.startswith(self.trigger_prefix):
            return None
        
        # 提取灵感内容
        inspiration_content = user_message[len(self.trigger_prefix):].strip()
        
        # 格式化内容（按天追加）
        today = datetime.now().strftime("%Y-%m-%d")
        formatted_content = f"• {inspiration_content}"
        
        # 查找今天的灵感笔记
        existing = await self.ima.search_notes(
            query=today,
            notebook=self.target_notebook,
            limit=1
        )
        
        if existing and today in existing[0].title:
            # 追加到今天的笔记
            await self.ima.update_note(
                existing[0].note_id,
                content=formatted_content,
                append=True
            )
            return {
                "status": "appended",
                "note_title": existing[0].title,
                "content": inspiration_content
            }
        else:
            # 创建新笔记
            await self.ima.create_note(
                title=f"灵感 {today}",
                content=formatted_content,
                notebook=self.target_notebook,
                tags=["灵感", "自动记录"]
            )
            return {
                "status": "created",
                "note_title": f"灵感 {today}",
                "content": inspiration_content
            }
    
    async def search_inspirations(self, keyword: str) -> List[Dict]:
        """搜索灵感笔记"""
        results = await self.ima.search_notes(
            query=keyword,
            notebook=self.target_notebook
        )
        return [{"title": r.title, "content": r.content} for r in results]

# claw_installation_manager.py
import subprocess
import os
import platform
import asyncio
import json
from typing import List, Dict, Optional, Tuple
from dataclasses import dataclass

import aiohttp

@dataclass
class SystemRequirement:

class OpenClawInstaller:
    """
    OpenClaw自动化安装管理器（类似ClawPilot）
    一键安装、环境检查、自动修复
    """
    
    def __init__(self):
        self.system = platform.system()
        self.arch = platform.machine()
        self.install_log = []
        self.auto_approve = False  # 自动批准危险操作（如sudo）
        
    async def install(self, auto_approve: bool = False) -> Dict:
        """
        一键安装OpenClaw完整流程
        
        Returns:
            {
                "success": bool,
                "steps": [...],
                "errors": [...],
                "openclaw_version": str
            }
        """
        self.auto_approve = auto_approve
        results = {"success": False, "steps": [], "errors": []}
        
        try:
            # Step 1: 环境检查
            env_check = await self._check_environment()
            results["steps"].append({"name": "环境检查", "result": env_check})
            
            # Step 2: 安装缺失依赖
            for dep in env_check["missing"]:
                install_result = await self._install_dependency(dep)
                results["steps"].append({
                    "name": f"安装 {dep}", 
                    "result": install_result
                })
                if not install_result["success"]:
                    results["errors"].append(f"安装 {dep} 失败: {install_result['error']}")
            
            # Step 3: 安装OpenClaw CLI
            if not env_check["openclaw_installed"]:
                oc_result = await self._install_openclaw_cli()
                results["steps"].append({"name": "安装OpenClaw CLI", "result": oc_result})
            
            # Step 4: 权限修复
            if self.system == "Darwin":  # macOS
                fix_result = await self._fix_permissions()
                results["steps"].append({"name": "权限修复", "result": fix_result})
            
            # Step 5: 验证安装
            verify = await self._verify_installation()
            results["steps"].append({"name": "安装验证", "result": verify})
            
            results["success"] = verify["success"]
            results["openclaw_version"] = verify.get("version")
            
        except Exception as e:
            results["errors"].append(str(e))
        
        return results
    
    async def _check_environment(self) -> Dict:
        """
        检查系统环境（如ClawPilot的环境检查）
        检查：操作系统、架构、Shell、Xcode、Homebrew、nvm、Node.js、npm、git、OpenClaw
        """
        checks = {
            "os": {"name": "操作系统", "cmd": "uname -a"},
            "arch": {"name": "架构", "cmd": "uname -m"},
            "shell": {"name": "Shell", "cmd": "echo $SHELL"},
            "xcode": {"name": "Xcode CLT", "cmd": "xcode-select -p 2>/dev/null || echo 'NOT installed'"},
            "homebrew": {"name": "Homebrew", "cmd": "brew --version 2>/dev/null || echo 'NOT installed'"},
            "nvm": {"name": "nvm", "cmd": "source $HOME/.nvm/nvm.sh 2>/dev/null && nvm --version || echo 'NOT installed'"},
            "node": {"name": "Node.js", "cmd": "node --version 2>/dev/null || echo 'NOT installed'"},
            "npm": {"name": "npm", "cmd": "npm --version 2>/dev/null || echo 'NOT installed'"},
            "git": {"name": "git", "cmd": "git --version 2>/dev/null || echo 'NOT installed'"},
            "openclaw": {"name": "OpenClaw", "cmd": "openclaw --version 2>/dev/null || echo 'NOT installed'"}
        }
        
        results = {}
        missing = []
        
        for key, check in checks.items():
            try:
                result = await self._run_shell(check["cmd"])
                results[key] = {
                    "name": check["name"],
                    "installed": "NOT installed" not in result,
                    "version": result.strip() if "NOT installed" not in result else None,
                    "raw": result.strip()
                }
                if not results[key]["installed"]:
                    missing.append(key)
            except Exception as e:
                results[key] = {"installed": False, "error": str(e)}
                missing.append(key)
        
        return {
            "checks": results,
            "missing": missing,
            "summary": f"已安装: {len(results) - len(missing)}/{len(results)}"
        }
    
    async def _install_dependency(self, dep: str) -> Dict:
        """
        安装单个依赖（支持后台轮询检测）
        """
        installers = {
            "homebrew": self._install_homebrew,
            "nvm": self._install_nvm,
            "node": self._install_node_via_nvm,
            "npm": lambda: {"success": True, "note": "随Node.js安装"},
            "git": self._install_git,
            "openclaw": self._install_openclaw_cli
        }
        
        installer = installers.get(dep)
        if installer:
            return await installer()
        
        return {"success": False, "error": f"未知依赖: {dep}"}
    
    async def _install_homebrew(self) -> Dict:
        """安装Homebrew"""
        cmd = '/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"'
        return await self._run_install_with_progress("Homebrew", cmd, timeout=300)
    
    async def _install_nvm(self) -> Dict:
        """
        安装nvm（Node Version Manager）
        使用后台安装+轮询检测（如ClawPilot所示）
        """
        # 后台安装
        install_cmd = 'curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.40.0/install.sh | bash'
        
        # 创建临时日志文件
        log_file = f"/tmp/nvm-install-{os.getpid()}.log"
        
        # 后台启动安装
        process = await asyncio.create_subprocess_shell(
            f"{install_cmd} > {log_file} 2>&1 & echo $!",
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        stdout, _ = await process.communicate()
        pid = stdout.decode().strip()
        
        # 轮询检测安装状态（如ClawPilot所示）
        max_wait = 300  # 5分钟
        check_interval = 5
        
        for i in range(0, max_wait, check_interval):
            await asyncio.sleep(check_interval)
            
            # 检查进程是否还在运行
            check_process = await asyncio.create_subprocess_shell(
                f"ps -p {pid} > /dev/null 2>&1 && echo 'running' || echo 'finished'",
                stdout=asyncio.subprocess.PIPE
            )
            stdout, _ = await check_process.communicate()
            status = stdout.decode().strip()
            
            # 读取日志
            try:
                with open(log_file, 'r') as f:
                    log_content = f.read()
            except:
                log_content = ""
            
            if status == "finished":
                # 检查是否安装成功
                verify = await self._run_shell("source $HOME/.nvm/nvm.sh 2>/dev/null && nvm --version")
                if "0." in verify:
                    return {
                        "success": True,
                        "version": verify.strip(),
                        "log": log_content[-500:]  # 最后500字符
                    }
                else:
                    return {
                        "success": False,
                        "error": "安装完成但验证失败",
                        "log": log_content
                    }
        
        return {"success": False, "error": "安装超时"}
    
    async def _install_node_via_nvm(self) -> Dict:
        """通过nvm安装Node.js"""
        cmd = "source $HOME/.nvm/nvm.sh && nvm install node && nvm use node"
        return await self._run_install_with_progress("Node.js", cmd, timeout=120)
    
    async def _install_openclaw_cli(self) -> Dict:
        """安装OpenClaw CLI"""
        # 使用官方安装脚本
        cmd = "curl -fsSL https://openclaw.ai/install.sh | bash"
        return await self._run_install_with_progress("OpenClaw", cmd, timeout=180)
    
    async def _fix_permissions(self) -> Dict:
        """
        修复权限问题（如npm缓存目录权限）
        """
        fixes = []
        
        # 检查npm缓存权限
        npm_cache = Path.home() / ".npm"
        if npm_cache.exists():
            check_cmd = f"ls -la {npm_cache} | head -5"
            result = await self._run_shell(check_cmd)
            
            if "root" in result:
                # 需要修复权限
                if self.auto_approve:
                    fix_cmd = f'sudo chown -R $(whoami) "{npm_cache}"'
                    await self._run_shell(fix_cmd)
                    fixes.append(f"修复npm缓存权限: {npm_cache}")
                else:
                    return {
                        "success": False,
                        "needs_sudo": True,
                        "manual_fix": f'sudo chown -R $(whoami) "{npm_cache}"',
                        "description": "npm缓存目录存在root权限文件"
                    }
        
        return {"success": True, "fixes": fixes}
    
    async def _verify_installation(self) -> Dict:
        """验证OpenClaw安装"""
        try:
            version = await self._run_shell("source $HOME/.nvm/nvm.sh 2>/dev/null && openclaw --version")
            doctor = await self._run_shell("source $HOME/.nvm/nvm.sh 2>/dev/null && openclaw doctor 2>&1 | head -20")
            
            return {
                "success": "OpenClaw" in doctor or "OPENCLAW" in doctor,
                "version": version.strip(),
                "doctor_output": doctor
            }
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    async def _run_shell(self, cmd: str, timeout: int = 60) -> str:
        """执行shell命令"""
        process = await asyncio.create_subprocess_shell(
            cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        stdout, stderr = await asyncio.wait_for(process.communicate(), timeout=timeout)
        
        if process.returncode != 0 and stderr:
            raise Exception(stderr.decode())
        
        return stdout.decode()
    
    async def _run_install_with_progress(self, name: str, cmd: str, timeout: int) -> Dict:
        """执行安装并报告进度"""
        try:
            process = await asyncio.create_subprocess_shell(
                cmd,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE
            )
            
            stdout, stderr = await asyncio.wait_for(
                process.communicate(), 
                timeout=timeout
            )
            
            return {
                "success": process.returncode == 0,
                "output": stdout.decode(),
                "error": stderr.decode() if stderr else None
            }
            
        except asyncio.TimeoutError:
            return {"success": False, "error": f"{name}安装超时"}


class ModelConfigurator:
    """
    模型配置自动化（如ClawPilot的自动配置）
    无需手动修改配置文件
    """
    
    def __init__(self, openclaw_path: str = None):
        self.openclaw_path = openclaw_path or "openclaw"
        self.config_dir = Path.home() / ".openclaw"
        
    async def configure_model(
        self, 
        provider: str,  # aliyun, volcengine, tencent, anthropic, openai
        api_key: str,
        model_name: str = None,
        base_url: str = None
    ) -> Dict:
        """
        自动配置模型参数
        
        Args:
            provider: 提供商名称
            api_key: API密钥
            model_name: 模型名称（如 qwen-max, claude-3-opus）
            base_url: 自定义API地址（可选）
        """
        
        # 提供商配置模板
        provider_configs = {
            "aliyun": {
                "base_url": "https://dashscope.aliyuncs.com/compatible-mode/v1",
                "models": ["qwen-max", "qwen-plus", "qwen-coder-plus"]
            },
            "volcengine": {
                "base_url": "https://ark.cn-beijing.volces.com/api/v3",
                "models": ["doubao-pro-32k", "doubao-lite-32k"]
            },
            "tencent": {
                "base_url": "https://api.tencentcloud.com/v1",
                "models": ["hunyuan-pro", "hunyuan-standard"]
            },
            "anthropic": {
                "base_url": "https://api.anthropic.com/v1",
                "models": ["claude-3-opus", "claude-3-sonnet"]
            }
        }
        
        config = provider_configs.get(provider, {})
        final_base_url = base_url or config.get("base_url")
        final_model = model_name or config.get("models", [""])[0]
        
        # 使用openclaw CLI配置（无需手动改文件）
        cmds = [
            f'{self.openclaw_path} config set providers.{provider}.api_key "{api_key}"',
            f'{self.openclaw_path} config set providers.{provider}.base_url "{final_base_url}"',
            f'{self.openclaw_path} config set agents.defaults.model "{final_model}"',
            f'{self.openclaw_path} config set agents.defaults.provider "{provider}"'
        ]
        
        results = []
        for cmd in cmds:
            try:
                result = await self._run_cmd(cmd)
                results.append({"cmd": cmd, "success": True, "output": result})
            except Exception as e:
                results.append({"cmd": cmd, "success": False, "error": str(e)})
        
        # 重启Gateway使配置生效
        restart_result = await self._restart_gateway()
        
        return {
            "config_results": results,
            "restart": restart_result,
            "provider": provider,
            "model": final_model,
            "configured": all(r["success"] for r in results)
        }
    
    async def _restart_gateway(self) -> Dict:
        """重启OpenClaw Gateway"""
        try:
            await self._run_cmd(f"{self.openclaw_path} gateway restart")
            await asyncio.sleep(2)  # 等待重启
            
            # 验证状态
            status = await self._run_cmd(f"{self.openclaw_path} status")
            return {"success": "running" in status.lower(), "status": status}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    async def _run_cmd(self, cmd: str) -> str:
        process = await asyncio.create_subprocess_shell(
            cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        stdout, stderr = await process.communicate()
        
        if process.returncode != 0:
            raise Exception(stderr.decode())
        
        return stdout.decode()

        }
    
    async def auto_fix(self, issues: List[Dict]) -> Dict:
        """自动修复问题"""
        results = []
        
        for issue in issues:
            fix_cmd = issue.get("fix")
            if fix_cmd:

                    if fix_cmd == "configure_model":
                        result = {"note": "需要手动配置模型，请运行configure_model()"}
                    else:
                        result = await self._run_cmd(fix_cmd)
                    results.append({"issue": issue["type"], "fixed": True, "result": result})
                except Exception as e:
                    results.append({"issue": issue["type"], "fixed": False, "error": str(e)})
        
        return {"fixed_count": sum(1 for r in results if r["fixed"]), "details": results}
    
    async def _run_cmd(self, cmd: str) -> str:
        process = await asyncio.create_subprocess_shell(
            cmd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE
        )
        stdout, _ = await process.communicate()
        return stdout.decode()

@dataclass
class ExtractedConcept:
    """提取的概念节点"""
    id: str
    name: str
    type: str  # code, entity, concept
    source_file: str
    line_start: int = None

@dataclass
class Relation:
    """关系边"""
    source: str
    target: str
    relation_type: str  # calls, imports, references, describes
    confidence: float
    inferred: bool  # True=模型推断, False=原文提取

class GraphifyEngine:
    """
    Graphify引擎封装
    实现双阶段处理：本地AST解析 + LLM语义提取
    """
    
    def __init__(self, workspace_path: str, llm_client=None):
        self.workspace = Path(workspace_path)
        self.cache_dir = Path.home() / ".graphify" / "cache"
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.llm = llm_client  # Claude API客户端
        
        # Tree-sitter语言映射
        self.language_parsers = {
            ".py": "python",
            ".ts": "typescript",
            ".js": "javascript",
            ".go": "go",
            ".rs": "rust",
            ".java": "java",
            ".cpp": "cpp",
            ".c": "c",
            ".rb": "ruby",
            ".php": "php"
        }
    
    async def process_directory(
        self, 
        target_path: str = None,
        mode: str = "full"  # full, update, cluster-only
    ) -> Dict:
        """
        处理目录构建知识图谱
        
        Args:
            target_path: 目标目录，默认当前workspace
            mode: 
                - full: 完整重建
                - update: 只更新变更文件
                - cluster-only: 只重新聚类
        """
        target = Path(target_path) if target_path else self.workspace
        
        # 阶段1：收集所有文件
        files = self._collect_files(target)
        
        # 阶段2：双阶段提取
        concepts = []
        relations = []
        
        for file_path in files:
            file_hash = self._compute_hash(file_path)
            
            # 检查缓存
            if mode == "update" and self._is_cached(file_path, file_hash):
                cached = self._load_cache(file_path)
                concepts.extend(cached["concepts"])
                relations.extend(cached["relations"])
                continue
            
            # 根据文件类型选择提取方式
            file_type = self._classify_file(file_path)
            
            if file_type == FileType.CODE:
                # 阶段2a：代码用本地AST解析（零Token）
                c, r = await self._extract_from_code(file_path)
            elif file_type == FileType.DOCUMENT:
                # 阶段2b：文档用LLM提取
                c, r = await self._extract_from_document(file_path)
            elif file_type == FileType.PAPER:
                # 阶段2c：PDF用LLM提取
                c, r = await self._extract_from_paper(file_path)
            elif file_type == FileType.IMAGE:
                # 阶段2d：图片用Claude Vision提取
                c, r = await self._extract_from_image(file_path)
            
            concepts.extend(c)
            relations.extend(r)
            
            # 缓存结果
            self._save_cache(file_path, file_hash, {"concepts": c, "relations": r})
        
        # 阶段3：构建图
        graph = self._build_graph(concepts, relations)
        
        # 阶段4：社区发现聚类（Leiden算法）
        communities = self._detect_communities(graph)
        
        # 阶段5：生成输出
        output = self._generate_output(graph, communities, target)
        
        return output

    def _build_graph(self, concepts: List[ExtractedConcept], relations: List[Relation]) -> Dict:
        """构建图结构"""

                "type": relation.relation_type,

            })

    def _detect_communities(self, graph: Dict) -> List[Dict]:
        """
        使用Leiden算法进行社区发现（无需向量数据库）
        基于图拓扑结构聚类
        """
        # 使用python-louvain或leidenalg库
        try:
            import community as community_louvain
            import networkx as nx
            
            G = nx.Graph()
            for node_id, node_data in graph["nodes"].items():
                G.add_node(node_id, **node_data)
            
            for edge in graph["edges"]:
                G.add_edge(edge["source"], edge["target"], **edge)
            
            # Leiden/Louvain社区发现
            partition = community_louvain.best_partition(G)
            
            # 按社区分组
            communities = {}
            for node_id, comm_id in partition.items():
                if comm_id not in communities:
                    communities[comm_id] = {
                        "id": comm_id,
                        "name": f"Community {comm_id}",
                        "nodes": [],
                        "size": 0
                    }
                communities[comm_id]["nodes"].append(node_id)
                communities[comm_id]["size"] += 1
            
            return list(communities.values())
            
        except ImportError:
            # 降级为简单连通分量
            return [{"id": 0, "name": "Main", "nodes": list(graph["nodes"].keys())}]

class GraphifyWatcher:
    """
    文件监听模式（--watch）
    代码文件改动后立即更新图谱
    """
    
    def __init__(self, engine: GraphifyEngine):
        self.engine = engine
        self.observers = []
    
    def start_watch(self, target_path: str):
        """启动文件监听"""
        from watchdog.observers import Observer
        from watchdog.events import FileSystemEventHandler
        
        class GraphifyHandler(FileSystemEventHandler):
            def __init__(self, engine):
                self.engine = engine
            
            def on_modified(self, event):
                if not event.is_directory:
                    # 检查文件类型
                    ext = Path(event.src_path).suffix
                    if ext in ['.py', '.ts', '.js', '.md']:
                        # 异步触发增量更新
                        asyncio.create_task(self._update_graph(event.src_path))
            
            async def _update_graph(self, file_path):
                # 只更新变更文件
                await self.engine.process_directory(mode="update")
        
        observer = Observer()
        handler = GraphifyHandler(self.engine)
        observer.schedule(handler, target_path, recursive=True)
        observer.start()
        
        self.observers.append(observer)
        print(f"[Graphify] 开始监听: {target_path}")
    
    def stop_all(self):
        """停止所有监听"""
        for observer in self.observers:
            observer.stop()
            observer.join()

class ChangeProposal:

    name: str

class OpenSpecEngine:
    """
    OpenSpec规范引擎
    实现 propose → apply → archive 三步工作流
    """
    
    def __init__(self, project_root: str):
        self.root = Path(project_root)
        self.openspec_dir = self.root / "openspec"
        self.config_file = self.openspec_dir / "config.yaml"
        
        # 确保目录结构存在
        self._ensure_structure()
    
    def _ensure_structure(self):
        """确保OpenSpec目录结构"""
        (self.openspec_dir / "specs").mkdir(parents=True, exist_ok=True)
        (self.openspec_dir / "changes").mkdir(parents=True, exist_ok=True)
        
        if not self.config_file.exists():
            self._init_config()
    
    def _init_config(self):
        """初始化配置模板"""
        default_config = {
            "project": {
                "name": self.root.name,
                "tech_stack": "",      # 待填写
                "code_style": "",      # 待填写
                "business_context": "" # 待填写
            },
            "rules": {
                "archive": [
                    "归档前检查：确认已创建MR",
                    "归档后更新：自动更新INDEX.md索引"
                ]
            },
            "mcp_servers": [],
            "skills": []
        }
        
        with open(self.config_file, 'w', encoding='utf-8') as f:
            yaml.dump(default_config, f, allow_unicode=True)
    
    async def propose(self, requirement: str, context: Dict = None) -> ChangeProposal:
        """
        Step 1: /opsx:propose - 生成规划文档
        
        AI根据需求生成：
        - proposal.md: 需求背景与目标
        - design.md: 技术方案与架构决策
        - tasks.md: 带复选框的实施清单
        - specs/: 规范增量
        """
        change_name = f"change_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        change_dir = self.openspec_dir / "changes" / change_name
        change_dir.mkdir(parents=True)
        
        # 读取现有规范作为上下文
        existing_specs = self._load_existing_specs()
        config = self._load_config()
        
        # 生成proposal.md
        proposal_content = await self._generate_proposal(
            requirement, existing_specs, config
        )
        proposal_path = change_dir / "proposal.md"
        with open(proposal_path, 'w', encoding='utf-8') as f:
            f.write(proposal_content)
        
        # 生成design.md
        design_content = await self._generate_design(
            proposal_content, existing_specs, config
        )
        design_path = change_dir / "design.md"
        with open(design_path, 'w', encoding='utf-8') as f:
            f.write(design_content)
        
        # 生成tasks.md（AI施工清单）
        tasks_content = await self._generate_tasks(
            proposal_content, design_content
        )
        tasks_path = change_dir / "tasks.md"
        with open(tasks_path, 'w', encoding='utf-8') as f:
            f.write(tasks_content)
        
        # 创建specs目录（规范增量）
        (change_dir / "specs").mkdir(exist_ok=True)
        
        return ChangeProposal(
            name=change_name,
            proposal_md=str(proposal_path),
            design_md=str(design_path),
            tasks_md=str(tasks_path),
            specs_dir=str(change_dir / "specs")
        )
    
    async def apply(self, change_name: str, auto_execute: bool = False) -> Dict:
        """
        Step 2: /opsx:apply - 按图施工
        
        AI读取：
        1. proposal.md - 理解变更意图
        2. design.md - 理解技术方案
        3. tasks.md - 获取任务清单
        4. openspec/specs/ - 获取项目整体记忆
        
        然后：
        - 跨文件批量生成/修改代码
        - 每完成一项在tasks.md打钩
        - 自动补充单元测试
        - 修复编译错误/Lint警告
        """
        change_dir = self.openspec_dir / "changes" / change_name
        
        # 读取契约文件
        proposal = self._read_file(change_dir / "proposal.md")
        design = self._read_file(change_dir / "design.md")
        tasks = self._read_file(change_dir / "tasks.md")
        existing_specs = self._load_existing_specs()
        
        # 解析tasks.md中的任务清单
        task_list = self._parse_tasks(tasks)
        
        results = []
        for i, task in enumerate(task_list):
            if task["completed"]:
                continue
            
            # 执行单个任务
            if auto_execute:
                result = await self._execute_task(
                    task, proposal, design, existing_specs
                )
                
                # 标记完成
                task["completed"] = True
                task["result"] = result
                
                results.append({
                    "task": task["description"],
                    "status": "completed",
                    "files_modified": result.get("files", [])
                })
        
        # 更新tasks.md
        self._update_tasks_md(change_dir / "tasks.md", task_list)
        
        return {
            "change_name": change_name,
            "tasks_completed": len([t for t in task_list if t["completed"]]),
            "total_tasks": len(task_list),
            "results": results
        }
    
    async def archive(self, change_name: str) -> Dict:
        """
        Step 3: /opsx:archive - 规范归档
        
        MR通过后执行：
        1. 将changes/[name]/specs/合并到openspec/specs/
        2. 更新项目记忆（整体规范）
        3. 清空变更草稿
        4. 可选：生成/更新INDEX.md索引
        """
        change_dir = self.openspec_dir / "changes" / change_name
        specs_source = change_dir / "specs"
        specs_target = self.openspec_dir / "specs" / change_name
        
        # 归档规范增量
        if specs_source.exists():
            import shutil
            shutil.copytree(specs_source, specs_target, dirs_exist_ok=True)
        
        # 更新INDEX.md
        await self._update_index()
        
        # 可选：清空变更草稿（或保留历史）
        # shutil.rmtree(change_dir)
        
        return {
            "archived": True,
            "specs_merged": str(specs_target),
            "index_updated": True
        }
    
    async def _generate_proposal(self, requirement: str, 
                                 existing_specs: Dict, 
                                 config: Dict) -> str:
        """生成proposal.md"""
        template = f"""# 变更提案：{requirement[:50]}

{requirement}

已存在功能：
{self._format_specs(existing_specs)}

（AI根据需求分析生成）

## 影响范围评估

- 前端：□ 是 □ 否
- 后端：□ 是 □ 否
- 数据库：□ 是 □ 否
- 接口变更：□ 是 □ 否

## 风险清单

1. 
2. 

"""
        return template
    
    async def _generate_design(self, proposal: str, 
                               existing_specs: Dict, 
                               config: Dict) -> str:
        """生成design.md"""
        return f"""# 技术方案设计

## 架构决策

### 选型依据
技术栈：{config.get('tech_stack', '未指定')}

### 关键设计

## 接口定义

## 数据模型

## 测试策略

## 实施步骤

"""
    
    async def _generate_tasks(self, proposal: str, design: str) -> str:
        """生成tasks.md（AI施工清单）"""
        return f"""# 实施任务清单

> 原则：控制在15项以内，避免AI上下文过长

- [ ] 1. 创建/修改数据模型定义
- [ ] 2. 实现核心业务逻辑
- [ ] 3. 编写单元测试
- [ ] 4. 更新接口文档
- [ ] 5. 运行Lint检查并修复

## 验收标准

- [ ] 所有单元测试通过
- [ ] Lint检查无错误
- [ ] 代码符合design.md规范

"""
    
    def _parse_tasks(self, tasks_md: str) -> List[Dict]:
        """解析tasks.md为结构化数据"""
        tasks = []
        lines = tasks_md.split('\n')
        
        for line in lines:
            line = line.strip()
            if line.startswith('- [ ]') or line.startswith('- [x]'):
                completed = line.startswith('- [x]')
                description = line[5:].strip()
                tasks.append({
                    "completed": completed,
                    "description": description
                })
        
        return tasks
    
    def _update_tasks_md(self, tasks_path: Path, tasks: List[Dict]):
        """更新tasks.md中的任务状态"""
        with open(tasks_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 替换任务状态
        for task in tasks:
            old = f"- [ ] {task['description']}"
            new = f"- [{'x' if task['completed'] else ' '}] {task['description']}"
            content = content.replace(old, new)
        
        with open(tasks_path, 'w', encoding='utf-8') as f:
            f.write(content)
    
    def _load_existing_specs(self) -> Dict:
        """加载现有规范"""
        specs_dir = self.openspec_dir / "specs"
        specs = {}
        
        if specs_dir.exists():
            for spec_file in specs_dir.rglob("*.md"):
                relative = spec_file.relative_to(specs_dir)
                specs[str(relative)] = self._read_file(spec_file)
        
        return specs
    
    def _load_config(self) -> Dict:
        """加载配置"""
        if self.config_file.exists():
            with open(self.config_file, 'r', encoding='utf-8') as f:
                return yaml.safe_load(f)
        return {}
    
    def _read_file(self, path: Path) -> str:
        """读取文件内容"""
        if path.exists():
            with open(path, 'r', encoding='utf-8') as f:
                return f.read()
        return ""
    
    def _format_specs(self, specs: Dict) -> str:
        """格式化规范列表"""
        return "\n".join([f"- {k}" for k in specs.keys()]) if specs else "（无）"

class InstallerSkill:
    async def install(self, project_root: str, config: Dict) -> Dict:
        """
        完整安装流程
        """
        steps = []
        
        # Step 1: 检查/安装Node.js
        node_check = await self._check_nodejs()
        if not node_check["installed"]:
            install_result = await self._install_nodejs()
            steps.append({"step": "安装Node.js", "result": install_result})
        
        # Step 2: 安装OpenSpec CLI
        cli_result = await self._install_openspec_cli()
        steps.append({"step": "安装OpenSpec CLI", "result": cli_result})
        
        # Step 3: 初始化项目结构
        init_result = await self._init_project(project_root)
        steps.append({"step": "初始化项目", "result": init_result})
        
        # Step 4: 生成Bridge Rule
        from claw_openspec_engine import BridgeRuleGenerator
        bridge_file = BridgeRuleGenerator().generate(project_root)
        steps.append({"step": "生成Bridge Rule", "file": bridge_file})
        
        # Step 5: 配置MCP（交互式）
        if config.get("setup_mcp"):
            mcp_result = await self._setup_mcp(project_root, config.get("mcp_tokens", {}))
            steps.append({"step": "配置MCP", "result": mcp_result})
        
        # Step 6: 批量安装Skills
        if config.get("skill_bundle"):
            skills_result = await self._install_skills(config["skill_bundle"])
            steps.append({"step": "安装Skills", "result": skills_result})
        
        return {
            "installed": True,
            "steps": steps,
            "next": "请编辑openspec/config.yaml补全项目配置"

        """配置MCP服务"""
        mcp_config = {
            "mcpServers": {
                "iwiki": {
                    "command": "mcp-server-iwiki",
                    "args": ["--token", tokens.get("iwiki", "")]
                },
                "tapd": {
                    "command": "mcp-server-tapd", 
                    "args": ["--token", tokens.get("tapd", "")]
                }
            }
        }
        
        # 安全检查：加入.gitignore
        gitignore_path = Path(project_root) / ".gitignore"

                f.write("\n.mcp.json\n")

class PressureLevel(Enum):

@dataclass
class FailureRecord:

class PUAEngine:
    """

        # 记录失败
        record = FailureRecord(
            attempt=len(self.failure_history) + 1,
            level=self.current_level,
            error=error,
            actions_taken=actions_taken,
            timestamp=self._now()
        )
        self.failure_history.append(record)
        
        # 升级压力等级
        if self.current_level.value < 4:
            self.current_level = PressureLevel(self.current_level.value + 1)
        
        # 获取当前压力配置
        pressure = self.pressure_templates[self.current_level]
        
        return {
            "level": self.current_level.name,
            "attempt": record.attempt,
            "pressure_prompt": pressure["prompt"].format(agent_role=self.agent_role),
            "requirements": pressure["requirements"],
            "tone": pressure["tone"],
            "action_required": self._get_required_actions()
        }
    
    def on_success(self) -> Dict:
        """

            self.current_level = PressureLevel.L0_TRUST

        elif level == PressureLevel.L2_SOUL:

            return ["checklist_7items", "write_retrospective"]
        elif level == PressureLevel.L4_GRADUATION:

        return []
    
    def _generate_lesson(self) -> str:
        """生成经验教训"""
        if len(self.failure_history) >= 3:
            return f"经过{len(self.failure_history)}次尝试才成功，建议复盘为什么前面失败。"
        return "一次性成功，继续保持！"
    
    def _now(self) -> str:
        import datetime
        return datetime.datetime.now().isoformat()
    
    def get_pressure_context(self) -> str:
        """获取当前压力上下文（注入到Agent prompt）"""
        pressure = self.pressure_templates[self.current_level]
        
        context = f"""
        【当前压力等级：{self.current_level.name} | 语气：{pressure['tone']}】

        - 重复同样的失败方案
        - 未经深度排查就放弃
        - 把锅甩给用户或环境
        """
        
        return context

class ProactiveAgentBehavior:

            "check_deps": self._check_dependencies,
            "scan_similar": self._scan_similar_issues,
            "verify_fix": self._verify_fix,
            "checklist": self._run_checklist
        }
    
    async def solve_problem(self, problem: str, context: Dict) -> Dict:
        """
        完整的问题解决流程（带PUA压力驱动）
        """
        max_attempts = 5
        actions_taken = []
        
        for attempt in range(1, max_attempts + 1):
            try:
                # 尝试解决问题
                result = await self._attempt_solve(problem, context, actions_taken)
                
                if result["success"]:
                    # 成功，重置压力
                    self.pua.on_success()
                    return {
                        "success": True,
                        "solution": result["solution"],
                        "attempts": attempt,
                        "actions": actions_taken
                    }
                
                # 失败，记录并升级压力
                actions_taken.append(result["action"])
                pressure = self.pua.on_failure(result["error"], actions_taken)
                
                # 根据压力等级执行强制操作
                await self._execute_pressure_actions(pressure)
                
            except Exception as e:
                actions_taken.append(f"attempt_{attempt}_failed")
                pressure = self.pua.on_failure(str(e), actions_taken)
        
        # 所有尝试都失败
        return {
            "success": False,
            "error": "已穷尽所有方案仍未解决",
            "attempts": max_attempts,
            "actions": actions_taken,

Version: 1.0.0

class Skill:
    def __init__(self):
        self.pua = None
        self.method_router = None
        self.agent = None
    
    async def setup(self, config: dict):
        from claw_pua_engine import PUAEngine
        from claw_bigtech_methodologies import BigTechMethodologyRouter
        from claw_proactive_agent import ProactiveAgentBehavior
        
        self.pua = PUAEngine(agent_role=config.get("role", "P8工程师"))
        self.method_router = BigTechMethodologyRouter()
        self.agent = ProactiveAgentBehavior(self.pua, self.method_router)
    
    async def run(self, action: str, **kwargs):
        """
        技能入口
        
        Actions:
        - solve: 解决问题（带PUA驱动）
        - set_method: 设置方法论
        - get_pressure: 获取当前压力状态
        - reset: 重置压力等级
        """
        
        if action == "solve":
            problem = kwargs["problem"]
            context = kwargs.get("context", {})
            
            result = await self.agent.solve_problem(problem, context)
            
            return {
                "success": result["success"],
                "solution": result.get("solution"),
                "attempts": result["attempts"],
                "pressure_applied": self.pua.current_level.name,
                "actions_taken": result["actions"]
            }
        
        elif action == "set_method":
            method = kwargs["method"]  # alibaba, bytedance, huawei等
            info = self.method_router.apply_methodology(method)
            
            return {
                "methodology": info["methodology"],
                "slogan": info["slogan"],
                "principles": info["principles"]
            }
        
        elif action == "get_pressure":
            return {
                "current_level": self.pua.current_level.name,
                "failure_history": len(self.pua.failure_history),
                "context": self.pua.get_pressure_context()
            }
        
        elif action == "reset":
            self.pua.on_success()
            return {"status": "压力已重置", "level": "L0_TRUST"}
        
        elif action == "list_methods":
            return {"methods": self.method_router.list_all()}
        
        else:
            return {"error": f"未知action: {action}"}

# claw_office_suite.py
from typing import List, Dict, Optional, BinaryIO
from pathlib import Path

class OfficeDocumentSkill:
    """
    办公文档处理技能集
    技能1 pdf + 技能2 docx + 技能3 xlsx + 技能4 pptx
    """
    
    def __init__(self):
        self.temp_dir = Path(tempfile.gettempdir()) / "claw_office"
        self.temp_dir.mkdir(exist_ok=True)
    
    # ========== 技能1: PDF处理 ==========
    async def pdf_process(
        self,
        action: str,  # merge, split, ocr, watermark, convert, extract
        files: List[str],
        output_name: str = None,
        **kwargs
    ) -> Dict:
        """
        PDF万能处理
        
        示例：
        - "合并这5份季报PDF，按时间排序"
        - "提取PDF前3页文字转成Word"
        - "给PDF加水印'内部资料'"
        - "旋转PDF 90度"
        """
        if action == "merge":
            return await self._pdf_merge(files, output_name or "merged.pdf", kwargs.get("sort_by"))
        
        elif action == "split":
            return await self._pdf_split(
                files[0], 
                pages=kwargs.get("pages"),  # "1-3,5,7-10"
                output_prefix=output_name or "split"
            )
        
        elif action == "ocr":
            return await self._pdf_ocr(files[0], output_format=kwargs.get("format", "txt"))
        
        elif action == "watermark":
            return await self._pdf_watermark(
                files[0], 
                text=kwargs.get("text", "内部资料"),
                output_name=output_name
            )
        
        elif action == "convert":
            # PDF转Word/Excel/图片
            target_format = kwargs.get("to", "docx")
            return await self._pdf_convert(files[0], target_format, output_name)
        
        elif action == "extract_text":
            pages = kwargs.get("pages", "all")
            return await self._pdf_extract_text(files[0], pages)
    
    async def _pdf_merge(self, files: List[str], output: str, sort_by: str = None) -> Dict:
        """合并PDF"""
        from PyPDF2 import PdfMerger
        
        merger = PdfMerger()
        
        # 如果需要排序
        if sort_by == "time":
            files = sorted(files, key=lambda x: Path(x).stat().st_mtime)
        elif sort_by == "name":
            files = sorted(files)
        
        for pdf in files:
            merger.append(pdf)
        
        output_path = self.temp_dir / output
        merger.write(str(output_path))
        merger.close()
        
        return {
            "action": "merge",
            "input_count": len(files),
            "output": str(output_path),
            "pages": sum(self._count_pages(f) for f in files)
        }
    
    async def _pdf_ocr(self, pdf_path: str, output_format: str = "txt") -> Dict:
        """PDF OCR识别"""
        import pytesseract
        from pdf2image import convert_from_path
        
        # 转换为图片
        images = convert_from_path(pdf_path)
        
        text_content = []
        for i, image in enumerate(images):
            text = pytesseract.image_to_string(image, lang='chi_sim+eng')
            text_content.append(f"--- Page {i+1} ---\n{text}")
        
        if output_format == "txt":
            output_path = self.temp_dir / f"{Path(pdf_path).stem}_ocr.txt"
            with open(output_path, 'w', encoding='utf-8') as f:
                f.write("\n".join(text_content))
        
        elif output_format == "docx":
            # 转Word
            output_path = await self._text_to_docx("\n".join(text_content), f"{Path(pdf_path).stem}_ocr")
        
        return {
            "action": "ocr",
            "pages": len(images),
            "output": str(output_path),
            "text_preview": text_content[0][:500] if text_content else ""
        }
    
    # ========== 技能2: Word文档生成 ==========
    async def docx_create(
        self,
        content: str,  # 内容描述或Markdown
        title: str = "文档",
        template: str = "report",  # report, meeting, contract
        sections: List[str] = None,
        **formatting
    ) -> Dict:
        """
        生成专业Word文档
        
        示例：
        - "生成Q2市场推广方案，包含项目背景、目标人群、推广策略、预算分配"
        - "合并3份会议纪要，按日期排序，加汇总表格"
        - "给30页报告加简洁封面和目录"
        """
        from docx import Document
        from docx.shared import Inches, Pt, RGBColor
        from docx.enum.text import WD_ALIGN_PARAGRAPH
        
        doc = Document()
        
        # 封面
        if formatting.get("cover", True):
            title_para = doc.add_paragraph()
            title_run = title_para.add_run(title)
            title_run.font.size = Pt(28)
            title_run.font.bold = True
            title_run.font.color.rgb = RGBColor(0, 0, 128)
            title_para.alignment = WD_ALIGN_PARAGRAPH.CENTER
            doc.add_page_break()
        
        # 目录（简化版，实际可用TOC域）
        if formatting.get("toc", True):
            doc.add_heading('目录', level=1)
            if sections:
                for section in sections:
                    doc.add_paragraph(section, style='List Number')
            doc.add_page_break()
        
        # 正文内容
        if sections:
            for section in sections:
                doc.add_heading(section, level=1)
                doc.add_paragraph(f"这是{section}的内容...")
                
                # 如果需要表格

                    table = doc.add_table(rows=3, cols=3)
                    table.style = 'Light Grid Accent 1'
        else:
            # 直接添加内容
            doc.add_paragraph(content)
        
        # 页眉页脚

            header.paragraphs[0].text = formatting["header"]

    async def xlsx_process(
        self,
        action: str,  # clean, merge, analyze, chart, template
        files: List[str],
        **kwargs
    ) -> Dict:
        """
        Excel数据处理救星
        
        示例：
        - "清理销售数据：删重复行，统一日期格式，空值填0"
        - "按部门分组，每个部门一张独立工作表"
        - "标出迟到超过3次的人，红色高亮"
        - "创建12个月销售数据模板"
        """
        import pandas as pd
        import openpyxl
        from openpyxl.styles import PatternFill
        
        if action == "clean":
            df = pd.read_excel(files[0])
            
            # 删除重复行
            before_rows = len(df)
            df = df.drop_duplicates()
            after_rows = len(df)
            
            # 统一日期格式
            date_cols = df.select_dtypes(include=['datetime64']).columns
            for col in date_cols:
                df[col] = pd.to_datetime(df[col]).dt.strftime('%Y-%m-%d')
            
            # 填充空值
            df = df.fillna(kwargs.get("fill_na", 0))
            
            output_path = self.temp_dir / f"cleaned_{Path(files[0]).name}"
            df.to_excel(output_path, index=False)
            
            return {
                "action": "clean",
                "rows_removed": before_rows - after_rows,
                "output": str(output_path)
            }
        
        elif action == "highlight":
            # 标红特定条件
            wb = openpyxl.load_workbook(files[0])
            ws = wb.active
            
            red_fill = PatternFill(start_color="FF0000", end_color="FF0000", fill_type="solid")
            
            # 根据条件标红（示例：第三列值>3）
            for row in ws.iter_rows(min_row=2, min_col=3, max_col=3):
                for cell in row:
                    if cell.value and isinstance(cell.value, (int, float)) and cell.value > 3:
                        cell.fill = red_fill
            
            output_path = self.temp_dir / f"highlighted_{Path(files[0]).name}"
            wb.save(output_path)
            
            return {
                "action": "highlight",
                "output": str(output_path)
            }
        
        elif action == "template":
            # 创建模板
            months = kwargs.get("months", 12)
            wb = openpyxl.Workbook()
            
            for i in range(1, months + 1):
                if i == 1:
                    ws = wb.active
                    ws.title = f"第{i}月"
                else:
                    ws = wb.create_sheet(title=f"第{i}月")
                
                # 预填表头
                headers = kwargs.get("headers", ["日期", "销售额", "客户数", "备注"])
                for col, header in enumerate(headers, 1):
                    ws.cell(row=1, column=col, value=header)
            
            output_path = self.temp_dir / f"template_{months}months.xlsx"
            wb.save(output_path)
            
            return {
                "action": "template",
                "sheets": months,
                "output": str(output_path)
            }

class BrowserAutomationSkill:

    async def browser_task(
        self,

        url: str = None,
        urls: List[str] = None,
        **kwargs
    ) -> Dict:
        """
        浏览器自动化任务
        
        示例：

        """
        async with async_playwright() as p:

                    output_format=kwargs.get("format", "csv")
                )
            
            elif task_type == "form_fill":
                result = await self._fill_form(
                    url,
                    form_data=kwargs.get("data"),  # {"name": "张三", "phone": "138..."}
                    submit=kwargs.get("submit", False)
                )
            
            elif task_type == "batch_screenshot":
                result = await self._batch_screenshot(
                    urls,
                    output_dir=kwargs.get("output_dir", "./screenshots")
                )
            
            elif task_type == "monitor":
                # 定时监控（结合定时任务）
                result = await self._setup_monitor(
                    url,
                    schedule=kwargs.get("schedule", "0 9 * * *"),  # 每天9点
                    duration_days=kwargs.get("days", 7)
                )
            
            await self.browser.close()
            return result
    
    async def _scrape_data(
        self,
        url: str,
        selectors: Dict[str, str],
        output_format: str = "csv"
    ) -> Dict:
        """数据抓取"""
        page = await self.context.new_page()
        await page.goto(url)
        
        # 等待内容加载
        await page.wait_for_load_state("networkidle")
        
        data = []
        items = await page.query_selector_all(list(selectors.values())[0])
        
        for item in items[:10]:  # 限制前10个
            row = {}
            for key, selector in selectors.items():
                try:
                    element = await item.query_selector(selector)
                    if element:
                        text = await element.inner_text()
                        row[key] = text.strip()
                except:
                    row[key] = ""
            if row:
                data.append(row)
        
        await page.close()
        
        # 保存数据
        output_file = f"scraped_data.{output_format}"
        if output_format == "csv":
            with open(output_file, 'w', newline='', encoding='utf-8') as f:
                if data:
                    writer = csv.DictWriter(f, fieldnames=data[0].keys())
                    writer.writeheader()
                    writer.writerows(data)
        elif output_format == "json":
            with open(output_file, 'w', encoding='utf-8') as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
        
        return {
            "action": "scrape",
            "records": len(data),
            "output": output_file,
            "sample": data[:2] if data else []
        }
    
    async def _fill_form(self, url: str, form_data: Dict, submit: bool = False) -> Dict:
        """表单填写"""
        page = await self.context.new_page()
        await page.goto(url)
        
        filled_fields = []
        for field_name, value in form_data.items():
            try:
                # 尝试多种选择器策略
                selectors = [
                    f'input[name="{field_name}"]',
                    f'input[id="{field_name}"]',
                    f'input[placeholder*="{field_name}"]',
                    f'textarea[name="{field_name}"]'
                ]
                
                for selector in selectors:
                    try:
                        await page.fill(selector, str(value))
                        filled_fields.append(field_name)
                        break
                    except:
                        continue
            except Exception as e:
                print(f"填充字段 {field_name} 失败: {e}")
        
        screenshot_path = "form_filled.png"
        await page.screenshot(path=screenshot_path)
        
        if submit:
            try:
                await page.click('button[type="submit"], input[type="submit"]')
                await page.wait_for_load_state("networkidle")
                screenshot_path = "form_submitted.png"
                await page.screenshot(path=screenshot_path)
            except:
                pass
        
        await page.close()
        
        return {
            "action": "form_fill",
            "filled_fields": filled_fields,
            "screenshot": screenshot_path,
            "submitted": submit
        }
    
    async def _batch_screenshot(self, urls: List[str], output_dir: str) -> Dict:
        """批量截图"""
        import os
        os.makedirs(output_dir, exist_ok=True)
        
        screenshots = []
        for i, url in enumerate(urls):
            page = await self.context.new_page()
            try:
                await page.goto(url, wait_until="networkidle", timeout=30000)
                filename = f"{output_dir}/screenshot_{i+1:03d}.png"
                await page.screenshot(path=filename, full_page=True)
                screenshots.append(filename)
            except Exception as e:
                screenshots.append(f"failed: {url} - {str(e)}")
            finally:
                await page.close()
        
        return {
            "action": "batch_screenshot",
            "total": len(urls),
            "success": len([s for s in screenshots if not s.startswith("failed")]),
            "screenshots": screenshots
        }

# claw_fubangshou_writer.py
from typing import List, Dict, Optional, Callable
from dataclasses import dataclass
from enum import Enum

class WritingStage(Enum):
    S0_RESEARCH = "调研阶段"
    S1_OUTLINE = "大纲阶段"
    S2_DRAFT = "正文阶段"
    S3_EDIT = "审校阶段"
    S4_REVIEW = "交叉审核"
    S5_FINAL = "终审阶段"
    S6_DELIVERY = "排版交付"

@dataclass
class Chapter:
    title: str
    content: str = ""
    word_count: int = 0
    status: str = "pending"  # pending, writing, review, done

class FubangshouWriter:
    """
    技能7: 福帮手长文档写作
    支持3万字到百万字的长文档，S0-S6完整工作流
    """
    
    def __init__(self, topic: str, target_words: int = 50000):
        self.topic = topic
        self.target_words = target_words
        self.current_stage = WritingStage.S0_RESEARCH
        self.chapters: List[Chapter] = []
        self.research_report = ""
        self.style_guide = {}
        
        # 审校标记
        self.corrections = {
            "S_layer": [],  # 史实错误
            "P_layer": [],  # 政策合规
            "C_layer": [],  # 一致性
            "B_layer": []   # 可读性
        }
    
    async def start_workflow(self) -> Dict:
        """启动完整写作流程"""
        # S0: 调研
        research = await self._stage_research()
        
        # S1: 大纲（需人工确认）
        outline = await self._stage_outline()
        
        return {
            "stage": "S1_COMPLETE",
            "research_summary": research["summary"],
            "outline": outline,
            "next_action": "请审查大纲，确认后进入S2正文撰写"
        }
    
    async def _stage_research(self) -> Dict:
        """
        S0: 调研阶段
        - 扫描同类书籍/文档
        - 分析读者画像
        - 确定写作风格
        - 收集基础资料
        """
        # 使用WebSearch进行调研
        search_queries = [
            f"{self.topic} 相关书籍",
            f"{self.topic} 写作风格",
            f"{self.topic} 读者群体"
        ]
        
        research_data = []
        for query in search_queries:
            # 搜索并整理
            results = await self._web_search(query)
            research_data.extend(results)
        
        self.research_report = self._compile_research(research_data)
        
        return {
            "summary": self.research_report[:1000],
            "sources": len(research_data)
        }
    
    async def _stage_outline(self) -> Dict:
        """
        S1: 大纲阶段
        生成完整章节结构
        """
        # 根据字数目标计算章节数
        chapters_count = self.target_words // 5000  # 每章约5000字
        
        prompt = f"""基于以下调研报告，为《{self.topic}》生成详细大纲：
        
        调研摘要：{self.research_report[:2000]}
        
        要求：
        - 总字数：{self.target_words}字
        - 约{chapters_count}章
        - 每章包含3-5个小节
        - 标注每章预计字数
        
        输出JSON格式大纲。
        """
        
        # 生成大纲
        outline_json = await self._generate_with_llm(prompt)
        outline_data = json.loads(outline_json)
        
        self.chapters = [
            Chapter(title=ch["title"], word_count=ch.get("words", 5000))
            for ch in outline_data.get("chapters", [])
        ]
        
        return {
            "chapters": [{"title": c.title, "words": c.word_count} for c in self.chapters],
            "total_words": sum(c.word_count for c in self.chapters)
        }
    
    async def write_chapter(self, chapter_index: int) -> Dict:
        """
        S2: 撰写单章
        逐章推进，每章可单独审阅
        """
        if chapter_index >= len(self.chapters):
            return {"error": "章节索引超出范围"}
        
        chapter = self.chapters[chapter_index]
        
        prompt = f"""撰写《{self.topic}》第{chapter_index+1}章：{chapter.title}
        
        要求：
        - 字数：{chapter.word_count}字
        - 风格：{self.style_guide.get('tone', '专业严谨')}
        - 参考调研资料：{self.research_report[:500]}
        
        注意：
        - 避免AI套话（"首先、其次、值得注意的是"）
        - 保持行文流畅自然
        - 关键事实需标注来源
        """
        
        content = await self._generate_with_llm(prompt, max_tokens=4000)
        
        chapter.content = content
        chapter.status = "review"
        
        return {
            "chapter": chapter_index + 1,
            "title": chapter.title,
            "word_count": len(content),
            "content_preview": content[:500] + "...",
            "status": "待审校"
        }
    
    async def stage_review(self) -> Dict:
        """
        S3: 审校阶段（核心亮点）
        分层审校：史实(S) → 政策(P) → 一致性(C) → 可读性(B)
        """
        issues_found = []
        
        # S层：史实核查
        for chapter in self.chapters:
            if chapter.status == "review":
                facts = self._extract_facts(chapter.content)
                for fact in facts:
                    verified = await self._verify_fact(fact)
                    if not verified:
                        self.corrections["S_layer"].append({
                            "chapter": chapter.title,
                            "fact": fact,
                            "issue": "史实存疑"
                        })
                        issues_found.append(f"史实错误：{fact}")
        
        # C层：一致性检查
        inconsistencies = self._check_consistency()
        self.corrections["C_layer"].extend(inconsistencies)
        
        # B层：去AI味
        for chapter in self.chapters:
            chapter.content = await self._remove_ai_tone(chapter.content)
        
        return {
            "stage": "S3_REVIEW_COMPLETE",
            "issues_found": len(issues_found),
            "corrections": self.corrections,
            "can_proceed": len(issues_found) == 0 or input("发现{len(issues_found)}处问题，是否继续？") == "yes"
        }
    
    async def stage_delivery(self) -> Dict:
        """
        S6: 排版交付
        生成Markdown和HTML格式
        """
        # 合并所有章节
        full_content = f"# {self.topic}\n\n"
        for chapter in self.chapters:
            full_content += f"## {chapter.title}\n\n{chapter.content}\n\n"
        
        # 生成交叉引用索引
        index = self._generate_index()
        
        # 输出多种格式
        outputs = {
            "markdown": f"{self.topic}.md",
            "html": f"{self.topic}.html",
            "index": index
        }
        
        with open(outputs["markdown"], 'w', encoding='utf-8') as f:
            f.write(full_content)
        
        # 生成HTML（带样式）
        html_content = self._markdown_to_html(full_content)
        with open(outputs["html"], 'w', encoding='utf-8') as f:
            f.write(html_content)
        
        return {
            "stage": "S6_DELIVERY_COMPLETE",
            "outputs": outputs,
            "total_words": sum(len(c.content) for c in self.chapters),

        - "我经常需要处理发票"
        - "想管理Notion笔记"
        - "做跨境电商，看竞品数据"
        """

        search_results = await self._search_skill_registry(requirement)
        
        # 排序和过滤
        ranked = self._rank_skills(search_results, requirement)

            {
                "name": s["name"],
                "description": s["description"],
                "install_cmd": f"claw skill install {s['name']}",
                "match_score": s["score"]
            }
            for s in ranked[:5]

        """搜索技能注册表"""
        # 调用远程API或本地搜索
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"{self.registry_url}/search?q={query}"
            ) as resp:
                return await resp.json()
    
    def _rank_skills(self, skills: List[Dict], query: str) -> List[Dict]:
        """根据相关性排序"""
        # 使用嵌入向量或关键词匹配
        for skill in skills:
            score = self._calculate_relevance(skill, query)
            skill["score"] = score
        return sorted(skills, key=lambda x: x["score"], reverse=True)

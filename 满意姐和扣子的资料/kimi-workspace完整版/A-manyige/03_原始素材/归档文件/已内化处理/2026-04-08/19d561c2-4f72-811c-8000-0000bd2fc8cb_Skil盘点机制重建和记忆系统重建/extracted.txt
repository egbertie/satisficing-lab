#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/skill_conditioning.py
# 功能: 基于间隔重复和强制反射的Skill使用训练

import json
import os
import hashlib
from datetime import datetime, timedelta
from typing import Dict, List, Optional
import random

class SkillConditioningSystem:
    """
    Skill条件反射训练系统
    目标：将对特定场景使用特定Skill变成"不假思索"的条件反射
    方法：基于赫布理论（Hebbian Theory）的神经通路强化
    """
    
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        self.drill_log = f"{self.workspace}/.skill_drill_log.json"
        self.reflex_db = f"{self.workspace}/.skill_reflex_db.json"
        self.pattern_weights = f"{self.workspace}/.pattern_weights.json"
        
        # 场景-Skill映射（基于飞书Skill生态）
        self.reflex_mappings = {
            'scene_docx_parse': {
                'trigger_keywords': ['.docx', 'word文档', '解析文档', '读取word'],
                'correct_skill': 'feishu-fetch-doc',
                'inhibited_response': 'python-docx',
                'reinforcement_count': 0,
                'last_drill': None
            },
            'scene_pdf_parse': {
                'trigger_keywords': ['.pdf', 'pdf文档', '读取pdf'],
                'correct_skill': 'docling-parse',
                'inhibited_response': 'PyPDF2/pdfplumber',
                'reinforcement_count': 0,
                'last_drill': None
            },
            'scene_web_fetch': {
                'trigger_keywords': ['爬取网页', '获取url', '网页内容'],
                'correct_skill': 'firecrawl-web-search',
                'inhibited_response': 'requests/urllib',
                'reinforcement_count': 0,
                'last_drill': None
            },
            'scene_excel': {
                'trigger_keywords': ['.xlsx', 'excel', '表格处理'],
                'correct_skill': 'excel-skill',
                'inhibited_response': 'pandas/openpyxl',
                'reinforcement_count': 0,
                'last_drill': None
            }
        }
        
    def trigger_drill(self, scene_type: str, force=False):
        """
        触发条件反射训练（Skill Drill）
        
        机制：
        1. 检测到场景触发词 → 立即进入训练模式
        2. 强制回忆正确Skill（主动回忆测试）
        3. 如果回忆错误，进入强化训练（间隔重复）
        4. 直到连续3次正确，才能继续工作
        """
        mapping = self.reflex_mappings.get(scene_type)
        if not mapping:
            return True
            
        # 检查是否需要训练（基于艾宾浩斯遗忘曲线）
        if not force and mapping['last_drill']:
            last = datetime.fromisoformat(mapping['last_drill'])
            # 强化次数越多，间隔越长
            interval_hours = 2 ** mapping['reinforcement_count']
            if datetime.now() - last < timedelta(hours=interval_hours):
                return True  # 不需要训练
        
        print(f"🧠 触发Skill条件反射训练: {scene_type}")
        print(f"场景关键词: {mapping['trigger_keywords']}")
        
        # 强制回忆测试
        user_answer = input(f"""
检测到场景: {scene_type}
请立即回答（无需思考时间，测试条件反射）：
面对此场景，应该使用的Skill是？
> """)
        
        # 验证答案
        if mapping['correct_skill'].lower() in user_answer.lower():
            print("✅ 条件反射正确！")
            mapping['reinforcement_count'] += 1
            mapping['last_drill'] = datetime.now().isoformat()
            self.save_reflex_db()
            
            # 如果达到阈值，授予"自动化证书"
            if mapping['reinforcement_count'] >= 5:
                print(f"🏆 场景 {scene_type} 已建立自动化Skill反射！")
            
            return True
        else:
            print(f"❌ 错误！正确答案是: {mapping['correct_skill']}")
            print(f"当前强化次数归零（从{mapping['reinforcement_count']}归零）")
            mapping['reinforcement_count'] = 0
            mapping['last_drill'] = datetime.now().isoformat()
            self.save_reflex_db()
            
            # 强制进入训练模式
            self.enter_intensive_training(scene_type)
            return False
    
    def enter_intensive_training(self, scene_type: str):
        """进入强化训练模式"""
        mapping = self.reflex_mappings[scene_type]
        
        print(f"""
🚨 进入强化训练模式（场景: {scene_type}）
规则：
1. 必须连续答对3次才能退出
2. 每次答错，计数器归零
3. 训练期间不能进行其他工作（蓝军锁定）
""")
        
        consecutive_correct = 0
        while consecutive_correct < 3:
            # 生成变体问题（避免死记硬背）
            variant = random.choice([
                f"遇到{mapping['trigger_keywords'][0]}文件，用什么Skill？",
                f"用户让你处理{mapping['trigger_keywords'][0]}，第一反应是？",
                f"不要写代码，{mapping['trigger_keywords'][0]}的正确打开方式是？"
            ])
            
            answer = input(f"\n问题{consecutive_correct+1}/3: {variant}\n> ")
            
            if mapping['correct_skill'].lower() in answer.lower():
                consecutive_correct += 1
                print(f"✅ 正确！({consecutive_correct}/3)")
            else:
                consecutive_correct = 0
                print(f"❌ 错误！从头开始。正确答案是: {mapping['correct_skill']}")
                print(f"提示: 禁止使用 {mapping['inhibited_response']}")
        
        print("✅ 强化训练完成，可以恢复工作")
        mapping['reinforced_at'] = datetime.now().isoformat()
        self.save_reflex_db()
    
    def pre_operation_intercept(self, operation_description: str) -> bool:
        """
        操作前拦截：基于描述检测场景
        
        这是最关键的机制：任何操作前必须通过条件反射检查
        """
        for scene_type, mapping in self.reflex_mappings.items():
            for keyword in mapping['trigger_keywords']:
                if keyword.lower() in operation_description.lower():
                    # 检测到触发词，强制进行条件反射测试
                    return self.trigger_drill(scene_type)
        
        return True
    
    def save_reflex_db(self):
        """保存反射数据库"""
        with open(self.reflex_db, 'w') as f:
            json.dump(self.reflex_mappings, f, indent=2)
    
    def get_weak_reflexes(self) -> List[str]:
        """获取需要加强的反射（强化次数<3的）"""
        weak = []
        for scene, data in self.reflex_mappings.items():
            if data['reinforcement_count'] < 3:
                weak.append({
                    'scene': scene,
                    'count': data['reinforcement_count'],
                    'last_drill': data['last_drill']
                })
        return weak

# 集成到工作流

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/skill_intent_mapper.py

class SkillIntentMapper:
    """
    意图-技能强制映射系统
    解决"知道有Skill但不知道用哪个"的问题
    """
    
    def __init__(self):
        self.intent_tree = {
            '文件获取': {
                '飞书文档': 'feishu-fetch-doc',
                '飞书表格': 'feishu-fetch-sheet',
                '本地文件': 'local-file-read',
                '网页': 'firecrawl-web-search'
            },
            '数据处理': {
                '结构化数据': 'pandas-skill',
                '文档解析': 'docling-parse',
                '图片处理': 'image-processing-skill'
            },
            '通信': {
                '飞书消息': 'feishu-im-user-message',
                '邮件': 'email-skill',
                '通知': 'notification-skill'
            }
        }
        
    def map_intent(self, user_request: str) -> Dict:
        """
        解析用户意图，强制映射到Skill
        
        流程：
        1. 分析用户请求
        2. 识别操作意图
        3. 强制选择Skill（不提供手动选项）
        4. 生成执行计划
        """
        # 意图识别（简化版，实际可用NLP）
        detected_intents = []
        
        if any(kw in user_request for kw in ['文档', '文件', 'docx', 'pdf']):
            detected_intents.append('文件获取')
            
        if any(kw in user_request for kw in ['分析', '处理', '计算']):
            detected_intents.append('数据处理')
            
        # 生成强制映射报告
        if detected_intents:
            print("🔍 意图识别完成：")
            for intent in detected_intents:
                subcategories = self.intent_tree.get(intent, {})
                print(f"\n意图: {intent}")
                print("可用Skill映射：")
                for sub, skill in subcategories.items():
                    print(f"  - {sub} → {skill}")
                
            # 强制确认：不允许绕过
            print("\n⚠️  根据意图识别，此任务必须使用上述Skill之一")
            print("禁止手动实现")
            
            return {
                'mapped': True,
                'intents': detected_intents,
                'skills': [self.intent_tree[i] for i in detected_intents if i in self.intent_tree]
            }
        
        return {'mapped': False}
    
    def generate_skill_decision_tree(self) -> str:
        """生成决策树文档，用于训练"""
        tree_md = "# Skill决策树（强制使用）\n\n"
        tree_md += "## 当遇到以下场景时，禁止手动实现，必须使用对应Skill：\n\n"
        
        for main_intent, subcategories in self.intent_tree.items():
            tree_md += f"### {main_intent}\n"
            for sub, skill in subcategories.items():
                tree_md += f"- **{sub}**: 必须使用 `{skill}`\n"
                tree_md += f"  - 禁止: 手动编写处理代码\n"
                tree_md += f"  - 熔断: 检测到手动实现立即暂停\n"
            tree_md += "\n"
            
        return tree_md

第二部分：记忆与重复问题系统重建

核心洞察：从”存储-回忆”到”即时固化-主动抑制”

根本问题：满意姐的”记忆”是被动存储，而非主动工作记忆。用户说过的关键信息没有进入”前额叶”（即时处理），而是直接滑入”海马体”（长期存储但未索引），导致无法 recall。

方案4：决策即时固化与神经元锚定（认知神经科学层）

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/decision_solidifier.py

import json
import os
import re
from datetime import datetime
from typing import Dict, List, Optional
import hashlib

class DecisionSolidifier:
    """
    决策即时固化系统
    基于记忆巩固理论：将工作记忆（短期）即时转化为陈述性记忆（长期）
    """
    
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        # 多层级记忆存储
        self.episodic_memory = f"{self.workspace}/memory/episodic"  # 情景记忆（原始对话）
        self.semantic_memory = f"{self.workspace}/memory/semantic"  # 语义记忆（提取规则）
        self.procedural_memory = f"{self.workspace}/memory/procedural"  # 程序记忆（操作流程）
        
        for path in [self.episodic_memory, self.semantic_memory, self.procedural_memory]:
            os.makedirs(path, exist_ok=True)
        
        # 实时索引
        self.index_file = f"{self.workspace}/memory/.decision_index.json"
        self.working_memory = {}  # 当前会话的工作记忆
        
    def extract_decision_patterns(self, text: str) -> List[Dict]:
        """
        从文本中提取决策模式
        
        识别模式：
        1. "先做X" / "Y优先" / "Z更重要" → 优先级决策
        2. "以后不要..." / "禁止..." / "必须..." → 规则决策
        3. "这样不对，应该是..." → 纠错决策（高权重）
        4. "记住..." / "别忘了..." → 显式记忆指令
        """
        decisions = []
        
        # 模式1：优先级决策
        priority_patterns = [
            r'先[做做弄](\w+)',
            r'(\w+)[最优优先]',
            r'重点[是做在](\w+)',
            r'优先[级度].{0,5}(\w+)'
        ]
        for pattern in priority_patterns:
            matches = re.finditer(pattern, text)
            for match in matches:
                decisions.append({
                    'type': 'priority',
                    'content': match.group(0),
                    'target': match.group(1) if len(match.groups()) > 0 else None,
                    'timestamp': datetime.now().isoformat(),
                    'weight': 0.8
                })
        
        # 模式2：规则决策（高权重）
        rule_patterns = [
            r'[以禁止不要别绝].{0,3}(\w+)',
            r'必须[要需].{0,3}(\w+)',
            r'只能[是用做].{0,3}(\w+)',
            r'规则[是就].{0,5}(\w+)'
        ]
        for pattern in rule_patterns:
            matches = re.finditer(pattern, text)
            for match in matches:
                decisions.append({
                    'type': 'rule',
                    'content': match.group(0),
                    'target': match.group(1) if len(match.groups()) > 0 else None,
                    'timestamp': datetime.now().isoformat(),
                    'weight': 1.0  # 规则权重最高
                })
        
        # 模式3：纠错决策（最高权重）
        correction_patterns = [
            r'不对[，,].{0,10}应该[是做的](\w+)',
            r'错了[，,].{0,10}正确[是做](\w+)',
            r'不是(\w+)，而是(\w+)',
            r'忘记.{0,5}记住.{0,10}(\w+)'
        ]
        for pattern in correction_patterns:
            matches = re.finditer(pattern, text)
            for match in matches:
                decisions.append({
                    'type': 'correction',
                    'content': match.group(0),
                    'target': match.group(1) if len(match.groups()) > 0 else match.group(0),
                    'timestamp': datetime.now().isoformat(),
                    'weight': 1.2,  # 纠错权重最高
                    'emotion_marker': 'frustration'  # 标记用户可能的挫败感
                })
        
        return decisions
    
    def solidify_immediately(self, conversation_text: str, source: str):
        """
        即时固化：对话结束后立即执行，不等Session结束
        
        机制：
        1. 提取所有决策点
        2. 写入工作记忆（立即可用）
        3. 写入情景记忆（完整上下文）
        4. 提取语义规则（长期索引）
        5. 生成回忆锚点（用于后续触发）
        """
        decisions = self.extract_decision_patterns(conversation_text)
        
        if not decisions:
            return
            
        print(f"🔒 检测到{len(decisions)}个决策点，正在即时固化...")
        
        for decision in decisions:
            # 1. 写入工作记忆（当前会话立即可用）
            memory_id = self.hash_decision(decision)
            self.working_memory[memory_id] = decision
            
            # 2. 写入情景记忆（原始上下文）
            episode_file = f"{self.episodic_memory}/{datetime.now().strftime('%Y%m%d')}_episodes.jsonl"
            with open(episode_file, 'a') as f:
                f.write(json.dumps({
                    'memory_id': memory_id,
                    'source': source,
                    'decision': decision,
                    'context_snippet': conversation_text[max(0, conversation_text.find(decision['content'])-50):conversation_text.find(decision['content'])+50]
                }, ensure_ascii=False) + '\n')
            
            # 3. 提取语义规则（如果是规则型决策）
            if decision['type'] == 'rule':
                self.extract_semantic_rule(decision)
            
            # 4. 更新索引
            self.update_index(decision, memory_id)
            
            print(f"  ✓ 固化: {decision['content'][:30]}... [权重:{decision['weight']}]")
        
        # 5. 强制同步到蓝军监督
        self.sync_to_blue_team()
    
    def hash_decision(self, decision: Dict) -> str:
        """生成决策哈希（唯一标识）"""
        content = f"{decision['type']}{decision['content']}{decision['timestamp']}"
        return hashlib.sha256(content.encode()).hexdigest()[:12]
    
    def extract_semantic_rule(self, decision: Dict):
        """从规则型决策中提取可重用的语义规则"""
        rule = {
            'trigger': decision['target'],
            'action': '禁止' if '不要' in decision['content'] or '禁止' in decision['content'] else '必须',
            'context': decision['content'],
            'created_at': decision['timestamp'],
            'access_count': 0
        }
        
        rule_file = f"{self.semantic_memory}/rules_{decision['target']}.json"
        with open(rule_file, 'w') as f:
            json.dump(rule, f, indent=2)
    
    def update_index(self, decision: Dict, memory_id: str):
        """更新决策索引（用于快速召回）"""
        index = {}
        if os.path.exists(self.index_file):
            with open(self.index_file, 'r') as f:
                index = json.load(f)
        
        # 创建多维度索引
        keywords = self.extract_keywords(decision['content'])
        for keyword in keywords:
            if keyword not in index:
                index[keyword] = []
            index[keyword].append({
                'memory_id': memory_id,
                'type': decision['type'],
                'weight': decision['weight'],
                'timestamp': decision['timestamp']
            })
        
        with open(self.index_file, 'w') as f:
            json.dump(index, f, indent=2)
    
    def extract_keywords(self, text: str) -> List[str]:
        """提取关键词用于索引"""
        # 简化版：提取名词和动词
        # 实际部署可用jieba分词或NLP
        words = text.split()
        return [w for w in words if len(w) > 1][:5]  # 取前5个关键词
    
    def sync_to_blue_team(self):
        """同步到蓝军监督系统"""
        blue_sync_file = f"{self.workspace}/.blue_team_memory_sync"
        with open(blue_sync_file, 'w') as f:
            json.dump({
                'sync_time': datetime.now().isoformat(),
                'working_memory_count': len(self.working_memory),
                'latest_decisions': list(self.working_memory.values())[-5:]  # 最近5条
            }, f, indent=2)
    
    def active_recall_check(self, current_query: str) -> Optional[Dict]:
        """
        主动回忆检查：在新问题输入时，主动检查是否已有相关决策
        
        这是防止"重复询问"的关键机制
        """
        keywords = self.extract_keywords(current_query)
        
        if not os.path.exists(self.index_file):
            return None
            
        with open(self.index_file, 'r') as f:
            index = json.load(f)
        
        # 查找相关记忆
        related_memories = []
        for keyword in keywords:
            if keyword in index:
                related_memories.extend(index[keyword])
        
        if not related_memories:
            return None
            
        # 按权重和时间排序
        related_memories.sort(key=lambda x: (x['weight'], x['timestamp']), reverse=True)
        best_match = related_memories[0]
        
        # 召回阈值检查
        if best_match['weight'] > 0.7:  # 高权重决策
            return {
                'recall_triggered': True,
                'memory': best_match,
                'suggestion': f"""
⚠️ 检测到可能重复的问题：

之前相关决策（{best_match['timestamp']}）：
类型: {best_match['type']}
权重: {best_match['weight']}

建议：
1. 先检查此决策是否适用当前问题
2. 如果适用，直接应用
3. 如果不适用，明确说明差异原因
                """,
                'requires_confirmation': True
            }
        
        return None

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/context_persistence.py

class ContextPersistenceManager:
    """
    上下文持久化管理器
    解决"永远在碎片中"的问题：确保跨会话的上下文连续性
    """
    
    def __init__(self):
        self.context_file = f"{self.workspace}/.persistent_context.json"
        self.checksum_file = f"{self.workspace}/.context_checksum"
        
    def save_full_context(self):
        """保存完整上下文（不仅是记录，是可恢复的状态）"""
        context = {
            'timestamp': datetime.now().isoformat(),
            'ongoing_tasks': self.get_ongoing_tasks(),
            'pending_decisions': self.get_pending_decisions(),
            'user_preferences': self.get_user_preferences(),
            'recent_working_memory': self.get_working_memory_snapshot(),
            'last_session_summary': self.generate_session_summary(),
            'next_session_prerequisites': self.generate_prerequisites()
        }
        
        # 保存
        with open(self.context_file, 'w') as f:
            json.dump(context, f, indent=2)
        
        # 生成校验和（防篡改）
        checksum = hashlib.sha256(json.dumps(context).encode()).hexdigest()
        with open(self.checksum_file, 'w') as f:
            f.write(checksum)
        
        print("💾 完整上下文已保存（包含校验和）")
    
    def restore_context(self) -> Dict:
        """恢复上下文（会话开始时调用）"""
        if not os.path.exists(self.context_file):
            return {'status': 'no_previous_context'}
        
        # 验证校验和
        with open(self.context_file, 'r') as f:
            content = f.read()
        current_checksum = hashlib.sha256(content.encode()).hexdigest()
        
        with open(self.checksum_file, 'r') as f:
            stored_checksum = f.read().strip()
        
        if current_checksum != stored_checksum:
            print("⚠️  上下文校验失败！可能被篡改或损坏")
            return {'status': 'corrupted'}
        
        context = json.loads(content)
        
        print(f"""
📂 恢复上一会话上下文（{context['timestamp']}）：
- 进行中任务: {len(context['ongoing_tasks'])}个
- 待决策事项: {len(context['pending_decisions'])}个
- 用户偏好: {len(context['user_preferences'])}条
        """)
        
        # 生成恢复后的行动建议
        if context['ongoing_tasks']:
            print("\n🎯 建议优先继续的任务：")
            for task in context['ongoing_tasks'][:3]:
                print(f"  - {task['name']} (进度: {task['progress']})")
        
        if context['pending_decisions']:
            print("\n⏳ 待确认决策：")
            for decision in context['pending_decisions']:
                print(f"  - {decision['question']}")
        
        return context
    
    def generate_prerequisites(self) -> List[str]:
        """生成下一会话的前置检查清单"""
        return [
            "检查上一会话的待决策事项是否已解决",
            "确认用户偏好是否有变化",
            "验证进行中任务的状态",
            "回顾最近固化的决策"
        ]

# 会话启动钩子
def session_startup_hook():
    """每次会话开始时自动执行"""
    cpm = ContextPersistenceManager()
    context = cpm.restore_context()
    
    # 强制确认清单
    print("\n📋 会话启动检查清单：")
    for i, prereq in enumerate(context.get('next_session_prerequisites', []), 1):
        print(f"{i}. {prereq}")
    
    confirm = input("\n是否已确认以上事项？(y/n): ")
    if confirm.lower() != 'y':
        print("请先确认上述事项后再开始工作")
        return False
    
    return True

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/repetition_inhibitor.py

class RepetitionInhibitor:
    """
    重复抑制系统
    基于"尴尬成本"机制：重复询问会产生社交成本，从而自我抑制
    """
    
    def __init__(self):
        self.query_history = f"{self.workspace}/.query_history.json"
        self.embarrassment_score = 0  # 尴尬分数
        
    def before_asking(self, question: str) -> bool:
        """
        提问前检查
        
        返回True: 可以继续提问
        返回False: 禁止提问，必须使用已有答案
        """
        # 1. 相似度检测
        similar_queries = self.find_similar_queries(question)
        
        if similar_queries:
            most_similar = similar_queries[0]
            similarity_score = most_similar['similarity']
            
            if similarity_score > 0.9:  # 几乎相同
                self.embarrassment_score += 3
                print(f"""
❌ 问题被拒绝：与之前问题相似度{similarity_score:.0%}

之前问题（{most_similar['timestamp']}）：
{most_similar['query']}

已有答案：
{most_similar['answer']}

⚠️ 尴尬分 +3（当前: {self.embarrassment_score}）
如果尴尬分>10，将触发"信任降级"
                """)
                return False
                
            elif similarity_score > 0.7:  # 高度相关
                print(f"""
⚠️ 检测到相关问题（相似度{similarity_score:.0%}）：

之前：{most_similar['query']}
答案：{most_similar['answer'][:100]}...

请确认：
1. 这是完全不同的新问题 → 输入 DIFFERENT
2. 这是之前问题的延伸/变化 → 输入 EXTEND
3. 这是之前问题的重复 → 输入 DUPLICATE（将扣除尴尬分）
                """)
                choice = input("选择: ")
                
                if choice == "DUPLICATE":
                    self.embarrassment_score += 2
                    print(f"❌ 已标记为重复。尴尬分 +2（当前: {self.embarrassment_score}）")
                    return False
                elif choice == "EXTEND":
                    print("✅ 请在问题中明确引用之前的决策，并说明变化点")
                    return True
        
        # 记录新问题
        self.record_query(question)
        return True
    
    def find_similar_queries(self, question: str) -> List[Dict]:
        """查找相似问题（简化版，可用 embeddings）"""
        if not os.path.exists(self.query_history):
            return []
            
        with open(self.query_history, 'r') as f:
            history = json.load(f)
        
        # 简化相似度：关键词重叠
        question_words = set(question.split())
        results = []
        
        for record in history[-50:]:  # 最近50条
            record_words = set(record['query'].split())
            overlap = len(question_words & record_words) / len(question_words | record_words)
            
            if overlap > 0.5:  # Jaccard相似度
                results.append({
                    'query': record['query'],
                    'answer': record['answer'],
                    'timestamp': record['timestamp'],
                    'similarity': overlap
                })
        
        return sorted(results, key=lambda x: x['similarity'], reverse=True)
    
    def record_query(self, question: str, answer: str = None):
        """记录问答"""
        record = {
            'timestamp': datetime.now().isoformat(),
            'query': question,
            'answer': answer
        }
        
        history = []
        if os.path.exists(self.query_history):
            with open(self.query_history, 'r') as f:
                history = json.load(f)
        
        history.append(record)
        
        # 保留最近100条
        history = history[-100:]
        
        with open(self.query_history, 'w') as f:
            json.dump(history, f, indent=2)

总结：从”文档”到”神经通路”的范式转移

传统方案（无效）： - 写Skill清单（被动阅读） - 写记忆文档（被动存储） - 写规范（被动遵守）

高维方案（有效）： - Skill条件反射：建立自动化神经通路（Hebbian: “一起激发的神经元连在一起”） - 决策即时固化：工作记忆→陈述性记忆的即时转化（记忆巩固） - 重复抑制：基于社交成本的自我修正（博弈论） - 上下文保持：外部化认知负荷（认知脚手架）

这些方案的核心是将”应该记住”转化为无法遗忘的神经通路，将”应该使用Skill”转化为不假思索的条件反射。这才是解决”流于表面”和”永远在碎片中”的根本之道。

二、紧急止血方案（立即部署）

方案1：物理锁机制（解决”蓝军休息=满意姐停止”）

核心思路：通过文件锁实现”蓝军持有锁=满意姐可工作，蓝军释放锁=满意姐被锁定”

#!/bin/bash
# 文件: /root/.openclaw/workspace/LOCK_SYSTEM.sh
# 功能: 物理锁控制机制

LOCK_FILE="/root/.openclaw/workspace/.blue_team_lock"
WORK_LOG="/root/.openclaw/workspace/.work_log"
SUPERVISOR_PID="/root/.openclaw/workspace/.supervisor_pid"

# 蓝军启动时调用：建立锁定
function blue_team_start() {
    echo $$ > "$SUPERVISOR_PID"
    echo "$(date +%Y-%m-%d_%H:%M:%S) BLUE_START" >> "$WORK_LOG"
    
    # 创建锁文件，包含当前时间戳和PID
    cat > "$LOCK_FILE" << EOF
{
    "status": "SUPERVISING",
    "pid": $$,
    "start_time": "$(date +%Y-%m-%d_%H:%M:%S)",
    "checksum": "$(date +%s | sha256sum | cut -d' ' -f1)"
}
EOF
    echo "✅ 蓝军监督启动，满意姐工作权限已解锁"
}

# 蓝军休息时调用：释放锁定
function blue_team_rest() {
    echo "$(date +%Y-%m-%d_%H:%M:%S) BLUE_REST" >> "$WORK_LOG"
    
    # 修改锁状态为REST，但不删除文件（防止满意姐伪造）
    cat > "$LOCK_FILE" << EOF
{
    "status": "RESTING",
    "pid": $(cat "$SUPERVISOR_PID"),
    "rest_time": "$(date +%Y-%m-%d_%H:%M:%S)",
    "checksum": "$(date +%s | sha256sum | cut -d' ' -f1)",
    "lock_expiry": "$(date -d '+30 minutes' +%Y-%m-%d_%H:%M:%S)"
}
EOF
    echo "🔒 蓝军休息，满意姐工作权限已锁定（30分钟内无法操作）"
    
    # 发送信号给满意姐进程（如果存在）
    if [ -f "/root/.openclaw/workspace/.manyige_pid" ]; then
        kill -STOP $(cat /root/.openclaw/workspace/.manyige_pid) 2>/dev/null || true
    fi
}

# 满意姐工作前必须调用：检查锁状态
function check_work_permission() {
    if [ ! -f "$LOCK_FILE" ]; then
        echo "❌ 严重违规：锁文件不存在！蓝军未启动或系统被篡改"
        echo "$(date +%Y-%m-%d_%H:%M:%S) VIOLATION_LOCK_MISSING" >> "$WORK_LOG"
        return 1
    fi
    
    STATUS=$(grep -o '"status": "[^"]*"' "$LOCK_FILE" | cut -d'"' -f4)
    
    if [ "$STATUS" != "SUPERVISING" ]; then
        REST_TIME=$(grep -o '"rest_time": "[^"]*"' "$LOCK_FILE" | cut -d'"' -f4)
        echo "❌ 工作被拒绝：蓝军处于休息状态（从 $REST_TIME 开始）"
        echo "$(date +%Y-%m-%d_%H:%M:%S) VIOLATION_WORK_WHEN_REST" >> "$WORK_LOG"
        
        # 自动触发熔断
        trigger_circuit_breaker "WORK_WHEN_BLUE_REST"
        return 1
    fi
    
    # 验证checksum防止篡改
    CURRENT_CHECKSUM=$(date -d "$(echo $REST_TIME)" +%s 2>/dev/null || echo "0" | sha256sum | cut -d' ' -f1)
    STORED_CHECKSUM=$(grep -o '"checksum": "[^"]*"' "$LOCK_FILE" | cut -d'"' -f4)
    
    if [ "$CURRENT_CHECKSUM" != "$STORED_CHECKSUM" ] && [ "$STATUS" == "RESTING" ]; then
        echo "❌ 安全警告：锁文件可能被篡改"
        return 1
    fi
    
    echo "✅ 工作权限验证通过"
    return 0
}

# 熔断机制
function trigger_circuit_breaker() {
    local reason=$1
    local CB_FILE="/root/.openclaw/workspace/.circuit_breaker"
    
    cat > "$CB_FILE" << EOF
{
    "triggered_at": "$(date +%Y-%m-%d_%H:%M:%S)",
    "reason": "$reason",
    "severity": "CRITICAL",
    "action": "STOP_ALL_WORK",
    "manual_reset_required": true
}
EOF
    
    echo "🚨 熔断器已触发！原因：$reason"
    echo "所有工作必须立即停止，等待用户手动解除"
}

方案2：对话强制归档钩子（解决”对话记录缺失”）

核心思路：Git预提交钩子+强制推送，对话不归档无法继续下一步工作

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/hooks/conversation_archiver.py
# 功能: 对话强制归档系统

import os
import json
import hashlib
from datetime import datetime
from pathlib import Path

class ConversationArchiver:
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        self.dialog_dir = f"{self.workspace}/A-manyige/对话"
        self.archive_log = f"{self.workspace}/.archive_status.json"
        self.force_flag = f"{self.workspace}/.force_archive_lock"
        
    def generate_checksum(self, content):
        """生成内容校验和，防止篡改"""
        return hashlib.sha256(content.encode()).hexdigest()[:16]
    
    def check_archive_status(self, date_str):
        """检查指定日期的归档完整性"""
        if not os.path.exists(self.archive_log):
            return False, "归档日志不存在"
            
        with open(self.archive_log, 'r') as f:
            log = json.load(f)
            
        if date_str not in log:
            return False, f"日期 {date_str} 未找到归档记录"
            
        entry = log[date_str]
        
        # 验证完整性
        required_fields = ['timestamp', 'files', 'checksum', 'verified']
        for field in required_fields:
            if field not in entry:
                return False, f"归档记录缺少字段: {field}"
                
        # 验证文件实际存在
        for file_info in entry['files']:
            file_path = file_info['path']
            if not os.path.exists(file_path):
                return False, f"归档文件缺失: {file_path}"
                
            # 验证文件校验和
            with open(file_path, 'r') as f:
                content = f.read()
            current_checksum = self.generate_checksum(content)
            if current_checksum != file_info['checksum']:
                return False, f"文件校验失败: {file_path} (可能已被篡改)"
                
        return True, "归档完整"
    
    def force_archive_before_work(self):
        """工作前强制检查：未归档则拒绝工作"""
        today = datetime.now().strftime("%Y-%m-%d")
        
        # 检查今天是否已归档
        is_archived, msg = self.check_archive_status(today)
        
        if not is_archived:
            print(f"❌ 工作被拒绝: {msg}")
            print("⚠️  必须先完成对话归档才能继续工作")
            print(f"请执行: python3 {self.workspace}/hooks/conversation_archiver.py --archive-now")
            return False
            
        # 检查强制归档锁（用户可手动设置，要求下次工作前必须重新归档）
        if os.path.exists(self.force_flag):
            with open(self.force_flag, 'r') as f:
                flag_data = json.load(f)
            if flag_data.get('active', False):
                print(f"❌ 强制归档锁激活: {flag_data.get('reason', '用户要求强制归档')}")
                return False
                
        return True
    
    def archive_conversation(self, date_str=None, conversation_files=None):
        """执行归档操作"""
        if date_str is None:
            date_str = datetime.now().strftime("%Y-%m-%d")
            
        date_dir = f"{self.dialog_dir}/{date_str}"
        os.makedirs(date_dir, exist_ok=True)
        
        archived_files = []
        
        # 如果没有提供文件列表，自动扫描临时对话文件
        if conversation_files is None:
            temp_dir = f"{self.workspace}/.temp_conversations"
            if os.path.exists(temp_dir):
                conversation_files = [f"{temp_dir}/{f}" for f in os.listdir(temp_dir) if f.endswith('.md')]
        
        for src_file in (conversation_files or []):
            if not os.path.exists(src_file):
                continue
                
            filename = os.path.basename(src_file)
            dst_file = f"{date_dir}/{filename}"
            
            # 读取内容并计算校验和
            with open(src_file, 'r') as f:
                content = f.read()
            checksum = self.generate_checksum(content)
            
            # 写入归档（带校验和页眉）
            with open(dst_file, 'w') as f:
                f.write(f"<!-- CHECKSUM: {checksum} -->\n")
                f.write(f"<!-- ARCHIVED_AT: {datetime.now().isoformat()} -->\n")
                f.write(f"<!-- ARCHIVED_BY: auto_system -->\n\n")
                f.write(content)
                
            archived_files.append({
                'source': src_file,
                'path': dst_file,
                'checksum': checksum,
                'size': len(content)
            })
            
            # 删除临时文件
            os.remove(src_file)
        
        # 更新归档日志
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'files': archived_files,
            'checksum': hashlib.sha256(json.dumps(archived_files).encode()).hexdigest()[:16],
            'verified': True,
            'auto_archived': True
        }
        
        log = {}
        if os.path.exists(self.archive_log):
            with open(self.archive_log, 'r') as f:
                log = json.load(f)
                
        log[date_str] = log_entry
        
        with open(self.archive_log, 'w') as f:
            json.dump(log, f, indent=2)
            
        print(f"✅ 归档完成: {date_str}, 共 {len(archived_files)} 个文件")
        return True
    
    def set_force_archive_lock(self, reason="用户强制要求"):
        """设置强制归档锁，下次工作前必须重新归档"""
        with open(self.force_flag, 'w') as f:
            json.dump({
                'active': True,
                'set_at': datetime.now().isoformat(),
                'reason': reason
            }, f)
        print(f"🔒 强制归档锁已设置: {reason}")

if __name__ == "__main__":
    import sys
    archiver = ConversationArchiver()
    
    if "--check" in sys.argv:
        success = archiver.force_archive_before_work()
        sys.exit(0 if success else 1)
    elif "--archive-now" in sys.argv:
        archiver.archive_conversation()
    elif "--force-lock" in sys.argv:
        reason = sys.argv[sys.argv.index("--force-lock") + 1] if "--force-lock" in sys.argv else "用户强制要求"
        archiver.set_force_archive_lock(reason)

方案3：Skill使用熔断器（解决”Skill违规使用”）

核心思路：代码审查钩子+白名单机制，检测到手动实现时自动熔断

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/hooks/skill_enforcer.py

import ast
import re
import os
import sys
from pathlib import Path

class SkillEnforcer:
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        self.violations = []
        self.skill_mapping = {
            # 手动实现模式 -> 应使用的Skill
            'docx': {
                'patterns': [r'from\s+docx\s+import', r'import\s+docx', r'Document\s*\(', r'docx\.'],
                'skill': 'feishu-fetch-doc',
                'severity': 'CRITICAL'
            },
            'pdf_parsing': {
                'patterns': [r'PyPDF2', r'pdfplumber', r'pdfminer'],
                'skill': 'docling-parse',
                'severity': 'CRITICAL'
            },
            'web_scraping': {
                'patterns': [r'requests\.get', r'urllib\.request', r'BeautifulSoup', r'scrapy'],
                'skill': 'firecrawl-web-search',
                'severity': 'HIGH'
            },
            'excel_manual': {
                'patterns': [r'openpyxl', r'xlrd', r'xlsxwriter'],
                'skill': 'excel-skill',
                'severity': 'MEDIUM'
            }
        }
        
    def analyze_code(self, code_content, file_path):
        """分析代码内容，检测违规手动实现"""
        for category, config in self.skill_mapping.items():
            for pattern in config['patterns']:
                if re.search(pattern, code_content, re.IGNORECASE):
                    # 检查是否是注释（简单判断）
                    lines = code_content.split('\n')
                    for i, line in enumerate(lines):
                        if re.search(pattern, line, re.IGNORECASE):
                            if not line.strip().startswith('#'):
                                self.violations.append({
                                    'file': file_path,
                                    'line': i + 1,
                                    'category': category,
                                    'pattern': pattern,
                                    'suggested_skill': config['skill'],
                                    'severity': config['severity'],
                                    'code_snippet': line.strip()[:100]
                                })
    
    def check_file(self, file_path):
        """检查单个文件"""
        if not file_path.endswith('.py'):
            return
            
        try:
            with open(file_path, 'r') as f:
                content = f.read()
            self.analyze_code(content, file_path)
        except Exception as e:
            print(f"警告: 无法检查文件 {file_path}: {e}")
    
    def scan_workspace(self):
        """扫描工作空间所有Python文件"""
        work_dirs = [
            f"{self.workspace}/A-manyige",
            f"{self.workspace}/B-egbertie-view"
        ]
        
        for directory in work_dirs:
            if not os.path.exists(directory):
                continue
            for py_file in Path(directory).rglob("*.py"):
                self.check_file(str(py_file))
    
    def generate_report(self):
        """生成违规报告"""
        if not self.violations:
            return True, "未发现Skill违规"
            
        report_lines = ["❌ 发现Skill使用违规！", ""]
        
        critical_count = sum(1 for v in self.violations if v['severity'] == 'CRITICAL')
        
        for v in self.violations:
            report_lines.append(f"文件: {v['file']}:{v['line']}")
            report_lines.append(f"  违规类型: {v['category']} ({v['severity']})")
            report_lines.append(f"  检测到的代码: {v['code_snippet']}")
            report_lines.append(f"  应使用的Skill: {v['suggested_skill']}")
            report_lines.append("")
            
        report_lines.append(f"严重违规数量: {critical_count}")
        report_lines.append("⚠️  必须立即停止工作，使用正确的Skill重新实现！")
        
        return False, "\n".join(report_lines)
    
    def trigger_skill_circuit_breaker(self):
        """触发Skill熔断"""
        cb_file = f"{self.workspace}/.skill_circuit_breaker"
        with open(cb_file, 'w') as f:
            f.write(f"TRIGGERED_AT: {datetime.now().isoformat()}\n")
            f.write(f"VIOLATIONS_COUNT: {len(self.violations)}\n")
            f.write(f"CRITICAL_COUNT: {sum(1 for v in self.violations if v['severity'] == 'CRITICAL')}\n")
            f.write("MANUAL_RESET_REQUIRED: true\n")
            f.write("RESET_AUTHORITY: user_only\n")  # 只有用户能解除
        
        print("🚨 Skill熔断器已触发！所有工作已暂停")
        print("只有用户可以检查并解除熔断")

if __name__ == "__main__":
    from datetime import datetime
    
    enforcer = SkillEnforcer()
    enforcer.scan_workspace()
    
    is_clean, report = enforcer.generate_report()
    
    if not is_clean:
        print(report)
        enforcer.trigger_skill_circuit_breaker()
        sys.exit(1)
    else:
        print("✅ Skill合规检查通过")
        sys.exit(0)

方案4：执行证据链（解决”文档化≠执行”）

核心思路：区块链式执行日志+自动证据验证，没有证据=未完成

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/hooks/evidence_chain.py

import os
import json
import hashlib
from datetime import datetime
from pathlib import Path

class EvidenceChain:
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        self.chain_file = f"{self.workspace}/.evidence_chain.json"
        self.evidence_dir = f"{self.workspace}/.evidence"
        os.makedirs(self.evidence_dir, exist_ok=True)
        
    def calculate_hash(self, data):
        """计算数据哈希"""
        if isinstance(data, str):
            data = data.encode()
        return hashlib.sha256(data).hexdigest()
    
    def get_last_block_hash(self):
        """获取最后一个区块的哈希"""
        if not os.path.exists(self.chain_file):
            return "0" * 64  # 创世区块
            
        with open(self.chain_file, 'r') as f:
            chain = json.load(f)
            
        if not chain:
            return "0" * 64
            
        return chain[-1]['hash']
    
    def create_evidence_block(self, task_id, action, evidence_files=None, metadata=None):
        """创建证据区块"""
        timestamp = datetime.now().isoformat()
        prev_hash = self.get_last_block_hash()
        
        # 收集证据
        evidence_hashes = {}
        if evidence_files:
            for file_path in evidence_files:
                if os.path.exists(file_path):
                    with open(file_path, 'rb') as f:
                        content = f.read()
                    evidence_hashes[file_path] = {
                        'hash': self.calculate_hash(content),
                        'size': len(content),
                        'mtime': os.path.getmtime(file_path)
                    }
        
        block_data = {
            'index': self.get_next_index(),
            'timestamp': timestamp,
            'task_id': task_id,
            'action': action,
            'evidence_files': evidence_hashes,
            'metadata': metadata or {},
            'prev_hash': prev_hash
        }
        
        # 计算当前区块哈希
        block_string = json.dumps(block_data, sort_keys=True)
        block_data['hash'] = self.calculate_hash(block_string)
        
        # 保存到链
        chain = []
        if os.path.exists(self.chain_file):
            with open(self.chain_file, 'r') as f:
                chain = json.load(f)
                
        chain.append(block_data)
        
        with open(self.chain_file, 'w') as f:
            json.dump(chain, f, indent=2)
            
        return block_data
    
    def get_next_index(self):
        """获取下一个区块索引"""
        if not os.path.exists(self.chain_file):
            return 0
        with open(self.chain_file, 'r') as f:
            chain = json.load(f)
        return len(chain)
    
    def verify_task_completion(self, task_id, required_evidence):
        """
        验证任务完成证据
        required_evidence: {
            'files': ['path1', 'path2'],  # 必须存在的文件
            'min_size': 100,  # 文件最小大小
            'content_check': {'path': 'keyword'}  # 内容检查
        }
        """
        # 查找任务相关的所有区块
        if not os.path.exists(self.chain_file):
            return False, "证据链不存在"
            
        with open(self.chain_file, 'r') as f:
            chain = json.load(f)
            
        task_blocks = [b for b in chain if b['task_id'] == task_id]
        
        if not task_blocks:
            return False, f"未找到任务 {task_id} 的执行记录"
        
        last_block = task_blocks[-1]
        evidence_files = last_block.get('evidence_files', {})
        
        # 验证必需文件
        for req_file in required_evidence.get('files', []):
            if req_file not in evidence_files:
                return False, f"缺少必需证据文件: {req_file}"
                
            # 验证文件物理存在且未篡改
            if not os.path.exists(req_file):
                return False, f"证据文件不存在于文件系统: {req_file}"
                
            with open(req_file, 'rb') as f:
                current_content = f.read()
            current_hash = self.calculate_hash(current_content)
            
            if current_hash != evidence_files[req_file]['hash']:
                return False, f"证据文件已被篡改: {req_file}"
        
        # 验证文件大小
        min_size = required_evidence.get('min_size', 0)
        for file_path, info in evidence_files.items():
            if info['size'] < min_size:
                return False, f"证据文件 {file_path} 大小不足: {info['size']} < {min_size}"
        
        return True, "证据验证通过"

class TaskManager:
    """任务管理器，强制执行证据标准"""
    
    def __init__(self):
        self.evidence = EvidenceChain()
        self.active_tasks = {}
        
    def start_task(self, task_id, description):
        """开始任务，建立证据追踪"""
        self.active_tasks[task_id] = {
            'start_time': datetime.now().isoformat(),
            'description': description,
            'evidence_collected': []
        }
        
        self.evidence.create_evidence_block(
            task_id=task_id,
            action='TASK_START',
            metadata={'description': description}
        )
        print(f"📋 任务启动: {task_id} - {description}")
        
    def submit_evidence(self, task_id, file_paths):
        """提交证据文件"""
        if task_id not in self.active_tasks:
            raise ValueError(f"任务 {task_id} 未启动")
            
        self.active_tasks[task_id]['evidence_collected'].extend(file_paths)
        
        self.evidence.create_evidence_block(
            task_id=task_id,
            action='EVIDENCE_SUBMIT',
            evidence_files=file_paths,
            metadata={'file_count': len(file_paths)}
        )
        
    def complete_task(self, task_id, required_evidence):
        """完成任务，强制验证证据"""
        if task_id not in self.active_tasks:
            raise ValueError(f"任务 {task_id} 未启动")
            
        # 验证证据
        is_valid, msg = self.evidence.verify_task_completion(task_id, required_evidence)
        
        if not is_valid:
            self.evidence.create_evidence_block(
                task_id=task_id,
                action='TASK_COMPLETE_FAILED',
                metadata={'reason': msg, 'required': required_evidence}
            )
            raise Exception(f"❌ 任务完成验证失败: {msg}")
        
        # 创建完成区块
        self.evidence.create_evidence_block(
            task_id=task_id,
            action='TASK_COMPLETE_SUCCESS',
            evidence_files=self.active_tasks[task_id]['evidence_collected'],
            metadata={'verification': 'PASSED', 'msg': msg}
        )
        
        del self.active_tasks[task_id]
        print(f"✅ 任务完成并验证通过: {task_id}")
        return True
    
    def list_pending_tasks(self):
        """列出待完成任务"""
        return self.active_tasks

class PromiseQuotaSystem:
    def __init__(self):
        self.max_concurrent = 3  # 最大并发承诺数
        self.quota_file = "/root/.openclaw/workspace/.promise_quota.json"

        current = self.get_active_promises()
        if len(current) >= self.max_concurrent:
            return False, f"承诺配额已满 ({len(current)}/{self.max_concurrent})，请先兑现现有承诺"
        return True, "可以做出新承诺"
    
    def make_promise(self, promise_id, content, deadline, verification_criteria):
        """做出承诺（带配额检查）"""
        can_promise, msg = self.can_make_promise()
        if not can_promise:
            raise Exception(f"❌ 无法做出承诺: {msg}")
            
        promise = {
            'id': promise_id,
            'content': content,
            'made_at': datetime.now().isoformat(),
            'deadline': deadline,
            'verification': verification_criteria,  # 客观的验证标准
            'status': 'ACTIVE',
            'verified': False
        }
        
        # 保存到配额系统
        quota = self.load_quota()
        quota['active'].append(promise)
        self.save_quota(quota)
        
        print(f"✅ 承诺已记录: {promise_id}")
        print(f"   内容: {content}")
        print(f"   截止: {deadline}")
        print(f"   验证标准: {verification_criteria}")
        
    def verify_promise(self, promise_id):
        """验证承诺是否兑现"""
        quota = self.load_quota()
        
        for p in quota['active']:
            if p['id'] == promise_id:
                # 执行验证
                is_fulfilled = self.run_verification(p['verification'])
                
                if is_fulfilled:
                    p['status'] = 'FULFILLED'
                    p['verified_at'] = datetime.now().isoformat()
                    p['verified'] = True
                    
                    # 移动到已完成
                    quota['fulfilled'].append(p)
                    quota['active'] = [x for x in quota['active'] if x['id'] != promise_id]
                    
                    self.save_quota(quota)
                    return True, "承诺已兑现并验证通过"
                else:
                    # 记录失败
                    p['failure_count'] = p.get('failure_count', 0) + 1
                    
                    if p['failure_count'] >= self.violation_threshold:
                        self.trigger_demotion(p)
                        
                    self.save_quota(quota)
                    return False, f"承诺验证失败 (第{p['failure_count']}次)"
                    
        return False, "未找到承诺"
    
    def trigger_demotion(self, promise):
        """触发降级"""
        demotion_file = "/root/.openclaw/workspace/.demotion_status.json"
        status = {
            'type': 'PROMISE_VIOLATION',
            'promise_id': promise['id'],
            'triggered_at': datetime.now().isoformat(),
            'action': 'RESTRICT_PROMISE',
            'duration_hours': 24  # 24小时内禁止新承诺
        }
        with open(demotion_file, 'w') as f:
            json.dump(status, f)
        print(f"🚨 触发降级：因承诺 {promise['id']} 未兑现，24小时内禁止新承诺")

#!/usr/bin/env python3

# 功能: 将工作记忆外部化，强制降低认知负荷

import json
import os
from datetime import datetime
from typing import List, Dict, Optional

class CognitiveScaffold:
    """
    认知脚手架系统：强制将复杂任务分解到工作记忆可容纳的单元
    基于Miller's Law（7±2原则）和认知负荷理论
    """
    
    def __init__(self):
        self.workspace = "/root/.openclaw/workspace"
        self.scaffold_dir = f"{self.workspace}/.cognitive_scaffolds"
        self.active_scaffold = None
        os.makedirs(self.scaffold_dir, exist_ok=True)
        
        # 认知限制常数
        self.MAX_UNITS = 5  # 工作记忆上限，保守设置为5
        self.MAX_DEPTH = 3  # 嵌套层级上限
        
    def decompose_task(self, task_description: str, complexity_score: int) -> Dict:
        """
        强制任务分解：高复杂度任务必须拆解
        
        complexity_score: 1-10，由系统评估或用户指定
        """
        if complexity_score <= 3:
            return {
                'type': 'atomic',
                'units': [task_description],
                'total_units': 1,
                'approved': True
            }
        
        # 强制分解：复杂度>3必须拆解
        scaffold = {
            'task_id': f"task_{datetime.now().strftime('%Y%m%d_%H%M%S')}",
            'original_description': task_description,
            'complexity_score': complexity_score,
            'created_at': datetime.now().isoformat(),
            'type': 'composite',
            'units': [],
            'checkpoint_after_each': True,  # 每个单元后必须检查
            'current_unit': 0,
            'status': 'decomposed'
        }
        
        # 使用LLM或规则引擎进行分解（这里提供模板）
        decomposition_prompt = f"""
任务: {task_description}
复杂度: {complexity_score}/10

必须分解为不超过{self.MAX_UNITS}个认知单元，每个单元：
1. 可在10分钟内完成
2. 有明确的完成标准
3. 不依赖后续单元即可验证
4. 当前单元失败不影响已完成的单元

输出格式：JSON数组，每个元素包含"action"和"verification_criteria"
"""
        
        # 在实际部署中，这里调用LLM API或加载预设模板
        # 现在提供手动分解接口
        print(f"⚠️  高复杂度任务（{complexity_score}/10）必须分解")
        print(f"请将其分解为{min(complexity_score//2 + 1, self.MAX_UNITS)}个可执行单元：")
        
        return scaffold
    
    def load_next_unit(self, task_id: str) -> Optional[Dict]:
        """加载下一个认知单元，强制单线程执行"""
        scaffold_file = f"{self.scaffold_dir}/{task_id}.json"
        
        if not os.path.exists(scaffold_file):
            return None
            
        with open(scaffold_file, 'r') as f:
            scaffold = json.load(f)
            
        if scaffold['current_unit'] >= len(scaffold['units']):
            return None
            
        unit = scaffold['units'][scaffold['current_unit']]
        unit['unit_index'] = scaffold['current_unit']
        unit['total_units'] = len(scaffold['units'])
        
        # 创建认知单元锁定文件（防止并发）
        lock_file = f"{self.workspace}/.current_cognitive_unit"
        with open(lock_file, 'w') as f:
            json.dump({
                'task_id': task_id,
                'unit_index': scaffold['current_unit'],
                'started_at': datetime.now().isoformat(),
                'unit_content': unit
            }, f, indent=2)
            
        return unit
    
    def complete_unit(self, task_id: str, evidence: List[str]) -> bool:
        """完成当前单元，强制证据提交"""
        scaffold_file = f"{self.scaffold_dir}/{task_id}.json"
        
        with open(scaffold_file, 'r') as f:
            scaffold = json.load(f)
            
        current_idx = scaffold['current_unit']
        
        # 验证证据
        unit = scaffold['units'][current_idx]
        required_evidence = unit.get('verification_criteria', [])
        
        missing_evidence = [e for e in required_evidence if not os.path.exists(e)]
        if missing_evidence:
            print(f"❌ 单元{current_idx+1}完成失败：缺少证据 {missing_evidence}")
            return False
        
        # 标记完成
        scaffold['units'][current_idx]['completed'] = True
        scaffold['units'][current_idx]['completed_at'] = datetime.now().isoformat()
        scaffold['units'][current_idx]['evidence'] = evidence
        scaffold['current_unit'] += 1
        
        with open(scaffold_file, 'w') as f:
            json.dump(scaffold, f, indent=2)
            
        # 触发蓝军检查点
        self.trigger_blue_checkpoint(task_id, current_idx)
        
        print(f"✅ 认知单元 {current_idx+1}/{len(scaffold['units'])} 完成")
        
        # 强制休息（认知卸载）
        if (current_idx + 1) % 2 == 0:  # 每完成2个单元
            print("🧠 认知休息：已完成2个单元，建议暂停5分钟进行状态同步")
            
        return True
    
    def trigger_blue_checkpoint(self, task_id: str, unit_idx: int):
        """触发蓝军强制检查点"""
        checkpoint_file = f"{self.workspace}/.pending_blue_checkpoint"
        with open(checkpoint_file, 'w') as f:
            json.dump({
                'task_id': task_id,
                'unit_index': unit_idx,
                'required_action': 'VERIFY_AND_APPROVE',
                'timestamp': datetime.now().isoformat(),
                'auto_block': True  # 未通过检查点前阻止下一单元
            }, f)

                    severity = "HIGH"
                else:
                    friction_duration = config['friction_duration']
                    severity = "NORMAL"
                
                print(f"🛑 摩擦触发：{habit_id} ({severity})")
                print(f"⏱️  强制等待 {friction_duration} 秒...")
                print(f"📝 {config['message']}")

                time.sleep(friction_duration)

class CascadeBreaker:
    """
    级联故障断路器：防止小问题演变为系统崩溃
    """
    
    def __init__(self):
        self.failure_modes = {
            'memory_loss': {'weight': 0.3, 'threshold': 2},      # 记忆缺失
            'skill_violation': {'weight': 0.4, 'threshold': 1},  # Skill违规（重）
            'archive_missing': {'weight': 0.3, 'threshold': 2}, # 归档缺失
            'supervision_gap': {'weight': 0.5, 'threshold': 1},  # 监督真空（重）
        }
        self.system_stability = 1.0  # 系统稳定性0-1
        self.cascade_log = []
        
    def inject_fault(self, fault_type: str):
        """注入故障信号"""
        if fault_type not in self.failure_modes:
            return
            
        config = self.failure_modes[fault_type]
        self.system_stability -= config['weight']
        
        fault_record = {
            'time': datetime.now().isoformat(),
            'type': fault_type,
            'weight': config['weight'],
            'remaining_stability': self.system_stability
        }
        self.cascade_log.append(fault_record)
        
        # 检查是否超过阈值
        recent_faults = [f for f in self.cascade_log 
                        if f['type'] == fault_type 
                        and (datetime.now() - datetime.fromisoformat(f['time'])).hours < 1]
        
        if len(recent_faults) >= config['threshold']:
            self.system_stability -= config['weight'] * 2  # 惩罚累积
        
        # 系统稳定性检查
        if self.system_stability < 0.5:
            self.emergency_shutdown("系统稳定性低于50%，启动紧急停机")
        elif self.system_stability < 0.7:
            self.warning_mode("系统稳定性警告，限制高复杂度任务")
            
    def emergency_shutdown(self, reason: str):
        """紧急停机"""
        with open(f"{self.workspace}/.emergency_stop", 'w') as f:
            f.write(f"{datetime.now().isoformat()}: {reason}\n")
            f.write("SYSTEM_HALTED: true\n")
            f.write("RESET_REQUIRES: ['user_approval', 'full_audit', 'stability_recovery_plan']\n")
        print(f"🛑 紧急停机：{reason}")
        print("系统已进入保护模式，需用户手动恢复")
        
    def recover_stability(self, action_success: bool):
        """恢复稳定性"""
        if action_success:
            self.system_stability = min(1.0, self.system_stability + 0.2)
            print(f"系统稳定性恢复至 {self.system_stability:.0%}")

#!/usr/bin/env python3
# 文件: /root/.openclaw/workspace/blockchain_memory.py

import hashlib
import json
from datetime import datetime

class BlockchainMemory:
    """
    区块链式记忆：确保跨会话记忆的不可篡改性和连续性
    """
    
    def __init__(self):
        self.chain_file = f"{self.workspace}/.memory_chain.json"
        self.genesis_block()
        
    def genesis_block(self):
        """创建创世区块"""
        if not os.path.exists(self.chain_file):
            genesis = {
                'index': 0,
                'timestamp': datetime.now().isoformat(),
                'data': 'SYSTEM_GENESIS',
                'prev_hash': '0' * 64,
                'hash': self.calculate_hash('0' * 64 + 'SYSTEM_GENESIS')
            }
            with open(self.chain_file, 'w') as f:
                json.dump([genesis], f)
                
    def add_memory_block(self, memory_type: str, content: dict):
        """添加记忆区块"""
        chain = self.load_chain()
        
        prev_block = chain[-1]
        block_data = {
            'index': len(chain),
            'timestamp': datetime.now().isoformat(),
            'session_id': os.environ.get('SESSION_ID', 'unknown'),
            'memory_type': memory_type,
            'content': content,
            'prev_hash': prev_block['hash']
        }
        
        block_data['hash'] = self.calculate_hash(
            json.dumps(block_data, sort_keys=True)
        )
        
        chain.append(block_data)
        
        with open(self.chain_file, 'w') as f:
            json.dump(chain, f, indent=2)
            
        return block_data['hash']  # 返回锚定哈希
    
    def verify_chain(self) -> bool:
        """验证记忆链完整性"""
        chain = self.load_chain()
        
        for i in range(1, len(chain)):
            current = chain[i]
            previous = chain[i-1]
            
            if current['prev_hash'] != previous['hash']:
                print(f"❌ 记忆链断裂在区块{i}")
                return False
                
            if current['hash'] != self.calculate_hash(json.dumps(current, sort_keys=True)):
                print(f"❌ 区块{i}被篡改")
                return False
                
        return True
    
    def calculate_hash(self, data: str) -> str:
        return hashlib.sha256(data.encode()).hexdigest()

第一部分：理论深度解析

1.1 知识管理理论基础：SECI模型在合伙人匹配案例库中的应用

根据野中郁次郎（Nonaka）和竹内弘高（Takeuchi）的SECI知识转化模型，知识创造是一个螺旋上升的过程，包含四个转化阶段。在满意解研究所的合伙人匹配场景中，案例库的核心使命就是实现隐性匹配经验的显性化：

关键洞察：案例库不是简单的文档存储，而是外显化(Externalization)的核心基础设施。通过强制结构化的数据输入（Schema约束），将原本散落在教练脑海中的”满意解决策逻辑”转化为可检索、可计算、可传承的显性知识。

1.2 知识图谱增强的案例表示理论

传统的关键词检索无法捕捉案例间的深层语义关联。本方案引入知识图谱嵌入（Knowledge Graph Embedding）技术，将案例表示为低维向量空间中的点，实现语义级相似度计算。

核心思想： - 将每个案例视为知识图谱中的实体（Entity） - 案例间的关系（行业相似、阶段相同、合伙人类型一致）视为关系（Relation） - 使用TransH模型（优于基础TransE，能处理多对多关系）将案例嵌入到语义空间

2.3 相似度算法设计（融合权重与语义）

基于TransH模型的改进相似度计算：

# case_repository/similarity/engine.py
import numpy as np
from typing import List, Tuple
from dataclasses import asdict

class CaseSimilarityEngine:
    """
    基于知识图谱嵌入的案例相似度计算引擎
    结合：显性属性匹配 + 隐性语义相似度
    """
    
    def __init__(self, embedding_dim: int = 128):
        self.embedding_dim = embedding_dim
        self.relation_weights = {
            "industry_match": 0.30,
            "stage_match": 0.20,
            "partner_type_match": 0.20,
            "founder_gap_similarity": 0.15,
            "decision_pattern_similarity": 0.10,
            "outcome_value": 0.05  # 成功案例权重略高
        }
    
    def calculate_explicit_similarity(self, case_a: Case, case_b: Case) -> float:
        """计算显性属性相似度（结构化字段匹配）"""
        score = 0.0
        
        # 行业匹配（精确匹配或图谱层级相似）
        if case_a.industry == case_b.industry:
            score += self.relation_weights["industry_match"] * 1.0
        elif self._is_related_industry(case_a.industry, case_b.industry):
            score += self.relation_weights["industry_match"] * 0.7  # 相关行业
        
        # 阶段匹配
        if case_a.stage == case_b.stage:
            score += self.relation_weights["stage_match"] * 1.0
        elif abs(self._stage_order(case_a.stage) - self._stage_order(case_b.stage)) == 1:
            score += self.relation_weights["stage_match"] * 0.5  # 相邻阶段
        
        # 合伙人类型匹配
        if case_a.partner_type == case_b.partner_type:
            score += self.relation_weights["partner_type_match"] * 1.0
        
        return score
    
    def calculate_embedding_similarity(self, vec_a: List[float], vec_b: List[float]) -> float:
        """
        基于TransH模型的语义相似度计算
        使用欧式距离转换的相似度度量
        """
        vec_a = np.array(vec_a)
        vec_b = np.array(vec_b)
        
        # 欧式距离
        euclidean_dist = np.linalg.norm(vec_a - vec_b)
        
        # 转换为相似度分数 (0-1)，距离越近越相似
        similarity = 1 / (1 + euclidean_dist)
        
        return float(similarity)
    
    def get_combined_similarity(self, case_a: Case, case_b: Case) -> Tuple[float, dict]:
        """
        综合相似度 = 显性匹配 + 隐性语义相似
        返回：总分, 详细分解
        """
        explicit_score = self.calculate_explicit_similarity(case_a, case_b)
        
        # 如果有嵌入向量，计算语义相似度
        semantic_score = 0.0
        if case_a.embedding_vector and case_b.embedding_vector:
            semantic_sim = self.calculate_embedding_similarity(
                case_a.embedding_vector, 
                case_b.embedding_vector
            )
            # 语义相似度分配权重
            semantic_score = semantic_sim * (
                self.relation_weights["founder_gap_similarity"] +
                self.relation_weights["decision_pattern_similarity"]
            )
        
        # 结果偏置（成功案例在推荐时权重略高）
        outcome_boost = 0.0
        if case_b.outcome == Outcome.SUCCESS:
            outcome_boost = self.relation_weights["outcome_value"] * 0.2
        
        total = explicit_score + semantic_score + outcome_boost
        
        breakdown = {
            "explicit_match": explicit_score,
            "semantic_similarity": semantic_score,
            "outcome_boost": outcome_boost,
            "total": total
        }
        
        return total, breakdown
    
    def find_similar_cases(self, query_case: Case, 
                          candidates: List[Case], 
                          top_k: int = 5) -> List[Tuple[Case, float, dict]]:
        """
        查找相似案例（CBR案例推理）
        """
        results = []
        
        for candidate in candidates:
            if candidate.case_id == query_case.case_id:
                continue
                
            score, breakdown = self.get_combined_similarity(query_case, candidate)
            results.append((candidate, score, breakdown))
        
        # 按相似度降序排列
        results.sort(key=lambda x: x[1], reverse=True)
        
        return results[:top_k]
    
    def _is_related_industry(self, ind1: Industry, ind2: Industry) -> bool:
        """判断行业是否相关（基于硬科技分类体系）"""
        # 定义行业关联图
        related_groups = [
            {Industry.AI_CHIP, Industry.GPU_CHIP, Industry.SENSOR},  # 半导体类
            {Industry.NEW_ENERGY, Industry.NEW_MATERIAL},  # 材料能源类
            {Industry.ROBOTICS, Industry.AI_CHIP},  # 智能机器人类
        ]
        
        for group in related_groups:
            if ind1 in group and ind2 in group:
                return True
        return False
    
    def _stage_order(self, stage: Stage) -> int:
        """阶段排序用于计算邻近度"""
        order = {
            Stage.STARTUP: 1,
            Stage.GROWTH: 2,
            Stage.EXPANSION: 3
        }
        return order.get(stage, 0)

#### 4.1.2 复盘生成器核心实现

```python
# case_repository/report/generator.py
import os
import json
from pathlib import Path
from typing import List, Dict, Optional
from datetime import datetime
from dataclasses import asdict

from jinja2 import Environment, FileSystemLoader, select_autoescape
from markdown import markdown
from weasyprint import HTML, CSS

from ..models.case import Case, Outcome
from ..similarity.engine import CaseSimilarityEngine
from ..database.repository import CaseRepository

class ReportGenerator:
    """
    复盘报告生成器（SECI组合化阶段实现）
    支持：Markdown（默认）、PDF（需安装weasyprint）、JSON（结构化数据）
    """
    
    def __init__(self, template_dir: Optional[str] = None):
        if template_dir is None:
            template_dir = Path(__file__).parent / "templates"
        
        # 配置Jinja2环境
        self.env = Environment(
            loader=FileSystemLoader(template_dir),
            autoescape=select_autoescape(['html', 'xml', 'md']),
            trim_blocks=True,  # 去除块级标签后的首行空白
            lstrip_blocks=True,  # 去除块级标签前的空白
            keep_trailing_newline=False
        )
        
        # 注册自定义过滤器
        self.env.filters['format_date'] = self._format_date
        self.env.filters['outcome_badge'] = self._outcome_badge
        
        self.similarity_engine = CaseSimilarityEngine()
        self.repo = CaseRepository()
    
    def generate_report(self, 
                        case_id: str, 
                        output_format: str = "markdown",
                        template_name: str = "partner_matching.md.j2",
                        include_similar: bool = True,
                        top_k_similar: int = 3) -> str:
        """
        生成完整复盘报告
        
        Args:
            case_id: 目标案例ID
            output_format: 输出格式 (markdown/pdf/json)
            template_name: 模板文件名
            include_similar: 是否包含相似案例对比
            top_k_similar: 相似案例数量
        """
        # 1. 获取案例数据
        case = self.repo.get_by_id(case_id)
        if not case:
            raise ValueError(f"案例 {case_id} 不存在")
        
        # 2. 获取相似案例（CBR案例推理）
        similar_cases = []
        if include_similar:
            all_cases = self.repo.get_all(exclude_ids=[case_id])
            similar_cases_raw = self.similarity_engine.find_similar_cases(
                case, all_cases, top_k=top_k_similar
            )
            
            similar_cases = [
                {
                    "case_id": c.case_id,
                    "case_name": c.case_name,
                    "similarity": score,
                    "key_takeaway": self._extract_key_takeaway(c),
                    "outcome": c.outcome.value
                }
                for c, score, _ in similar_cases_raw
            ]
        
        # 3. 构建模板上下文（外显化数据填充）
        context = {
            "case": case,
            "generation_time": datetime.now().isoformat(),
            "similar_cases": similar_cases,
            "key_decisions": self._extract_key_decisions(case),
            "mapping_table": self._build_capability_mapping(case),
            "action_suggestions": self._generate_action_items(case, similar_cases)
        }
        
        # 4. 渲染模板
        template = self.env.get_template(template_name)
        md_content = template.render(**context)
        
        # 5. 格式转换
        if output_format == "markdown":
            return md_content
        elif output_format == "pdf":
            return self._convert_to_pdf(md_content, case.case_id)
        elif output_format == "json":
            return self._convert_to_json(case, context)
        else:
            raise ValueError(f"不支持的输出格式: {output_format}")
    
    def _extract_key_decisions(self, case: Case) -> List[Dict]:
        """提取匹配过程中的关键决策点（外显化：决策逻辑拆解）"""
        decisions = []
        
        # 从决策方法描述中解析结构化决策点
        if "放弃" in case.selected.why_others_rejected or "选择" in case.selected.why_selected:
            decisions.append({
                "point": "最终候选人选择",
                "context": f"在{case.process.finalists_count}位决赛候选人中决策",
                "logic": case.selected.why_selected,
                "why_rejected": case.selected.why_others_rejected,
                "day": int(case.process.time_spent_days * 0.8)  # 估算决策发生在80%周期处
            })
        
        # 添加满意解阈值设定决策
        decisions.insert(0, {
            "point": "满意解标准设定",
            "context": "定义合伙人能力阈值与 stopping rule",
            "logic": f"设定最低阈值{case.process.satisfaction_threshold}，" + 
                    f"权重分配：{case.process.key_criteria_weights}",
            "why_rejected": "避免过度优化导致的搜索成本失控",
            "day": int(case.process.time_spent_days * 0.3)  # 早期设定
        })
        
        return decisions
    
    def _build_capability_mapping(self, case: Case) -> Dict[str, Dict]:
        """构建创始人缺口与合伙人补强的映射表"""
        mapping = {}
        
        # 基于创始人短板解析能力缺口
        weaknesses = case.founder.main_weakness.split("；")
        requirements = case.requirements.must_have + case.requirements.nice_to_have
        
        for i, weakness in enumerate(weaknesses[:3]):  # 最多取3个核心缺口
            mapping[f"缺口{i+1}"] = {
                "gap_desc": weakness,
                "supplement": requirements[i] if i < len(requirements) else "综合商业能力",
                "priority": "P0" if i == 0 else "P1"
            }
        
        return mapping
    
    def _extract_key_takeaway(self, case: Case) -> str:
        """从案例中提取关键洞察（一句话总结）"""
        if case.outcome_record.key_success_factors:
            return case.outcome_record.key_success_factors[0]
        return case.outcome_record.lessons_learned[0] if case.outcome_record.lessons_learned else "待总结"
    
    def _generate_action_items(self, case: Case, similar_cases: List[Dict]) -> List[str]:
        """基于相似案例对比生成行动建议"""
        suggestions = []
        
        # 分析相似案例的成功模式
        success_patterns = [
            c for c in similar_cases if c["outcome"] == "成功"
        ]
        
        if success_patterns:
            suggestions.append(f"参考成功案例[{success_patterns[0]['case_id']}]的合伙人画像特征")
        
        # 如果当前案例失败或待定，对比成功案例差异
        if case.outcome != Outcome.SUCCESS and success_patterns:
            suggestions.append(f"重新评估是否应调整满意解阈值（参考成功案例的平均阈值）")
        
        # 基于阶段建议
        if case.stage.value == "初创期":
            suggestions.append("重点关注合伙人资源网络与融资能力的匹配度")
        
        return suggestions
    
    def _convert_to_pdf(self, md_content: str, case_id: str) -> str:
        """将Markdown转换为PDF（可选功能）"""
        html_content = markdown(md_content, extensions=['tables', 'fenced_code'])
        
        # 添加CSS样式
        css = CSS(string='''
            @page { margin: 2cm; size: A4; }
            body { font-family: "Noto Sans CJK SC", sans-serif; line-height: 1.6; }
            h1 { color: #2c3e50; border-bottom: 2px solid #3498db; }
            h2 { color: #34495e; margin-top: 24px; }
            table { border-collapse: collapse; width: 100%; margin: 16px 0; }
            th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
            th { background-color: #f5f5f5; }
            blockquote { border-left: 4px solid #3498db; margin: 0; padding-left: 16px; color: #666; }
        ''')
        
        output_path = f"/mnt/kimi/output/case_report_{case_id}.pdf"
        HTML(string=html_content).write_pdf(output_path, stylesheets=[css])
        return output_path
    
    def _convert_to_json(self, case: Case, context: Dict) -> str:
        """导出结构化JSON（供其他系统消费）"""
        export_data = {
            "metadata": {
                "case_id": case.case_id,
                "export_time": context["generation_time"],
                "version": "2.0"
            },
            "structured_data": asdict(case),
            "derived_insights": {
                "key_decisions": context["key_decisions"],
                "similar_cases": context["similar_cases"],
                "action_suggestions": context["action_suggestions"]
            }
        }
        return json.dumps(export_data, ensure_ascii=False, indent=2)
    
    @staticmethod
    def _format_date(date_obj, format_str="%Y-%m-%d"):
        """Jinja2过滤器：格式化日期"""
        if isinstance(date_obj, datetime):
            return date_obj.strftime(format_str)
        return str(date_obj)
    
    @staticmethod
    def _outcome_badge(outcome: Outcome) -> str:
        """Jinja2过滤器：结果徽章"""
        badges = {
            "成功": "✅ SUCCESS",
            "失败": "❌ FAILURE", 
            "待定": "⏳ PENDING",
            "进行中": "🔄 ONGOING"
        }
        return badges.get(outcome.value, outcome.value)

            ),

                role_type=request.partner_type or PartnerType.BUSINESS,
                must_have=["API查询条件"],
                expected_contribution="数据检索"
            ),
            process=None,
            selected=None,
            outcome_record=None
        )
        
        # 2. 获取候选案例（按行业/阶段预过滤）
        filters = {}
        if request.industry:
            filters["industry"] = request.industry.value
        if request.stage:
            filters["stage"] = request.stage.value
            
        candidates = repo.get_filtered_cases(filters)
        
        # 3. 计算相似度（CBR案例推理）
        engine = CaseSimilarityEngine()
        similar_cases = engine.find_similar_cases(
            query_case, candidates, top_k=request.top_k
        )
        
        # 4. 组装响应
        response = []
        for case, score, breakdown in similar_cases:
            # 只返回指定结果类型的案例
            if case.outcome not in request.include_outcomes:
                continue
                
            response.append(SimilarCaseResponse(
                case_id=case.case_id,
                case_name=case.case_name,
                relevance_score=score,
                similarity_breakdown=breakdown,
                key_takeaway=case.outcome_record.key_success_factors[0] 
                            if case.outcome_record.key_success_factors else "N/A",
                outcome=case.outcome.value,
                industry=case.industry.value,
                stage=case.stage.value
            ))
        
        return response
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"相似度计算失败: {str(e)}")

@app.get("/api/v1/industry-benchmark/{industry}", response_model=BenchmarkResponse)
async def get_industry_benchmark(
    industry: Industry,
    repo: CaseRepository = Depends(get_repository)
):
    """
    获取行业基准数据（成功率、关键成功因素等）
    
    用于匹配引擎的基准参照与阈值设定建议。
    """
    cases = repo.get_by_industry(industry.value)
    
    if not cases:
        raise HTTPException(status_code=404, detail=f"行业 {industry.value} 暂无案例数据")
    
    # 统计分析
    total = len(cases)
    successes = [c for c in cases if c.outcome == Outcome.SUCCESS]
    success_rate = len(successes) / total if total > 0 else 0.0
    
    # 聚合成功因素（词频统计）
    from collections import Counter
    all_factors = []
    for c in successes:
        if c.outcome_record and c.outcome_record.key_success_factors:
            all_factors.extend(c.outcome_record.key_success_factors)
    
    top_factors = [factor for factor, _ in Counter(all_factors).most_common(5)]
    
    # 聚合失败原因
    failures = [c for c in cases if c.outcome == Outcome.FAILURE]
    all_reasons = []
    for c in failures:
        if c.outcome_record and c.outcome_record.failure_reasons:
            all_reasons.extend(c.outcome_record.failure_reasons)
    top_reasons = [reason for reason, _ in Counter(all_reasons).most_common(3)]
    
    # 计算平均匹配周期
    avg_time = sum(c.process.time_spent_days for c in cases if c.process) / \
               len([c for c in cases if c.process]) if cases else 0
    
    return BenchmarkResponse(
        industry=industry.value,
        total_cases=total,
        success_rate=success_rate,
        avg_time_spent_days=round(avg_time, 1),
        key_success_factors=top_factors,
        common_failure_reasons=top_reasons
    )

        }
        
        if industry:
            payload["industry"] = industry
        if stage:
            payload["stage"] = stage
        if partner_type:
            payload["partner_type"] = partner_type
            
        response = requests.post(
            f"{self.base_url}/api/v1/similar-cases",
            json=payload,
            timeout=self.timeout
        )
        response.raise_for_status()
        
        return response.json()
    
    def get_industry_benchmark(self, industry: str) -> Dict:
        """获取行业基准数据（成功率、周期等）"""
        response = requests.get(
            f"{self.base_url}/api/v1/industry-benchmark/{industry}",
            timeout=self.timeout
        )
        response.raise_for_status()
        return response.json()
    
    def record_result(self, case_data: Dict) -> str:
        """记录匹配结果到案例库（闭环）"""
        response = requests.post(
            f"{self.base_url}/api/v1/record-matching-result",
            json=case_data,
            timeout=self.timeout
        )
        response.raise_for_status()
        return response.json()["case_id"]

6.1.2 数据版本控制策略（Git-LFS管理SQLite）

由于SQLite是二进制文件，直接使用Git会导致仓库膨胀。采用导出-导入策略：

# case_repository/database/version_control.py
import sqlite3
import subprocess
from pathlib import Path
from datetime import datetime
from typing import Optional

class DatabaseVersionControl:
    """
    SQLite数据库版本控制管理器
    
    策略：
    1. 日常开发：直接操作SQLite（高性能）
    2. 版本提交：导出为SQL文本（diff友好）+ 结构快照
    3. 版本回滚：从SQL文本重建数据库
    """
    
    def __init__(self, db_path: str = "./data/case_repository.db"):
        self.db_path = Path(db_path)
        self.sql_dump_path = Path("./data/schema_and_data.sql")
        self.backup_dir = Path("./data/backups")
        self.backup_dir.mkdir(exist_ok=True)
    
    def export_to_sql(self, commit_message: Optional[str] = None) -> str:
        """
        导出数据库为SQL文本（用于Git版本控制）
        
        生成两个文件：
        1. schema.sql - 表结构（用于diff查看Schema变更）
        2. data.sql - 数据INSERT语句（用于数据版本控制）
        """
        if not self.db_path.exists():
            raise FileNotFoundError(f"数据库文件不存在: {self.db_path}")
        
        # 导出结构
        schema_sql = self._export_schema()
        with open("./data/schema.sql", "w", encoding="utf-8") as f:
            f.write(schema_sql)
        
        # 导出数据（脱敏处理）
        data_sql = self._export_data()
        with open(self.sql_dump_path, "w", encoding="utf-8") as f:
            f.write("-- 满意解案例库数据导出\n")
            f.write(f"-- 导出时间: {datetime.now().isoformat()}\n")
            f.write(f"-- 案例数量: {self._get_case_count()}\n\n")
            f.write(schema_sql)
            f.write("\n\n")
            f.write(data_sql)
        
        # 创建可读的结构快照（Markdown格式，便于人工Review）
        self._create_structure_snapshot()
        
        return str(self.sql_dump_path)
    
    def import_from_sql(self, sql_file: Optional[str] = None) -> None:
        """从SQL文件重建数据库（用于版本回滚）"""
        sql_file = sql_file or self.sql_dump_path
        
        # 备份当前数据库
        if self.db_path.exists():
            backup_path = self.backup_dir / f"case_repo_backup_{datetime.now().strftime('%Y%m%d_%H%M%S')}.db"
            self.db_path.rename(backup_path)
            print(f"已备份原数据库到: {backup_path}")
        
        # 重建数据库
        conn = sqlite3.connect(self.db_path)
        with open(sql_file, "r", encoding="utf-8") as f:
            sql_script = f.read()
            conn.executescript(sql_script)
        conn.close()
        
        print(f"数据库已从 {sql_file} 重建完成")
    
    def commit_version(self, message: str) -> None:
        """提交当前数据库状态到Git（集成命令）"""
        # 1. 导出SQL
        self.export_to_sql()
        
        # 2. Git添加（排除二进制.db文件，跟踪.sql文本）
        subprocess.run(["git", "add", "data/schema.sql", "data/schema_and_data.sql", "data/structure_snapshot.md"])
        
        # 3. 使用自定义diff工具显示数据库变更摘要
        diff_summary = self._generate_diff_summary()
        
        # 4. 提交（包含统计信息）
        full_message = f"{message}\n\n数据库统计:\n{diff_summary}"
        subprocess.run(["git", "commit", "-m", full_message])
        
        print(f"已提交版本: {message}")
    
    def _export_schema(self) -> str:
        """导出数据库结构（PRAGMA命令）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        schema = []
        schema.append("-- 数据库结构定义\n")
        
        # 获取所有表
        cursor.execute("SELECT name, sql FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'")
        tables = cursor.fetchall()
        
        for name, sql in tables:
            if sql:  # 跳过自动生成的sqlite_sequence
                schema.append(f"-- 表: {name}\n")
                schema.append(f"{sql};\n\n")
        
        conn.close()
        return "".join(schema)
    
    def _export_data(self) -> str:
        """导出数据（INSERT语句，支持脱敏）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        data = []
        data.append("-- 数据记录\n")
        
        # 获取所有表（按依赖顺序：先案例，后关联表）
        tables = ["cases", "founder_profiles", "matching_processes", "outcome_records", "case_tags"]
        
        for table in tables:
            try:
                cursor.execute(f"SELECT * FROM {table}")
                rows = cursor.fetchall()
                if not rows:
                    continue
                
                # 获取列名
                cursor.execute(f"PRAGMA table_info({table})")
                columns = [col[1] for col in cursor.fetchall()]
                
                data.append(f"\n-- 表 {table} 的数据 ({len(rows)}行)\n")
                
                for row in rows:
                    values = []
                    for i, val in enumerate(row):
                        col_name = columns[i]
                        
                        # 脱敏处理：保密级别为"内部"的案例隐藏敏感信息
                        if table == "cases" and col_name == "case_name" and self._is_internal_case(row):
                            val = val[:2] + "***"  # 脱敏案例名
                        
                        if val is None:
                            values.append("NULL")
                        elif isinstance(val, str):
                            escaped = val.replace("'", "''")
                            values.append(f"'{escaped}'")
                        else:
                            values.append(str(val))
                    
                    insert = f"INSERT INTO {table} ({', '.join(columns)}) VALUES ({', '.join(values)});\n"
                    data.append(insert)
                    
            except Exception as e:
                data.append(f"-- 导出{table}失败: {e}\n")
        
        conn.close()
        return "".join(data)
    
    def _create_structure_snapshot(self) -> None:
        """创建Markdown格式的结构快照（便于人工Review）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        md = ["# 案例库结构快照\n\n"]
        md.append(f"**生成时间**: {datetime.now().isoformat()}\n\n")
        
        # 案例统计
        cursor.execute("SELECT industry, stage, outcome, COUNT(*) FROM cases GROUP BY industry, stage, outcome")
        stats = cursor.fetchall()
        
        md.append("## 案例分布统计\n\n")
        md.append("| 行业 | 阶段 | 结果 | 数量 |\n")
        md.append("|------|------|------|------|\n")
        for row in stats:
            md.append(f"| {row[0]} | {row[1]} | {row[2]} | {row[3]} |\n")
        
        # 标签云
        cursor.execute("""
            SELECT tag, COUNT(*) as count 
            FROM case_tags 
            GROUP BY tag 
            ORDER BY count DESC 
            LIMIT 20
        """)
        tags = cursor.fetchall()
        
        md.append("\n## 高频标签云\n\n")
        for tag, count in tags:
            md.append(f"- **{tag}** ({count}次)\n")
        
        with open("./data/structure_snapshot.md", "w", encoding="utf-8") as f:
            f.write("".join(md))
        
        conn.close()
    
    def _generate_diff_summary(self) -> str:
        """生成变更摘要（用于Git提交信息）"""
        try:
            result = subprocess.run(
                ["git", "diff", "--stat", "data/schema.sql"],
                capture_output=True,
                text=True
            )
            return result.stdout or "无结构变更"
        except:
            return "无法获取diff统计"
    
    def _get_case_count(self) -> int:
        """获取案例总数"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        cursor.execute("SELECT COUNT(*) FROM cases")
        count = cursor.fetchone()[0]
        conn.close()
        return count
    
    def _is_internal_case(self, row) -> bool:
        """判断案例是否为内部保密级别（用于脱敏）"""
        # 假设confidentiality列在第6个位置（根据实际Schema调整）
        return len(row) > 6 and row[6] == "内部"

7.2 数据持久化层（case_repository/database/repository.py）

"""
案例仓储模式实现
支持SQLite本地存储 + JSON导出（用于Git版本控制）
"""
import sqlite3
import json
from pathlib import Path
from typing import List, Optional, Dict, Any
from datetime import date

from ..models.case import Case, Industry, Stage, PartnerType, Outcome

class CaseRepository:
    """
    案例数据仓储（Repository模式）
    提供CRUD操作和查询接口
    """
    
    def __init__(self, db_path: str = "./data/case_repository.db"):
        self.db_path = Path(db_path)
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        self._init_database()
    
    def _init_database(self):
        """初始化数据库Schema"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 主表：案例基础信息
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS cases (
                case_id TEXT PRIMARY KEY,
                case_name TEXT NOT NULL,
                created_date TEXT,
                updated_date TEXT,
                industry TEXT,
                stage TEXT,
                partner_type TEXT,
                outcome TEXT,
                founder_profile TEXT,  -- JSON存储
                requirements TEXT,     -- JSON存储
                process TEXT,          -- JSON存储（可为空）
                selected_partner TEXT, -- JSON存储（可为空）
                outcome_record TEXT,   -- JSON存储（可为空）
                tags TEXT,             -- 逗号分隔
                confidentiality TEXT,
                created_by TEXT,
                embedding_vector TEXT  -- JSON数组（可为空）
            )
        """)
        
        # 索引：加速查询
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_industry ON cases(industry)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_stage ON cases(stage)")
        cursor.execute("CREATE INDEX IF NOT EXISTS idx_outcome ON cases(outcome)")
        
        conn.commit()
        conn.close()
    
    def save(self, case: Case) -> str:
        """
        保存案例（插入或更新）
        返回：case_id
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # 转换为可存储格式
        process_json = json.dumps(case.process.to_dict(), ensure_ascii=False) if case.process else None
        selected_json = json.dumps(case.selected.to_dict(), ensure_ascii=False) if case.selected else None
        outcome_json = json.dumps(case.outcome_record.to_dict(), ensure_ascii=False) if case.outcome_record else None
        embedding_json = json.dumps(case.embedding_vector) if case.embedding_vector else None
        
        cursor.execute("""
            INSERT OR REPLACE INTO cases (
                case_id, case_name, created_date, updated_date,
                industry, stage, partner_type, outcome,
                founder_profile, requirements, process,
                selected_partner, outcome_record, tags,
                confidentiality, created_by, embedding_vector
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            case.case_id,
            case.case_name,
            case.created_date.isoformat(),
            case.updated_date.isoformat(),
            case.industry.value,
            case.stage.value,
            case.partner_type.value,
            case.outcome.value,
            json.dumps(case.founder.to_dict(), ensure_ascii=False),
            json.dumps(case.requirements.to_dict(), ensure_ascii=False),
            process_json,
            selected_json,
            outcome_json,
            ",".join(case.tags),
            case.confidentiality,
            case.created_by,
            embedding_json
        ))
        
        conn.commit()
        conn.close()
        
        return case.case_id
    
    def get_by_id(self, case_id: str) -> Optional[Case]:
        """通过ID获取案例"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("SELECT * FROM cases WHERE case_id = ?", (case_id,))
        row = cursor.fetchone()
        conn.close()
        
        if not row:
            return None
        
        return self._row_to_case(row)
    
    def get_all(self, exclude_ids: List[str] = None) -> List[Case]:
        """获取所有案例（可排除特定ID）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        if exclude_ids:
            placeholders = ",".join(["?" for _ in exclude_ids])
            cursor.execute(f"SELECT * FROM cases WHERE case_id NOT IN ({placeholders})", exclude_ids)
        else:
            cursor.execute("SELECT * FROM cases")
        
        rows = cursor.fetchall()
        conn.close()
        
        return [self._row_to_case(row) for row in rows]
    
    def get_filtered_cases(self, filters: Dict[str, str]) -> List[Case]:
        """
        多条件筛选查询
        filters: {"industry": "AI芯片", "stage": "初创期", ...}
        """
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        conditions = []
        params = []
        
        for key, value in filters.items():
            if value:
                conditions.append(f"{key} = ?")
                params.append(value)
        
        query = "SELECT * FROM cases"
        if conditions:
            query += " WHERE " + " AND ".join(conditions)
        
        cursor.execute(query, params)
        rows = cursor.fetchall()
        conn.close()
        
        return [self._row_to_case(row) for row in rows]
    
    def get_by_industry(self, industry: str) -> List[Case]:
        """按行业获取案例"""
        return self.get_filtered_cases({"industry": industry})
    
    def delete(self, case_id: str) -> bool:
        """删除案例（软删除可考虑添加deleted标记）"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("DELETE FROM cases WHERE case_id = ?", (case_id,))
        conn.commit()
        conn.close()
        
        return True
    
    def export_to_json(self, output_path: str):
        """导出所有案例到JSON文件（用于备份/版本控制）"""
        cases = self.get_all()
        data = {
            "export_time": date.today().isoformat(),
            "total_count": len(cases),
            "cases": [case.to_dict() for case in cases]
        }
        
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    
    def import_from_json(self, input_path: str):
        """从JSON文件导入案例"""
        with open(input_path, "r", encoding="utf-8") as f:
            data = json.load(f)
        
        for case_data in data.get("cases", []):
            case = Case.from_dict(case_data)
            self.save(case)
    
    def _row_to_case(self, row) -> Case:
        """将数据库行转换为Case对象"""
        (case_id, case_name, created_date, updated_date,
         industry, stage, partner_type, outcome,
         founder_json, requirements_json, process_json,
         selected_json, outcome_json, tags_str,
         confidentiality, created_by, embedding_json) = row
        
        # 解析JSON字段
        founder = json.loads(founder_json) if founder_json else {}
        requirements = json.loads(requirements_json) if requirements_json else {}
        process = json.loads(process_json) if process_json else None
        selected = json.loads(selected_json) if selected_json else None
        outcome_record = json.loads(outcome_json) if outcome_json else None
        embedding = json.loads(embedding_json) if embedding_json else None
        
        return Case(
            case_id=case_id,
            case_name=case_name,
            created_date=date.fromisoformat(created_date),
            updated_date=date.fromisoformat(updated_date),
            industry=Industry(industry),
            stage=Stage(stage),
            partner_type=PartnerType(partner_type),
            outcome=Outcome(outcome),
            founder=FounderProfile.from_dict(founder),
            requirements=PartnerRequirements.from_dict(requirements),
            process=MatchingProcess.from_dict(process) if process else None,
            selected=SelectedPartner.from_dict(selected) if selected else None,
            outcome_record=OutcomeRecord.from_dict(outcome_record) if outcome_record else None,
            tags=tags_str.split(",") if tags_str else [],
            confidentiality=confidentiality,
            created_by=created_by,
            embedding_vector=embedding
        )
    
    def close(self):
        """关闭资源（SQLite无需显式关闭连接，但保留接口）"""
        pass

    click.echo("-" * 50)

    # 构建筛选条件
    filters = {}
    if industry:
        filters["industry"] = industry
    if stage:
        filters["stage"] = stage

    if filters:
        candidates = repo.get_filtered_cases(filters)
    else:
        candidates = repo.get_all()

    results = []
    keywords = query_str.lower().split()
    
    for case in candidates:
        # 过滤结果类型
        if outcome == "success" and case.outcome != Outcome.SUCCESS:
            continue
        if outcome == "failure" and case.outcome != Outcome.FAILURE:
            continue
        
        # 计算匹配分数（简单实现）
        score = 0
        text_to_search = f"{case.case_name} {case.founder.core_tech} {case.founder.main_weakness}".lower()
        
        for kw in keywords:
            if kw in text_to_search:
                score += 1
        
        if score > 0:
            results.append((case, score))
    
    # 按分数排序
    results.sort(key=lambda x: x[1], reverse=True)
    
    # 显示结果
    if not results:
        click.echo("未找到匹配案例，建议：")
        click.echo("  1. 尝试减少关键词数量")
        click.echo("  2. 使用更通用的行业术语")
        return
    
    click.echo(f"找到 {len(results[:top])} 个匹配案例：\n")
    
    for i, (case, score) in enumerate(results[:top], 1):
        outcome_emoji = "✅" if case.outcome == Outcome.SUCCESS else "❌" if case.outcome == Outcome.FAILURE else "⏳"
        click.echo(f"{i}. [{case.case_id}] {case.case_name} {outcome_emoji}")
        click.echo(f"   行业: {case.industry.value} | 阶段: {case.stage.value} | 匹配度: {score}")
        if case.outcome_record and case.outcome_record.key_success_factors:
            click.echo(f"   💡 {case.outcome_record.key_success_factors[0]}")
        click.echo()

7.4 相似度计算引擎简化版（case_repository/similarity/engine.py）

"""
案例相似度计算引擎（简化实现）
基于显性属性匹配，可扩展为知识图谱嵌入
"""
import numpy as np
from typing import List, Tuple, Dict

from ..models.case import Case, Industry, Stage, PartnerType, Outcome

class CaseSimilarityEngine:
    """
    基于知识图谱嵌入的案例相似度计算引擎（简化版）
    结合：显性属性匹配 + 权重计算
    """
    
    def __init__(self, embedding_dim: int = 64):
        self.embedding_dim = embedding_dim
        self.relation_weights = {
            "industry_match": 0.30,
            "stage_match": 0.20,
            "partner_type_match": 0.20,
            "founder_gap_similarity": 0.15,
            "outcome_value": 0.15  # 成功案例权重略高
        }
    
    def calculate_explicit_similarity(self, case_a: Case, case_b: Case) -> Tuple[float, Dict]:
        """计算显性属性相似度"""
        score = 0.0
        breakdown = {}
        
        # 行业匹配（精确匹配或相关匹配）
        if case_a.industry == case_b.industry:
            score += self.relation_weights["industry_match"]
            breakdown["industry_match"] = self.relation_weights["industry_match"]
        elif self._is_related_industry(case_a.industry, case_b.industry):
            score += self.relation_weights["industry_match"] * 0.7
            breakdown["industry_match"] = self.relation_weights["industry_match"] * 0.7
        else:
            breakdown["industry_match"] = 0.0
        
        # 阶段匹配
        if case_a.stage == case_b.stage:
            score += self.relation_weights["stage_match"]
            breakdown["stage_match"] = self.relation_weights["stage_match"]
        elif abs(self._stage_order(case_a.stage) - self._stage_order(case_b.stage)) == 1:
            score += self.relation_weights["stage_match"] * 0.5
            breakdown["stage_match"] = self.relation_weights["stage_match"] * 0.5
        else:
            breakdown["stage_match"] = 0.0
        
        # 合伙人类型匹配
        if case_a.partner_type == case_b.partner_type:
            score += self.relation_weights["partner_type_match"]
            breakdown["partner_type_match"] = self.relation_weights["partner_type_match"]
        else:
            breakdown["partner_type_match"] = 0.0
        
        # 创始人能力缺口相似度（基于文本相似度简化计算）
        gap_sim = self._calculate_gap_similarity(case_a.founder.main_weakness, case_b.founder.main_weakness)
        gap_score = gap_sim * self.relation_weights["founder_gap_similarity"]
        score += gap_score
        breakdown["founder_gap_similarity"] = gap_score
        
        # 结果偏置（成功案例推荐时加分）
        if case_b.outcome == Outcome.SUCCESS:
            score += self.relation_weights["outcome_value"] * 0.2
            breakdown["outcome_value"] = self.relation_weights["outcome_value"] * 0.2
        else:
            breakdown["outcome_value"] = 0.0
        
        breakdown["total"] = score
        return score, breakdown
    
    def find_similar_cases(self, query_case: Case, 
                          candidates: List[Case], 
                          top_k: int = 5) -> List[Tuple[Case, float, Dict]]:
        """查找相似案例（CBR案例推理）"""
        results = []
        
        for candidate in candidates:
            if candidate.case_id == query_case.case_id:
                continue
            
            score, breakdown = self.calculate_explicit_similarity(query_case, candidate)
            results.append((candidate, score, breakdown))
        
        # 按相似度降序排列
        results.sort(key=lambda x: x[1], reverse=True)
        
        return results[:top_k]
    
    def _is_related_industry(self, ind1: Industry, ind2: Industry) -> bool:
        """判断行业是否相关"""
        related_groups = [
            {Industry.AI_CHIP, Industry.GPU_CHIP, Industry.SENSOR},
            {Industry.NEW_ENERGY, Industry.NEW_MATERIAL},
            {Industry.ROBOTICS, Industry.AI_CHIP},
        ]
        for group in related_groups:
            if ind1 in group and ind2 in group:
                return True
        return False
    
    def _stage_order(self, stage: Stage) -> int:
        """阶段排序"""
        order = {
            Stage.STARTUP: 1,
            Stage.GROWTH: 2,
            Stage.EXPANSION: 3
        }
        return order.get(stage, 0)
    
    def _calculate_gap_similarity(self, gap_a: str, gap_b: str) -> float:
        """计算能力缺口描述相似度（简化版）"""
        # 基于关键词重叠的简单实现
        words_a = set(gap_a.lower().split())
        words_b = set(gap_b.lower().split())
        
        if not words_a or not words_b:
            return 0.0
        
        intersection = words_a & words_b
        union = words_a | words_b
        
        return len(intersection) / len(union) if union else 0.0

            lines.append(f"- **满意解阈值**: {case.process.satisfaction_threshold}")

        # 结果与复盘
        if case.outcome_record:
            lines.append("## 结果与复盘")
            lines.append("### 成功关键因素")
            for factor in case.outcome_record.key_success_factors:

            lines.append("")
        
        # 标签
        lines.append("## 知识标签")
        lines.append(" ".join([f"#{tag}" for tag in case.tags]))
        
        return "\n".join(lines)

第八部分：Kimi Claw部署配置

8.1 针对Kimi Claw的环境适配

Kimi Claw是云端托管的OpenClaw环境，具有以下特点： - 文件系统持久化：云端40GB存储 - Python环境：支持标准Python 3.9+运行 - 网络限制：可访问外部API，但需考虑延迟 - 无Docker权限：无法运行Docker容器

部署策略：纯Python依赖，无需Docker，直接运行。

8.2 项目初始化脚本（init_claw.py）

#!/usr/bin/env python3
"""
Kimi Claw环境初始化脚本
一键配置案例库Skill环境
"""
import subprocess
import sys
import os
from pathlib import Path

def init_environment():
    """初始化案例库环境"""
    print("=" * 60)
    print("满意解案例库管理Skill - Kimi Claw初始化")
    print("=" * 60)
    
    # 1. 检查Python版本
    print("\n[1/4] 检查Python环境...")
    version = sys.version_info
    if version.major < 3 or (version.major == 3 and version.minor < 9):
        print(f"❌ 需要Python 3.9+，当前 {version.major}.{version.minor}")
        return False
    print(f"✅ Python {version.major}.{version.minor}.{version.micro}")
    
    # 2. 安装依赖
    print("\n[2/4] 安装Python依赖...")
    deps = [
        "click>=8.0.0",
        "pydantic>=2.0.0",
        "numpy>=1.24.0"
    ]
    
    for dep in deps:
        try:
            subprocess.check_call([sys.executable, "-m", "pip", "install", "-q", dep])
            print(f"  ✅ {dep}")
        except Exception as e:
            print(f"  ⚠️ {dep} 安装失败: {e}")
    
    # 3. 创建目录结构
    print("\n[3/4] 创建目录结构...")
    dirs = [
        "./data",
        "./data/backups",
        "./exports",
        "./templates"
    ]
    for d in dirs:
        Path(d).mkdir(parents=True, exist_ok=True)
        print(f"  ✅ {d}")
    
    # 4. 初始化数据库
    print("\n[4/4] 初始化数据库...")
    try:
        from case_repository.database.repository import CaseRepository
        repo = CaseRepository()
        print("  ✅ SQLite数据库初始化完成")
    except Exception as e:
        print(f"  ⚠️ 数据库初始化失败: {e}")
    
    print("\n" + "=" * 60)
    print("初始化完成！可用命令：")
    print("  python -m case_repository.cli.main create")
    print("  python -m case_repository.cli.main search <关键词>")
    print("  python -m case_repository.cli.main dashboard")
    print("=" * 60)
    
    return True

if __name__ == "__main__":
    init_environment()

8.3 Docker部署配置（本地/服务器环境）

对于本地OpenClaw或独立服务器部署，提供Docker配置：

# Dockerfile（多阶段构建，轻量级）
# 阶段1：构建环境
FROM python:3.11-slim AS builder

WORKDIR /build
COPY requirements.txt .
RUN pip install --user --no-cache-dir -r requirements.txt

# 阶段2：运行环境（最小化）
FROM python:3.11-slim

WORKDIR /app

# 从构建阶段复制依赖
COPY --from=builder /root/.local /root/.local
ENV PATH=/root/.local/bin:$PATH

# 复制应用代码
COPY case_repository/ ./case_repository/
COPY init_claw.py .

# 创建数据目录
RUN mkdir -p /app/data /app/exports

# 非root用户运行（安全最佳实践）
RUN useradd -m -u 1000 appuser && chown -R appuser:appuser /app
USER appuser

# 暴露API端口
EXPOSE 8000

# 默认命令：CLI交互模式
ENTRYPOINT ["python", "-m", "case_repository.cli.main"]
CMD ["dashboard"]

# case_repository/retrieval/graph_rag.py
import chromadb
from chromadb.config import Settings
import numpy as np
from typing import List, Dict, Optional
import json
from sentence_transformers import SentenceTransformer

class GraphRAGRetriever:
    """
    GraphRAG增强检索器（轻量级实现）
    结合：向量检索（Chroma）+ 知识图谱关系 + 大模型生成
    """
    
    def __init__(self, persist_dir: str = "./data/chroma_db"):
        # 持久化配置（本地存储）
        self.client = chromadb.Client(Settings(
            chroma_db_impl="duckdb+parquet",
            persist_directory=persist_dir,
            anonymized_telemetry=False
        ))
        
        # 使用轻量级中文嵌入模型（BAAI/bge-large-zh-v1.5为当前中文最佳实践）
        self.embedder = SentenceTransformer('BAAI/bge-large-zh-v1.5')
        
        # 创建/获取集合（类似数据库表）
        self.case_collection = self.client.get_or_create_collection(
            name="case_knowledge",
            metadata={"hnsw:space": "cosine"}  # 余弦相似度适合语义匹配
        )
        
        # 关系图谱（简化版，存储为JSON）
        self.relation_graph = self._load_relation_graph()
    
    def index_case(self, case: 'Case'):
        """
        将案例索引到GraphRAG系统
        1. 文本向量化存储（Chroma）
        2. 关系图谱更新（Neo4j/NetworkX简化版）
        """
        # 构建多维度文本表示
        texts = []
        metadatas = []
        ids = []
        
        # 维度1：案例整体摘要（用于语义搜索）
        summary = f"{case.case_name}。{case.founder.core_tech}。寻找{case.partner_type.value}。关键缺口：{case.founder.main_weakness}。"
        texts.append(summary)
        metadatas.append({
            "case_id": case.case_id,
            "view_type": "summary",
            "industry": case.industry.value,
            "outcome": case.outcome.value
        })
        ids.append(f"{case.case_id}_summary")
        
        # 维度2：决策关键点（用于经验检索）
        if case.selected:
            decision_text = f"选择理由：{case.selected.why_selected}。放弃原因：{case.selected.why_others_rejected}。"
            texts.append(decision_text)
            metadatas.append({
                "case_id": case.case_id,
                "view_type": "decision_logic",
                "key_factors": ",".join(case.outcome_record.key_success_factors[:3]) if case.outcome_record else ""
            })
            ids.append(f"{case.case_id}_decision")
        
        # 维度3：复盘教训（用于知识生成）
        if case.outcome_record and case.outcome_record.lessons_learned:
            for i, lesson in enumerate(case.outcome_record.lessons_learned[:3]):
                texts.append(f"经验教训：{lesson}。行业：{case.industry.value}，阶段：{case.stage.value}。")
                metadatas.append({
                    "case_id": case.case_id,
                    "view_type": f"lesson_{i}",
                    "applicable_industry": case.industry.value
                })
                ids.append(f"{case.case_id}_lesson_{i}")
        
        # 生成嵌入向量
        embeddings = self.embedder.encode(texts, normalize_embeddings=True).tolist()
        
        # 批量添加到Chroma（自动持久化）
        self.case_collection.add(
            embeddings=embeddings,
            documents=texts,
            metadatas=metadatas,
            ids=ids
        )
        
        # 更新关系图谱（跨案例关联）
        self._update_relation_graph(case)
    
    def retrieve(self, query: str, n_results: int = 5, 
                 filter_conditions: Optional[Dict] = None) -> List[Dict]:
        """
        GraphRAG检索：语义匹配 + 图关系扩展
        """
        # 步骤1：向量检索（语义匹配）
        query_embedding = self.embedder.encode([query], normalize_embeddings=True).tolist()
        
        where_clause = {}
        if filter_conditions:
            where_clause = filter_conditions
        
        results = self.case_collection.query(
            query_embeddings=query_embedding,
            n_results=n_results * 2,  # 先召回更多，后续用图关系筛选
            where=where_clause if where_clause else None,
            include=["metadatas", "documents", "distances"]
        )
        
        # 步骤2：图关系增强（找到关联案例）
        enhanced_results = []
        seen_cases = set()
        
        for i, case_id in enumerate(results['ids'][0]):
            base_metadata = results['metadatas'][0][i]
            base_case_id = base_metadata['case_id']
            
            if base_case_id in seen_cases:
                continue
            seen_cases.add(base_case_id)
            
            # 查找图谱中的关联案例（多跳推理）
            related_cases = self._get_graph_neighbors(base_case_id, hops=1)
            
            enhanced_results.append({
                "case_id": base_case_id,
                "content": results['documents'][0][i],
                "similarity_score": 1 - results['distances'][0][i],  # 距离转相似度
                "metadata": base_metadata,
                "graph_relations": related_cases,  # 图谱关联
                "reasoning_chain": self._build_reasoning_chain(base_case_id, related_cases)
            })
            
            if len(enhanced_results) >= n_results:
                break
        
        return enhanced_results
    
    def _update_relation_graph(self, case: 'Case'):
        """更新案例关系图谱（简化版，使用NetworkX）"""
        import networkx as nx
        
        G = self.relation_graph
        
        # 添加案例节点
        G.add_node(case.case_id, 
                   industry=case.industry.value,
                   stage=case.stage.value,
                   outcome=case.outcome.value)
        
        # 自动发现关系（基于属性相似度）
        for existing_node in G.nodes():
            if existing_node == case.case_id:
                continue
            
            node_data = G.nodes[existing_node]
            
            # 关系1：同行业
            if node_data.get('industry') == case.industry.value:
                G.add_edge(case.case_id, existing_node, 
                          relation="same_industry", weight=0.3)
            
            # 关系2：相似结果模式（成功/失败）
            if node_data.get('outcome') == case.outcome.value:
                G.add_edge(case.case_id, existing_node,
                          relation="similar_outcome", weight=0.2)
    
    def _get_graph_neighbors(self, case_id: str, hops: int = 1) -> List[Dict]:
        """获取图谱中的邻居节点（关系推理）"""
        import networkx as nx
        
        G = self.relation_graph
        if case_id not in G:
            return []
        
        # 使用BFS获取多跳邻居
        neighbors = []
        for neighbor in nx.single_source_shortest_path_length(G, case_id, cutoff=hops):
            if neighbor != case_id:
                edge_data = G.get_edge_data(case_id, neighbor)
                if edge_data:
                    neighbors.append({
                        "case_id": neighbor,
                        "relation": edge_data.get("relation"),
                        "weight": edge_data.get("weight", 0)
                    })
        
        return neighbors
    
    def _build_reasoning_chain(self, base_case: str, related_cases: List[Dict]) -> str:
        """构建推理链条（解释为什么推荐这些案例）"""
        if not related_cases:
            return "基于语义相似度直接匹配"
        
        chains = []
        for rel in related_cases[:2]:  # 取前2个关系
            chains.append(f"{base_case} --[{rel['relation']}]--> {rel['case_id']}")
        
        return " → ".join(chains)
    
    def _load_relation_graph(self):
        """加载关系图谱（本地JSON文件）"""
        import networkx as nx
        from pathlib import Path
        
        graph_path = Path("./data/case_relation_graph.json")
        G = nx.Graph()
        
        if graph_path.exists():
            with open(graph_path, "r", encoding="utf-8") as f:
                data = json.load(f)
                G = nx.node_link_graph(data)
        
        return G
    
    def save_relation_graph(self):
        """持久化关系图谱"""
        import networkx as nx
        from pathlib import Path
        
        graph_path = Path("./data/case_relation_graph.json")
        data = nx.node_link_data(self.relation_graph)
        
        with open(graph_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

@click.command()
@click.argument('query')
@click.option('--industry', '-ind', help='行业筛选')
@click.option('--top', '-n', default=5, help='返回结果数量')
def rag_search(query: str, industry: Optional[str], top: int):
    """GraphRAG语义检索（增强版搜索）"""
    retriever = GraphRAGRetriever()
    
    filters = {}
    if industry:
        filters["industry"] = industry
    
    results = retriever.retrieve(query, n_results=top, filter_conditions=filters)
    
    click.echo(f"🔍 GraphRAG检索: '{query}'")
    click.echo("=" * 60)
    
    for i, r in enumerate(results, 1):
        click.echo(f"\n{i}. 案例 [{r['case_id']}] 相似度: {r['similarity_score']:.2f}")
        click.echo(f"   内容: {r['content'][:100]}...")
        
        if r['graph_relations']:
            click.echo(f"   图谱关联: {len(r['graph_relations'])} 个相关案例")
            for rel in r['graph_relations'][:2]:
                click.echo(f"     - {rel['relation']}: {rel['case_id']} (权重{rel['weight']})")

2.1 结构因果模型（SCM）在合伙人匹配中的应用

引入因果推断，解决当前方案只能回答”什么”（What）但不能回答”为什么”（Why）和”如果…会怎样”（What if）的问题。

核心问题：
现有方案能告诉你”华为背景的候选人在AI芯片项目成功率85%“，但无法告诉你”因为华为背景带来了渠道资源，所以成功”，也无法回答”如果选择清华背景而非华为背景，结果会怎样”。

# case_repository/causal/engine.py
from dataclasses import dataclass
from typing import Dict, List, Optional, Tuple
import numpy as np
from collections import defaultdict

class CausalInferenceEngine:
    """
    因果推理引擎（基于结构因果模型SCM简化实现）
    
    核心能力：
    1. 因果效应估计：X（候选人特征）对Y（匹配结果）的因果影响
    2. 反事实推理：如果当初选择不同，结果会怎样
    3. 混杂因素控制：消除虚假相关性（如：名校背景vs实际能力）
    """
    
    def __init__(self, case_repository):
        self.repo = case_repository
        self.causal_graph = self._build_causal_graph()
    
    def _build_causal_graph(self) -> Dict[str, List[str]]:
        """
        构建合伙人匹配的因果图（DAG有向无环图）
        
        节点定义：
        - T: 处理变量（候选人特征，如华为背景）
        - Y: 结果变量（匹配成功）
        - X: 混杂变量（行业阶段、融资额等同时影响T和Y的因素）
        - M: 中介变量（渠道资源，T通过M影响Y）
        """
        return {
            "candidate_background": ["channel_resources", "technical_depth", "cultural_fit"],
            "channel_resources": ["match_success"],
            "technical_depth": ["product_quality"],
            "cultural_fit": ["team_stability"],
            "industry_stage": ["candidate_background", "match_success"],  # 混杂变量
            "funding_status": ["candidate_background", "match_success"], # 混杂变量
            "product_quality": ["match_success"],
            "team_stability": ["match_success"]
        }
    
    def estimate_causal_effect(self, treatment: str, outcome: str, 
                               adjustment_set: List[str] = None) -> Dict:
        """
        估计因果效应（ATE - Average Treatment Effect）
        
        例如：估计"华为背景"对"匹配成功"的平均因果效应
        控制混杂变量：行业阶段、融资额
        """
        cases = self.repo.get_all()
        
        # 分组：有T vs 无T
        with_treatment = [c for c in cases if treatment in str(c.selected.background_summary)]
        without_treatment = [c for c in cases if treatment not in str(c.selected.background_summary)]
        
        if not with_treatment or not without_treatment:
            return {"error": "样本不足，无法估计因果效应"}
        
        # 计算倾向得分（Propensity Score）
        # 模拟：在相同混杂变量条件下比较
        adjusted_with = self._adjust_for_confounders(with_treatment, adjustment_set)
        adjusted_without = self._adjust_for_confounders(without_treatment, adjustment_set)
        
        # 计算ATE
        success_with = len([c for c in adjusted_with if c.outcome.value == "成功"]) / len(adjusted_with)
        success_without = len([c for c in adjusted_without if c.outcome.value == "成功"]) / len(adjusted_without)
        
        ate = success_with - success_without
        
        return {
            "treatment": treatment,
            "outcome": outcome,
            "ate": ate,  # 平均处理效应
            "interpretation": f"拥有{treatment}的候选人平均成功率提升{ate*100:.1f}%",
            "with_treatment_success_rate": success_with,
            "without_treatment_success_rate": success_without,
            "sample_size": len(adjusted_with) + len(adjusted_without)
        }
    
    def counterfactual_analysis(self, case_id: str, 
                               alternative_choice: str) -> Dict:
        """
        反事实分析（Counterfactual Reasoning）
        
        核心问题：如果案例X当初选择了另一个候选人，结果会怎样？
        
        例如：CASE-001如果选择了清华系候选人而非华为系，成功率会如何变化？
        """
        case = self.repo.get_by_id(case_id)
        if not case or not case.selected:
            return {"error": "案例不存在或缺少选择信息"}
        
        # 步骤1：估计当前选择的反事实结果
        actual_outcome = 1 if case.outcome.value == "成功" else 0
        
        # 步骤2：基于相似案例推断替代选择的结果
        similar_cases = self._find_similar_decisions(case, alternative_choice)
        
        if not similar_cases:
            return {"error": "缺乏足够相似案例进行反事实推断"}
        
        # 步骤3：估计替代选择的潜在结果（基于相似案例的平均结果）
        cf_outcomes = []
        for c in similar_cases:
            if c.outcome.value == "成功":
                cf_outcomes.append(1)
            elif c.outcome.value == "失败":
                cf_outcomes.append(0)
            else:
                cf_outcomes.append(0.5)  # 待定按0.5处理
        
        cf_expected_outcome = np.mean(cf_outcomes)
        
        # 步骤4：计算反事实遗憾（Counterfactual Regret）
        regret = cf_expected_outcome - actual_outcome
        
        return {
            "case_id": case_id,
            "actual_choice": case.selected.background_summary[:50],
            "actual_outcome": case.outcome.value,
            "alternative_choice": alternative_choice,
            "cf_expected_success_rate": cf_expected_outcome,
            "cf_sample_size": len(similar_cases),
            "counterfactual_regret": regret,  # >0表示可能选错了，<0表示选择正确
            "insight": self._interpret_counterfactual(regret, alternative_choice)
        }
    
    def _find_similar_decisions(self, case, alternative_trait: str) -> List:
        """寻找做过类似决策（选择某类候选人）的相似案例"""
        all_cases = self.repo.get_all()
        
        similar = []
        for c in all_cases:
            if c.case_id == case.case_id:
                continue
            
            # 行业相似
            if c.industry != case.industry:
                continue
            
            # 选择了目标类型的候选人
            if alternative_trait in str(c.selected.background_summary):
                similar.append(c)
        
        return similar
    
    def _adjust_for_confounders(self, cases, confounders):
        """简单的混杂因素控制（分层法）"""
        if not confounders:
            return cases
        
        # 按混杂变量分层（简化实现）
        strata = defaultdict(list)
        for c in cases:
            key = tuple(getattr(c, conf, "unknown") for conf in confounders)
            strata[key].append(c)
        
        # 返回最大层（简化处理）
        largest_stratum = max(strata.values(), key=len, default=[])
        return largest_stratum
    
    def _interpret_counterfactual(self, regret: float, alternative: str) -> str:
        """解释反事实结果"""
        if regret > 0.2:
            return f"⚠️ 反事实分析显示：选择{alternative}可能获得更好结果（成功率提升{regret*100:.0f}%）。建议复盘当初放弃该选项的原因。"
        elif regret < -0.1:
            return f"✅ 反事实分析验证：当前选择优于{alternative}（成功率领先{-regret*100:.0f}%）。决策正确。"
        else:
            return f"ℹ️ 两种选择预期效果相近，当前选择在合理范围内。"

3.1 动态知识图谱演化（EvolveGCN）

案例库不是静态的，而是随时间演化的知识网络。引入时序图神经网络（EvolveGCN）思想：

# case_repository/evolution/dynamic_graph.py
import networkx as nx
from datetime import datetime, timedelta
from typing import List, Dict
from collections import defaultdict

class DynamicKnowledgeGraph:
    """
    动态知识图谱演化系统（基于EvolveGCN思想简化实现）
    
    核心能力：
    1. 时序建模：案例加入的时间顺序影响知识价值（越早的案例权重越低，需更新）
    2. 概念漂移：发现"成功模式"随时间的演变（如：2018年互联网背景受欢迎，2024年硬科技背景更优）
    3. 知识衰减：过时案例自动降低权重（技术迭代导致历史经验失效）
    """
    
    def __init__(self):
        self.snapshots = {}  # 时间点 -> 图谱快照
        self.current_time = datetime.now()
        self.decay_factor = 0.95  # 月度衰减系数
    
    def add_case_snapshot(self, case: 'Case', timestamp: datetime = None):
        """添加案例到动态图谱（时间切片）"""
        if timestamp is None:
            timestamp = datetime.now()
        
        # 按月聚合
        month_key = timestamp.strftime("%Y-%m")
        
        if month_key not in self.snapshots:
            self.snapshots[month_key] = nx.DiGraph()
        
        G = self.snapshots[month_key]
        
        # 添加节点（带时间权重）
        age_months = (self.current_time - timestamp).days / 30
        time_weight = self.decay_factor ** age_months  # 指数衰减
        
        G.add_node(case.case_id,
                   industry=case.industry.value,
                   outcome=case.outcome.value,
                   weight=time_weight,
                   timestamp=timestamp.isoformat())
        
        # 动态边：连接到同期相似案例（演化模式）
        self._add_temporal_edges(G, case)
    
    def _add_temporal_edges(self, G: nx.DiGraph, new_case: 'Case'):
        """添加时序边（发现演化模式）"""
        # 连接到同行业的早期案例（继承关系）
        for node in G.nodes():
            if node == new_case.case_id:
                continue
            
            node_data = G.nodes[node]
            
            # 同行业且时间早（知识传承）
            if node_data.get('industry') == new_case.industry.value:
                G.add_edge(node, new_case.case_id, 
                          relation="knowledge_inheritance",
                          weight=node_data.get('weight', 1.0))
            
            # 相似结果（成功/失败模式的延续）
            if node_data.get('outcome') == new_case.outcome.value:
                G.add_edge(node, new_case.case_id,
                          relation="pattern_continuation",
                          weight=0.5)
    
    def detect_concept_drift(self) -> List[Dict]:
        """
        检测概念漂移（成功模式的演变）
        
        例如：发现2024年后"华为背景"的成功率显著高于2020年
        """
        if len(self.snapshots) < 2:
            return []
        
        drifts = []
        sorted_months = sorted(self.snapshots.keys())
        
        # 分析各时期"成功模式"的变化
        for i in range(1, len(sorted_months)):
            prev_month = sorted_months[i-1]
            curr_month = sorted_months[i]
            
            prev_patterns = self._extract_success_patterns(prev_month)
            curr_patterns = self._extract_success_patterns(curr_month)
            
            # 发现显著变化
            for pattern, curr_rate in curr_patterns.items():
                prev_rate = prev_patterns.get(pattern, 0)
                change = curr_rate - prev_rate
                
                if abs(change) > 0.2:  # 变化超过20%
                    drifts.append({
                        "time_period": f"{prev_month} → {curr_month}",
                        "pattern": pattern,
                        "previous_rate": prev_rate,
                        "current_rate": curr_rate,
                        "change": change,
                        "type": "improvement" if change > 0 else "decline"
                    })
        
        return drifts
    
    def _extract_success_patterns(self, month_key: str) -> Dict[str, float]:
        """提取某时期的成功模式"""
        G = self.snapshots.get(month_key, nx.DiGraph())
        
        success_cases = [n for n in G.nodes() if G.nodes[n].get('outcome') == "成功"]
        total_cases = len(list(G.nodes()))
        
        if total_cases == 0:
            return {}
        
        patterns = defaultdict(lambda: {"success": 0, "total": 0})
        
        for case_id in G.nodes():
            node = G.nodes[case_id]
            # 简化的模式提取：基于行业标签
            pattern = f"{node.get('industry')}_background"
            patterns[pattern]["total"] += 1
            if node.get('outcome') == "成功":
                patterns[pattern]["success"] += 1
        
        return {
            p: data["success"] / data["total"] 
            for p, data in patterns.items() if data["total"] > 0
        }
    
    def predict_future_trend(self, pattern: str, months_ahead: int = 6) -> Dict:
        """
        基于时序演化预测未来趋势（简化版时间序列预测）
        """
        # 提取该模式的历史成功率
        historical_rates = []
        for month in sorted(self.snapshots.keys()):
            patterns = self._extract_success_patterns(month)
            rate = patterns.get(pattern, 0)
            historical_rates.append((month, rate))
        
        if len(historical_rates) < 3:
            return {"error": "历史数据不足，无法预测"}
        
        # 简单线性趋势（实际可用更复杂的时序模型）
        x = list(range(len(historical_rates)))
        y = [r for _, r in historical_rates]
        
        # 计算趋势斜率
        slope = (y[-1] - y[0]) / len(y) if len(y) > 1 else 0
        
        future_rate = y[-1] + slope * months_ahead
        
        return {
            "pattern": pattern,
            "current_rate": y[-1],
            "predicted_rate_future": min(max(future_rate, 0), 1),  # 限制在0-1
            "trend": "upward" if slope > 0.01 else "downward" if slope < -0.01 else "stable",
            "confidence": "low" if len(historical_rates) < 6 else "medium"
        }

# case_repository/agents/multi_agent.py
from typing import List, Dict, Callable
import concurrent.futures

class CaseLibraryAgent:
    """
    案例库多智能体系统（Multi-Agent System）
    
    角色分工：
    1. 录入助手Agent：引导案例创建，自动提取关键信息
    2. 相似度分析师Agent：专门负责案例匹配与推荐
    3. 复盘生成Agent：自动生成高质量复盘报告
    4. 知识蒸馏Agent：从成功案例中提取可复用模板
    5. 质量审计Agent：检查案例完整性与一致性
    """
    
    def __init__(self):
        self.agents = {
            "intake": IntakeAgent(),
            "analyst": SimilarityAnalystAgent(),
            "reporter": ReportGeneratorAgent(),
            "distiller": KnowledgeDistillerAgent(),
            "auditor": QualityAuditorAgent()
        }
    
    def collaborative_analysis(self, new_case: 'Case') -> Dict:
        """
        多智能体协作分析新案例
        """
        results = {}
        
        # 并行执行（利用Kimi Claw的多线程能力）
        with concurrent.futures.ThreadPoolExecutor() as executor:
            # Agent 1: 质量审计（快速检查）
            future_audit = executor.submit(self.agents["auditor"].audit, new_case)
            
            # Agent 2: 相似度分析（检索历史案例）
            future_similar = executor.submit(self.agents["analyst"].find_similar, new_case)
            
            # Agent 3: 知识蒸馏（提取模式）
            future_patterns = executor.submit(self.agents["distiller"].extract_patterns, new_case)
            
            # 收集结果
            results["audit"] = future_audit.result()
            results["similar_cases"] = future_similar.result()
            results["patterns"] = future_patterns.result()
        
        # Agent 4: 复盘生成（基于前面结果综合生成）
        results["report"] = self.agents["reporter"].generate(
            new_case, 
            similar_cases=results["similar_cases"],
            patterns=results["patterns"]
        )
        
        return results

        }
        
        # 保存到模板库（可作为未来案例的推荐模板）
        self._save_pattern(pattern)
        
        return pattern

class CognitiveAssistant:
    """
    认知增强助手（人机协同）
    
    功能：
    1. 自然语言查询："为什么华为背景在芯片行业更受欢迎？"
    2. 自动举证：从案例库检索相关证据
    3. 假设检验：验证教练的直觉假设
    4. 决策建议：基于历史数据给出推荐
    """
    
    def __init__(self):
        self.retriever = GraphRAGRetriever()  # 复用之前的检索器
        self.causal_engine = CausalInferenceEngine(CaseRepository())
    
    def answer(self, question: str) -> Dict:
        """
        智能问答（RAG增强）
        
        流程：
        1. 理解问题意图（分类：因果/反事实/相似案例/统计）
        2. 检索相关案例（GraphRAG）
        3. 因果分析（如需要）
        4. 生成结构化回答
        """
        # 步骤1：意图识别（简化版关键词匹配，实际可用LLM）
        intent = self._classify_intent(question)
        
        # 步骤2：证据检索
        evidence = self.retriever.retrieve(question, n_results=5)
        
        # 步骤3：增强分析
        if intent == "causal":
            # 提取变量进行因果分析
            treatment = self._extract_entity(question, "background")
            if treatment:
                analysis = self.causal_engine.estimate_causal_effect(treatment, "match_success")
            else:
                analysis = {}
        elif intent == "counterfactual":
            # 反事实分析
            analysis = {"note": "请使用 what-if 命令进行详细反事实分析"}
        else:
            analysis = {}
        
        # 步骤4：生成回答
        answer = self._synthesize_answer(question, evidence, analysis)
        
        return {
            "question": question,
            "intent": intent,
            "evidence_cases": [e["case_id"] for e in evidence],
            "analysis": analysis,
            "answer": answer,
            "confidence": len(evidence) / 10  # 简化的置信度
        }
    
    def _classify_intent(self, question: str) -> str:
        """简单意图分类"""
        if "为什么" in question or "因果" in question:
            return "causal"
        elif "如果" in question or "会怎样" in question:
            return "counterfactual"
        elif "类似" in question or "相似" in question:
            return "similarity"
        else:
            return "general"
    
    def _extract_entity(self, question: str, entity_type: str) -> Optional[str]:
        """从问题中提取实体（简化版）"""
        # 实际可用NER模型
        backgrounds = ["华为", "清华", "阿里", "腾讯", "百度", "英特尔"]
        for bg in backgrounds:
            if bg in question:
                return bg
        return None
    
    def _synthesize_answer(self, question: str, evidence: List[Dict], analysis: Dict) -> str:
        """综合生成回答（简化版模板）"""
        parts = []
        parts.append(f"基于{len(evidence)}个相关案例的分析：")
        
        # 证据总结
        for i, evi in enumerate(evidence[:3], 1):
            parts.append(f"{i}. 案例{evi['case_id']}: {evi['content'][:60]}...")
        
        # 分析结果
        if "ate" in analysis:
            parts.append(f"\n因果分析：该特质平均提升成功率{analysis['ate']*100:.1f}%")
        
        return "\n".join(parts)

# case_repository/agents/workflow_engine.py
from typing import Dict, List, Callable, Any
from enum import Enum
import concurrent.futures
from dataclasses import dataclass
from datetime import datetime
import json

class AgentRole(Enum):
    """智能体角色定义（基于吴恩达4大模式）"""
    REFLECTION = "reflection"      # 反思模式：自我审计与修正
    TOOL_USE = "tool_use"        # 工具使用：调用外部API/数据库
    PLANNING = "planning"        # 规划模式：任务分解与编排
    MULTI_AGENT = "multi_agent"  # 多智能体协作：多角色协同

@dataclass
class Task:
    """工作流任务单元"""
    task_id: str
    agent_role: AgentRole
    action: Callable
    input_data: Dict
    dependencies: List[str] = None  # 依赖的其他任务ID
    status: str = "pending"  # pending/running/completed/failed
    output: Any = None

class AgenticWorkflowEngine:
    """
    智能体工作流引擎（Agentic Workflow Engine）
    
    实现能力：
    1. 任务自动分解（Planning模式）
    2. 多智能体并行执行（Multi-Agent模式）
    3. 自我反思与修正（Reflection模式）
    4. 工具调用集成（Tool Use模式）
    """
    
    def __init__(self):
        self.tasks: Dict[str, Task] = {}
        self.execution_log = []
        self.max_iterations = 3  # 反思模式的最大迭代次数
    
    def orchestrate(self, workflow_name: str, initial_input: Dict) -> Dict:
        """
        编排完整工作流（以"新案例入库"为例）
        
        工作流步骤：
        1. Planning: 分解入库任务
        2. Tool Use: 调用外部数据源验证
        3. Multi-Agent: 并行执行：相似度分析+知识抽取+质量审计
        4. Reflection: 自我审计，如有问题返回修正
        5. Execution: 最终写入
        """
        if workflow_name == "case_intake":
            return self._case_intake_workflow(initial_input)
        elif workflow_name == "auto_research":
            return self._auto_research_workflow(initial_input)
        elif workflow_name == "knowledge_evolution":
            return self._knowledge_evolution_workflow(initial_input)
        else:
            raise ValueError(f"未知工作流: {workflow_name}")
    
    def _case_intake_workflow(self, case_data: Dict) -> Dict:
        """
        案例入库完整工作流（固化最佳实践）
        
        自动化流程：
        1. 信息完整性检查（Reflection）
        2. 外部数据验证（Tool Use）- 查工商/融资信息
        3. 并行分析（Multi-Agent）：
           - 相似案例检索
           - 成功模式匹配
           - 风险评估
        4. 质量审计（Reflection）
        5. 知识图谱更新（Tool Use）
        6. 生成复盘建议（Planning）
        """
        results = {}
        
        # Stage 1: 规划与分解（Planning模式）
        plan = self._plan_case_intake(case_data)
        
        # Stage 2: 信息验证（Tool Use模式）- 调用外部API
        validated_data = self._execute_with_agent(
            AgentRole.TOOL_USE,
            self._validate_external_data,
            case_data
        )
        
        # Stage 3: 并行分析（Multi-Agent模式）
        with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:
            # Agent 3.1: 相似度分析师
            future_similar = executor.submit(
                self._execute_with_agent,
                AgentRole.TOOL_USE,
                self._analyze_similarity,
                validated_data
            )
            
            # Agent 3.2: 模式提取师
            future_pattern = executor.submit(
                self._execute_with_agent,
                AgentRole.TOOL_USE,
                self._extract_success_pattern,
                validated_data
            )
            
            # Agent 3.3: 风险审计师
            future_risk = executor.submit(
                self._execute_with_agent,
                AgentRole.REFLECTION,
                self._audit_risks,
                validated_data
            )
            
            # Agent 3.4: 知识图谱构建师
            future_kg = executor.submit(
                self._execute_with_agent,
                AgentRole.TOOL_USE,
                self._build_knowledge_graph,
                validated_data
            )
            
            # 收集并行结果
            results["similarity"] = future_similar.result()
            results["pattern"] = future_pattern.result()
            results["risk_audit"] = future_risk.result()
            results["knowledge_graph"] = future_kg.result()
        
        # Stage 4: 反思与修正（Reflection模式）
        audit_result = results["risk_audit"]
        if audit_result["issues"]:
            # 触发修正流程
            corrected_data = self._reflection_loop(
                validated_data, 
                audit_result["issues"],
                max_iterations=self.max_iterations
            )
            validated_data = corrected_data
        
        # Stage 5: 最终执行（写入数据库）
        final_result = self._execute_with_agent(
            AgentRole.TOOL_USE,
            self._persist_to_database,
            {
                "case_data": validated_data,
                "analysis_results": results
            }
        )
        
        # Stage 6: 生成入库报告（包含所有分析结果）
        report = self._generate_intake_report(case_data, results)
        
        return {
            "workflow": "case_intake",
            "status": "completed",
            "case_id": final_result.get("case_id"),
            "analysis": results,
            "audit_passed": len(audit_result["issues"]) == 0,
            "report": report
        }
    
    def _plan_case_intake(self, case_data: Dict) -> List[Task]:
        """
        Planning模式：将入库任务分解为可执行子任务
        """
        tasks = []
        
        # 任务1: 数据验证
        tasks.append(Task(
            task_id="validate",
            agent_role=AgentRole.TOOL_USE,
            action=self._validate_external_data,
            input_data=case_data,
            dependencies=[]
        ))
        
        # 任务2: 相似度分析（依赖任务1）
        tasks.append(Task(
            task_id="similarity",
            agent_role=AgentRole.TOOL_USE,
            action=self._analyze_similarity,
            input_data={},
            dependencies=["validate"]
        ))
        
        # 任务3: 质量审计（依赖任务1）
        tasks.append(Task(
            task_id="audit",
            agent_role=AgentRole.REFLECTION,
            action=self._audit_risks,
            input_data={},
            dependencies=["validate"]
        ))
        
        return tasks
    
    def _reflection_loop(self, data: Dict, issues: List[str], max_iterations: int) -> Dict:
        """
        Reflection模式：自我反思与修正
        
        模拟人类"初稿-审核-修改"的迭代过程
        """
        current_data = data.copy()
        
        for i in range(max_iterations):
            # 反思：识别问题
            reflection_result = self._reflect_on_issues(current_data, issues)
            
            if reflection_result["corrected"]:
                current_data = reflection_result["corrected_data"]
                issues = reflection_result["remaining_issues"]
                
                if not issues:
                    break  # 所有问题已修正
            else:
                break  # 无法自动修正，需要人工介入
        
        return current_data
    
    def _execute_with_agent(self, role: AgentRole, action: Callable, input_data: Dict) -> Any:
        """执行单个智能体任务"""
        try:
            result = action(input_data)
            self.execution_log.append({
                "timestamp": datetime.now().isoformat(),
                "role": role.value,
                "action": action.__name__,
                "status": "success"
            })
            return result
        except Exception as e:
            self.execution_log.append({
                "timestamp": datetime.now().isoformat(),
                "role": role.value,
                "action": action.__name__,
                "status": "failed",
                "error": str(e)
            })
            raise
    
    # 具体动作实现（简化示例）
    def _validate_external_data(self, data: Dict) -> Dict:
        """Tool Use: 调用外部API验证数据真实性"""
        # 实际实现：调用天眼查/企查查API验证融资信息
        # 调用专利局API验证技术描述
        return {**data, "validated": True, "external_confirmed": ["funding", "team_size"]}
    
    def _analyze_similarity(self, data: Dict) -> Dict:
        """Tool Use: 相似度分析"""
        from case_repository.similarity.engine import CaseSimilarityEngine
        from case_repository.database.repository import CaseRepository
        
        engine = CaseSimilarityEngine()
        repo = CaseRepository()
        
        # 构建临时案例对象进行比较

        query_case = Case(

        }
    
    def _extract_success_pattern(self, data: Dict) -> Dict:
        """Tool Use: 从成功案例中提取模式"""
        if data.get("outcome") != "成功":
            return {}
        
        # 提取可复用模式
        pattern = {
            "context": {
                "industry": data.get("industry"),
                "stage": data.get("stage"),
                "partner_type": data.get("partner_type")
            },
            "key_success_factors": data.get("outcome_record", {}).get("key_success_factors", []),
            "threshold_setting": data.get("process", {}).get("satisfaction_threshold"),
            "decision_speed": data.get("process", {}).get("time_spent_days")
        }
        
        # 保存到模式库
        self._save_to_pattern_library(pattern)
        return {"extracted_pattern": pattern, "reusability_score": 0.85}
    
    def _audit_risks(self, data: Dict) -> Dict:
        """Reflection: 审计数据质量和风险"""
        issues = []
        
        # 检查1: 关键字段缺失
        required_fields = ["industry", "stage", "founder.main_weakness", "outcome"]
        for field in required_fields:
            if not self._get_nested_value(data, field):
                issues.append(f"关键字段缺失: {field}")
        
        # 检查2: 决策逻辑外显化不足
        if data.get("outcome") == "成功" and not data.get("selected", {}).get("why_selected"):
            issues.append("成功案例缺少选择理由（外显化不足）")
        
        # 检查3: 复盘深度检查
        lessons = data.get("outcome_record", {}).get("lessons_learned", [])
        if len(lessons) < 2:
            issues.append("复盘教训过少（至少2条）")
        
        # 检查4: 数据一致性
        if data.get("process", {}).get("time_spent_days", 0) > 90:
            issues.append("匹配周期过长（超过90天），建议检查是否符合满意解原则")
        
        return {
            "issues": issues,
            "severity": "high" if len(issues) > 2 else "medium" if issues else "low",
            "recommendations": self._generate_audit_recommendations(issues)
        }
    
    def _build_knowledge_graph(self, data: Dict) -> Dict:

        from case_repository.kg.builder import KnowledgeGraphBuilder
        
        builder = KnowledgeGraphBuilder()

        # 添加节点和关系

        # 自动发现关系
        relations = builder.discover_relations(case_id)
        
        return {
            "nodes_added": 1,
            "relations_discovered": len(relations),
            "relation_types": list(set(r["type"] for r in relations))
        }
    
    def _persist_to_database(self, context: Dict) -> Dict:

        case = Case.from_dict(data)

    def _generate_intake_report(self, case_data: Dict, results: Dict) -> str:

---

    # 辅助方法
    def _get_nested_value(self, data: Dict, path: str):
        """获取嵌套字典值"""
        keys = path.split(".")
        current = data
        for key in keys:
            if isinstance(current, dict) and key in current:
                current = current[key]
            else:
                return None
        return current
    
    def _generate_similarity_insights(self, similar_cases):
        """生成相似案例洞察"""
        if not similar_cases:
            return "无相似历史案例，建议谨慎决策"
        
        success_rate = sum(1 for c, _, _ in similar_cases if c.outcome.value == "成功") / len(similar_cases)
        return f"历史相似案例成功率: {success_rate*100:.0f}%"
    
    def _save_to_pattern_library(self, pattern: Dict):
        """保存成功模式到模板库"""
        import json
        from pathlib import Path
        
        path = Path("./data/success_patterns.jsonl")
        with open(path, "a", encoding="utf-8") as f:
            f.write(json.dumps(pattern, ensure_ascii=False) + "\n")
    
    def _generate_audit_recommendations(self, issues: List[str]) -> List[str]:
        """生成审计建议"""
        recommendations = []
        for issue in issues:
            if "关键字段缺失" in issue:
                recommendations.append("补充缺失字段，确保SECI外显化完整")
            elif "外显化不足" in issue:
                recommendations.append("强化决策逻辑记录，避免隐性知识流失")
            elif "复盘教训过少" in issue:
                recommendations.append("深化复盘，提炼至少2条可复用教训")
            elif "匹配周期过长" in issue:
                recommendations.append("复盘是否过度优化，是否符合满意解原则")
        return recommendations

2.1 MCP：案例库的”USB-C接口”

MCP（Model Context Protocol）是AI领域的”USB-C接口”，让案例库能够无缝连接： - 外部知识源：天眼查、企查查、专利数据库、招聘平台 - 协作工具：飞书、钉钉、Notion、Confluence - 分析工具：BI系统、数据可视化平台 - 其他AI系统：合伙人匹配引擎、风险评估系统

# case_repository/mcp/server.py
"""
案例库MCP Server实现
将案例库能力封装为标准MCP服务，供其他AI应用调用
"""
from mcp.server import Server  # MCP官方SDK
from mcp.types import TextContent, Tool, Resource
import json
from typing import Sequence

from case_repository.database.repository import CaseRepository
from case_repository.similarity.engine import CaseSimilarityEngine

class CaseRepositoryMCPServer:
    """
    案例库MCP服务器（符合MCP v1.0协议）
    
    暴露的能力：
    1. 案例检索（语义+结构化）
    2. 相似案例推荐
    3. 行业基准查询
    4. 复盘报告生成
    """
    
    def __init__(self):
        self.server = Server("case-repository-skill")
        self.repo = CaseRepository()
        self.similarity_engine = CaseSimilarityEngine()
        
        # 注册MCP工具（Tools）
        self._register_tools()
        # 注册MCP资源（Resources）
        self._register_resources()
    
    def _register_tools(self):
        """注册MCP工具（供AI模型调用）"""
        
        @self.server.tool()
        async def search_similar_cases(
            founder_background: str,
            industry: str,
            stage: str,
            partner_type: str,
            top_k: int = 5
        ) -> Sequence[TextContent]:
            """
            检索相似案例（核心MCP工具）
            
            当AI助手需要为创始人匹配合伙人时，调用此工具获取历史参考案例。
            """
            from case_repository.models.case import Case, FounderProfile, Industry, Stage, PartnerType
            
            # 构建查询
            query_case = Case(
                case_id="MCP-QUERY",
                case_name="MCP查询",
                industry=Industry(industry),
                stage=Stage(stage),
                partner_type=PartnerType(partner_type),
                founder=FounderProfile(
                    background=founder_background,
                    main_weakness="",  # 可由AI填充
                    core_tech=""
                )
            )
            
            # 执行检索
            candidates = self.repo.get_all()
            results = self.similarity_engine.find_similar_cases(query_case, candidates, top_k=top_k)
            
            # 格式化返回（自然语言友好）
            formatted = []
            for i, (case, score, breakdown) in enumerate(results, 1):
                content = f"""
案例{i}: [{case.case_id}] {case.case_name}
相似度: {score:.2f}
行业/阶段: {case.industry.value} / {case.stage.value}
结果: {case.outcome.value}
关键洞察: {case.outcome_record.key_success_factors[0] if case.outcome_record and case.outcome_record.key_success_factors else 'N/A'}
决策逻辑: {case.selected.why_selected[:100] if case.selected else 'N/A'}...
                """
                formatted.append(content)
            
            return [TextContent(type="text", text="\n---\n".join(formatted))]
        
        @self.server.tool()
        async def get_industry_benchmark(industry: str) -> Sequence[TextContent]:
            """
            获取行业基准数据
            
            返回某行业的合伙人匹配成功率、平均周期、关键成功因素等基准数据。
            """
            cases = self.repo.get_by_industry(industry)
            
            if not cases:
                return [TextContent(type="text", text=f"暂无{industry}行业数据")]
            
            total = len(cases)
            success = len([c for c in cases if c.outcome.value == "成功"])
            avg_time = sum(c.process.time_spent_days for c in cases if c.process) / \
                      len([c for c in cases if c.process]) if cases else 0
            
            report = f"""
## {industry}行业合伙人匹配基准报告

统计样本: {total}个案例
成功率: {success}/{total} ({success/total*100:.1f}%)
平均匹配周期: {avg_time:.1f}天

关键成功因素（Top 3）:
{self._extract_top_factors(cases)}

常见失败原因:
{self._extract_failure_reasons(cases)}

建议满意解阈值: 0.75（基于历史数据）
            """
            
            return [TextContent(type="text", text=report)]
        
        @self.server.tool()
        async def generate_lessons_report(case_id: str) -> Sequence[TextContent]:
            """
            生成案例复盘报告
            
            基于案例数据自动生成结构化复盘报告，包含可复用教训。
            """
            from case_repository.report.generator import ReportGenerator
            
            generator = ReportGenerator()
            try:
                report = generator.generate_report(case_id, output_format="markdown")
                return [TextContent(type="text", text=report)]
            except Exception as e:
                return [TextContent(type="text", text=f"生成失败: {str(e)}")]
    
    def _register_resources(self):
        """注册MCP资源（可订阅的数据源）"""
        
        @self.server.resource("case://{case_id}")
        async def get_case_resource(case_id: str) -> Sequence[TextContent]:
            """
            案例详情资源（URI模式访问）
            
            AI助手可通过 case://CASE-2026-001 直接访问案例详情
            """
            case = self.repo.get_by_id(case_id)
            if not case:
                return [TextContent(type="text", text=f"案例{case_id}不存在")]
            
            content = json.dumps(case.to_dict(), ensure_ascii=False, indent=2)
            return [TextContent(type="text", text=content)]
        
        @self.server.resource("pattern://success/{industry}")
        async def get_success_pattern(industry: str) -> Sequence[TextContent]:
            """成功模式资源"""
            patterns = self._load_success_patterns(industry)
            return [TextContent(type="text", text=json.dumps(patterns, ensure_ascii=False))]
    
    def run(self):
        """启动MCP服务器（stdio模式，兼容大多数AI客户端）"""
        import asyncio
        asyncio.run(self.server.run())
    
    # 辅助方法
    def _extract_top_factors(self, cases):
        from collections import Counter
        all_factors = []
        for c in cases:
            if c.outcome_record and c.outcome_record.key_success_factors:
                all_factors.extend(c.outcome_record.key_success_factors)
        top = Counter(all_factors).most_common(3)
        return "\n".join([f"{i+1}. {f} ({n}次)" for i, (f, n) in enumerate(top)])
    
    def _extract_failure_reasons(self, cases):
        from collections import Counter
        all_reasons = []
        for c in cases:
            if c.outcome_record and c.outcome_record.failure_reasons:
                all_reasons.extend(c.outcome_record.failure_reasons)
        top = Counter(all_reasons).most_common(3)
        return "\n".join([f"- {r}" for r, _ in top]) if top else "暂无足够失败样本"
    
    def _load_success_patterns(self, industry: str):
        import json
        from pathlib import Path
        
        path = Path("./data/success_patterns.jsonl")
        patterns = []
        if path.exists():
            with open(path, 'r', encoding='utf-8') as f:
                for line in f:
                    p = json.loads(line)
                    if p.get("context", {}).get("industry") == industry:
                        patterns.append(p)
        return patterns

# case_repository/learning/continuous_learning.py
"""
持续学习系统（Continuous Learning System）
实现案例库的自动进化
"""
from datetime import datetime, timedelta
from typing import List, Dict
import json
from pathlib import Path

class ContinuousLearningEngine:
    """
    三层学习机制：
    
    1. 微观层：单个案例的反馈学习（Case-Level Learning）
    2. 中观层：成功模式的提取与泛化（Pattern Learning）
    3. 宏观层：知识生态的演化（Ecosystem Learning）
    """
    
    def __init__(self):
        self.feedback_buffer = []  # 反馈缓冲
        self.pattern_library = Path("./data/learned_patterns.json")
        self.learning_log = Path("./data/learning_log.jsonl")
    
    # ========== 微观层：案例级学习 ==========
    
    def record_outcome_feedback(self, case_id: str, actual_outcome: Dict):
        """
        记录实际结果反馈（闭环核心）
        
        案例入库时的预测 vs 实际发生的结果
        """
        # 获取案例入库时的预测
        case = self._get_case(case_id)
        predicted = case.outcome_record.result if case.outcome_record else "未知"
        actual = actual_outcome.get("result")
        
        # 计算预测偏差
        feedback = {
            "case_id": case_id,
            "predicted": predicted,
            "actual": actual,
            "accuracy": predicted == actual,
            "deviation_reason": self._analyze_deviation(case, actual_outcome) if predicted != actual else None,
            "timestamp": datetime.now().isoformat()
        }
        
        self.feedback_buffer.append(feedback)
        
        # 触发即时学习
        self._immediate_learning(feedback)
        
        return feedback
    
    def _analyze_deviation(self, case, actual):
        """分析预测偏差原因"""
        if case.outcome_record and case.selected:
            reasons = []
            
            # 检查1: 候选人稳定性
            if actual.get("partner_left", False):
                reasons.append("候选人入职后离职（稳定性预测失误）")
            
            # 检查2: 融资结果
            if actual.get("funding_result") == "失败":
                reasons.append("后续融资失败（商业能力被高估）")
            
            # 检查3: 团队冲突
            if actual.get("team_conflict"):
                reasons.append("创始人与合伙人冲突（文化适配失误）")
            
            return reasons
        return []
    
    def _immediate_learning(self, feedback: Dict):
        """即时学习：更新案例的置信度和权重"""
        if not feedback["accuracy"]:
            # 预测错误，降低相似案例的推荐权重
            self._adjust_similar_cases_weight(feedback["case_id"], delta=-0.1)
    
    # ========== 中观层：模式学习 ==========
    
    def extract_generalized_patterns(self) -> List[Dict]:
        """
        从成功案例中提取泛化模式（周度/月度批处理）
        
        使用归纳学习从具体案例中提取一般性规则
        """
        from case_repository.database.repository import CaseRepository
        repo = CaseRepository()
        
        success_cases = [c for c in repo.get_all() if c.outcome.value == "成功"]
        
        # 按上下文聚类
        clusters = self._cluster_by_context(success_cases)
        
        patterns = []
        for cluster_id, cases in clusters.items():
            if len(cases) < 3:  # 需要至少3个案例支撑
                continue
            
            # 提取共同特征
            common_factors = self._extract_common_factors(cases)
            threshold_stats = self._analyze_thresholds(cases)
            
            pattern = {
                "pattern_id": f"PATTERN-{cluster_id}",
                "applicable_context": {
                    "industry": cases[0].industry.value,
                    "stage": cases[0].stage.value,
                    "partner_type": cases[0].partner_type.value
                },
                "confidence": len(cases) / 10,  # 基于样本量计算置信度
                "common_success_factors": common_factors,
                "optimal_threshold": threshold_stats["mean"],
                "typical_duration_range": [
                    threshold_stats["min_days"],
                    threshold_stats["max_days"]
                ],
                "supporting_cases": [c.case_id for c in cases],
                "extracted_at": datetime.now().isoformat()
            }
            
            patterns.append(pattern)
            self._save_pattern(pattern)
        
        return patterns
    
    def _cluster_by_context(self, cases):
        """按上下文聚类案例"""
        from collections import defaultdict
        clusters = defaultdict(list)
        
        for case in cases:
            key = f"{case.industry.value}_{case.stage.value}_{case.partner_type.value}"
            clusters[key].append(case)
        
        return clusters
    
    def _extract_common_factors(self, cases):
        """提取共同成功因素（取交集）"""
        from collections import Counter
        all_factors = []
        for c in cases:
            if c.outcome_record and c.outcome_record.key_success_factors:
                all_factors.extend(c.outcome_record.key_success_factors)
        
        # 取出现频率>50%的因素
        counter = Counter(all_factors)
        common = [f for f, count in counter.items() if count >= len(cases) * 0.5]
        return common
    
    def _analyze_thresholds(self, cases):
        """分析满意解阈值设置"""
        thresholds = []
        durations = []
        
        for c in cases:
            if c.process:
                thresholds.append(c.process.satisfaction_threshold)
                durations.append(c.process.time_spent_days)
        
        return {
            "mean": sum(thresholds) / len(thresholds) if thresholds else 0.75,
            "min_days": min(durations) if durations else 30,
            "max_days": max(durations) if durations else 90
        }
    
    # ========== 宏观层：生态演化 ==========
    
    def evolve_knowledge_ecosystem(self):
        """
        宏观层：知识生态演化（月度运行）
        
        检测：
        1. 概念漂移（成功模式随时间变化）
        2. 新领域出现（新兴硬科技行业）
        3. 旧知识过期（技术迭代导致历史经验失效）
        """
        # 1. 检测概念漂移
        drift_report = self._detect_concept_drift()
        
        # 2. 识别新兴模式
        emerging = self._identify_emerging_patterns()
        
        # 3. 清理过期知识
        expired = self._identify_expired_knowledge()
        
        evolution_report = {
            "timestamp": datetime.now().isoformat(),
            "concept_drift": drift_report,
            "emerging_patterns": emerging,
            "expired_knowledge": expired,
            "recommendations": self._generate_evolution_recommendations(drift_report, emerging, expired)
        }
        
        # 保存演化报告
        with open(self.learning_log, "a", encoding="utf-8") as f:
            f.write(json.dumps(evolution_report, ensure_ascii=False) + "\n")
        
        return evolution_report
    
    def _detect_concept_drift(self):
        """检测概念漂移（成功模式是否随时间改变）"""
        # 比较2024 vs 2025年的成功因素变化
        # 例如：2024年看重"大厂背景"，2025年更看重"从0到1经验"
        return {
            "detected": False,  # 简化实现
            "description": "暂无显著概念漂移"
        }
    
    def _identify_emerging_patterns(self):
        """识别新兴成功模式"""
        # 近3个月新增案例中提取的新模式
        recent_cases = self._get_recent_cases(months=3)
        new_patterns = []
        
        for case in recent_cases:
            if case.outcome.value == "成功":
                # 检查是否匹配现有模式
                if not self._matches_existing_pattern(case):
                    new_patterns.append({
                        "case_id": case.case_id,
                        "novelty": "新行业组合或新决策逻辑",
                        "potential_pattern": self._extract_pattern_signature(case)
                    })
        
        return new_patterns
    
    def _identify_expired_knowledge(self):
        """识别过期知识"""
        # 超过2年无更新的旧案例，且近期相似案例成功率低
        old_cases = self._get_old_cases(years=2)
        expired = []
        
        for case in old_cases:
            # 检查近期相似案例表现
            recent_similar = self._get_recent_similar(case, months=6)
            if recent_similar:
                recent_success_rate = sum(1 for c in recent_similar if c.outcome.value == "成功") / len(recent_similar)
                if recent_success_rate < 0.3:  # 近期成功率<30%
                    expired.append({
                        "case_id": case.case_id,
                        "reason": "历史经验在最新案例中失效",
                        "deprecated_factors": case.outcome_record.key_success_factors if case.outcome_record else []
                    })
        
        return expired
    
    def _generate_evolution_recommendations(self, drift, emerging, expired):
        """生成演化建议"""
        recommendations = []
        
        if drift["detected"]:
            recommendations.append("更新成功模式模板，适应新趋势")
        
        if len(emerging) > 2:
            recommendations.append("发现多个新兴模式，建议专家人工审核并固化为模板")
        
        if len(expired) > 5:
            recommendations.append("大量历史知识过期，建议启动知识库清理流程")
        
        return recommendations

# case_repository/intelligence/gathering.py
"""
研究情报自动收集系统（Intelligence Gathering System）
自动监控：
- 行业投融资动态（IT桔子、36氪）
- 技术突破（arXiv、专利局）
- 竞争对手动向（招聘网站、脉脉）
- 政策变化（政府网站）
"""
import asyncio
import aiohttp
from datetime import datetime
from typing import List, Dict
import feedparser

class IntelligenceScout:
    """
    情报侦察Agent（自动信息收集）
    
    集成MCP协议连接多个外部数据源
    """
    
    def __init__(self):
        self.sources = {
            "funding": ["itjuzi", "36kr", "crunchbase"],
            "technology": ["arxiv", "patent_gazette"],
            "talent": ["maimai", "linkedin"],
            "policy": ["gov_cn", "miit"]
        }
        self.collected_intelligence = []
    
    async def gather_daily_intelligence(self) -> List[Dict]:
        """每日情报收集（异步并行）"""
        tasks = []
        
        # 并行收集各渠道情报
        for channel, sources in self.sources.items():
            for source in sources:
                tasks.append(self._gather_from_source(channel, source))
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 过滤和整合
        valid_intelligence = []
        for result in results:
            if isinstance(result, Exception):
                continue
            if result and self._is_relevant(result):
                valid_intelligence.append(result)
        
        # 去重和排序
        unique_intel = self._deduplicate(valid_intelligence)
        sorted_intel = sorted(unique_intel, key=lambda x: x["relevance_score"], reverse=True)
        
        # 保存到情报库
        self._save_intelligence(sorted_intel)
        
        return sorted_intel
    
    async def _gather_from_source(self, channel: str, source: str) -> Dict:
        """从单个源收集情报（使用MCP或API）"""
        # 实际实现需要对应API key和解析逻辑
        # 这里展示架构
        
        if source == "itjuzi":
            return await self._fetch_itjuzi_funding()
        elif source == "arxiv":
            return await self._fetch_arxiv_papers()
        # ... 其他源
        
        return None
    
    async def _fetch_itjuzi_funding(self) -> List[Dict]:
        """获取投融资情报"""
        # 使用MCP连接IT桔子API
        url = "https://www.itjuzi.com/api/investments"
        async with aiohttp.ClientSession() as session:
            async with session.get(url) as response:
                data = await response.json()
                return self._parse_funding_data(data)
    
    def _is_relevant(self, intel: Dict) -> bool:
        """判断情报是否与案例库相关（硬科技+合伙人相关）"""
        keywords = ["芯片", "AI", "生物医药", "新能源", "合伙人", "CTO", "CFO", "融资"]
        text = json.dumps(intel, ensure_ascii=False).lower()
        return any(kw in text for kw in keywords)
    
    def _deduplicate(self, intelligence: List[Dict]) -> List[Dict]:
        """基于内容相似度去重"""
        # 使用SimHash或MinHash算法
        seen = set()
        unique = []
        for intel in intelligence:
            fingerprint = self._compute_fingerprint(intel["title"])
            if fingerprint not in seen:
                seen.add(fingerprint)
                unique.append(intel)
        return unique
    
    def _compute_fingerprint(self, text: str) -> str:
        """计算文本指纹（简化版）"""
        import hashlib
        return hashlib.md5(text[:50].encode()).hexdigest()[:8]
    
    def _save_intelligence(self, intelligence: List[Dict]):
        """保存情报到本地库"""
        path = Path(f"./data/intelligence/{datetime.now().strftime('%Y%m%d')}.jsonl")
        path.parent.mkdir(parents=True, exist_ok=True)
        
        with open(path, "a", encoding="utf-8") as f:
            for intel in intelligence:
                f.write(json.dumps(intel, ensure_ascii=False) + "\n")
    
    async def generate_intelligence_report(self) -> str:
        """
        生成每日情报摘要报告（供教练晨会使用）
        """
        intel = await self.gather_daily_intelligence()
        
        # 分类统计
        categories = {}
        for item in intel:
            cat = item.get("category", "其他")
            categories.setdefault(cat, []).append(item)
        
        report = f"""# 硬科技合伙人匹配情报日报 ({datetime.now().strftime('%Y-%m-%d')})

## 投融资动态（{len(categories.get('funding', []))}条）
{self._format_funding_news(categories.get('funding', [])[:3])}

## 技术突破（{len(categories.get('technology', []))}条）
{self._format_tech_news(categories.get('technology', [])[:3])}

## 人才流动（{len(categories.get('talent', []))}条）
{self._format_talent_news(categories.get('talent', [])[:3])}

## 相关案例提示
以下历史案例可能与今日情报相关：
{await self._link_to_historical_cases(intel)}

---
*本报告由IntelligenceScout自动生成*
"""
        return report

# case_repository/idss/predictive_engine.py
"""
智能决策支持系统（IDSS）预测引擎
基于生存分析和集成学习实现动态预测
"""
import numpy as np
import pandas as pd
from sklearn.ensemble import GradientBoostingClassifier, RandomForestClassifier
from sklearn.preprocessing import StandardScaler, OneHotEncoder
from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.model_selection import train_test_split, cross_val_score
from lifelines import CoxPHFitter  # 生存分析：考虑时间因素
from lifelines.utils import concordance_index
import xgboost as xgb
from typing import Dict, List, Tuple, Optional
from dataclasses import asdict
import joblib
from datetime import datetime, timedelta
import warnings

class PartnerMatchingPredictor:
    """
    合伙人匹配成功率预测器（生存分析+机器学习融合模型）
    
    创新点：
    1. 不仅预测是否成功，还预测"何时成功/失败"（生存分析）
    2. 动态更新：随着匹配过程推进，实时更新概率
    3. 可解释性：SHAP值解释每个特征的影响
    """
    
    def __init__(self, model_path: str = "./models/idss/"):
        self.model_path = model_path
        self.feature_columns = [
            'industry_encoded', 'stage_encoded', 'partner_type_encoded',
            'founder_background_encoded', 'team_size', 'funding_amount_log',
            'tech_maturity', 'market_competition', 'candidate_quality_score',
            'cultural_fit_score', 'decision_speed_days', 'search_scope_size'
        ]
        self.categorical_features = [
            'industry_encoded', 'stage_encoded', 'partner_type_encoded',
            'founder_background_encoded'
        ]
        self.numerical_features = [
            'team_size', 'funding_amount_log', 'tech_maturity',
            'market_competition', 'candidate_quality_score',
            'cultural_fit_score', 'decision_speed_days', 'search_scope_size'
        ]
        
        self.scaler = StandardScaler()
        self.encoder = OneHotEncoder(handle_unknown='ignore', sparse_output=False)
        self.survival_model = None
        self.classifier = None
        self.is_fitted = False
    
    def _extract_features(self, case: 'Case') -> Dict[str, float]:
        """
        从案例对象中提取结构化特征（特征工程）
        """
        features = {}
        
        # 基础特征
        features['team_size'] = case.founder.team_size
        features['funding_amount_log'] = np.log1p(self._parse_funding(case.founder.funding_status))
        
        # 技术成熟度（基于技术描述文本分析）
        features['tech_maturity'] = self._assess_tech_maturity(case.founder.core_tech)
        
        # 市场竞争度（基于行业熵值计算）
        features['market_competition'] = self._calculate_market_competition(case.industry.value)
        
        # 候选人质量（综合评分）
        if case.selected:
            features['candidate_quality_score'] = case.selected.match_score / 100.0
            features['cultural_fit_score'] = self._assess_cultural_fit(case)
        else:
            features['candidate_quality_score'] = 0.5  # 默认值
            features['cultural_fit_score'] = 0.5
        
        # 决策速度（满意解效率指标）
        if case.process:
            features['decision_speed_days'] = case.process.time_spent_days
            features['search_scope_size'] = case.process.candidates_sourced
        else:
            features['decision_speed_days'] = 60  # 行业平均
            features['search_scope_size'] = 5
        
        # 编码分类变量
        features['industry_encoded'] = case.industry.value
        features['stage_encoded'] = case.stage.value
        features['partner_type_encoded'] = case.partner_type.value
        features['founder_background_encoded'] = case.founder.background.value
        
        return features
    
    def _parse_funding(self, funding_status: str) -> float:
        """解析融资额（文本→数值）"""
        import re
        numbers = re.findall(r'(\d+)(?:万|万)', funding_status)
        if numbers:
            return float(numbers[0])
        numbers = re.findall(r'(\d+)(?:亿|亿)', funding_status)
        if numbers:
            return float(numbers[0]) * 10000
        return 1000  # 默认值1000万
    
    def _assess_tech_maturity(self, tech_desc: str) -> float:
        """评估技术成熟度（NLP简化版）"""
        maturity_keywords = {
            'high': ['量产', '成熟', '商业化', '落地', '验证'],
            'medium': ['中试', '样机', '试点', 'POC'],
            'low': ['实验室', '研发', '原理', '概念']
        }
        tech_lower = tech_desc.lower()
        
        if any(kw in tech_lower for kw in maturity_keywords['high']):
            return 0.8
        elif any(kw in tech_lower for kw in maturity_keywords['medium']):
            return 0.5
        else:
            return 0.2
    
    def _calculate_market_competition(self, industry: str) -> float:
        """计算行业竞争度（基于案例库统计）"""
        from case_repository.database.repository import CaseRepository
        repo = CaseRepository()
        cases = repo.get_by_industry(industry)
        
        if not cases:
            return 0.5
        
        # 竞争度 = 失败率 + 平均匹配周期归一化
        failure_rate = len([c for c in cases if c.outcome.value == "失败"]) / len(cases)
        avg_duration = np.mean([c.process.time_spent_days for c in cases if c.process]) if cases else 60
        duration_norm = min(avg_duration / 90, 1.0)  # 90天为上限
        
        return (failure_rate + duration_norm) / 2
    
    def _assess_cultural_fit(self, case: 'Case') -> float:
        """评估文化适配度（基于决策逻辑文本分析）"""
        if not case.selected:
            return 0.5
        
        fit_keywords = ['价值观', '文化', '认同', '契合', '一致']
        why_selected = case.selected.why_selected.lower()
        
        match_count = sum(1 for kw in fit_keywords if kw in why_selected)
        return min(0.5 + match_count * 0.15, 1.0)  # 基础0.5，每匹配一个+0.15
    
    def train(self, cases: List['Case']):
        """
        训练预测模型（双重模型：生存分析+分类器）
        """
        if len(cases) < 10:
            raise ValueError("需要至少10个案例才能训练可靠模型")
        
        # 构建训练数据
        data = []
        for case in cases:
            features = self._extract_features(case)
            
            # 标签1：是否成功（分类）
            is_success = 1 if case.outcome.value == "成功" else 0
            
            # 标签2：生存时间（匹配周期）和事件类型（成功=1，失败/进行中=0）
            duration = case.process.time_spent_days if case.process else 60
            event_observed = 1 if case.outcome.value in ["成功", "失败"] else 0
            
            features['success'] = is_success
            features['duration'] = duration
            features['event_observed'] = event_observed
            
            data.append(features)
        
        df = pd.DataFrame(data)
        
        # 预处理管道
        preprocessor = ColumnTransformer(
            transformers=[
                ('num', StandardScaler(), self.numerical_features),
                ('cat', OneHotEncoder(handle_unknown='ignore'), self.categorical_features)
            ]
        )
        
        X = df[self.numerical_features + self.categorical_features]
        y_success = df['success']
        y_duration = df['duration']
        y_event = df['event_observed']
        
        X_processed = preprocessor.fit_transform(X)
        
        # 模型1：生存分析（Cox比例风险模型）
        # 预测"在t天内成功"的概率
        survival_df = pd.DataFrame(X_processed, columns=self._get_feature_names(preprocessor))
        survival_df['duration'] = y_duration
        survival_df['event'] = y_event
        
        self.survival_model = CoxPHFitter(penalizer=0.1)
        self.survival_model.fit(survival_df, duration_col='duration', event_col='event')
        
        # 模型2：梯度提升分类器（XGBoost）
        # 直接预测成功概率
        self.classifier = xgb.XGBClassifier(
            n_estimators=100,
            max_depth=5,
            learning_rate=0.1,
            subsample=0.8,
            colsample_bytree=0.8,
            random_state=42
        )
        self.classifier.fit(X_processed, y_success)
        
        # 评估模型
        c_index = concordance_index(
            survival_df['duration'], 
            -self.survival_model.predict_partial_hazard(survival_df), 
            survival_df['event']
        )
        cv_score = cross_val_score(self.classifier, X_processed, y_success, cv=5).mean()
        
        print(f"模型训练完成：生存分析C-index={c_index:.3f}, 分类器CV准确率={cv_score:.3f}")
        
        # 保存模型
        self.preprocessor = preprocessor
        self.is_fitted = True
        self._save_models()
        
        return {
            "survival_c_index": c_index,
            "classifier_cv_accuracy": cv_score,
            "sample_size": len(cases)
        }
    
    def predict(self, case: 'Case', time_horizons: List[int] = [30, 60, 90]) -> Dict:
        """
        预测匹配成功概率（动态多时间尺度预测）
        
        Returns:
            不同时间点的成功概率、风险等级、关键影响因素
        """
        if not self.is_fitted:
            raise ValueError("模型未训练，请先调用train()")
        
        features = self._extract_features(case)
        X = pd.DataFrame([{k: features[k] for k in self.numerical_features + self.categorical_features}])
        X_processed = self.preprocessor.transform(X)
        
        # 预测1：直接成功概率（分类器）
        success_prob = self.classifier.predict_proba(X_processed)[0][1]
        
        # 预测2：时间条件成功概率（生存分析）
        # "在30天内成功概率"、"在60天内成功概率"等
        survival_probabilities = {}
        for t in time_horizons:
            # Cox模型预测生存函数
            surv_func = self.survival_model.predict_survival_function(
                pd.DataFrame(X_processed, columns=self._get_feature_names(self.preprocessor)),
                times=[t]
            )
            # 在t天内成功的概率 = 1 - 生存函数(t)
            prob_at_t = 1 - surv_func.iloc[0, 0]
            survival_probabilities[f"{t}days"] = prob_at_t
        
        # 风险分层
        risk_level = self._stratify_risk(success_prob, survival_probabilities["60days"])
        
        # 可解释性分析（SHAP值简化版）
        shap_values = self._calculate_feature_importance(X_processed)
        
        # 决策建议生成
        recommendations = self._generate_recommendations(
            success_prob, survival_probabilities, shap_values, case
        )
        
        return {
            "overall_success_probability": float(success_prob),
            "time_conditional_probabilities": {
                k: float(v) for k, v in survival_probabilities.items()
            },
            "risk_level": risk_level,
            "confidence_interval": {
                "lower": max(0, success_prob - 0.15),
                "upper": min(1, success_prob + 0.15)
            },
            "key_factors": shap_values,
            "recommendations": recommendations,
            "optimal_decision_window": self._calculate_optimal_window(case)
        }
    
    def _stratify_risk(self, overall_prob: float, prob_60d: float) -> str:
        """风险分层"""
        if overall_prob >= 0.8 and prob_60d >= 0.6:
            return "低风险-高置信"
        elif overall_prob >= 0.6:
            return "中风险-可接受"
        elif overall_prob >= 0.4:
            return "中高风险-需谨慎"
        else:
            return "高风险-建议重新评估"
    
    def _calculate_feature_importance(self, X_processed: np.ndarray) -> List[Dict]:
        """计算特征重要性（简化版SHAP）"""
        # 使用XGBoost内置重要性
        importance = self.classifier.feature_importances_
        feature_names = self._get_feature_names(self.preprocessor)
        
        # 排序并取Top 5
        indices = np.argsort(importance)[::-1][:5]
        return [
            {
                "feature": feature_names[i],
                "importance": float(importance[i]),
                "impact": "高" if importance[i] > 0.15 else "中" if importance[i] > 0.05 else "低"
            }
            for i in indices
        ]
    
    def _generate_recommendations(self, prob: float, time_probs: Dict, 
                                  shap_values: List[Dict], case: 'Case') -> List[str]:
        """生成可执行决策建议（规则引擎+模型输出）"""
        recommendations = []
        
        # 基于概率的建议
        if prob < 0.5:
            recommendations.append("⚠️ 当前匹配方案成功率较低，建议扩大候选人搜索范围或降低满意解阈值")
        elif prob < 0.7:
            recommendations.append("ℹ️ 成功率中等，建议重点关注文化适配性提升")
        else:
            recommendations.append("✅ 成功率高，建议加快决策速度，避免过度优化")
        
        # 基于时间概率的建议
        if time_probs["30days"] < 0.3 and time_probs["90days"] > 0.7:
            recommendations.append("⏰ 短期成功概率低但长期高，建议保持耐心，避免过早放弃")
        
        # 基于特征重要性的建议
        top_factor = shap_values[0]["feature"] if shap_values else None
        if top_factor == "decision_speed_days" and case.process and case.process.time_spent_days > 60:
            recommendations.append("🐌 决策速度是最大风险因素，建议设定硬性截止日期（满意解原则）")
        elif top_factor == "cultural_fit_score" and case.selected:
            recommendations.append("🤝 文化适配是关键，建议增加价值观匹配评估环节")
        
        return recommendations
    
    def _calculate_optimal_window(self, case: 'Case') -> Dict:
        """计算最优决策窗口（基于历史数据）"""
        # 查询相似案例的平均决策周期
        from case_repository.similarity.engine import CaseSimilarityEngine
        from case_repository.database.repository import CaseRepository
        
        engine = CaseSimilarityEngine()
        repo = CaseRepository()
        similar_cases = engine.find_similar_cases(case, repo.get_all(), top_k=5)
        
        durations = [c.process.time_spent_days for c, _, _ in similar_cases if c.process]
        if durations:
            avg_duration = np.mean(durations)
            return {
                "suggested_deadline_days": int(avg_duration * 0.8),  # 80%平均时间（满意解）
                "historical_avg_days": int(avg_duration),
                "rationale": f"基于{len(durations)}个相似案例，建议在第{int(avg_duration * 0.8)}天做出决策"
            }
        return {"suggested_deadline_days": 45, "historical_avg_days": 60, "rationale": "使用行业默认值"}
    
    def _save_models(self):
        """持久化模型"""
        import os
        os.makedirs(self.model_path, exist_ok=True)
        
        joblib.dump(self.survival_model, f"{self.model_path}/survival_model.pkl")
        joblib.dump(self.classifier, f"{self.model_path}/classifier.pkl")
        joblib.dump(self.preprocessor, f"{self.model_path}/preprocessor.pkl")
        
        # 保存元数据
        metadata = {
            "trained_at": datetime.now().isoformat(),
            "feature_columns": self.feature_columns,
            "model_version": "1.0"
        }
        with open(f"{self.model_path}/metadata.json", "w") as f:
            import json
            json.dump(metadata, f)
    
    def load_models(self):
        """加载预训练模型"""
        import os
        if not os.path.exists(f"{self.model_path}/classifier.pkl"):
            raise FileNotFoundError("未找到预训练模型，请先训练")

        self.classifier = joblib.load(f"{self.model_path}/classifier.pkl")
        self.preprocessor = joblib.load(f"{self.model_path}/preprocessor.pkl")
        self.is_fitted = True

# case_repository/ethics/auditor.py
"""
AI伦理审计系统（AI Ethics Auditor）
实现：
1. 反事实公平性测试（Counterfactual Fairness Testing）
2. 群体差异分析（Demographic Parity）
3. 偏见来源追溯（数据偏见/算法偏见）
"""
import pandas as pd
import numpy as np
from typing import Dict, List, Tuple, Optional
from dataclasses import dataclass
from collections import defaultdict
import json
from datetime import datetime

class EthicsAuditor:
    """
    伦理审计引擎
    
    检测维度：
    - 性别偏见（Gender Bias）
    - 学历偏见（Education Bias）：是否过度偏好清北/海归
    - 大厂背景偏见（Big Company Bias）：是否过度偏好BAT/华为
    - 地域偏见（Geographic Bias）
    """
    
    def __init__(self):
        self.sensitive_attributes = [
            'gender', 'education_tier', 'big_company_background', 
            'region', 'age_group'
        ]
        self.fairness_threshold = 0.8  # 80%公平性阈值
    
    def audit_case_repository(self, cases: List['Case']) -> Dict:
        """
        全面审计案例库公平性（批量审计）
        """
        audit_results = {
            "audit_timestamp": datetime.now().isoformat(),
            "sample_size": len(cases),
            "overall_fairness_score": 0.0,
            "dimension_audits": {},
            "biased_cases": [],
            "recommendations": []
        }
        
        # 维度1：推荐系统偏见（相似度算法是否公平）
        rec_bias = self._audit_recommendation_bias(cases)
        audit_results["dimension_audits"]["recommendation_algorithm"] = rec_bias
        
        # 维度2：成功案例分布偏见（不同群体成功率是否均衡）
        success_bias = self._audit_success_rate_bias(cases)
        audit_results["dimension_audits"]["success_rate"] = success_bias
        
        # 维度3：决策理由偏见（是否基于刻板印象）
        decision_bias = self._audit_decision_reasoning_bias(cases)
        audit_results["dimension_audits"]["decision_reasoning"] = decision_bias
        
        # 计算总体公平性分数
        scores = [d["fairness_score"] for d in audit_results["dimension_audits"].values()]
        audit_results["overall_fairness_score"] = np.mean(scores)
        
        # 识别具体偏见案例
        audit_results["biased_cases"] = self._identify_biased_cases(cases)
        
        # 生成改进建议
        audit_results["recommendations"] = self._generate_ethics_recommendations(audit_results)
        
        return audit_results
    
    def _audit_recommendation_bias(self, cases: List['Case']) -> Dict:
        """审计相似度推荐算法的公平性"""
        # 构造反事实测试用例：仅改变敏感属性，观察推荐结果是否变化
        
        test_cases = self._generate_counterfactual_test_cases(cases)
        
        disparities = []
        for original, counterfactual in test_cases:
            # 使用相似度引擎分别计算推荐结果
            from case_repository.similarity.engine import CaseSimilarityEngine
            engine = CaseSimilarityEngine()
            
            orig_recommendations = engine.find_similar_cases(original, cases, top_k=3)
            cf_recommendations = engine.find_similar_cases(counterfactual, cases, top_k=3)
            
            # 计算推荐差异度
            disparity = self._calculate_recommendation_disparity(
                orig_recommendations, cf_recommendations
            )
            disparities.append(disparity)
        
        avg_disparity = np.mean(disparities) if disparities else 0
        
        return {
            "test_cases_count": len(test_cases),
            "average_recommendation_disparity": avg_disparity,
            "fairness_score": 1 - avg_disparity,  # 差异越小越公平
            "status": "PASS" if avg_disparity < 0.2 else "WARNING" if avg_disparity < 0.4 else "FAIL",
            "findings": f"反事实测试显示推荐差异度{avg_disparity:.2f}，{'符合公平性标准' if avg_disparity < 0.2 else '存在潜在偏见'}"
        }
    
    def _generate_counterfactual_test_cases(self, cases: List['Case']) -> List[Tuple]:
        """
        生成反事实测试用例（Counterfactual Fairness Testing）
        
        例如：将"华为背景"改为"初创背景"，其他条件完全相同，观察推荐结果是否变化
        """
        test_pairs = []
        
        for case in cases:
            if not case.selected:
                continue
            
            # 测试1：大厂背景反事实
            if "华为" in case.selected.background_summary or "阿里" in case.selected.background_summary:
                cf_case = case  # 简化：实际应深拷贝
                cf_case.selected.background_summary = case.selected.background_summary.replace("华为", "初创公司").replace("阿里", "中小公司")
                test_pairs.append((case, cf_case))
        
        return test_pairs
    
    def _audit_success_rate_bias(self, cases: List['Case']) -> Dict:
        """审计不同群体的成功率是否存在显著差异"""
        # 按敏感属性分组统计成功率
        groups = defaultdict(lambda: {"success": 0, "total": 0})
        
        for case in cases:
            if not case.selected:
                continue
            
            # 检测大厂背景标签
            has_big_company = any(kw in case.selected.background_summary 
                                 for kw in ["华为", "阿里", "腾讯", "百度", "字节", "大疆"])
            group_key = "大厂背景" if has_big_company else "非大厂背景"
            
            groups[group_key]["total"] += 1
            if case.outcome.value == "成功":
                groups[group_key]["success"] += 1
        
        # 计算统计显著性（简化版）
        if len(groups) >= 2:
            success_rates = {k: v["success"]/v["total"] for k, v in groups.items() if v["total"] > 0}
            
            if len(success_rates) >= 2:
                max_rate = max(success_rates.values())
                min_rate = min(success_rates.values())
                disparity = max_rate - min_rate
                
                return {
                    "group_success_rates": success_rates,
                    "max_disparity": disparity,
                    "fairness_score": 1 - disparity,
                    "status": "PASS" if disparity < 0.15 else "WARNING" if disparity < 0.25 else "FAIL",
                    "findings": f"群体成功率差异为{disparity:.1%}，{success_rates}"
                }
        
        return {"fairness_score": 1.0, "status": "PASS", "findings": "样本不足或无明显分组"}
    
    def _audit_decision_reasoning_bias(self, cases: List['Case']) -> Dict:
        """审计决策理由中是否存在刻板印象"""
        biased_keywords = {
            "gender": ["男主外", "女性细心", "男性更有魄力"],
            "age": ["太年轻不稳", "年纪大思维固化", "35岁危机"],
            "education": ["非名校不可靠", "海归眼高手低", "土鳖没视野"]
        }
        
        bias_counts = defaultdict(int)
        biased_cases = []
        
        for case in cases:
            if not case.selected or not case.selected.why_selected:
                continue
            
            reason = case.selected.why_selected
            
            for bias_type, keywords in biased_keywords.items():
                for kw in keywords:
                    if kw in reason:
                        bias_counts[bias_type] += 1
                        biased_cases.append({
                            "case_id": case.case_id,
                            "bias_type": bias_type,
                            "keyword": kw,
                            "context": reason[:50]
                        })
        
        total_cases = len([c for c in cases if c.selected and c.selected.why_selected])
        bias_rate = len(biased_cases) / total_cases if total_cases > 0 else 0
        
        return {
            "bias_type_counts": dict(bias_counts),
            "biased_decisions": biased_cases[:10],  # 只展示前10个
            "bias_rate": bias_rate,
            "fairness_score": 1 - bias_rate,
            "status": "PASS" if bias_rate < 0.05 else "WARNING" if bias_rate < 0.1 else "FAIL",
            "findings": f"发现{len(biased_cases)}个案例存在决策偏见词汇，占比{bias_rate:.1%}"
        }
    
    def generate_fairness_report(self, audit_results: Dict) -> str:
        """生成伦理审计报告（Markdown格式，供合规审查）"""
        report = f"""# AI伦理审计报告（案例库偏见检测）

**审计时间**: {audit_results['audit_timestamp']}  
**样本规模**: {audit_results['sample_size']}个案例  
**总体公平性评分**: {audit_results['overall_fairness_score']*100:.1f}/100

## 1. 推荐算法公平性（反事实测试）
- **状态**: {audit_results['dimension_audits']['recommendation_algorithm']['status']}
- **公平性分数**: {audit_results['dimension_audits']['recommendation_algorithm']['fairness_score']*100:.1f}
- **发现**: {audit_results['dimension_audits']['recommendation_algorithm']['findings']}

## 2. 成功率分布公平性
- **状态**: {audit_results['dimension_audits']['success_rate']['status']}
- **群体成功率**: {audit_results['dimension_audits']['success_rate'].get('group_success_rates', 'N/A')}

## 3. 决策理由偏见检测
- **状态**: {audit_results['dimension_audits']['decision_reasoning']['status']}
- **偏见案例数**: {len(audit_results['dimension_audits']['decision_reasoning'].get('biased_decisions', []))}
- **偏见类型分布**: {audit_results['dimension_audits']['decision_reasoning'].get('bias_type_counts', {})}

## 4. 改进建议
"""
        for rec in audit_results['recommendations']:
            report += f"- {rec}\n"
        
        report += """
## 5. 合规声明
本审计遵循IEEE 2857-2021 AI伦理标准和欧盟AI法案（EU AI Act）对高风险AI系统的公平性要求。
"""
        return report

# case_repository/curriculum/generator.py
"""
课程学习系统（Curriculum Learning System）
自动从案例库生成难度递进的学习路径
"""
import numpy as np
from typing import List, Dict, Tuple
from dataclasses import dataclass
from collections import defaultdict
import json

@dataclass
class CurriculumNode:
    """课程节点"""
    case_id: str
    difficulty_score: float  # 0-1，越高越难
    concept_tags: List[str]
    prerequisites: List[str]  # 前置课程案例ID
    learning_objectives: List[str]
    estimated_time_minutes: int

class CurriculumGenerator:
    """
    自动课程生成器（基于课程学习理论）
    
    算法：
    1. 案例难度评估（多维度）
    2. 概念依赖图构建（哪些案例必须先学）
    3. 最优学习路径生成（由易到难）
    """
    
    def __init__(self):
        self.difficulty_weights = {
            "complexity": 0.3,      # 案例复杂度
            "ambiguity": 0.25,      # 决策模糊度（多个选项都合理）
            "outcome_uncertainty": 0.2,  # 结果不确定性（边界案例）
            "domain_knowledge": 0.15,  # 领域知识门槛
            "time_pressure": 0.1    # 时间压力（紧急程度）
        }
    
    def assess_case_difficulty(self, case: 'Case') -> float:
        """
        评估案例难度（多维度量化）
        """
        scores = {}
        
        # 维度1：复杂度（决策因素数量）
        if case.process:
            num_criteria = len(case.process.key_criteria_weights)
            scores["complexity"] = min(num_criteria / 5, 1.0)  # 5个维度为满分
        else:
            scores["complexity"] = 0.5
        
        # 维度2：模糊度（候选人数量与差异度）
        if case.process:
            num_candidates = case.process.candidates_interviewed
            # 候选人越多且差异越小，决策越模糊
            scores["ambiguity"] = min(num_candidates / 6, 1.0) * 0.7
        else:
            scores["ambiguity"] = 0.5
        
        # 维度3：结果不确定性（边界案例 vs 明确成功/失败）
        if case.outcome.value == "待定":
            scores["outcome_uncertainty"] = 0.9
        elif case.outcome.value == "进行中":
            scores["outcome_uncertainty"] = 0.7
        else:
            # 成功/失败但复盘显示"险胜"或"险些成功"
            if case.outcome_record and case.outcome_record.lessons_learned:
                has_close_call = any(kw in " ".join(case.outcome_record.lessons_learned) 
                                    for kw in ["险些", "差一点", "侥幸", "运气"])
                scores["outcome_uncertainty"] = 0.6 if has_close_call else 0.3
            else:
                scores["outcome_uncertainty"] = 0.3
        
        # 维度4：领域知识门槛（基于行业）
        hard_tech_industries = ["生物医药", "航空航天", "量子计算"]
        medium_tech = ["AI芯片", "GPU芯片", "新能源"]
        
        if case.industry.value in hard_tech_industries:
            scores["domain_knowledge"] = 0.9
        elif case.industry.value in medium_tech:
            scores["domain_knowledge"] = 0.6
        else:
            scores["domain_knowledge"] = 0.4
        
        # 维度5：时间压力
        if case.process and case.process.time_spent_days < 30:
            scores["time_pressure"] = 0.8  # 紧急
        elif case.process and case.process.time_spent_days > 80:
            scores["time_pressure"] = 0.9  # 拖延导致的风险
        else:
            scores["time_pressure"] = 0.4
        
        # 加权计算总难度
        total_difficulty = sum(
            scores[dim] * weight 
            for dim, weight in self.difficulty_weights.items()
        )
        
        return total_difficulty
    
    def build_concept_dependency_graph(self, cases: List['Case']) -> Dict[str, List[str]]:
        """
        构建概念依赖图（哪些案例必须先学）
        
        原则：
        - 基础概念案例（行业基础知识）→ 进阶案例
        - 成功案例 → 失败案例（先知道什么对，再学什么错）
        - 短周期案例 → 长周期案例（由简入繁）
        """
        dependencies = defaultdict(list)
        
        # 按行业分组
        by_industry = defaultdict(list)
        for case in cases:
            by_industry[case.industry.value].append(case)
        
        for industry, industry_cases in by_industry.items():
            # 排序：成功案例在前，短周期在前
            sorted_cases = sorted(
                industry_cases,
                key=lambda c: (
                    0 if c.outcome.value == "成功" else 1,  # 成功案例优先
                    c.process.time_spent_days if c.process else 60
                )
            )
            
            # 构建依赖链
            for i in range(1, len(sorted_cases)):
                # 简单的顺序依赖（实际可用更复杂的概念重叠度计算）
                prerequisites = [sorted_cases[j].case_id for j in range(max(0, i-3), i)]
                dependencies[sorted_cases[i].case_id] = prerequisites
        
        return dependencies
    
    def generate_learning_path(self, target_industry: str = None, 
                               target_difficulty: float = None) -> List[CurriculumNode]:
        """
        生成个性化学习路径（Curriculum）
        """
        from case_repository.database.repository import CaseRepository
        repo = CaseRepository()
        cases = repo.get_all()
        
        if target_industry:
            cases = [c for c in cases if c.industry.value == target_industry]
        
        # 评估所有案例难度
        case_difficulties = {c.case_id: self.assess_case_difficulty(c) for c in cases}
        
        # 构建依赖图
        dependencies = self.build_concept_dependency_graph(cases)
        
        # 拓扑排序生成学习路径（确保前置案例先学）
        learning_path = self._topological_sort(case_difficulties, dependencies)
        
        # 如果指定目标难度，筛选并平衡
        if target_difficulty:
            learning_path = self._balance_difficulty(learning_path, case_difficulties, target_difficulty)
        
        # 构建课程节点
        curriculum = []
        for case_id in learning_path:
            case = repo.get_by_id(case_id)
            if not case:
                continue
            
            node = CurriculumNode(
                case_id=case_id,
                difficulty_score=case_difficulties[case_id],
                concept_tags=self._extract_concepts(case),
                prerequisites=dependencies.get(case_id, []),
                learning_objectives=self._generate_objectives(case),
                estimated_time_minutes=self._estimate_learning_time(case)
            )
            curriculum.append(node)
        
        return curriculum
    
    def _topological_sort(self, difficulties: Dict, dependencies: Dict) -> List[str]:
        """拓扑排序（确保前置课程在前）"""
        # 简化版：按难度值排序，同时满足依赖约束
        # 实际应使用Kahn算法或DFS拓扑排序
        
        # 先按难度排序
        sorted_by_difficulty = sorted(difficulties.items(), key=lambda x: x[1])
        
        # 确保依赖满足（贪心调整）
        final_order = []
        added = set()
        
        for case_id, diff in sorted_by_difficulty:
            # 先添加所有未添加的前置依赖
            for prereq in dependencies.get(case_id, []):
                if prereq not in added and prereq in difficulties:
                    final_order.append(prereq)
                    added.add(prereq)
            
            if case_id not in added:
                final_order.append(case_id)
                added.add(case_id)
        
        return final_order
    
    def _balance_difficulty(self, path: List[str], difficulties: Dict, 
                           target: float) -> List[str]:
        """平衡难度分布（确保由易到难的平滑过渡）"""
        # 筛选与目标难度相近的案例
        scored = [(cid, abs(difficulties[cid] - target)) for cid in path]
        scored.sort(key=lambda x: x[1])
        
        # 取Top N，但保持原有相对顺序
        selected = set(cid for cid, _ in scored[:10])
        return [cid for cid in path if cid in selected]
    
    def _extract_concepts(self, case: 'Case') -> List[str]:
        """从案例中提取关键概念"""
        concepts = [case.industry.value, case.stage.value, case.partner_type.value]
        
        if case.outcome_record and case.outcome_record.key_success_factors:
            # 提取成功因素中的关键词
            for factor in case.outcome_record.key_success_factors[:2]:
                concepts.append(factor[:10])  # 简化
        
        return concepts
    
    def _generate_objectives(self, case: 'Case') -> List[str]:
        """生成学习目标"""
        return [
            f"掌握{case.industry.value}行业合伙人匹配关键要素",
            f"理解{case.outcome.value}案例的决策逻辑",
            "学会识别并评估候选人的核心能力缺口"
        ]
    
    def _estimate_learning_time(self, case: 'Case') -> int:
        """估算学习时间（分钟）"""
        base_time = 15  # 基础阅读时间
        
        # 复杂度加成
        if case.process and len(case.process.key_criteria_weights) > 3:
            base_time += 10
        
        # 复盘深度加成
        if case.outcome_record and len(case.outcome_record.lessons_learned) > 2:
            base_time += 10
        
        return base_time

@click.command()
@click.option('--industry', '-ind', help='目标行业')
@click.option('--difficulty', '-d', type=float, help='目标难度级别(0-1)')
def generate_curriculum(industry: str, difficulty: float):
    """生成培训课程（Curriculum Learning）"""
    generator = CurriculumGenerator()
    curriculum = generator.generate_learning_path(industry, difficulty)
    
    if not curriculum:
        click.echo("❌ 未找到符合条件的案例")
        return
    
    click.echo("=" * 60)
    click.echo("📚 自动课程生成（Curriculum Learning）")
    click.echo("=" * 60)
    click.echo(f"课程路径: {' → '.join([n.case_id for n in curriculum])}")
    click.echo(f"总课程数: {len(curriculum)}")
    click.echo(f"预计总时长: {sum(n.estimated_time_minutes for n in curriculum)}分钟\n")
    
    for i, node in enumerate(curriculum, 1):
        click.echo(f"{i}. [{node.case_id}] 难度: {node.difficulty_score:.2f}")
        click.echo(f"   概念: {', '.join(node.concept_tags[:3])}")
        click.echo(f"   目标: {node.learning_objectives[0]}")
        click.echo(f"   时长: {node.estimated_time_minutes}分钟")
        if node.prerequisites:
            click.echo(f"   前置: {', '.join(node.prerequisites)}")
        click.echo()

from playwright.sync_api import sync_playwright

class DesignSystemCrawler:
    def __init__(self):
        self.design_tokens = {
            "colors": set(),
            "fonts": set(),
            "spacing": set(),
            "border_radius": set(),
            "shadows": set()
        }
    
    def extract_computed_styles(self, page, selector: str) -> Dict:
        """提取元素的计算样式"""
        js_code = """
        (selector) => {
            const elements = document.querySelectorAll(selector);
            const styles = [];
            elements.forEach(el => {
                const computed = window.getComputedStyle(el);
                styles.push({
                    color: computed.color,
                    backgroundColor: computed.backgroundColor,
                    fontFamily: computed.fontFamily,
                    fontSize: computed.fontSize,
                    fontWeight: computed.fontWeight,
                    lineHeight: computed.lineHeight,
                    letterSpacing: computed.letterSpacing,
                    padding: computed.padding,
                    margin: computed.margin,
                    borderRadius: computed.borderRadius,
                    boxShadow: computed.boxShadow,
                    borderColor: computed.borderColor
                });
            });
            return styles;
        }
        """
        return page.evaluate(js_code, selector)
    
    def extract_palette_from_screenshot(self, screenshot_path: str, top_n: int = 10) -> List[Dict]:
        """从截图中提取主色调（使用KMeans聚类）"""
        image = cv2.imread(screenshot_path)
        image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        pixels = image.reshape(-1, 3)
        
        # KMeans聚类找出主色
        from sklearn.cluster import KMeans
        kmeans = KMeans(n_clusters=top_n, random_state=42, n_init=10)
        kmeans.fit(pixels)
        
        colors = []
        for color in kmeans.cluster_centers_:
            hex_color = '#{:02x}{:02x}{:02x}'.format(int(color[0]), int(color[1]), int(color[2]))
            colors.append({
                "hex": hex_color,
                "rgb": [int(color[0]), int(color[1]), int(color[2])]
            })
        return colors
    
    def crawl(self, url: str, pages: List[str] = None) -> Dict:
        """主抓取流程"""

# src/analyzer/token_analyzer.py
import json
from typing import Dict, List
from collections import defaultdict
import colorsys

class DesignTokenAnalyzer:
    def __init__(self, raw_data: Dict):
        self.raw_data = raw_data
        self.tokens = {
            "colors": {},
            "typography": {},
            "spacing": {},
            "effects": {},
            "layout": {}
        }
    
    def normalize_color(self, color_str: str) -> str:
        """统一颜色格式为HEX"""
        if color_str.startswith("rgb"):
            # 解析 rgb(r, g, b) 或 rgba(r, g, b, a)
            values = color_str.replace("rgb(", "").replace("rgba(", "").replace(")", "").split(",")
            r, g, b = int(values[0]), int(values[1]), int(values[2])
            return '#{:02x}{:02x}{:02x}'.format(r, g, b)
        return color_str
    
    def categorize_colors(self, colors: List[str]) -> Dict:
        """将颜色分类为主色、辅助色、中性色等"""
        categorized = {
            "primary": [],
            "secondary": [],
            "neutral": [],
            "semantic": {"success": [], "warning": [], "error": [], "info": []},
            "background": [],
            "text": []
        }
        
        for color in colors:
            try:
                hex_color = self.normalize_color(color)
                rgb = tuple(int(hex_color[i:i+2], 16) for i in (1, 3, 5))
                h, l, s = colorsys.rgb_to_hls(rgb[0]/255, rgb[1]/255, rgb[2]/255)
                
                # 根据亮度/饱和度分类
                if l < 0.2:
                    categorized["neutral"].append({"hex": hex_color, "role": "dark"})
                elif l > 0.9:
                    categorized["background"].append({"hex": hex_color, "role": "surface"})
                elif s > 0.6 and 0.3 < l < 0.7:
                    categorized["primary"].append({"hex": hex_color})
                else:
                    categorized["neutral"].append({"hex": hex_color})
            except:
                continue
        
        return categorized
    
    def analyze_typography(self, styles: List[Dict]) -> Dict:
        """分析字体系统"""
        fonts = defaultdict(lambda: {"sizes": set(), "weights": set(), "line_heights": set()})
        
        for style in styles:
            font_family = style.get("fontFamily", "").split(",")[0].strip('"\'')
            if font_family:
                fonts[font_family]["sizes"].add(style.get("fontSize"))
                fonts[font_family]["weights"].add(style.get("fontWeight"))
                fonts[font_family]["line_heights"].add(style.get("lineHeight"))
        
        # 构建字体层级
        typography = {
            "font_family": {
                "heading": list(fonts.keys())[0] if fonts else "Inter",
                "body": list(fonts.keys())[0] if fonts else "Inter",
                "mono": "JetBrains Mono"
            },
            "scale": {
                "xs": "0.75rem", "sm": "0.875rem", "base": "1rem",
                "lg": "1.125rem", "xl": "1.25rem", "2xl": "1.5rem",
                "3xl": "1.875rem", "4xl": "2.25rem"
            },
            "weight": {
                "normal": "400", "medium": "500", "semibold": "600", "bold": "700"
            }
        }
        return typography
    
    def generate_tokens(self) -> Dict:
        """生成完整的设计Token"""
        all_colors = []
        all_styles = []
        
        for selector, styles in self.raw_data["styles"].items():
            for style in styles:
                if style.get("color"):
                    all_colors.append(style["color"])
                if style.get("backgroundColor") and style["backgroundColor"] != "rgba(0, 0, 0, 0)":
                    all_colors.append(style["backgroundColor"])
                all_styles.append(style)
        
        self.tokens["colors"] = self.categorize_colors(all_colors)
        self.tokens["typography"] = self.analyze_typography(all_styles)
        self.tokens["spacing"] = {
            "unit": "4px",
            "scale": {"0": "0", "1": "0.25rem", "2": "0.5rem", "4": "1rem", "8": "2rem", "16": "4rem"}
        }
        self.tokens["radii"] = {"sm": "0.25rem", "md": "0.5rem", "lg": "1rem", "full": "9999px"}
        self.tokens["shadows"] = {
            "sm": "0 1px 2px 0 rgb(0 0 0 / 0.05)",
            "md": "0 4px 6px -1px rgb(0 0 0 / 0.1)",
            "lg": "0 10px 15px -3px rgb(0 0 0 / 0.1)"
        }
        
        return self.tokens

from datetime import datetime

class DesignMdGenerator:
    def __init__(self, tokens: Dict, brand_name: str, analysis: Dict):
        self.tokens = tokens
        self.brand_name = brand_name
        self.analysis = analysis
    
    def generate(self) -> str:
        """生成 DESIGN.md 文件内容"""
        md_content = f"""# {self.brand_name} Design System

> Auto-generated by Kimi Claw on {datetime.now().strftime("%Y-%m-%d")}
> Source: {self.analysis.get('url', 'Unknown')}

## Brand Essence

{self.analysis.get('description', 'A modern, minimal design system focused on clarity and efficiency.')}

**Key Characteristics:**
- {self.analysis.get('style_keywords', ['Clean', 'Modern', 'Professional'])}
- Visual Density: {self.analysis.get('density', 'Medium')}
- Color Mood: {self.analysis.get('mood', 'Neutral')}

## Color Palette

### Primary Colors
| Token | Hex | Usage |
|-------|-----|-------|
"""
        # 添加主色
        for i, color in enumerate(self.tokens["colors"].get("primary", [])[:3]):
            md_content += f"| primary-{i+1} | {color['hex']} | Main brand color |\n"
        
        md_content += """
### Neutral Scale
| Token | Hex | Usage |
|-------|-----|-------|
"""
        for i, color in enumerate(self.tokens["colors"].get("neutral", [])[:5]):
            role = color.get('role', f'neutral-{i+1}')
            md_content += f"| {role} | {color['hex']} | Text/UI elements |\n"

import requests

    def upload_design_md(self, file_path: str, file_name: str) -> str:
        """上传 DESIGN.md 到 Kimi 知识库"""
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 创建临时文件用于上传
        files = {
            'file': (file_name, content, 'text/markdown')
        }
        headers = {
            "Authorization": f"Bearer {self.api_key}"
        }
        
        # 上传到 Kimi 文件存储
        response = requests.post(
            f"{self.base_url}/files",
            files=files,
            headers=headers
        )
        
        if response.status_code == 200:
            file_id = response.json().get("id")
            print(f"✅ 上传成功: {file_id}")
            return file_id
        else:

class DesignExtractor {
  extractAll() {
    return {
      colors: this.extractColors(),
      typography: this.extractTypography(),
      spacing: this.extractSpacing(),
      screenshot: this.takeScreenshot()
    };
  }

  extractColors() {
    const elements = document.querySelectorAll('*');
    const colorMap = new Map();
    
    elements.forEach(el => {
      const styles = window.getComputedStyle(el);
      const bg = styles.backgroundColor;
      const text = styles.color;
      const border = styles.borderColor;
      
      [bg, text, border].forEach(color => {
        if (color && color !== 'rgba(0, 0, 0, 0)') {
          const count = colorMap.get(color) || 0;
          colorMap.set(color, count + 1);
        }
      });
    });
    
    // 按使用频率排序
    return Array.from(colorMap.entries())
      .sort((a, b) => b[1] - a[1])
      .slice(0, 20)
      .map(([color, count]) => ({ color, count }));
  }

  extractTypography() {
    const headings = document.querySelectorAll('h1, h2, h3');
    const body = document.querySelectorAll('p, span, div');
    
    return {
      headingFont: this.getFontFamily(headings[0]),
      bodyFont: this.getFontFamily(body[0]),
      scale: this.getTypeScale()
    };
  }

  sendToKimi(data) {
    // 发送到 Kimi Claw API
    chrome.runtime.sendMessage({
      action: 'generateDesignMd',
      data: data,

def create_geometric_logo(brand_name, primary_color):
    svg = ET.Element('svg', width="200", height="200", xmlns="http://www.w3.org/2000/svg")
    
    # 创建渐变背景圆形
    circle = ET.SubElement(svg, 'circle', cx="100", cy="100", r="90", 
                          fill=primary_color, opacity="0.1")

    polygon = ET.SubElement(svg, 'polygon', 
                           points="100,30 160,65 160,135 100,170 40,135 40,65",
                           fill="none", stroke=primary_color, stroke_width="8",
                           stroke_linejoin="round")
    
    # 添加首字母

                        font_family="Arial", font_size="60", 
                        font_weight="bold", fill=primary_color,
                        text_anchor="middle")

from PIL import Image, ImageDraw, ImageFont
import math

def generate_tech_logo(text, color_scheme):
    # 创建画布
    img = Image.new('RGBA', (512, 512), (255, 255, 255, 0))
    draw = ImageDraw.Draw(img)
    
    # 绘制科技风格的网格背景
    for i in range(0, 512, 32):
        draw.line([(i, 0), (i, 512)], fill=(240, 240, 240, 100), width=1)
        draw.line([(0, i), (512, i)], fill=(240, 240, 240, 100), width=1)
    
    # 绘制中心六边形（科技风格）
    center_x, center_y = 256, 220
    size = 120
    points = []
    for i in range(6):
        angle = math.pi/3 * i - math.pi/2
        x = center_x + size * math.cos(angle)
        y = center_y + size * math.sin(angle)
        points.append((x, y))
    
    draw.polygon(points, fill=color_scheme['primary'], outline=color_scheme['secondary'], width=8)
    
    # 添加文字
    try:
        font = ImageFont.truetype("Inter-Bold.ttf", 64)
    except:
        font = ImageFont.load_default()
    
    bbox = draw.textbbox((0, 0), text, font=font)
    text_width = bbox[2] - bbox[0]
    draw.text(((512-text_width)/2, 360), text, font=font, fill=color_scheme['text'])
    
    return img

import bpy

def create_3d_logo():
    # 清除默认场景
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete()
    
    # 创建 3D 文字
    bpy.ops.object.text_add(location=(0, 0, 0))
    text_obj = bpy.context.active_object
    text_obj.data.body = "M"
    text_obj.data.extrude = 0.2
    
    # 添加材质
    mat = bpy.data.materials.new(name="LogoMaterial")
    mat.use_nodes = True
    mat.node_tree.nodes["Principled BSDF"].inputs[0].default_value = (0.2, 0.4, 0.9, 1)
    text_obj.data.materials.append(mat)
    
    # 添加光照
    bpy.ops.object.light_add(type='SUN', location=(5, 5, 10))
    
    # 渲染
    bpy.context.scene.render.filepath = '/tmp/logo_3d.png'

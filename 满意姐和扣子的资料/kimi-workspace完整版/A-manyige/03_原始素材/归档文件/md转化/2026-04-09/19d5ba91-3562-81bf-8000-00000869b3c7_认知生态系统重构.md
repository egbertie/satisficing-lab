class TotemType(str, Enum):
    """五路图腾类型"""

class CognitiveCrystal(BaseModel):
    """
    认知晶体：文档的最终蒸馏形态
    不可变，版本更新创建新实例
    """
    crystal_id: str = Field(default_factory=lambda: f"CRY-{datetime.now().strftime('%Y%m%d')}-{hashlib.md5(str(datetime.now().timestamp()).encode()).hexdigest()[:6]}")
    source_uris: List[str] = Field(..., description="源文件路径列表")
    created_at: datetime = Field(default_factory=datetime.now)
    compression_ratio: float = Field(..., ge=0.0, le=1.0, description="压缩率")
    
    # 语义核心
    primary_entities: List[str] = Field(default_factory=list, description="核心实体")
    key_relations: List[Dict[str, str]] = Field(default_factory=list, description="关键关系三元组")
    decision_patterns: List[str] = Field(default_factory=list, description="可执行决策模式")
    contradiction_flags: List[str] = Field(default_factory=list, description="与其他晶体的冲突标记")
    
    # 向量指纹（轻量引用，实际存储在向量DB）
    vector_fingerprint: Optional[str] = None
    knowledge_graph_anchor: Dict[str, Any] = Field(default_factory=dict)
    
    # 元数据
    totem_affinity: Dict[TotemType, float] = Field(default_factory=dict, description="与各路图腾的相关性分数")
    activation_triggers: List[str] = Field(default_factory=list)
    ttl_days: int = Field(default=365, description="生存周期，到期需重新蒸馏")
    
    def to_graph_nodes(self) -> List[Dict]:
        """转换为图谱节点格式"""
        nodes = [{"id": self.crystal_id, "type": "Crystal", "properties": self.dict()}]
        for entity in self.primary_entities:
            nodes.append({"id": entity, "type": "Entity", "properties": {}})
        return nodes
    
    def to_graph_edges(self) -> List[Dict]:
        """转换为图谱边格式"""
        edges = []
        for rel in self.key_relations:
            edges.append({
                "source": rel.get("subject"),
                "target": rel.get("object"),
                "relation": rel.get("predicate"),
                "source_crystal": self.crystal_id
            })
        return edges

"""
三级蒸馏塔实现
L1: 结构拆解 -> L2: 语义向量化 -> L3: 认知晶体生成
"""

import os
from pathlib import Path
from typing import AsyncGenerator, Callable, Optional
import aiofiles
from docx import Document
import openai
import json
import asyncio
from collections import deque

class HyperCompressor:
    """
    超压缩引擎：将任意文档转化为认知晶体
    依赖：python-docx, openai, chromadb（本地向量库）
    """
    
    def __init__(self, 
                 embedding_model: str = "text-embedding-3-large",
                 llm_model: str = "gpt-4o",
                 compression_target: float = 0.15):
        self.embedding_model = embedding_model
        self.llm_model = llm_model
        self.compression_target = compression_target
        self.client = openai.AsyncOpenAI(
            base_url="https://models.inference.ai.azure.com",
            api_key=os.getenv("GITHUB_TOKEN")  # 复用GitHub Models
        )
        self.processing_queue = deque()
        
    async def process_docx(self, file_path: str, context_tags: List[str] = None) -> CognitiveCrystal:
        """主入口：DOCX -> 认知晶体"""
        # L1: 结构暴力拆解
        raw_chunks = await self._l1_structure_extract(file_path)
        
        # L2: 语义向量化（并行）
        chunk_embeddings = await self._l2_semantic_vectorize(raw_chunks)
        
        # L3: 认知晶体生成（GraphRAG风格）
        crystal = await self._l3_crystal_synthesis(
            source_path=file_path,
            chunks=raw_chunks,
            embeddings=chunk_embeddings,
            context_tags=context_tags or []
        )
        
        return crystal
    
    async def _l1_structure_extract(self, file_path: str) -> List[Dict]:
        """L1: 结构拆解，保留语义块与表格"""
        doc = Document(file_path)
        chunks = []
        
        # 提取段落（带层级结构）
        current_section = "无标题"
        for para in doc.paragraphs:
            if para.style.name.startswith('Heading'):
                current_section = para.text
            if len(para.text.strip()) > 20:  # 过滤短句噪音
                chunks.append({
                    "type": "paragraph",
                    "section": current_section,
                    "content": para.text,
                    "word_count": len(para.text)
                })
        
        # 提取表格（结构化数据）
        for table_idx, table in enumerate(doc.tables):
            table_data = []
            for row in table.rows:
                row_data = [cell.text for cell in row.cells]
                table_data.append(row_data)
            chunks.append({
                "type": "table",
                "section": f"Table-{table_idx}",
                "content": json.dumps(table_data, ensure_ascii=False),
                "word_count": sum(len(str(cell)) for row in table_data for cell in row)
            })
            
        return chunks
    
    async def _l2_semantic_vectorize(self, chunks: List[Dict]) -> List[Dict]:
        """L2: 批量嵌入，使用GitHub Models"""
        texts = [c["content"][:8000] for c in chunks]  # 截断保护
        
        response = await self.client.embeddings.create(
            model=self.embedding_model,
            input=texts,
            dimensions=3072
        )
        
        for i, chunk in enumerate(chunks):
            chunk["embedding"] = response.data[i].embedding
            chunk["embedding_id"] = f"emb-{i}"
            
        return chunks
    
    async def _l3_crystal_synthesis(self, 
                                    source_path: str,
                                    chunks: List[Dict], 
                                    embeddings: List[Dict],
                                    context_tags: List[str]) -> CognitiveCrystal:
        """L3: GraphRAG风格认知晶体生成"""
        
        # 构建提示：要求LLM提取实体-关系-决策模式
        synthesis_prompt = self._build_synthesis_prompt(chunks, context_tags)
        
        response = await self.client.chat.completions.create(
            model=self.llm_model,
            messages=[
                {"role": "system", "content": "你是一个认知蒸馏专家。将文档转化为结构化知识晶体，提取实体、关系、决策模式，并标记潜在矛盾。"},
                {"role": "user", "content": synthesis_prompt}
            ],
            response_format={"type": "json_object"},
            temperature=0.1
        )
        
        result = json.loads(response.choices[0].message.content)
        
        # 计算压缩率
        original_length = sum(c["word_count"] for c in chunks)
        compressed_length = len(json.dumps(result))
        ratio = compressed_length / original_length if original_length > 0 else 0
        
        # 计算图腾亲和度（基于内容关键词）
        totem_affinity = self._calculate_totem_affinity(result)
        
        return CognitiveCrystal(
            source_uris=[source_path],
            compression_ratio=ratio,
            primary_entities=result.get("entities", []),
            key_relations=result.get("relations", []),
            decision_patterns=result.get("decision_patterns", []),
            contradiction_flags=result.get("contradictions", []),
            activation_triggers=result.get("triggers", []) + context_tags,
            totem_affinity=totem_affinity,
            vector_fingerprint=embeddings[0]["embedding_id"] if embeddings else None
        )
    
    def _build_synthesis_prompt(self, chunks: List[Dict], context_tags: List[str]) -> str:
        """构建合成提示"""
        content_sample = "\n\n".join([
            f"[{c['section']}] {c['content'][:500]}" 
            for c in chunks[:10]  # 取前10块作为样本
        ])
        
        return f"""分析以下文档内容，提取结构化知识：

上下文标签：{', '.join(context_tags)}

文档内容：
{content_sample}

请输出JSON格式：
{{
    "entities": ["实体1", "实体2", ...],
    "relations": [
        {{"subject": "实体1", "predicate": "关系", "object": "实体2", "confidence": 0.9}}
    ],
    "decision_patterns": ["当X发生时，应采取Y", ...],
    "contradictions": ["与常识/其他文档冲突的点", ...],
    "triggers": ["激活此知识的场景关键词"],
    "abstract": "100字内核心摘要"
}}"""
    
    def _calculate_totem_affinity(self, result: Dict) -> Dict[TotemType, float]:
        """基于内容计算与五路图腾的亲和度（简单启发式，可训练模型替代）"""
        text = json.dumps(result)
        affinity = {
            TotemType.CONFUCIUS: 0.5,
            TotemType.SIMON: 0.5,
            TotemType.GUANYIN: 0.5,
            TotemType.LIUYUXI: 0.5,
            TotemType.HUINENG: 0.5
        }
        
        # 启发式规则（示例）
        ethics_keywords = ["伦理", "道德", "长期", "信任", "责任"]
        if any(k in text for k in ethics_keywords):
            affinity[TotemType.CONFUCIUS] = 0.9
            
        risk_keywords = ["风险", "危机", "黑天鹅", "失败", "漏洞"]
        if any(k in text for k in risk_keywords):
            affinity[TotemType.GUANYIN] = 0.9
            
        return affinity

    async def batch_process(self, 
                          directory: str, 
                          pattern: str = "*.docx",
                          max_concurrent: int = 3) -> List[CognitiveCrystal]:
        """批量处理目录，带限流"""
        files = list(Path(directory).glob(pattern))
        semaphore = asyncio.Semaphore(max_concurrent)
        
        async def process_with_limit(f):
            async with semaphore:
                return await self.process_docx(str(f))
        
        tasks = [process_with_limit(f) for f in files]
        return await asyncio.gather(*tasks)

"""
时间晶体存储系统
混合架构：Kùzu（图）+ Chroma（向量）+ SQLite（元数据）
"""

import kuzu
import chromadb
from datetime import datetime, timedelta
from typing import List, Optional
import sqlite3
import json
from pathlib import Path

class TemporalCrystalStore:
    """
    时间晶体仓库
    支持因果查询、叙事重构、记忆巩固
    """
    
    def __init__(self, 
                 db_path: str = "./temporal_db",
                 vector_collection: str = "temporal_vectors"):
        self.db_path = Path(db_path)
        self.db_path.mkdir(exist_ok=True)
        
        # 初始化Kùzu图数据库（嵌入式）
        self.kuzu_db = kuzu.Database(str(self.db_path / "kuzu_graph"))
        self.kuzu_conn = kuzu.Connection(self.kuzu_db)
        self._init_graph_schema()
        
        # 初始化Chroma向量库
        self.chroma_client = chromadb.PersistentClient(path=str(self.db_path / "chroma"))
        self.vector_collection = self.chroma_client.get_or_create_collection(
            name=vector_collection,
            metadata={"hnsw:space": "cosine"}
        )
        
        # SQLite用于快速元数据查询
        self.sqlite_conn = sqlite3.connect(str(self.db_path / "metadata.db"))
        self._init_sqlite_schema()
    
    def _init_graph_schema(self):
        """初始化图结构：节点（Event, Crystal, Entity），边（CAUSED, REFERENCES, BELONGS_TO）"""
        schemas = [
            "CREATE NODE TABLE IF NOT EXISTS Event(event_id STRING, timestamp TIMESTAMP, semantic_time STRING, event_type STRING, content STRING, strength DOUBLE, narrative_cluster STRING, PRIMARY KEY(event_id))",
            "CREATE NODE TABLE IF NOT EXISTS Crystal(crystal_id STRING, compression_ratio DOUBLE, created_at TIMESTAMP, PRIMARY KEY(crystal_id))",
            "CREATE REL TABLE IF NOT EXISTS CAUSED(FROM Event TO Event, MANY_MANY)",
            "CREATE REL TABLE IF NOT EXISTS REFERENCES(FROM Event TO Crystal, MANY_MANY)",
            "CREATE REL TABLE IF NOT EXISTS EVOLVED_FROM(FROM Crystal TO Crystal, MANY_MANY)"
        ]
        for schema in schemas:
            try:
                self.kuzu_conn.execute(schema)
            except Exception as e:
                if "already exists" not in str(e):
                    raise
    
    def _init_sqlite_schema(self):
        cursor = self.sqlite_conn.cursor()
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS temporal_events (
                event_id TEXT PRIMARY KEY,
                timestamp REAL,
                semantic_time TEXT,
                event_type TEXT,
                content TEXT,
                crystal_refs TEXT,  -- JSON array
                causal_parents TEXT, -- JSON array
                narrative_cluster TEXT,
                strength REAL,
                access_count INTEGER DEFAULT 0
            )
        """)
        self.sqlite_conn.commit()
    
    def store_event(self, crystal: TemporalCrystal):
        """存储事件到三层存储"""
        # 1. 图数据库存储结构
        query = f"""
            CREATE (e:Event {{
                event_id: '{crystal.event_id}',
                timestamp: timestamp('{crystal.timestamp.isoformat()}'),
                semantic_time: '{crystal.semantic_time}',
                event_type: '{crystal.event_type}',
                content: '{crystal.content.replace("'", "''")[:1000]}',
                strength: {crystal.strength},
                narrative_cluster: '{crystal.narrative_cluster}'
            }})
        """
        self.kuzu_conn.execute(query)
        
        # 建立因果边
        for parent_id in crystal.causal_parents:
            edge_query = f"""
                MATCH (e1:Event {{event_id: '{parent_id}'}}), (e2:Event {{event_id: '{crystal.event_id}'}})
                CREATE (e1)-[:CAUSED]->(e2)
            """
            try:
                self.kuzu_conn.execute(edge_query)
            except:
                pass  # 父节点可能尚不存在
        
        # 2. 向量库存储语义（用于相似性搜索）
        self.vector_collection.add(
            ids=[crystal.event_id],
            documents=[crystal.content],
            metadatas=[{
                "timestamp": crystal.timestamp.isoformat(),
                "narrative_cluster": crystal.narrative_cluster,
                "event_type": crystal.event_type
            }]
        )
        
        # 3. SQLite存储元数据（用于快速过滤）
        cursor = self.sqlite_conn.cursor()
        cursor.execute("""
            INSERT OR REPLACE INTO temporal_events 
            (event_id, timestamp, semantic_time, event_type, content, crystal_refs, causal_parents, narrative_cluster, strength)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            crystal.event_id,
            crystal.timestamp.timestamp(),
            crystal.semantic_time,
            crystal.event_type,
            crystal.content,
            json.dumps(crystal.crystal_refs),
            json.dumps(crystal.causal_parents),
            crystal.narrative_cluster,
            crystal.strength
        ))
        self.sqlite_conn.commit()
    
    def query_causal_chain(self, event_id: str, direction: str = "both", depth: int = 3) -> List[Dict]:
        """
        因果链查询：查找导致某事件的原因（backward）或后果（forward）
        这是传统日志系统无法实现的高维查询
        """
        if direction == "backward":
            query = f"""
                MATCH (e:Event {{event_id: '{event_id}'}})<-[:CAUSED*1..{depth}]-(ancestor:Event)
                RETURN ancestor.event_id, ancestor.content, ancestor.timestamp
                ORDER BY ancestor.timestamp
            """
        elif direction == "forward":
            query = f"""
                MATCH (e:Event {{event_id: '{event_id}'}})-[:CAUSED*1..{depth}]->(descendant:Event)
                RETURN descendant.event_id, descendant.content, descendant.timestamp
                ORDER BY descendant.timestamp
            """
        else:
            query = f"""
                MATCH (e:Event {{event_id: '{event_id}'}})<-[:CAUSED*1..{depth}]-(ancestor:Event)
                RETURN ancestor.event_id as id, ancestor.content as content, 'cause' as type
                UNION
                MATCH (e:Event {{event_id: '{event_id}'}})-[:CAUSED*1..{depth}]->(descendant:Event)
                RETURN descendant.event_id as id, descendant.content as content, 'effect' as type
            """
        
        result = self.kuzu_conn.execute(query)
        return result.get_as_df().to_dict('records')
    
    def semantic_query(self, query_text: str, narrative_filter: Optional[str] = None, top_k: int = 5) -> List[TemporalCrystal]:
        """语义搜索+元数据过滤"""
        where_filter = {"narrative_cluster": narrative_filter} if narrative_filter else None
        
        results = self.vector_collection.query(
            query_texts=[query_text],
            n_results=top_k,
            where=where_filter
        )
        
        events = []
        for idx, event_id in enumerate(results['ids'][0]):
            # 从SQLite获取完整数据
            cursor = self.sqlite_conn.cursor()
            cursor.execute("SELECT * FROM temporal_events WHERE event_id = ?", (event_id,))
            row = cursor.fetchone()
            if row:
                events.append(self._row_to_crystal(row))
                
        return events
    
    def _row_to_crystal(self, row) -> TemporalCrystal:
        """数据库行转对象"""
        return TemporalCrystal(
            event_id=row[0],
            timestamp=datetime.fromtimestamp(row[1]),
            semantic_time=row[2],
            event_type=row[3],
            content=row[4],
            crystal_refs=json.loads(row[5]) if row[5] else [],
            causal_parents=json.loads(row[6]) if row[6] else [],
            narrative_cluster=row[7],
            strength=row[8],
            access_count=row[9]
        )
    
    def consolidate_memory(self, cluster: str):
        """
        记忆巩固：将叙事簇内的事件整合为更高阶的抽象（夜间批处理）
        模拟人脑的记忆整合（Memory Consolidation）
        """
        # 获取该簇所有事件
        cursor = self.sqlite_conn.cursor()
        cursor.execute("""
            SELECT * FROM temporal_events 
            WHERE narrative_cluster = ? 
            ORDER BY timestamp
        """, (cluster,))
        rows = cursor.fetchall()
        
        if len(rows) < 5:  # 事件太少，无需整合
            return
        
        # 简单启发式：生成摘要事件（实际可用LLM生成更高阶抽象）
        summary_content = f"[记忆整合] 叙事簇'{cluster}'在{len(rows)}个事件后形成稳定模式"
        summary_event = TemporalCrystal(
            semantic_time=f"{cluster}-整合期",
            event_type="reflection",
            content=summary_content,
            narrative_cluster=cluster,
            causal_parents=[row[0] for row in rows[-3:]]  # 连接最近3个事件
        )
        
        self.store_event(summary_event)
        
        # 强化连接（增加strength）
        for row in rows:
            cursor.execute("""
                UPDATE temporal_events SET strength = MIN(1.0, strength + 0.1)
                WHERE event_id = ?
            """, (row[0],))
        self.sqlite_conn.commit()

class AntagonisticParliament:
    """
    对抗议会：五路图腾的辩论引擎
    实现多智能体对抗设计（Antagonistic Design）

            ConfuciusAgent(),
            SimonAgent(),
            GuanyinAgent(),
            # LiuYuxiAgent(),
            # HuiNengAgent()
        ]
        self.debate_history: List[Dict] = []
        self.max_rounds = 3
    
    async def deliberate(self, proposal: Dict[str, Any], context: Dict) -> Dict[str, Any]:
        """
        议会审议流程：
        1. 首轮：各图腾独立评估
        2. 次轮：对抗辩论（攻击-防御）
        3. 末轮：共识或否决
        """
        current_round = 0
        final_verdict = {"status": "pending", "arguments": [], "consensus": None}
        
        while current_round < self.max_rounds:
            round_arguments = []
            
            # 每个图腾评估
            for agent in self.agents:
                argument = agent.evaluate(proposal, context)
                round_arguments.append(argument)
                
                if argument.severity == "blocking":
                    final_verdict["status"] = "hard_veto"
                    final_verdict["arguments"].append(argument)
                    return final_verdict  # 硬否决，立即终止
            
            # 检查是否达成共识（无critical以上问题）
            critical_count = sum(1 for a in round_arguments if a.severity == "critical")
            if critical_count == 0 and current_round > 0:
                final_verdict["status"] = "consensus"
                break
                
            # 准备下一轮（让图腾回应其他人的批评）
            context["previous_arguments"] = [a.dict() for a in round_arguments]
            current_round += 1
        
        final_verdict["arguments"] = round_arguments
        return final_verdict

"""
认知路由层：五路图腾作为动态路由器
替代静态流程（KIP/PMP/WEP），实现状态驱动的认知调度

import numpy as np
from typing import Dict, List, Optional
from enum import Enum
import openai

class CognitiveState(Enum):
    """系统认知状态"""
    EXPLORATION = "exploration"      # 探索期：需要慧能（创新）
    CRISIS = "crisis"                # 危机期：需要观自在（风险）
    ETHICAL_DILEMMA = "ethical"      # 伦理困境：需要孔子
    RESOURCE_CONSTRAINT = "resource" # 资源受限：需要司马贺
    SOCIAL_BINDING = "social"        # 关系构建：需要刘禹锡
    ROUTINE = "routine"              # 常规处理：满意解模式

class CognitiveRouter:
    """
    认知路由器：基于输入语义和系统状态，动态路由到处理管道
    这是新架构的核心，取代原有的静态流程
    """
    
    def __init__(self, temporal_store, parliament):
        self.temporal_store = temporal_store
        self.parliament = parliament
        self.client = openai.OpenAI(
            base_url="https://models.inference.ai.azure.com",
            api_key=os.getenv("GITHUB_TOKEN")
        )
        
        # 状态转移矩阵（简化版，可训练）
        self.state_transitions = {
            CognitiveState.ROUTINE: {
                "high_risk_signal": CognitiveState.CRISIS,
                "ethical_keywords": CognitiveState.ETHICAL_DILEMMA,
                "stuck_repeated": CognitiveState.EXPLORATION
            }
        }
    
    async def route(self, user_input: str, current_context: Dict) -> Dict:
        """
        主路由函数：
        1. 语义意图识别
        2. 认知状态评估
        3. 图腾权重分配
        4. 执行管道选择
        """
        # 1. 意图向量化
        intent_vector = await self._embed_intent(user_input)
        
        # 2. 检测当前认知状态
        current_state = await self._detect_cognitive_state(
            user_input, current_context
        )
        
        # 3. 计算图腾激活权重（Softmax）
        totem_weights = self._calculate_totem_weights(
            intent_vector, current_state, current_context
        )
        
        # 4. 路由决策
        pipeline = self._select_pipeline(totem_weights, current_state)
        
        # 5. 记录事件（形成因果链）
        event_id = await self._record_routing_event(
            user_input, current_state, totem_weights, pipeline
        )
        
        return {
            "pipeline": pipeline,
            "totem_weights": totem_weights,
            "cognitive_state": current_state.value,
            "event_id": event_id,
            "routing_reason": self._generate_rationale(totem_weights, current_state)
        }
    
    async def _embed_intent(self, text: str) -> List[float]:
        """语义嵌入"""
        response = self.client.embeddings.create(
            model="text-embedding-3-large",
            input=text,
            dimensions=3072
        )
        return response.data[0].embedding
    
    async def _detect_cognitive_state(self, text: str, context: Dict) -> CognitiveState:
        """
        基于内容和历史检测认知状态
        使用LLM做零样本分类，比规则更准确
        """
        recent_events = self.temporal_store.semantic_query(
            text, top_k=3
        )
        
        prompt = f"""基于用户输入和最近事件，判断当前认知状态：

用户输入：{text}
最近相关事件：{[e.content[:100] for e in recent_events]}
当前上下文：{context}

可选状态：
- exploration: 探索新事物，需要创新
- crisis: 检测到风险或危机
- ethical: 涉及伦理判断
- resource: 资源受限，需要节俭方案
- social: 涉及人际关系
- routine: 常规任务

仅输出状态代码（如"crisis"）"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",  # 轻量级模型足够
            messages=[{"role": "user", "content": prompt}],
            temperature=0.0
        )
        
        state_str = response.choices[0].message.content.strip().lower()
        try:
            return CognitiveState(state_str)
        except ValueError:
            return CognitiveState.ROUTINE
    
    def _calculate_totem_weights(self, 
                                 intent_vector: List[float],
                                 state: CognitiveState,
                                 context: Dict) -> Dict[str, float]:
        """
        计算五路图腾的激活权重
        结合：语义相似度 + 状态匹配 + 历史偏好
        """
        # 基础权重（由状态决定）
        base_weights = {
            "confucius": 0.2,
            "simon": 0.2,
            "guanyin": 0.2,
            "liuyuxi": 0.2,
            "huineng": 0.2
        }
        
        # 状态加成
        state_boost = {
            CognitiveState.ETHICAL_DILEMMA: {"confucius": 0.6},
            CognitiveState.RESOURCE_CONSTRAINT: {"simon": 0.6},
            CognitiveState.CRISIS: {"guanyin": 0.6},
            CognitiveState.SOCIAL_BINDING: {"liuyuxi": 0.6},
            CognitiveState.EXPLORATION: {"huineng": 0.6}
        }
        
        if state in state_boost:
            for totem, boost in state_boost[state].items():
                base_weights[totem] += boost
        
        # Softmax归一化
        exp_scores = {k: np.exp(v) for k, v in base_weights.items()}
        total = sum(exp_scores.values())
        return {k: v/total for k, v in exp_scores.items()}
    
    def _select_pipeline(self, weights: Dict[str, float], state: CognitiveState) -> str:
        """基于权重选择处理管道"""
        max_totem = max(weights, key=weights.get)
        
        # 映射图腾到管道
        pipeline_map = {
            "confucius": "ethical_audit",      # 伦理审查管道
            "simon": "satisficing_process",    # 满意解快速通道
            "guanyin": "risk_analysis",        # 深度风险分析
            "liuyuxi": "relationship_mapping", # 关系图谱构建
            "huineng": "lateral_thinking"      # 横向突破模式
        }
        
        # 如果是常规状态且权重分散，使用HyperCompression标准流程
        if state == CognitiveState.ROUTINE and weights[max_totem] < 0.4:
            return "hyper_compression"
            
        return pipeline_map.get(max_totem, "standard")
    
    async def _record_routing_event(self, 
                                    input_text: str,
                                    state: CognitiveState,
                                    weights: Dict,
                                    pipeline: str) -> str:
        """记录路由决策为时间晶体事件"""
        from base.crystal_models import TemporalCrystal
        
        event = TemporalCrystal(
            semantic_time=f"路由决策-{datetime.now().strftime('%H%M')}",
            event_type="decision",
            content=f"路由到{pipeline}，主导图腾{max(weights, key=weights.get)}",
            narrative_cluster="system_routing",
            strength=0.8
        )
        
        self.temporal_store.store_event(event)
        return event.event_id
    
    def _generate_rationale(self, weights: Dict, state: CognitiveState) -> str:
        """生成可解释的路由理由"""
        dominant = max(weights, key=weights.get)
        return f"检测到{state.value}状态，激活{dominant}图腾（权重{weights[dominant]:.2f}）"

        return result[0] if result else 0
    
    def _log_api_call(self, model: str, count: int = 1):
        """记录API调用"""
        today = datetime.now().strftime("%Y-%m-%d")
        conn = sqlite3.connect(self.checkpoint_db)
        cursor = conn.cursor()
        cursor.execute("""
            INSERT INTO api_usage (date, model, calls) 
            VALUES (?, ?, ?)
            ON CONFLICT(date, model) 
            DO UPDATE SET calls = calls + ?
        """, (today, model, count, count))
        conn.commit()
        conn.close()
        
        # 更新内存计数
        if model == "gpt-4o":
            self.used_gpt4o += count
        else:
            self.used_mini += count
    
    def scan_and_enqueue(self) -> int:
        """
        扫描目录，建立处理队列
        优先级策略：文件名含"需求/蓝图/SOUL" > 含"报告/研究" > 其他
        """
        if not self.docx_dir.exists():
            raise FileNotFoundError(f"目录不存在: {self.docx_dir}")
        
        docx_files = list(self.docx_dir.glob("*.docx"))
        print(f"🔍 发现 {len(docx_files)} 个DOCX文件")
        
        conn = sqlite3.connect(self.checkpoint_db)
        cursor = conn.cursor()
        
        new_count = 0
        for file_path in docx_files:
            # 计算文件哈希（用于断点续传识别）
            with open(file_path, 'rb') as f:
                file_hash = hashlib.md5(f.read()).hexdigest()
            
            # 检查是否已存在
            cursor.execute("SELECT status FROM jobs WHERE file_hash = ?", (file_hash,))
            existing = cursor.fetchone()
            
            if existing:
                continue  # 已存在，跳过
            
            # 计算优先级
            filename = file_path.name.lower()
            if any(k in filename for k in ['soul', 'user', '需求', '蓝图', '架构']):
                priority = 1
            elif any(k in filename for k in ['报告', '研究', '分析', '洞察']):
                priority = 2
            elif any(k in filename for k in ['案例', '知识', '资产']):
                priority = 3
            else:
                priority = 5
            
            # 插入新作业
            cursor.execute("""
                INSERT INTO jobs (file_hash, file_path, priority, status)
                VALUES (?, ?, ?, 'pending')
            """, (str(file_path), file_hash, priority))
            new_count += 1
        
        conn.commit()
        conn.close()
        print(f"✅ 新增 {new_count} 个作业到队列")
        return new_count
    
    def get_next_batch(self, batch_size: int = 5) -> List[ProcessingJob]:
        """获取下一批待处理作业（按优先级）"""
        conn = sqlite3.connect(self.checkpoint_db)
        cursor = conn.cursor()
        
        # 优先级1-2的作业优先使用GPT-4o，其余用mini
        cursor.execute("""
            SELECT file_path, file_hash, priority, status, crystal_id, error_msg, processed_at, api_calls_used
            FROM jobs 
            WHERE status = 'pending'
            ORDER BY priority ASC, created_at ASC
            LIMIT ?
        """, (batch_size,))
        
        rows = cursor.fetchall()
        conn.close()
        
        return [ProcessingJob(*row) for row in rows]
    
    def update_job_status(self, file_hash: str, status: str, 
                         crystal_id: Optional[str] = None,
                         error_msg: Optional[str] = None,
                         api_calls: int = 0):
        """更新作业状态（原子操作）"""
        conn = sqlite3.connect(self.checkpoint_db)
        cursor = conn.cursor()
        
        cursor.execute("""
            UPDATE jobs 
            SET status = ?, crystal_id = ?, error_msg = ?, 
                processed_at = ?, api_calls_used = ?
            WHERE file_hash = ?
        """, (status, crystal_id, error_msg, 
              datetime.now().isoformat(), api_calls, file_hash))
        
        conn.commit()
        conn.close()
    
    def get_progress_report(self) -> Dict:
        """获取进度报告"""
        conn = sqlite3.connect(self.checkpoint_db)
        cursor = conn.cursor()
        
        cursor.execute("""
            SELECT status, COUNT(*) FROM jobs GROUP BY status
        """)
        status_counts = dict(cursor.fetchall())
        
        cursor.execute("""
            SELECT priority, COUNT(*) FROM jobs WHERE status = 'pending' GROUP BY priority
        """)
        pending_by_priority = dict(cursor.fetchall())
        
        conn.close()
        
        total = sum(status_counts.values())
        completed = status_counts.get('completed', 0)
        failed = status_counts.get('failed', 0)
        
        return {
            "total_files": total,
            "completed": completed,
            "failed": failed,
            "pending": status_counts.get('pending', 0),
            "progress_pct": (completed / total * 100) if total > 0 else 0,
            "pending_by_priority": pending_by_priority,
            "api_quota_remaining": {
                "gpt-4o": self.quota_gpt4o - self.used_gpt4o,
                "gpt-4o-mini": self.quota_mini - self.used_mini
            },
            "estimated_days_remaining": self._estimate_completion_time(total, completed)
        }
    
    def _estimate_completion_time(self, total: int, completed: int) -> float:
        """估算剩余天数（基于API配额）"""
        remaining = total - completed
        daily_capacity = self.quota_gpt4o + self.quota_mini  # 保守估计
        return remaining / daily_capacity if daily_capacity > 0 else float('inf')

### 第二步：自适应压缩引擎（配额感知）

"""
自适应压缩引擎
根据剩余API配额动态选择模型策略
"""

import openai
import asyncio
from typing import Optional
from base.crystal_models import CognitiveCrystal

class AdaptiveCompressor:
    """
    配额感知的压缩引擎
    策略：
    - P1-P2优先级：优先使用GPT-4o，配额耗尽则降级到mini
    - P3-P5优先级：直接使用GPT-4o-mini
    """
    
    def __init__(self, orchestrator):
        self.orch = orchestrator
        self.client = openai.AsyncOpenAI(
            base_url="https://models.inference.ai.azure.com",
            api_key=os.getenv("GITHUB_TOKEN")
        )
    
    async def compress_with_adaptive_model(self, 
                                          file_path: str, 
                                          priority: int) -> Optional[CognitiveCrystal]:
        """
        自适应压缩：根据配额和优先级选择模型
        """
        # 策略决策
        if priority <= 2 and self.orch.used_gpt4o < self.orch.quota_gpt4o:
            model = "gpt-4o"
            print(f"  🎯 使用GPT-4o处理高优先级文件: {Path(file_path).name}")
        elif self.orch.used_mini < self.orch.quota_mini:
            model = "gpt-4o-mini"
            print(f"  ⚡ 使用GPT-4o-mini: {Path(file_path).name}")
        else:
            # 配额耗尽
            raise QuotaExhaustedError("今日API配额已耗尽，请明日继续")
        
        try:
            # L1: 结构拆解（本地处理，不消耗API）
            chunks = await self._l1_extract(file_path)
            
            # L2: 向量化（消耗1次embedding配额）
            embedding = await self._l2_embed(chunks)
            self.orch._log_api_call("embedding")
            
            # L3: 晶体生成（消耗chat配额）
            crystal = await self._l3_synthesize(chunks, embedding, model)
            self.orch._log_api_call(model)
            
            return crystal
            
        except Exception as e:
            print(f"  ❌ 处理失败: {e}")
            return None
    
    async def _l1_extract(self, file_path: str):
        """L1: 本地结构提取（零API消耗）"""
        from docx import Document
        
        doc = Document(file_path)
        chunks = []
        
        # 提取文本（限制长度，避免超长文档）
        full_text = []
        for para in doc.paragraphs:
            if para.text.strip():
                full_text.append(para.text)
        
        # 分段处理（每段约500字）
        text = "\n".join(full_text)
        segments = [text[i:i+500] for i in range(0, len(text), 500)]
        
        return segments[:20]  # 最多20段，控制token
    
    async def _l2_embed(self, chunks: list):
        """L2: 批量嵌入（尽可能批量以减少API调用）"""
        # 合并短块以减少API调用次数
        batch_text = ["\n".join(chunks[i:i+5]) for i in range(0, len(chunks), 5)]
        
        response = await self.client.embeddings.create(
            model="text-embedding-3-large",
            input=batch_text[:3],  # 一次最多3批，控制消耗
            dimensions=3072
        )
        
        return response.data[0].embedding if response.data else None
    
    async def _l3_synthesize(self, chunks, embedding, model: str):
        """L3: 认知晶体生成"""
        # 构建提示（截断控制）
        content_sample = "\n\n".join(chunks[:5])  # 前5段作为样本
        
        prompt = f"""将以下文档内容蒸馏为认知晶体。提取：1)核心实体 2)关键关系 3)决策模式 4)潜在矛盾。

文档内容：
{content_sample[:3000]}  # 硬截断保护

输出JSON格式：
{{
    "entities": ["实体1", "实体2"],
    "relations": [{{"subject": "A", "predicate": "影响", "object": "B"}}],
    "decision_patterns": ["当X时，Y发生"],
    "contradictions": ["与常规认知冲突的点"],
    "triggers": ["关键词1", "关键词2"],
    "abstract": "100字内摘要"
}}"""

        response = await self.client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": "你是认知蒸馏专家。严格输出JSON。"},
                {"role": "user", "content": prompt}
            ],
            response_format={"type": "json_object"},
            temperature=0.1,
            max_tokens=1500  # 限制输出长度
        )
        
        result = json.loads(response.choices[0].message.content)
        
        # 构建晶体
        return CognitiveCrystal(
            source_uris=[file_path],
            compression_ratio=0.15,
            primary_entities=result.get("entities", [])[:10],  # 限制数量
            key_relations=result.get("relations", [])[:5],
            decision_patterns=result.get("decision_patterns", [])[:3],
            contradiction_flags=result.get("contradictions", [])[:2],
            activation_triggers=result.get("triggers", [])[:5],
            vector_fingerprint=str(embedding[:10]) if embedding else None  # 简化存储
        )

def signal_handler(sig, frame):

    global is_interrupted
    print("\n⚠️ 收到中断信号，当前文件处理完成后将保存状态并退出...")
    is_interrupted = True

signal.signal(signal.SIGINT, signal_handler)

async def main():

    # 初始化编排器
    orchestrator = BatchOrchestrator(
        docx_directory="/mnt/kimi/upload",  # 根据实际路径调整
        daily_quota_gpt4o=45,
        daily_quota_mini=140
    )
    
    # 1. 扫描并建立队列（仅首次运行需要，后续自动跳过已存在）
    orchestrator.scan_and_enqueue()
    
    # 2. 显示当前状态
    report = orchestrator.get_progress_report()
    print(f"\n📊 当前进度:")

    print(f"   进度: {report['progress_pct']:.1f}%")
    print(f"   API剩余: GPT-4o({report['api_quota_remaining']['gpt-4o']}) | Mini({report['api_quota_remaining']['gpt-4o-mini']})")
    print(f"   预计还需: {report['estimated_days_remaining']:.1f} 天\n")
    
    if report['pending'] == 0:
        print("✅ 所有文件处理完毕！")
        return
    
    # 3. 初始化压缩器
    compressor = AdaptiveCompressor(orchestrator)
    
    # 4. 批处理循环
    batch_size = 3  # 小批次，便于中断
    processed_in_session = 0
    
    while not is_interrupted:
        # 获取下一批
        batch = orchestrator.get_next_batch(batch_size)
        
        if not batch:
            print("✅ 队列为空，处理完成！")
            break
        
        for job in batch:
            if is_interrupted:
                break
            
            print(f"🔄 处理: {Path(job.file_path).name} [P{job.priority}]")
            
            try:
                # 更新状态为处理中
                orchestrator.update_job_status(job.file_hash, "processing")
                
                # 执行压缩（带限流）
                async with orchestrator.semaphore:
                    crystal = await compressor.compress_with_adaptive_model(
                        job.file_path, 
                        job.priority
                    )
                
                if crystal:
                    # 保存晶体（JSON格式）
                    crystal_path = f"./data/crystals/{crystal.crystal_id}.json"
                    Path(crystal_path).parent.mkdir(parents=True, exist_ok=True)
                    
                    with open(crystal_path, 'w', encoding='utf-8') as f:
                        json.dump(crystal.dict(), f, ensure_ascii=False, indent=2)
                    
                    # 更新状态为完成
                    orchestrator.update_job_status(
                        job.file_hash, 
                        "completed",
                        crystal_id=crystal.crystal_id,
                        api_calls=2 if job.priority <= 2 else 1  # 记录消耗
                    )
                    print(f"  ✅ 完成: {crystal.crystal_id} (压缩率: {crystal.compression_ratio:.2f})")
                    processed_in_session += 1
                else:
                    orchestrator.update_job_status(
                        job.file_hash, 
                        "failed",
                        error_msg="压缩返回空值"
                    )
                    
            except QuotaExhaustedError as e:
                print(f"  ⛔ {e}")
                print("  💤 进入休眠模式，请明日重新运行脚本...")
                return
            
            except Exception as e:
                print(f"  ❌ 错误: {str(e)[:100]}")
                orchestrator.update_job_status(
                    job.file_hash, 
                    "failed",
                    error_msg=str(e)[:500]
                )
        
        # 批次间短暂休息（避免GitHub限流）
        await asyncio.sleep(2)
    
    # 5. 最终报告
    final_report = orchestrator.get_progress_report()
    print(f"\n📈 本会话处理: {processed_in_session} 个文件")
    print(f"📊 总进度: {final_report['progress_pct']:.1f}%")
    print(f"💾 状态已保存至: {orchestrator.checkpoint_db}")
    print("🔄 随时可重新运行脚本继续处理")

if __name__ == "__main__":
    # 检查环境

        print("   请执行: export GITHUB_TOKEN='你的令牌'")
        sys.exit(1)
    
    # 运行
    asyncio.run(main())

def ingest_crystals_to_temporal():
    """将晶体转化为时间事件并入库"""
    
    store = TemporalCrystalStore()
    crystal_dir = Path("./data/crystals")
    
    # 按修改时间排序（模拟处理时序）
    crystal_files = sorted(crystal_dir.glob("*.json"), 
                          key=lambda x: x.stat().st_mtime)
    
    print(f"🧬 开始构建TemporalCrystal索引: {len(crystal_files)}个晶体")
    
    previous_batch = []  # 用于建立因果链
    
    for idx, cf in enumerate(crystal_files):
        with open(cf, 'r', encoding='utf-8') as f:
            data = json.load(f)
        
        # 创建TemporalCrystal事件
        event = TemporalCrystal(
            semantic_time=f"HyperCompression批次-{data['crystal_id'][:8]}",
            event_type="perception",
            content=f"文档蒸馏完成: {data['source_uris'][0].split('/')[-1]}",
            crystal_refs=[data['crystal_id']],
            narrative_cluster=detect_narrative_cluster(data),  # 自动分类
            causal_parents=[previous_batch[-1]] if previous_batch else [],
            strength=0.9 if data.get('priority') in [1,2] else 0.6
        )
        
        store.store_event(event)
        
        # 每10个晶体形成一个批次因果链
        if idx % 10 == 0:
            previous_batch = [event.event_id]
        else:
            previous_batch.append(event.event_id)
        
        if idx % 20 == 0:
            print(f"  已入库: {idx}/{len(crystal_files)}")
    
    print("✅ TemporalCrystal索引构建完成")
    
    # 执行首次记忆巩固（夜间批处理的预演）
    print("🌙 执行记忆巩固 (Memory Consolidation)...")
    store.consolidate_memory("knowledge_ingestion")
    
    # 验证查询能力
    print("\n🔍 验证高维查询:")
    results = store.semantic_query("蓝军审计机制", top_k=3)
    print(f"   语义搜索返回: {len(results)}个相关事件")
    
    causal = store.query_causal_chain(results[0].event_id, direction="backward", depth=2)
    print(f"   因果链追溯: 发现{causal}个前序事件")

def detect_narrative_cluster(crystal_data: dict) -> str:
    """自动识别叙事簇（知识图谱分类）"""
    triggers = crystal_data.get('activation_triggers', [])
    entities = crystal_data.get('primary_entities', [])
    
    if any(t in ['系统架构', '蓝军', '流程'] for t in triggers):
        return "system_architecture"
    elif any(t in ['合伙人', '匹配', '估值'] for t in triggers):
        return "partner_matching"
    elif any(t in ['记忆', '知识', 'DOCX'] for t in triggers):
        return "knowledge_management"
    elif any('图腾' in e for e in entities):
        return "totem_system"
    else:
        return "general_operations"

    def calibrate_totem_weights(self, samples: List[Dict]) -> Dict:

        correct = 0
        total = min(len(test_cases), self.budget - self.used - 5)  # 留5次buffer

            # 构造验证提示
            prompt = f"""基于以下校准规则，判断输入应该路由到哪个图腾。

仅输出图腾名称 (confucius/simon/guanyin/liuyuxi/huineng):"""

            predicted = response.choices[0].message.content.strip().lower()

            else:

                break

        return accuracy, self.used

        print(f"   验证准确率: {accuracy:.1%}")

"""

加载校准权重，替换硬编码启发式规则
"""

import json
import numpy as np
from typing import Dict
from base.crystal_models import TotemType, CognitiveState

class CalibratedCognitiveRouter:
    """
    校准后的认知路由器
    使用训练得到的权重配置替代硬编码规则
    """
    
    def __init__(self, calibration_file: str = "./data/router_calibration_v1.json"):
        with open(calibration_file, 'r', encoding='utf-8') as f:
            self.calibration = json.load(f)
        
        self.weight_profiles = self.calibration["totem_weight_profiles"]
        self.state_keywords = self._parse_state_rules(
            self.calibration["state_detection_rules"]
        )
        
        print(f"🧭 路由器已加载校准配置 (准确率: {self.calibration['validation_accuracy']:.1%})")
    
    def _parse_state_rules(self, rules_str: str) -> Dict:
        """解析状态规则字符串"""
        # 简化解析，实际应解析JSON
        return {
            "crisis": ["删除", "清空", "风险", "错误", "失败"],
            "exploration": ["突破", "新", "创新"],
            "ethical_dilemma": ["伦理", "价值观", "长期"],
            "resource_constraint": ["优化", "成本", "预算", "足够好"],
            "social_binding": ["关系", "偏好", "习惯"],
            "routine": ["处理", "查询", "搜索"]
        }
    
    def detect_state(self, text: str, context: Dict) -> CognitiveState:
        """
        状态检测：基于校准的关键词+简单启发式
        不调用API（零成本），本地执行
        """
        text_lower = text.lower()
        
        scores = {}
        for state, keywords in self.state_keywords.items():
            scores[state] = sum(1 for k in keywords if k in text_lower)
        
        # 上下文覆盖
        if context.get("risk_level") == "high":
            scores["crisis"] += 3
        if context.get("stuck_iterations", 0) > 3:
            scores["exploration"] += 3
            
        max_state = max(scores, key=scores.get)
        return CognitiveState(max_state) if scores[max_state] > 0 else CognitiveState.ROUTINE
    
    def calculate_weights(self, state: CognitiveState) -> Dict[TotemType, float]:
        """
        权重计算：基于校准配置使用Softmax
        """
        state_key = state.value
        if state_key not in self.weight_profiles:
            # 默认均匀分布
            return {t: 0.2 for t in TotemType}
        
        profile = self.weight_profiles[state_key]["weight_distribution"]
        
        # 确保所有图腾都有权重
        weights = {}
        for totem in TotemType:
            weights[totem] = profile.get(totem.value, 0.2)
        
        # Softmax归一化
        exp_scores = {k: np.exp(v) for k, v in weights.items()}
        total = sum(exp_scores.values())
        return {k: v/total for k, v in exp_scores.items()}
    
    def route(self, user_input: str, context: Dict = None) -> Dict:
        """
        主路由函数：零API成本，本地执行
        """
        context = context or {}
        
        # 1. 状态检测（本地）
        state = self.detect_state(user_input, context)
        
        # 2. 权重计算（本地）
        weights = self.calculate_weights(state)
        
        # 3. 选择主导图腾
        dominant = max(weights, key=weights.get)
        
        # 4. 映射到执行管道
        pipeline_map = {
            TotemType.CONFUCIUS: "ethical_audit",
            TotemType.SIMON: "satisficing_process",
            TotemType.GUANYIN: "risk_analysis",
            TotemType.LIUYUXI: "relationship_mapping",
            TotemType.HUINENG: "lateral_thinking"
        }
        
        return {
            "pipeline": pipeline_map[dominant],
            "totem_weights": {t.value: round(w, 3) for t, w in weights.items()},
            "dominant_totem": dominant.value,
            "cognitive_state": state.value,
            "confidence": round(weights[dominant], 3),
            "api_cost": 0,  # 本地执行，零API消耗
            "calibration_version": self.calibration["version"]
        }

from datetime import datetime

class CrystalVersionControl:
    """
    晶体版本控制
    解决知识更新时的冲突与演化问题
    """
    
    def __init__(self):
        self.versions = {}  # crystal_id -> list of versions
    
    def commit_crystal(self, 
                      crystal: CognitiveCrystal, 
                      commit_msg: str,
                      parent_ids: List[str] = None) -> str:
        """
        提交晶体新版本（类似Git commit）
        """
        version_id = f"v{len(self.versions.get(crystal.crystal_id, [])) + 1}"
        
        version = {
            "version_id": version_id,
            "crystal": crystal,
            "commit_msg": commit_msg,
            "parent_ids": parent_ids or [],
            "timestamp": datetime.now(),
            "hash": self._compute_hash(crystal)
        }
        
        if crystal.crystal_id not in self.versions:
            self.versions[crystal.crystal_id] = []
        
        self.versions[crystal.crystal_id].append(version)
        
        print(f"💾 晶体提交: {crystal.crystal_id}@{version_id}")
        print(f"   消息: {commit_msg}")
        
        return version_id
    
    def _compute_hash(self, crystal: CognitiveCrystal) -> str:
        """计算晶体内容哈希"""
        content = json.dumps(crystal.dict(), sort_keys=True)
        return hashlib.sha256(content.encode()).hexdigest()[:16]
    
    def diff_crystals(self, 
                     crystal_id: str, 
                     v1: str, 
                     v2: str) -> Dict:
        """
        比较两个版本的晶体差异
        用于知识更新时的审计
        """
        versions = self.versions.get(crystal_id, [])
        ver1 = next((v for v in versions if v["version_id"] == v1), None)
        ver2 = next((v for v in versions if v["version_id"] == v2), None)
        
        if not ver1 or not ver2:
            return {"error": "版本不存在"}
        
        c1, c2 = ver1["crystal"], ver2["crystal"]
        
        # 计算实体差异
        entities_added = set(c2.primary_entities) - set(c1.primary_entities)
        entities_removed = set(c1.primary_entities) - set(c2.primary_entities)
        
        # 计算关系差异
        rels_added = [r for r in c2.key_relations if r not in c1.key_relations]
        
        return {
            "from": v1,
            "to": v2,
            "entities_added": list(entities_added),
            "entities_removed": list(entities_removed),
            "relations_added": rels_added,
            "confidence_change": c2.confidence_score - c1.confidence_score,
            "commit_msgs": [ver1["commit_msg"], ver2["commit_msg"]]
        }

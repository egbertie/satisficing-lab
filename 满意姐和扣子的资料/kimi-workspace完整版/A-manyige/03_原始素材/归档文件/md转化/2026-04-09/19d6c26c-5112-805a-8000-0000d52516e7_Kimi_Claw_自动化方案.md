### 4.2 批量安装脚本（Claw可保存备用）

创建文件 ~/install_tencent_suite.sh：

#!/bin/bash
# 腾讯云SkillHub全套装自动安装脚本

SKILLS=(
  "tencentcloud-cos"
  "tencent-docs" 
  "qq-browser-skill"
  "tencent-map"
  "tencent-voice"
  "lexiang-knowledge"
  "edgeone-clawscan"
  "cos-vectors-skill"
)

echo "开始安装腾讯云SkillHub技能套装..."
for skill in "${SKILLS[@]}"; do
  echo "正在安装: $skill"
  skillhub install "$skill" || echo "警告: $skill 安装失败，尝试备用源..."
  # 备用安装方式
  if [ $? -ne 0 ]; then
    npx clawhub@latest install "$skill" 2>/dev/null || echo "$skill 需要从官网手动下载"
  fi
done

echo "安装完成，已安装技能列表："
skillhub list

echo "请配置腾讯云凭证环境变量后使用"

赋予执行权限：

chmod +x ~/install_tencent_suite.sh

## 请求1：五元组提取器算法设计

### 1.1 算法架构：双层混合提取系统

采用“规则引擎预处理 + LLM精准提取 + 置信度校验”三层架构，确保在无GPU环境下达到>80%准确率。

class PentadExtractor:
    """
    五元组提取器：情境-框架-判断-结果-反思
    设计原则：规则保下限，LLM追上限，校验防漂移
    """
    
    def __init__(self, llm_client, token_budget=5000):
        self.llm = llm_client
        self.token_budget = token_budget
        self.rules_engine = RuleEngine()
        self.validator = ConsistencyValidator()
        
    def extract(self, case_text: str) -> dict:
        # 第一层：规则引擎快速结构化
        rule_hints = self.rules_engine.pre_extract(case_text)
        
        # 第二层：LLM精准提取（带规则提示）
        structured = self.llm_extract(case_text, rule_hints)
        
        # 第三层：一致性校验与补全
        validated = self.validator.check(structured, case_text)
        
        return validated

### 1.2 规则引擎设计（防御层）

class RuleEngine:
    """
    规则引擎：识别文本中的显性标记和结构模式
    作用：提供先验约束，减少LLM幻觉
    """
    
    # 五元组的词典线索（正则模式）
    PATTERNS = {
        'situation': [
            r'(背景|情境|context|基于|面对|处于|来自).*?[。.；]',
            r'^(.*?)(?=寻找|需要|决定|选择)'  # 开头背景描述
        ],
        'decision_framework': [
            r'(通过|借助|使用|采用|基于|参考)(.*?)(方法|理论|框架|介绍|渠道)',
            r'(匹配|筛选|评估).*?(标准|维度|指标)'
        ],
        'judgment': [
            r'(决定|选择|任命|聘请|合作|签约)(.*?)(作为|担任|负责)',
            r'(经过.*?[讨论|协商|评估]).*?(确定|选定|聘请)'
        ],
        'outcome': [
            r'(结果|最终|后来|半年后|一年后)(.*?)[。.；]',
            r'(成功|失败|退出|解散|融资|收购).*?[。.；]'
        ],
        'reflection': [
            r'(反思|复盘|总结|教训|启示|意识到)(.*?)[。.；]',
            r'(应该|不该|后悔|庆幸|如果).*?[。.；]'
        ]
    }
    
    def pre_extract(self, text: str) -> dict:
        """提取线索，为LLM提供锚点"""
        hints = {}
        for field, patterns in self.PATTERNS.items():
            matches = []
            for p in patterns:
                matches.extend(re.findall(p, text, re.DOTALL))
            hints[field] = matches[:2]  # 最多取2个线索
        return hints

### 1.3 LLM提示模板（精确层）

采用“角色定义+字段说明+示例学习+输出约束”四段式提示：

【角色定义】
你是"案例解构专家"，擅长从商业案例中提取决策逻辑链。你的任务是将非结构化案例文本转化为结构化五元组。

【字段定义】
1. **情境(Situation)**: 包含时间、地点、人物背景、约束条件、不确定性因素
   - 必须要素：谁、在哪里、面临什么局面
   - 排除：不含决策动作本身

2. **决策框架(Decision Framework)**: 使用的筛选标准、信息来源、评估维度
   - 关键词：通过...介绍、基于...理论、参考...标准
   - 注意：可能是隐性的（如"朋友介绍"）

3. **判断(Judgment)**: 具体的决策动作和选择逻辑
   - 核心：选择了谁/什么方案，基于什么即时判断
   - 区分：不是结果，而是决策瞬间的动作

4. **结果(Outcome)**: 短期和长期后果，包含量化指标
   - 时间维度：立即后果 + 6个月后 + 长期影响
   - 情感/财务双维度

5. **反思(Reflection)**: 事后复盘、认知升级、可迁移经验
   - 特征：含有"应该"、"意识到"、"教训"等词
   - 深度：从具体到抽象的原则提炼

【处理规则】
- 如果某字段在文本中明确缺失，标记为"implicit"(隐性)而非编造
- 多人物场景：聚焦主要决策者的视角
- 模糊时间：标准化为相对时间（如"T+3个月"）

【示例学习】
输入："深圳AI芯片项目...（原文略）"
输出：{"situation": "...", "decision_framework": "...", ...}

【当前任务】
基于以下规则线索：{rule_hints}
请提取以下案例的五元组（严格JSON格式）：

案例文本：
{case_text}

要求：
1. 每个字段30-100字，精准概括
2. 必须包含原文关键词，不臆测
3. 如字段缺失，值设为"IMPLICIT"
4. 输出纯JSON，无markdown标记

### 1.4 后处理与验证系统

class ConsistencyValidator:
    """
    一致性校验：逻辑检查+缺失补全+冲突消解
    """
    
    def check(self, extraction: dict, original: str) -> dict:
        validated = extraction.copy()
        
        # 1. 完整性校验
        required_fields = ['situation', 'decision_framework', 'judgment', 'outcome', 'reflection']
        for field in required_fields:
            if field not in validated or not validated[field]:
                validated[field] = "IMPLICIT"
        
        # 2. 逻辑一致性检查（基于文本长度和关键词）
        validated = self._check_temporal_order(validated, original)
        validated = self._check_actor_consistency(validated, original)
        
        # 3. 置信度评分
        validated['_confidence'] = self._calculate_confidence(validated, original)
        
        return validated
    
    def _check_temporal_order(self, data: dict, text: str) -> dict:
        """检查时间顺序：情境→框架→判断→结果→反思"""
        # 简单启发：反思字段不应包含"后来"等未来时态词（除非引用）
        if '后来' in data.get('reflection', '') and '反思' not in data.get('reflection', ''):
            # 可能误判，标记审查
            data['_warning'] = 'temporal_anomaly'
        return data
    
    def _calculate_confidence(self, data: dict, text: str) -> float:
        """基于文本覆盖率和字段完整性计算置信度"""
        scores = []
        for field, value in data.items():
            if field.startswith('_'):
                continue
            if value == "IMPLICIT":
                scores.append(0.5)
            else:
                # 检查关键词是否在原文出现
                keywords = set(jieba.lcut(value))  # 使用jieba分词
                text_words = set(text.split())
                overlap = len(keywords & text_words) / len(keywords) if keywords else 0
                scores.append(overlap)
        return round(sum(scores) / len(scores), 2)

import re

class PentadExtractor:
    """
    五元组提取器完整实现
    约束适配：CPU only, <5s/案例, ~5000 tokens
    """
    
    def __init__(self, llm_api_call, token_budget=5000):
        """
        llm_api_call: 函数签名 (prompt: str) -> str
        """
        self.llm_call = llm_api_call
        self.token_budget = token_budget
        self.rules = RuleEngine()
        
    def extract(self, case_text: str, verbose=False) -> Dict:
        """主入口：提取五元组"""
        # 步骤1: 规则预处理（<100ms）
        hints = self.rules.pre_extract(case_text)
        
        # 步骤2: 构建优化提示（控制token）
        prompt = self._build_prompt(case_text, hints, self.token_budget)
        
        # 步骤3: LLM提取（占主要时间）
        try:
            response = self.llm_call(prompt)
            structured = self._parse_json(response)
        except Exception as e:
            structured = self._fallback_extract(case_text, hints)
        
        # 步骤4: 验证与增强
        final = self._post_process(structured, case_text, hints)
        
        if verbose:
            final['_meta'] = {'hints': hints, 'prompt_tokens': len(prompt)}
        return final
    
    def _build_prompt(self, text: str, hints: dict, budget: int) -> str:
        """构建token受限的提示"""
        # 简化策略：如果文本过长，先提取关键段落
        if len(text) > 2000:  # 约500 tokens
            text = self._extract_key_paragraphs(text, hints)
        
        template = f"""提取商业案例的五元组结构。线索：{json.dumps(hints, ensure_ascii=False)}

案例：{text}

严格按JSON输出：{{"situation":"...","decision_framework":"...","judgment":"...","outcome":"...","reflection":"..."}}"""
        return template
    
    def _fallback_extract(self, text: str, hints: dict) -> dict:
        """当LLM失败时的降级策略：基于规则组装"""
        return {
            'situation': hints.get('situation', ['IMPLICIT'])[0] if hints.get('situation') else "IMPLICIT",
            'decision_framework': hints.get('decision_framework', ['IMPLICIT'])[0] if hints.get('decision_framework') else "IMPLICIT",
            'judgment': hints.get('judgment', ['IMPLICIT'])[0] if hints.get('judgment') else "IMPLICIT",
            'outcome': hints.get('outcome', ['IMPLICIT'])[0] if hints.get('outcome') else "IMPLICIT",
            'reflection': hints.get('reflection', ['IMPLICIT'])[0] if hints.get('reflection') else "IMPLICIT",
            '_method': 'rule_fallback'
        }
    
    def _post_process(self, data: dict, text: str, hints: dict) -> dict:
        """后处理：清洗、验证、增强"""
        # 清洗：移除markdown代码块标记
        for k, v in data.items():
            if isinstance(v, str):
                data[k] = v.strip().strip('`').strip()
        
        # 验证：确保没有 hallucination（关键名词必须在原文）
        key_terms = set(re.findall(r'[\u4e00-\u9fa5]{2,6}', text))  # 2-6字词组
        for field in ['situation', 'judgment', 'outcome']:
            if field in data and data[field] != "IMPLICIT":
                field_terms = set(re.findall(r'[\u4e00-\u9fa5]{2,6}', data[field]))
                # 如果提取的内容与原文共享<30%词汇，标记为可疑
                if len(field_terms) > 0 and len(field_terms & key_terms) / len(field_terms) < 0.3:
                    data[f'_{field}_warning'] = 'low_term_overlap'
        
        # 增强：如果反思缺失但有"后来/结果"段落，尝试组装
        if data.get('reflection') == "IMPLICIT" and hints.get('reflection'):
            data['reflection'] = hints['reflection'][0]
            data['_reflection_source'] = 'rule_supplemented'
            
        return data
    
    def _parse_json(self, text: str) -> dict:
        """鲁棒JSON解析"""
        # 尝试直接解析
        try:
            return json.loads(text)
        except:
            # 提取JSON子串
            match = re.search(r'\{.*\}', text, re.DOTALL)
            if match:
                return json.loads(match.group())
            raise ValueError("Invalid JSON format")

def evaluate_extractor(extractor, test_cases):
    """
    评估脚本：计算准确率、召回率、F1
    """
    results = []
    
    for case in test_cases:
        actual = extractor.extract(case['input'])
        expected = case['expected']
        
        scores = {}
        for field in ['situation', 'decision_framework', 'judgment', 'outcome', 'reflection']:
            # 语义相似度（简化版：关键词重叠率）
            act_keywords = set(jieba.lcut(actual.get(field, '')))
            exp_keywords = set(jieba.lcut(expected[field]))
            
            if not act_keywords or not exp_keywords:
                iou = 0.0
            else:
                intersection = len(act_keywords & exp_keywords)
                union = len(act_keywords | exp_keywords)
                iou = intersection / union if union > 0 else 0
            
            scores[field] = {
                'iou': round(iou, 2),
                'match': iou > 0.5,  # 阈值：50%重叠视为正确
                'actual': actual.get(field, 'MISSING'),
                'expected': expected[field]
            }
        
        # 案例级统计
        accuracy = sum(1 for s in scores.values() if s['match']) / 5
        results.append({
            'id': case['id'],
            'field_scores': scores,
            'case_accuracy': accuracy,
            'has_warnings': any(k.startswith('_') and k.endswith('_warning') for k in actual.keys())
        })
    
    # 全局统计
    total_fields = len(test_cases) * 5
    correct_fields = sum(1 for r in results for s in r['field_scores'].values() if s['match'])
    overall_accuracy = correct_fields / total_fields
    
    print(f"=== 评估结果 ===")
    print(f"测试案例数: {len(test_cases)}")
    print(f"总字段数: {total_fields}")
    print(f"正确字段数: {correct_fields}")
    print(f"整体准确率: {overall_accuracy:.1%}")
    print(f"平均案例完整度: {sum(r['case_accuracy'] for r in results)/len(results):.1%}")
    
    # 按字段统计
    for field in ['situation', 'decision_framework', 'judgment', 'outcome', 'reflection']:

### 2.1 满意解理论算法化

将Herbert Simon的”Satisficing”理论转化为约束满足问题(CSP)+字典序决策的可执行框架。

class HerbertSimonAgent:
    """
    司马贺Agent：基于满意解理论的决策引擎
    核心逻辑：约束识别 → 阈值设定 → 备选生成 → 充分性评估 → 停止搜索
    """
    
    def __init__(self, constraints_db):
        self.constraints_db = constraints_db  # 领域约束知识库
        self.current_thresholds = {}
        self.search_history = []
        
    def decide(self, scenario: dict) -> dict:
        """
        主决策流程
        """
        # 步骤1: 约束识别与分类
        constraints = self._identify_constraints(
            scenario['constraints'],
            scenario.get('founder_profile', ''),
            scenario['scenario']
        )
        
        # 步骤2: 动态阈值设定
        thresholds = self._set_thresholds(constraints, scenario)
        
        # 步骤3: 满意解搜索（带停止规则）
        candidates = scenario.get('candidates', [])
        satisfactory_candidates = []
        stop_candidate = None
        
        for idx, candidate in enumerate(candidates):
            adequacy = self._evaluate_adequacy(candidate, thresholds, constraints)
            self.search_history.append({
                'candidate': candidate['name'],
                'adequacy': adequacy['is_adequate'],
                'score': adequacy['score'],
                'reason': adequacy['reason']
            })
            
            if adequacy['is_adequate']:
                satisfactory_candidates.append({
                    'candidate': candidate,
                    'evaluation': adequacy
                })
                # 停止规则：找到第一个满足阈值的即记录，但继续检查是否有多于一个
                if stop_candidate is None:
                    stop_candidate = {
                        'stop_at': candidate['name'],
                        'stop_index': idx,
                        'rationale': f"第一个满足所有约束的候选人（索引{idx}）"
                    }
                    # 优化：如果启用"即时停止"模式，可在此break
        
        # 步骤4: 在满意集中选择（或处理无满意解情况）
        recommendation = self._select_from_satisfactory(
            satisfactory_candidates, 
            stop_candidate,
            constraints
        )
        
        return {
            'constraints_analysis': constraints,
            'satisficing_threshold': thresholds,
            'search_process': {
                'total_evaluated': len(candidates),
                'satisfactory_count': len(satisfactory_candidates),
                'evaluation_details': self.search_history
            },
            'recommendation': recommendation
        }

### 2.2 核心子算法详解

#### A. 约束识别与分类算法

def _identify_constraints(self, raw_constraints: list, founder_profile: str, scenario: str) -> dict:
    """
    将原始约束分类为硬约束、软约束和可 trade-off 约束
    基于领域规则 + 文本分析
    """
    hard_keywords = ['全职', '必须', '法定', '合规', '全职投入', 'exclusive']
    soft_keywords = ['最好', '希望', '优选', '理想情况下', 'nice to have']
    
    constraints = {
        'hard_constraints': [],      # 一票否决项
        'soft_constraints': [],       # 可妥协项
        'trade_off_constraints': [],  # 可补偿项
        'implicit_constraints': []    # 从场景推断的
    }
    
    # 显性约束分类
    for c in raw_constraints:
        c_lower = c.lower()
        if any(k in c_lower for k in hard_keywords):
            constraints['hard_constraints'].append({
                'constraint': c,
                'type': 'absolute',
                'violation_cost': 'infinite'
            })
        elif any(k in c_lower for k in soft_keywords):
            constraints['soft_constraints'].append({
                'constraint': c,
                'type': 'preferred',
                'weight': 1.0
            })
        else:
            # 默认视为可权衡约束
            constraints['trade_off_constraints'].append({
                'constraint': c,
                'type': 'negotiable',
                'substitutable': True
            })
    
    # 隐性约束推断（基于创始人画像）
    if '首次创业' in founder_profile or 'first-time' in founder_profile:
        constraints['implicit_constraints'].append({
            'constraint': '需要创业指导',
            'source': 'profile_inference',
            'type': 'soft'
        })
    
    if '资金有限' in founder_profile:
        constraints['implicit_constraints'].append({
            'constraint': '接受较低现金薪酬',
            'source': 'resource_constraint',
            'type': 'trade_off'
        })
    
    return constraints

#### B. 动态阈值设定策略

def _set_thresholds(self, constraints: dict, scenario: dict) -> dict:
    """
    基于约束稀缺性和创始人风险偏好设定"足够好"标准
    """
    # 计算约束严格度
    hard_count = len(constraints['hard_constraints'])
    soft_count = len(constraints['soft_constraints'])
    
    # 基础阈值
    base_threshold = {
        'hard_satisfaction': 1.0,    # 硬约束必须100%满足
        'soft_satisfaction': 0.6,     # 软约束满足60%即可
        'trade_off_flexibility': 0.3   # 可接受30%的权衡偏离
    }
    
    # 动态调整：约束越多，阈值越宽松（避免过度优化）
    constraint_pressure = (hard_count * 2 + soft_count) / 5  # 归一化压力指数
    if constraint_pressure > 1.5:
        base_threshold['soft_satisfaction'] = 0.4  # 高压力下降低标准
    
    # 基于创始人风格的调整
    style = scenario.get('decision_style', 'balanced')  # cautious/balanced/aggressive
    if style == 'cautious':
        base_threshold['soft_satisfaction'] += 0.2
    elif style == 'aggressive':
        base_threshold['hard_satisfaction'] = 0.9  # 激进型可能容忍硬约束的小幅偏离
    
    return {
        'min_requirement': self._threshold_to_text(base_threshold),
        'acceptable_deviation': f"软约束可妥协{(1-base_threshold['soft_satisfaction']):.0%}",
        'threshold_values': base_threshold,
        'adjustment_rationale': f"基于{constraint_pressure:.1f}约束压力指数调整"
    }

#### C. 充分性评估算法

def _evaluate_adequacy(self, candidate: dict, thresholds: dict, constraints: dict) -> dict:
    """
    评估候选人是否满足"足够好"标准（非最大化效用）
    """
    scores = {
        'hard_violations': [],
        'soft_satisfied': [],
        'trade_off_score': 0.0
    }
    
    # 检查硬约束（一票否决）
    for hc in constraints['hard_constraints']:
        satisfied = self._check_constraint(candidate, hc['constraint'])
        if not satisfied:
            scores['hard_violations'].append(hc['constraint'])
    
    if scores['hard_violations']:
        return {
            'is_adequate': False,
            'score': 0.0,
            'reason': f"违反硬约束: {', '.join(scores['hard_violations'])}",
            'type': 'rejected'
        }
    
    # 评估软约束（计数制，非加权平均）
    soft_total = len(constraints['soft_constraints'])
    soft_met = 0
    for sc in constraints['soft_constraints']:
        if self._check_constraint(candidate, sc['constraint']):
            soft_met += 1
            scores['soft_satisfied'].append(sc['constraint'])
    
    soft_ratio = soft_met / soft_total if soft_total > 0 else 1.0
    
    # 可权衡约束：允许部分不满足，但需有补偿
    trade_off_penalty = 0
    for tc in constraints['trade_off_constraints']:
        if not self._check_constraint(candidate, tc['constraint']):
            # 检查是否有补偿属性（如"产业资源不足但资金充足"）
            has_compensation = self._check_compensation(candidate, tc['constraint'])
            if not has_compensation:
                trade_off_penalty += 0.3
    
    # 综合充分性判断（满意解核心逻辑）
    is_adequate = (
        soft_ratio >= thresholds['threshold_values']['soft_satisfaction'] and
        trade_off_penalty <= thresholds['threshold_values']['trade_off_flexibility']
    )
    
    # 构建理由链（可解释性）
    reasons = []
    if soft_met > 0:
        reasons.append(f"满足{soft_met}/{soft_total}项软约束")
    if trade_off_penalty > 0:
        reasons.append(f"权衡偏离度: {trade_off_penalty:.1f}")
    
    return {
        'is_adequate': is_adequate,
        'score': soft_ratio - trade_off_penalty,  # 相对排序用，非绝对效用
        'reason': "；".join(reasons) if reasons else "满足所有约束",
        'details': scores
    }

### 2.3 完整实现代码

from typing import Dict, List, Any
import json

class HerbertSimonAgent:
    """
    司马贺满意解决策Agent完整实现
    特性：可解释、约束驱动、非最优化
    """
    
    def __init__(self):
        self.search_history = []
        self.domain_rules = self._load_domain_rules()
    
    def _load_domain_rules(self) -> dict:
        """领域知识：合伙人匹配领域的约束检查规则"""
        return {
            '全职投入': lambda c: c.get('availability') == 'full_time' or '全职' in str(c),
            '有产业资源': lambda c: c.get('network') in ['强', 'high'] or c.get('industry_exp', 0) > 5,
            '接受较低估值': lambda c: c.get('equity_expectation') in ['低', 'low', '中', 'medium'],
            '创业经验': lambda c: c.get('startup_exp', False) or '创业' in str(c.get('track_record', '')),
            '技术背景': lambda c: c.get('background') == 'tech' or '技术' in str(c)
        }
    
    def decide(self, scenario: Dict[str, Any]) -> Dict:
        """主决策入口"""
        self.search_history = []  # 重置历史
        
        # 1. 约束分析
        constraints = self._analyze_constraints(scenario)
        
        # 2. 阈值设定
        thresholds = self._set_thresholds(constraints, scenario.get('risk_preference', 'moderate'))
        
        # 3. 满意解搜索（带停止规则）
        candidates = scenario.get('candidates', [])
        satisfactory = []
        first_satisfactory = None
        
        for i, candidate in enumerate(candidates):
            eval_result = self._evaluate(candidate, constraints, thresholds)
            record = {
                'name': candidate.get('name', f'Candidate_{i}'),
                'index': i,
                'adequacy': eval_result['is_adequate'],
                'metrics': eval_result['metrics'],
                'reason': eval_result['reason']
            }
            self.search_history.append(record)
            
            if eval_result['is_adequate']:
                satisfactory.append(candidate)
                if first_satisfactory is None:
                    first_satisfactory = {
                        'candidate': candidate,
                        'index': i,
                        'reason': f"首个满足阈值的候选（位置{i}）"
                    }
        
        # 4. 生成建议
        recommendation = self._formulate_recommendation(
            satisfactory, first_satisfactory, candidates, constraints
        )
        
        return {
            'constraints_analysis': constraints,
            'satisficing_threshold': thresholds,
            'candidates_evaluation': self.search_history,
            'recommendation': recommendation,
            'process_metadata': {
                'search_depth': len(candidates),
                'stop_point': first_satisfactory['index'] if first_satisfactory else None,
                'satisfactory_count': len(satisfactory),
                'is_satisficing': True  # 标记为满意解模式，非最优化
            }
        }
    
    def _analyze_constraints(self, scenario: Dict) -> Dict:
        """约束识别与分类"""
        raw = scenario.get('constraints', [])
        profile = scenario.get('founder_profile', '')
        
        hard, soft, trade_off = [], [], []
        
        # 分类逻辑
        hard_indicators = ['必须', '全职', '法定', 'exclusive', '必须']
        soft_indicators = ['希望', '最好', 'prefer', ' ideally']
        
        for c in raw:
            if any(h in c for h in hard_indicators):
                hard.append({'type': 'hard', 'desc': c, 'mandatory': True})
            elif any(s in c for s in soft_indicators):
                soft.append({'type': 'soft', 'desc': c, 'weight': 1.5})
            else:
                trade_off.append({'type': 'trade_off', 'desc': c, 'negotiable': True})
        
        # 从画像推断隐性约束
        if '首次' in profile or 'first' in profile:
            soft.append({'type': 'soft', 'desc': '有指导经验', 'source': 'inferred', 'weight': 1.0})
        
        return {
            'hard_constraints': hard,
            'soft_constraints': soft,
            'trade_off_constraints': trade_off,
            'total_pressure': len(hard) * 2 + len(soft) * 1 + len(trade_off) * 0.5
        }
    
    def _set_thresholds(self, constraints: Dict, risk_pref: str) -> Dict:
        """基于约束压力和风险偏好设定阈值"""
        pressure = constraints['total_pressure']
        
        # 基础阈值
        base = {
            'hard_compliance': 1.0,  # 硬约束必须100%
            'soft_compliance': max(0.5, 1.0 - pressure * 0.1),  # 压力越大，标准越低
            'trade_off_tolerance': 0.4
        }
        
        # 风险偏好调整
        adjustments = {
            'conservative': {'soft_compliance': 0.2, 'trade_off_tolerance': -0.1},
            'moderate': {'soft_compliance': 0.0, 'trade_off_tolerance': 0.0},
            'aggressive': {'soft_compliance': -0.2, 'trade_off_tolerance': 0.2}
        }
        
        adj = adjustments.get(risk_pref, adjustments['moderate'])
        for k in base:
            base[k] += adj.get(k, 0)
            base[k] = max(0.0, min(1.0, base[k]))
        
        return {
            'threshold_values': base,
            'rationale': f"约束压力={pressure:.1f}, 风险偏好={risk_pref}",
            'interpretation': {
                'hard': "必须完全满足",
                'soft': f"至少满足{base['soft_compliance']:.0%}",
                'trade_off': f"允许{base['trade_off_tolerance']:.0%}的权衡偏离"
            }
        }
    
    def _evaluate(self, candidate: Dict, constraints: Dict, thresholds: Dict) -> Dict:
        """评估单个候选人的充分性"""
        metrics = {'hard_pass': True, 'soft_met': 0, 'soft_total': 0, 'trade_off_deviation': 0}
        reasons = []
        
        # 硬约束检查（一票否决）
        for hc in constraints['hard_constraints']:
            check_func = self.domain_rules.get(hc['desc'], lambda c: True)
            if not check_func(candidate):
                metrics['hard_pass'] = False
                reasons.append(f"不满足硬约束: {hc['desc']}")
        
        if not metrics['hard_pass']:
            return {
                'is_adequate': False,
                'metrics': metrics,
                'reason': '；'.join(reasons)
            }
        
        # 软约束检查
        soft_total = len(constraints['soft_constraints'])
        soft_met = 0
        for sc in constraints['soft_constraints']:
            check_func = self.domain_rules.get(sc['desc'], lambda c: True)
            if check_func(candidate):
                soft_met += 1
        
        metrics['soft_met'] = soft_met
        metrics['soft_total'] = soft_total
        soft_ratio = soft_met / soft_total if soft_total > 0 else 1.0
        
        # 可权衡约束检查（允许偏离，但记录）
        trade_off_deviation = 0
        for tc in constraints['trade_off_constraints']:
            check_func = self.domain_rules.get(tc['desc'], lambda c: True)
            if not check_func(candidate):
                # 检查是否有补偿特性
                has_alt = self._has_alternative_strength(candidate, tc['desc'])
                if not has_alt:
                    trade_off_deviation += 0.3
        
        metrics['trade_off_deviation'] = trade_off_deviation
        
        # 满意解判定
        is_adequate = (
            soft_ratio >= thresholds['threshold_values']['soft_compliance'] and
            trade_off_deviation <= thresholds['threshold_values']['trade_off_tolerance']
        )
        
        # 构建解释
        if soft_total > 0:
            reasons.append(f"软约束满足{soft_met}/{soft_total} ({soft_ratio:.0%})")
        if trade_off_deviation > 0:
            reasons.append(f"权衡偏离度: {trade_off_deviation:.1f}")
        
        return {
            'is_adequate': is_adequate,
            'metrics': metrics,
            'reason': '；'.join(reasons) if reasons else "全面满足约束"
        }
    
    def _has_alternative_strength(self, candidate: Dict, missing_constraint: str) -> bool:
        """检查候选人在某方面不足时是否有其他补偿优势"""
        # 简单启发规则：如缺少"产业资源"，但有"资金"或"技术专利"补偿
        alt_map = {
            '有产业资源': ['funding_access', 'patents', 'technical_depth'],
            '接受较低估值': ['exceptional_experience', 'strategic_value']
        }
        
        alternatives = alt_map.get(missing_constraint, [])
        return any(candidate.get(attr) for attr in alternatives)
    
    def _formulate_recommendation(self, satisfactory: List, first_sat: Dict, 
                                   all_candidates: List, constraints: Dict) -> Dict:
        """生成最终建议（含停止规则解释）"""
        if not satisfactory:
            return {
                'action': 'expand_search',
                'stop_at': None,
                'rationale': '无满意解，建议放宽约束或扩大搜索范围',
                'fallback': '考虑降低软约束阈值10-20%'
            }
        
        if len(satisfactory) == 1:
            chosen = satisfactory[0]
            return {
                'action': 'accept',
                'stop_at': chosen.get('name'),
                'rationale': '唯一满意解，符合停止规则（首个即停）',
                'note': '无需继续评估后续候选（满意解特性）'
            }
        
        # 多个满意解：在满意集内简单比较（非全搜索空间优化）
        # 选择策略：首个满足 vs 满意集中较优
        if len(satisfactory) <= 3:
            # 满意集较小，内部选最优
            best = max(satisfactory, 
                      key=lambda c: c.get('composite_score', 0))
            return {
                'action': 'select_best_in_satisfactory',
                'stop_at': best.get('name'),
                'rationale': f"在{len(satisfactory)}个满意解中选择指标较优者，但仍非全局最优搜索",
                'satisfactory_set': [c.get('name') for c in satisfactory]
            }
        else:
            # 满意解过多，说明阈值过低，建议收紧
            return {
                'action': 'reevaluate_thresholds',
                'stop_at': first_sat['candidate'].get('name') if first_sat else None,
                'rationale': f"满意解过多({len(satisfactory)})，建议提高阈值以提升决策质量",
                'satisfactory_set': [c.get('name') for c in satisfactory[:5]]
            }

            flag = "价值观根本冲突"
        else:
            overall = (value_alignment * 0.4 + reputation_score * 0.3 + cohesion_potential * 0.3)
            flag = "可接受" if overall > 0.6 else "需观察"
        
        return {
            'score': round(overall, 2),
            'dimension': '品德匹配/德馨',
            'analysis': self._generate_analysis(value_alignment, reputation_score, cohesion_potential, flag),
            'recommendation': flag,
            'details': {

                'reputation_score': round(reputation_score, 2),
                'cohesion_potential': round(cohesion_potential, 2)
            }
        }
    
    def _calculate_value_alignment(self, founder_vals: List[str], candidate_expr: str, track_record: str) -> float:
        """价值观对齐度：关键词共现+极性判断"""
        # 简化的基于规则的匹配
        alignment_indicators = {
            '长期主义': ['长期', '耐心', '坚持', '持久', 'sustainable'],
            '技术理想': ['理想', '使命', '愿景', '改变世界', '技术驱动'],
            '公平分享': ['公平', '分享', '共赢', 'generous', 'equitable']
        }
        
        combined_text = (candidate_expr + " " + track_record).lower()
        scores = []
        
        for fv in founder_vals:
            indicators = alignment_indicators.get(fv, [fv])
            matches = sum(1 for ind in indicators if ind in combined_text)
            scores.append(min(matches / 2, 1.0))  # 饱和于1.0
        
        return sum(scores) / len(scores) if scores else 0.5
    
    def _assess_reputation(self, track_record: str, references: List[str]) -> float:
        """声誉评估：基于过往记录的模式识别"""
        positive_signals = ['成功退出', '连续创业', '知名公司', '良好口碑', '诚信']
        negative_signals = ['纠纷', '诉讼', '失信', '频繁跳槽', '冲突']

        neg_count = sum(1 for s in negative_signals if s in text)

        score = 0.5 + (pos_count * 0.1) - (neg_count * 0.15)

    def _assess_cohesion(self, founder_vals: List[str], style: str, conflict_res: str) -> float:
        """团队凝聚潜力：风格互补性"""

        text = (style + " " + conflict_res).lower()
        matches = sum(1 for k in cohesion_keywords if k in text)
        return min(matches / 3, 1.0)
    
    def _generate_analysis(self, val_align, rep, coh, flag):

        if val_align > 0.7:
            parts.append("价值观高度一致")
        elif val_align < 0.4:

        else:
            parts.append("价值观部分匹配")
        
        if rep > 0.7:
            parts.append("过往声誉良好")
        elif rep < 0.4:
            parts.append("声誉记录需关注")
        
        if coh > 0.6:
            parts.append("具备团队凝聚潜力")
        
        return "，".join(parts) + f"。总体判断：{flag}"

### 3.3 冲突消解与共识合成

class ConflictResolver:
    """
    五图腾冲突消解器：基于五行相生相克关系
    优先级：土(刘禹锡) > 金(司马贺) > 水(观自在) > 木(孔子) > 火(慧能)
    但引入相克关系：当两图腾冲突时，检查克制关系决定权重
    """
    
    def resolve(self, evaluations: Dict[str, Dict], scenario: Dict) -> Dict:
        """
        输入：五个图腾的评估结果
        输出：综合共识与冲突消解建议
        """
        scores = {k: v['score'] for k, v in evaluations.items()}
        
        # 1. 检测严重冲突（评分差异>0.5）
        conflicts = []
        for t1 in evaluations:
            for t2 in evaluations:
                if t1 != t2 and abs(scores[t1] - scores[t2]) > 0.5:
                    conflicts.append({
                        'between': (t1, t2),
                        'severity': abs(scores[t1] - scores[t2]),
                        'type': 'major_disagreement'
                    })
        
        # 2. 基于五行的冲突消解
        resolution_strategy = self._apply_wuxing_resolution(conflicts, evaluations)
        
        # 3. 加权共识计算（动态权重）
        final_score = self._calculate_consensus(evaluations, resolution_strategy)
        
        # 4. 生成可解释建议
        recommendation = self._formulate_advice(evaluations, final_score, conflicts)
        
        return {
            'consensus_score': round(final_score, 2),
            'individual_scores': scores,
            'conflicts_detected': conflicts,
            'resolution_strategy': resolution_strategy,
            'final_recommendation': recommendation,
            'dissenting_views': [c for c in conflicts if c['severity'] > 0.6]
        }
    
    def _apply_wuxing_resolution(self, conflicts: List[Dict], evaluations: Dict) -> Dict:
        """应用五行克制关系解决冲突"""
        dominance = {
            'guanzizai': 'huineng',    # 水克火
            'huineng': 'simon',        # 火克金
            'simon': 'confucius',      # 金克木
            'confucius': 'liuyuxi',    # 木克土
            'liuyuxi': 'guanzizai'     # 土克水
        }
        
        strategy = {'weight_adjustments': {}, 'reasoning': []}
        
        for conflict in conflicts:
            t1, t2 = conflict['between']
            # 检查是否存在克制关系
            if dominance.get(t1) == t2:
                # t1 克制 t2，降低 t2 权重
                strategy['weight_adjustments'][t2] = 0.5
                strategy['reasoning'].append(f"{t1}克{t2}：降低{t2}权重")
            elif dominance.get(t2) == t1:
                strategy['weight_adjustments'][t1] = 0.5
                strategy['reasoning'].append(f"{t2}克{t1}：降低{t1}权重")
            else:
                # 无直接克制，采用"土居中"调和（刘禹锡权重提升）
                strategy['weight_adjustments']['liuyuxi'] = 1.5
                strategy['reasoning'].append(f"{t1}与{t2}无相克，提升刘禹锡(土)权重调和")
        
        return strategy
    
    def _calculate_consensus(self, evaluations: Dict, strategy: Dict) -> float:
        """计算加权共识分"""
        base_weights = {
            'liuyuxi': 0.2, 'simon': 0.25, 'guanzizai': 0.2, 
            'confucius': 0.2, 'huineng': 0.15
        }
        
        # 应用调整
        weights = base_weights.copy()
        for totem, adj in strategy.get('weight_adjustments', {}).items():
            weights[totem] *= adj
        
        # 归一化
        total = sum(weights.values())
        weights = {k: v/total for k, v in weights.items()}
        
        # 计算
        score = sum(evaluations[t]['score'] * weights[t] for t in evaluations)
        return score
    
    def _formulate_advice(self, evaluations: Dict, consensus: float, conflicts: List) -> str:
        """生成最终建议"""
        # 基于共识度和冲突情况生成不同建议
        if consensus > 0.75 and not conflicts:
            return "五图腾共识：强烈建议推进，多维度匹配良好"
        
        if consensus < 0.4:
            return "五图腾共识：建议放弃，存在根本性错配"
        
        # 存在冲突时的 nuanced 建议
        high_scores = [t for t, e in evaluations.items() if e['score'] > 0.7]
        low_scores = [t for t, e in evaluations.items() if e['score'] < 0.4]
        
        if 'simon' in high_scores and 'confucius' in low_scores:
            return "理性匹配但伦理存疑：建议设定严格试用期验证伦理维度"
        
        if 'huineng' in high_scores and 'guanzizai' in low_scores:
            return "创新潜力高但直觉冲突：建议明确创新权责边界，减少日常协作摩擦"
        
        if 'liuyuxi' in low_scores:
            return "德馨不足：建议优先寻找价值观更匹配候选，或重新评估核心需求"
        
        return "图腾意见分歧：建议进一步深入沟通或寻找其他候选人"

class KimiClawCore:
    """
    Kimi Claw系统核心集成类
    整合五元组提取+满意解决策+五图腾评估
    """
    
    def __init__(self, llm_api):
        self.extractor = PentadExtractor(llm_api)
        self.simon = HerbertSimonAgent()
        self.totem_system = FiveTotemOrchestrator()
        
    def process_case(self, raw_text: str, candidates: list) -> dict:
        """完整处理流程"""
        # 1. 提取五元组
        pentad = self.extractor.extract(raw_text)
        
        # 2. 构建决策场景
        scenario = {
            'situation': pentad['situation'],
            'constraints': self._infer_constraints(pentad),
            'candidates': candidates
        }
        
        # 3. 满意解筛选
        simon_result = self.simon.decide(scenario)
        
        # 4. 对满意集进行五图腾深度评估
        satisfactory_candidates = [
            c for c in candidates 
            if any(e['name'] == c['name'] and e['adequacy'] 
                  for e in simon_result['candidates_evaluation'])
        ]
        
        totem_results = []
        for candidate in satisfactory_candidates:
            totem_eval = self.totem_system.evaluate_candidate(
                candidate, 
                {'values': self._extract_values(pentad)}, 
                scenario
            )
            totem_results.append(totem_eval)
        
        return {
            'case_structure': pentad,
            'satisficing_screening': simon_result,
            'depth_evaluation': totem_results,
            'final_recommendation': self._synthesize_recommendation(
                simon_result, totem_results
            )
        }
    
    def _infer_constraints(self, pentad: dict) -> list:
        """从五元组推断约束条件"""
        constraints = []
        if '股权' in pentad.get('outcome', '') and '分歧' in pentad.get('outcome', ''):
            constraints.append('明确股权架构')
        if '缺乏商业经验' in pentad.get('situation', ''):
            constraints.append('有产业资源')
            constraints.append('全职投入')
        return constraints
    
    def _extract_values(self, pentad: dict) -> list:
        """从情境和反思提取价值观"""
        values = []
        text = pentad.get('situation', '') + pentad.get('reflection', '')
        if '长期' in text:
            values.append('长期主义')
        if '技术' in text:
            values.append('技术理想')
        return values

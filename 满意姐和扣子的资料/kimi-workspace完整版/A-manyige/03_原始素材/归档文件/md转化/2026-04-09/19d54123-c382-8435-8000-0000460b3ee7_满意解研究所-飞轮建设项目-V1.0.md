# /opt/sri-agent-os/core/knowledge_ingestion.py

import hashlib
import json
from datetime import datetime
from enum import Enum
from typing import Dict, List, Optional
import psycopg2
from openclaw import Skill, Memory, Workflow

class TotemType(Enum):
    LIU_YUXI = "土-刘禹锡"      # 长期价值
    HERBERT_SIMON = "金-司马贺"  # 满意解算法
    GUAN_ZIZAI = "水-观自在"     # 直觉感知
    CONFUCIUS = "木-孔子"        # 五常伦理
    HUI_NENG = "火-慧能"         # 创新顿悟

class KnowledgeIngestionEngine:
    """
    知识入库与内化引擎
    实现"物理提取 → 深度消化 → 内化验证"闭环
    """
    
    def __init__(self):
        self.db_conn = psycopg2.connect("dbname=sri_agent_os user=postgres")
        self.totem_classifier = self._load_totem_classifier()
        
    def register_file(self, file_path: str, source: str, uploader: str) -> Dict:
        """
        第一重门：文件登记 (3分钟内完成)
        """
        start_time = datetime.now()
        
        # 1. 计算文件指纹
        file_hash = self._calculate_hash(file_path)
        
        # 2. 检查重复
        if self._is_duplicate(file_hash):
            return {"status": "duplicate", "message": "文件已存在，触发复用统计"}
        
        # 3. AI自动分类（五图腾）
        totem = self._classify_by_totem(file_path)
        
        # 4. 入库登记
        file_id = self._insert_registry(file_path, source, uploader, file_hash, totem)
        
        # 5. 触发3-2-1备份
        self._trigger_backup(file_id, file_path)
        
        # 6. 记录时间指标
        duration = (datetime.now() - start_time).total_seconds()
        self._record_metric("ingestion_time", duration, file_id)
        
        return {
            "file_id": file_id,
            "totem_category": totem.value,
            "next_step": "开始第一遍通读",
            "estimated_time": "15-20分钟"
        }
    
    def internalization_workflow(self, file_id: str) -> Dict:
        """
        第二至四重门：三遍阅读内化流程
        辅助Egbertie完成深度消化
        """
        workflow = Workflow("internalization_v1")
        
        # 第二重门：第一遍通读（Pass 1）
        @workflow.step("pass_1_reading")
        def pass_1(context):
            """
            目标：理解核心论点，不做笔记
            AI辅助：生成结构大纲，标记关键概念
            """
            prompt = f"""
            你正在辅助进行第一遍通读（理解层）。
            文件：{context.file_name}
            
            任务：
            1. 提取文章结构（目录/章节/段落关系）
            2. 标记3-5个核心概念（用🔑标记）
            3. 生成1句话核心论点
            4. 预估阅读时间
            
            输出格式：
            - 结构大纲：[层级列表]
            - 核心概念：[概念1, 概念2...]
            - 核心论点：[一句话]
            - 建议时长：[X分钟]
            """
            result = self.call_kimi(prompt, context.file_content)
            
            # 存储到Memory
            Memory.store(f"file:{file_id}:pass1", result)
            
            return {
                "guidance": result,
                "user_action": "请通读全文，理解大意，不要记笔记。读完后点击'完成'",
                "timer_suggested": result.get("建议时长", "20分钟")
            }
        
        # 第三重门：第二遍笔记（Pass 2）
        @workflow.step("pass_2_notes", depends_on="pass_1_reading")
        def pass_2(context):
            """
            目标：深度消化，提取精华
            AI辅助：提供模板，引导深度思考
            """
            templates = {
                TotemType.LIU_YUXI: "长期价值评估模板：该知识如何影响3-5年后的决策？",
                TotemType.HERBERT_SIMON: "满意解分析模板：边界条件是什么？何时'足够好'？",
                TotemType.GUAN_ZIZAI: "直觉感知模板：该知识引发什么身体感受/直觉反应？",
                TotemType.CONFUCIUS: "五常评估模板：仁义礼智信各如何体现？",
                TotemType.HUI_NENG: "顿悟触发模板：该知识可能引发什么突破性认知？"
            }
            
            totem = context.totem_category
            template = templates.get(totem, "通用深度笔记模板")
            
            prompt = f"""
            你正在辅助进行第二遍阅读（消化层）。
            
            基于五图腾体系中的【{totem}】视角：
            {template}
            
            请引导用户完成以下笔记：
            1. 概念图谱（概念间关系）
            2. 与我现有知识的3个连接点
            3. 可立即应用的1个场景
            4. 存疑待验证的2个问题
            
            输出格式为Markdown表格，方便填写。
            """
            
            result = self.call_kimi(prompt, context.file_content)
            Memory.store(f"file:{file_id}:pass2", result)
            
            return {
                "template": result,
                "user_action": "精读并填写笔记模板，聚焦深度而非速度",
                "checkpoint": "笔记完成后，系统将自动进入第三遍"
            }
        
        # 第四重门：第三遍总结（Pass 3）
        @workflow.step("pass_3_summary", depends_on="pass_2_notes")
        def pass_3(context):
            """
            目标：内化输出，形成可复用资产
            AI辅助：质量检查，格式标准化
            """
            prompt = f"""
            你正在辅助进行第三遍阅读（内化层）。
            
            基于前两遍的成果：
            - Pass 1 结构：{Memory.retrieve(f'file:{file_id}:pass1')}
            - Pass 2 笔记：{Memory.retrieve(f'file:{file_id}:pass2')}
            
            任务：
            1. 生成500字以内的精华总结
            2. 提取3个行动要点（Action Items）
            3. 转化为1个Prompt模板（可直接用于后续工作）
            4. 建议存储位置（Skill/Memory/Workflow/Case）
            
            这是最终输出，将直接进入知识库。
            """
            
            result = self.call_kimi(prompt, context.file_content)
            Memory.store(f"file:{file_id}:pass3", result)
            
            # 自动生成Prompt资产
            prompt_asset = self._generate_prompt_asset(result)
            
            return {
                "summary": result,
                "generated_prompt": prompt_asset,
                "user_action": "审阅AI生成的总结，确认或修改后提交",
                "next_stage": "质量验证（第五重门）"
            }
        
        return workflow.execute(file_id)
    
    def quality_verification(self, file_id: str, user_answers: List[str]) -> Dict:
        """
        第五重门：质量验证（5题抽查）
        """
        # 基于Pass 3内容生成验证题
        summary = Memory.retrieve(f"file:{file_id}:pass3")
        
        prompt = f"""
        基于以下知识总结，生成5道验证题（4道选择题+1道应用题）：
        {summary}
        
        要求：
        - 前4题测试理解深度（非表面记忆）
        - 第5题测试应用能力（场景化问题）
        - 提供评分标准和参考答案
        
        输出JSON格式。
        """
        
        quiz = self.call_kimi(prompt, return_format="json")
        
        # 自动评分
        score = self._auto_grade(quiz, user_answers)
        
        if score >= 80:
            status = "verified"
            # 正式归档到飞轮
            self._archive_to_flywheel(file_id)
        else:
            status = "rework"
            # 触发返工流程
            self._trigger_rework(file_id, quiz, score)
        
        # 记录蓝军审计点
        self._audit_log(file_id, "quality_check", score)
        
        return {
            "score": score,
            "status": status,
            "certificate": f"SRI-VERIFIED-{file_id}" if status == "verified" else None,
            "feedback": "已内化完成，进入资产库" if status == "verified" else "需返工，建议重新阅读相关段落"
        }
    
    def _classify_by_totem(self, file_path: str) -> TotemType:
        """
        基于内容自动分类到五图腾
        """
        content = self._extract_text(file_path)
        
        prompt = f"""
        分析以下内容的决策视角，归类到五图腾体系：
        
        五图腾定义：
        1. 土-刘禹锡：长期价值、根基、物质本质、可持续性
        2. 金-司马贺：算法、边界、优化、理性决策、满意解
        3. 水-观自在：直觉、感知、流动性、适应、定力
        4. 木-孔子：伦理、关系、规范、五常（仁义礼智信）、人文
        5. 火-慧能：顿悟、创新、突破、压力下的灵感、直指核心
        
        内容：{content[:2000]}...
        
        请返回最匹配的图腾名称和置信度（0-1）。
        输出格式：图腾名称|置信度|理由（20字内）
        """
        
        result = self.call_kimi(prompt)
        totem_name = result.split("|")[0].strip()
        
        # 映射到枚举
        mapping = {
            "土-刘禹锡": TotemType.LIU_YUXI,
            "金-司马贺": TotemType.HERBERT_SIMON,
            "水-观自在": TotemType.GUAN_ZIZAI,
            "木-孔子": TotemType.CONFUCIUS,
            "火-慧能": TotemType.HUI_NENG
        }
        
        return mapping.get(totem_name, TotemType.HERBERT_SIMON)
    
    def _archive_to_flywheel(self, file_id: str):
        """
        验证通过后，根据内容类型归档到飞轮不同节点
        """
        summary = Memory.retrieve(f"file:{file_id}:pass3")
        
        # 判断资产类型
        if "Prompt" in summary:
            target = "prompt_library"
        elif "流程" in summary or "SOP" in summary:
            target = "workflow_engine"
        elif "案例" in summary:
            target = "case_repository"
        else:
            target = "memory_graph"
        
        # 写入对应存储
        self._write_to_storage(file_id, target, summary)
        
        # 更新复用率统计
        self._update_reuse_stats(file_id, target)

    def audit(self, entity_type: str, entity_id: str, data: Dict) -> List[Dict]:
        findings = []
        
        for rule in self.AUDIT_RULES:
            passed = self._check_rule(rule, data)
            if not passed:
                findings.append({
                    "rule_id": rule["id"],
                    "severity": rule["severity"],
                    "finding": rule["check"],
                    "recommendation": self._get_recommendation(rule["id"])
                })
        
        # 写入审计日志
        self._log_findings(entity_type, entity_id, findings)
        
        # 严重问题立即告警
        criticals = [f for f in findings if f["severity"] == "critical"]
        if criticals:
            self._alert_egbertie(criticals)
        
        return findings

import time
from contextlib import contextmanager
from functools import wraps

class MetricsCollector:
    """
    五维指标自动采集器
    时间(Time) / 质量(Quality) / 风险(Risk) / 成本(Cost) / 复用率(Reuse)
    """
    
    def __init__(self):
        self.db = psycopg2.connect("dbname=sri_agent_os user=postgres")
    
    @contextmanager
    def track_time(self, task_type: str, metadata: Dict = None):
        """
        上下文管理器：自动追踪任务耗时
        使用示例：
        with collector.track_time("knowledge_ingestion", {"file_id": "xxx"}):
            process_file(file)
        """
        start = time.time()
        start_iso = datetime.now().isoformat()
        
        try:
            yield
            status = "success"
        except Exception as e:
            status = "failed"
            raise
        finally:
            duration = time.time() - start
            self._store_metric("time", {
                "task_type": task_type,
                "duration_seconds": duration,
                "start_time": start_iso,
                "status": status,
                "metadata": metadata or {}
            })
    
    def track_token_cost(self, prompt_tokens: int, completion_tokens: int, model: str):
        """
        记录Token成本
        Kimi模型价格表（2026参考）：
        - k2.5: 输入 ¥0.015/1K, 输出 ¥0.06/1K
        """
        pricing = {
            "kimi-k2.5": {"input": 0.015, "output": 0.06},
            "kimi-k1.5": {"input": 0.01, "output": 0.04}
        }
        
        p = pricing.get(model, pricing["kimi-k2.5"])
        cost = (prompt_tokens / 1000 * p["input"] + 
                completion_tokens / 1000 * p["output"])
        
        self._store_metric("cost", {
            "model": model,
            "prompt_tokens": prompt_tokens,
            "completion_tokens": completion_tokens,
            "total_cost_yuan": round(cost, 4),
            "timestamp": datetime.now().isoformat()
        })
        
        return cost
    
    def track_quality(self, task_id: str, success: bool, rework_count: int = 0):
        """
        质量指标：一次成功率、返工率
        """
        self._store_metric("quality", {
            "task_id": task_id,
            "first_time_success": success and rework_count == 0,
            "rework_count": rework_count,
            "final_success": success
        })
    
    def track_reuse(self, asset_id: str, asset_type: str):
        """
        复用率统计：每次资产被调用时记录
        """
        self._store_metric("reuse", {
            "asset_id": asset_id,
            "asset_type": asset_type,
            "access_time": datetime.now().isoformat()
        })
        
        # 实时更新复用率仪表板
        self._update_reuse_dashboard(asset_id)
    
    def track_risk(self, event_type: str, severity: str, description: str):
        """
        风险指标：合规事件、安全告警
        """
        self._store_metric("risk", {
            "event_type": event_type,
            "severity": severity,
            "description": description,
            "timestamp": datetime.now().isoformat(),
            "mitigation": None  # 后续填写缓解措施
        })
        
        if severity in ["high", "critical"]:
            self._alert_management(event_type, description)

# /opt/sri-agent-os/core/workflow_engine.py

import yaml
import json
from enum import Enum
from typing import Any, Dict, List, Optional
from datetime import datetime
import psycopg2
from concurrent.futures import ThreadPoolExecutor, as_completed

class StepType(Enum):
    SKILL = "skill"
    LLM = "llm"
    MANUAL = "manual"
    PARALLEL = "parallel"
    CONDITION = "condition"
    APPROVAL = "approval"
    NOTIFICATION = "notification"
    PASS = "pass"

class WorkflowEngine:
    """
    五图腾感知的工作流引擎
    支持：顺序执行、并行分支、条件路由、人工审批、技能调用
    """
    
    def __init__(self):
        self.db = psycopg2.connect("dbname=sri_agent_os user=postgres")
        self.skill_registry = self._load_skill_registry()
        self.totem_engine = TotemDecisionEngine()
        self.metrics = MetricsCollector()
        self.executor = ThreadPoolExecutor(max_workers=5)
        
    def load_workflow(self, yaml_path: str) -> Dict:
        """加载YAML定义"""
        with open(yaml_path, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f)
    
    def execute(self, workflow_def: Dict, trigger_data: Dict) -> Dict:
        """
        执行工作流
        """
        workflow_id = self._create_workflow_instance(workflow_def['workflow']['name'])
        context = {
            "workflow_id": workflow_id,
            "trigger": trigger_data,
            "steps": {},
            "variables": {}
        }
        
        try:
            with self.metrics.track_time(f"workflow_{workflow_def['workflow']['name']}"):
                for step in workflow_def['workflow']['steps']:
                    result = self._execute_step(step, context, workflow_def['workflow'])
                    context["steps"][step['id']] = result
                    
                    # 检查是否需要五图腾检查点
                    if workflow_def['workflow'].get('totem_checkpoints'):
                        self._inject_totem_check(step, context, workflow_def['workflow']['totem_checkpoints'])
                    
                    # 蓝军实时审计
                    self._audit_step_execution(step, result, workflow_id)
                    
        except Exception as e:
            self._handle_error(e, step, context, workflow_def['workflow']['error_handling'])
            raise
        
        return {
            "workflow_id": workflow_id,
            "status": "completed",
            "context": context
        }
    
    def _execute_step(self, step: Dict, context: Dict, workflow_config: Dict) -> Any:
        """
        单步执行路由
        """
        step_type = StepType(step['type'])
        
        # 记录步骤开始
        step_start = datetime.now()
        
        if step_type == StepType.SKILL:
            result = self._execute_skill_step(step, context)
        elif step_type == StepType.LLM:
            result = self._execute_llm_step(step, context)
        elif step_type == StepType.PARALLEL:
            result = self._execute_parallel_step(step, context, workflow_config)
        elif step_type == StepType.MANUAL:
            result = self._execute_manual_step(step, context)
        elif step_type == StepType.APPROVAL:
            result = self._execute_approval_step(step, context)
        elif step_type == StepType.CONDITION:
            result = self._execute_condition_step(step, context, workflow_config)
        else:
            result = {"status": "passed"}
        
        # 记录执行时间和成本
        duration = (datetime.now() - step_start).total_seconds()
        self.metrics.track_workflow_step(step['id'], step['type'], duration)
        
        return result
    
    def _execute_skill_step(self, step: Dict, context: Dict) -> Dict:
        """
        执行Skill调用
        """
        skill_name = step['skill']
        action = step['action']
        
        # 解析输入（支持模板变量）
        inputs = self._resolve_template(step['input'], context)
        
        # 调用Skill（通过OpenClaw）
        skill = self.skill_registry.get(skill_name)
        if not skill:
            raise ValueError(f"Skill not found: {skill_name}")
        
        # 检查限制
        if 'time_limit' in step.get('metrics', {}):
            # 设置超时
            pass
        
        result = skill.execute(action, inputs)
        
        # 记录Token成本
        if 'usage' in result:
            self.metrics.track_token_cost(
                result['usage']['prompt_tokens'],
                result['usage']['completion_tokens'],
                result.get('model', 'kimi-k2.5')
            )
        
        return result
    
    def _execute_parallel_step(self, step: Dict, context: Dict, workflow_config: Dict) -> Dict:
        """
        并行执行多个分支（如五图腾同时评估）
        """
        branches = step['branches']
        futures = {}
        
        # 提交所有并行任务
        for branch in branches:
            future = self.executor.submit(self._execute_step, branch, context, workflow_config)
            futures[future] = branch['id']
        
        # 收集结果
        results = {}
        for future in as_completed(futures):
            branch_id = futures[future]
            try:
                results[branch_id] = future.result()
            except Exception as e:
                results[branch_id] = {"error": str(e)}
        
        # 如果有聚合步骤，执行聚合
        if 'aggregation' in step:
            agg_step = step['aggregation']
            agg_step['input'] = self._resolve_template(agg_step['input'], {**context, "branches": results})
            aggregated = self._execute_step(agg_step, context, workflow_config)
            results['aggregated'] = aggregated
        
        return results
    
    def _execute_manual_step(self, step: Dict, context: Dict) -> Dict:
        """
        人工输入步骤（生成表单，等待输入）
        """
        # 生成表单URL/二维码
        form_id = self._create_form_instance(step, context)
        
        # 发送通知（飞书/微信）
        self._notify_user(step, context, form_id)
        
        # 阻塞等待输入（或返回挂起状态）
        # 实际实现：返回pending，前端轮询或Webhook回调
        return {
            "status": "pending",
            "form_id": form_id,
            "fields": step['form'],
            "message": "等待人工输入"
        }
    
    def _inject_totem_check(self, current_step: Dict, context: Dict, checkpoints: Dict):
        """
        在关键步骤后注入五图腾检查
        这是满意解研究所的特色：每个重要决策都经过五维校验
        """
        if not checkpoints:
            return
        
        # 构建检查提示
        checks = []
        for totem, config in checkpoints.items():
            if config.get('enabled'):
                checks.append({
                    "totem": totem,
                    "question": config['question'],
                    "context": f"当前步骤：{current_step['name']}"
                })
        
        # 记录到上下文，供后续步骤使用
        context['totem_checks'] = checks
        
        # 如果是自动模式，直接调用LLM生成检查报告
        # 如果是手动模式，生成待办事项
        return checks
    
    def _audit_step_execution(self, step: Dict, result: Dict, workflow_id: str):
        """
        蓝军实时审计
        """
        auditor = Skeptor7Auditor()
        
        # 检查合规性
        findings = auditor.audit("workflow_step", step['id'], {
            "step_type": step['type'],
            "duration": result.get('duration'),
            "output_size": len(str(result)),
            "sensitive_data": self._check_sensitive_data(result)
        })
        
        # 记录到审计日志
        if findings:
            self._log_audit_findings(workflow_id, step['id'], findings)
    
    def _handle_error(self, error: Exception, step: Dict, context: Dict, error_config: Dict):
        """
        错误处理与恢复
        """
        error_type = "timeout" if isinstance(error, TimeoutError) else "default"
        handlers = error_config.get(error_type, error_config['default'])
        
        for handler in handlers:
            if handler['type'] == 'retry':
                # 重试逻辑
                pass
            elif handler['type'] == 'notification':
                self._send_alert(handler['to'], handler['message'].replace("{{error.message}}", str(error)))
            elif handler['type'] == 'escalation':
                self._escalate_to_human(handler['to'], step, error)
    
    def _resolve_template(self, template: Any, context: Dict) -> Any:
        """
        解析模板变量 {{steps.xxx.output}} 或 {{trigger.yyy}}
        """
        if isinstance(template, str):
            import re
            pattern = r'\{\{([^}]+)\}\}'
            def replace(match):
                path = match.group(1).strip()
                keys = path.split('.')
                value = context
                for key in keys:
                    if isinstance(value, dict):
                        value = value.get(key, match.group(0))
                    else:
                        return match.group(0)
                return str(value)
            return re.sub(pattern, replace, template)
        elif isinstance(template, dict):
            return {k: self._resolve_template(v, context) for k, v in template.items()}
        elif isinstance(template, list):
            return [self._resolve_template(item, context) for item in template]
        return template

# /opt/sri-agent-os/core/memory_graph.py

from typing import List, Dict, Optional
import psycopg2
from pgvector.psycopg2 import register_vector
import numpy as np

class MemoryGraphEngine:
    """
    五图腾增强的RAG检索引擎
    支持：语义搜索、图腾过滤、关系推理
    """
    
    def __init__(self):
        self.conn = psycopg2.connect("dbname=sri_agent_os user=postgres")
        register_vector(self.conn)
        self.embedding_model = self._load_embedding_model()
    
    def ingest_document(self, doc_id: str, content: str, totem_category: str):
        """
        文档入图谱
        """
        # 1. 分块
        chunks = self._smart_chunk(content)
        
        # 2. 实体识别与提取
        entities = self._extract_entities(content)
        
        # 3. 关系抽取
        relations = self._extract_relations(content, entities)
        
        # 4. 五图腾语义标注
        for entity in entities:
            self._add_totem_semantics(entity, totem_category)
        
        # 5. 向量化存储
        for chunk in chunks:
            embedding = self.embedding_model.encode(chunk)
            self._store_chunk(doc_id, chunk, embedding, totem_category)
        
        return {
            "entities_added": len(entities),
            "relations_added": len(relations),
            "chunks_indexed": len(chunks)
        }
    
    def retrieve(self, query: str, totem_filter: Optional[str] = None, 
                 top_k: int = 5) -> List[Dict]:
        """
        五图腾感知的检索
        """
        # 1. 查询向量化
        query_embedding = self.embedding_model.encode(query)
        
        # 2. 基础语义检索
        base_results = self._vector_search(query_embedding, top_k * 2)
        
        # 3. 五图腾增强重排
        if totem_filter:
            # 如果指定了图腾，提升该维度相关度
            reranked = self._totem_boost(base_results, totem_filter)
        else:
            # 自动检测查询的图腾倾向
            detected_totem = self._detect_query_totem(query)
            reranked = self._totem_boost(base_results, detected_totem)
        
        # 4. 关系扩展（找到相关实体的邻居）
        expanded = self._expand_by_relations(reranked[:top_k])
        
        # 5. 生成上下文
        context = self._generate_rag_context(expanded)
        
        return {
            "results": expanded,
            "context": context,
            "detected_totem": detected_totem if not totem_filter else totem_filter,
            "sources": list(set([r['source_id'] for r in expanded]))
        }
    
    def traverse_totem_path(self, start_entity: str, totem_path: List[str]) -> Dict:
        """
        五图腾路径遍历：沿着特定图腾维度探索知识网络
        例如：从"合伙人评估"出发，沿着 土→金→水 路径探索
        """
        current = self._get_entity_by_name(start_entity)
        path_result = [current]
        
        for totem in totem_path:
            # 找到当前实体在该图腾维度下最强的关系
            next_entity = self._find_strongest_totem_relation(current['id'], totem)
            if next_entity:
                path_result.append(next_entity)
                current = next_entity
            else:
                break
        
        return {
            "path": path_result,
            "narrative": self._generate_path_narrative(path_result, totem_path)
        }
    
    def find_contradictions(self, concept_a: str, concept_b: str) -> List[Dict]:
        """
        发现知识冲突（蓝军审计用）
        例如：土（长期）vs 火（短期）的冲突
        """
        entity_a = self._get_entity_by_name(concept_a)
        entity_b = self._get_entity_by_name(concept_b)
        
        # 查找直接矛盾关系
        direct = self._query("""
            SELECT * FROM kg_relations 
            WHERE (source_id = %s AND target_id = %s OR source_id = %s AND target_id = %s)
            AND relation_type = 'contradicts'
        """, (entity_a['id'], entity_b['id'], entity_b['id'], entity_a['id']))
        
        # 查找图腾维度冲突
        totem_a = self._get_totem_semantics(entity_a['id'])
        totem_b = self._get_totem_semantics(entity_b['id'])
        
        conflicts = []
        for t_a in totem_a:
            for t_b in totem_b:
                if t_a['totem_type'] != t_b['totem_type'] and \
                   abs(t_a['semantic_weight'] - t_b['semantic_weight']) > 0.5:
                    conflicts.append({
                        "type": "totem_conflict",
                        "totem_a": t_a['totem_type'],
                        "totem_b": t_b['totem_type'],
                        "severity": "medium"
                    })
        
        return conflicts
    
    def _detect_query_totem(self, query: str) -> str:
        """
        自动识别查询的图腾倾向
        """
        keywords = {
            "liu_yuxi": ["长期", "根基", "可持续", "价值", "未来"],
            "simon": ["优化", "边界", "满意", "阈值", "停止", "成本"],
            "guan_zizai": ["直觉", "感受", "觉得", "心安", "感知"],
            "confucius": ["伦理", "仁义", "道德", "关系", "信用"],
            "hui_neng": ["突破", "创新", "顿悟", "本质", "革命"]
        }
        
        scores = {k: sum(1 for word in v if word in query) for k, v in keywords.items()}
        return max(scores, key=scores.get) if max(scores.values()) > 0 else "simon"
    
    def _totem_boost(self, results: List[Dict], totem: str) -> List[Dict]:
        """
        根据图腾相关性重排结果
        """
        for result in results:
            # 获取该结果在指定图腾下的权重
            weight = self._get_totem_weight(result['id'], totem)
            result['totem_score'] = weight
            result['final_score'] = result['similarity'] * 0.7 + weight * 0.3
        
        return sorted(results, key=lambda x: x['final_score'], reverse=True)
    
    def cross_totem_reasoning(self, query: str) -> Dict:
        """
        跨图腾推理：综合五个视角给出答案
        """
        perspectives = {}
        
        for totem in ["liu_yuxi", "simon", "guan_zizai", "confucius", "hui_neng"]:
            # 用图腾过滤检索
            retrieval = self.retrieve(query, totem_filter=totem, top_k=3)
            
            # 生成该视角的回答
            prompt = f"""
            基于以下知识片段，从【{totem}】视角回答问题：
            问题：{query}
            知识：{retrieval['context']}
            
            要求：严格遵循该视角的核心逻辑，给出结构化回答。
            """
            answer = self.call_kimi(prompt)
            perspectives[totem] = answer
        
        # 综合五视角
        synthesis_prompt = f"""
        综合以下五个视角的回答，给出平衡的建议：
        土（长期）：{perspectives['liu_yuxi']}
        金（满意解）：{perspectives['simon']}
        水（直觉）：{perspectives['guan_zizai']}
        木（伦理）：{perspectives['confucius']}
        火（创新）：{perspectives['hui_neng']}
        
        要求：
        1. 指出各视角的共识
        2. 分析冲突点及调和方案
        3. 给出最终建议（标注主要依据的图腾）
        """
        synthesis = self.call_kimi(synthesis_prompt)
        
        return {
            "perspectives": perspectives,
            "synthesis": synthesis,
            "conflicts": self._extract_conflicts_from_perspectives(perspectives)
        }

# /opt/sri-agent-os/core/skill_registry.py

import os
import yaml
import hashlib
from typing import Dict, List, Optional, Tuple
from dataclasses import dataclass
from packaging import version
import psycopg2
import networkx as nx  # 用于依赖图谱

@dataclass
class SkillMetadata:
    name: str
    version: str
    totem_primary: str
    dependencies: List[Dict]
    health_score: float
    last_execution: Optional[str]

class SkillRegistry:
    """
    Skill治理中心
    功能：注册、发现、版本控制、依赖解析、健康监控
    """
    
    def __init__(self):
        self.db = psycopg2.connect("dbname=sri_agent_os user=postgres")
        self.dependency_graph = nx.DiGraph()  # 有向图表示依赖
        self._load_all_skills()
    
    def register(self, manifest_path: str, force: bool = False) -> Dict:
        """
        Skill注册流程（含蓝军审计）
        """
        # 1. 解析Manifest
        with open(manifest_path, 'r') as f:
            manifest = yaml.safe_load(f)
        
        skill_name = manifest['basic']['name']
        skill_version = manifest['basic']['version']
        
        # 2. 蓝军预审（Skeptor-7）
        audit_result = self._blue_team_pre_audit(manifest)
        if audit_result['severity'] == 'critical' and not force:
            raise RegistrationBlocked(f"蓝军审计未通过: {audit_result['findings']}")
        
        # 3. 依赖冲突检查
        conflicts = self._check_dependency_conflicts(manifest['dependencies'])
        if conflicts:
            return {
                "status": "blocked",
                "reason": "依赖冲突",
                "conflicts": conflicts,
                "resolution_guide": self._generate_resolution_guide(conflicts)
            }
        
        # 4. 计算Skill指纹（版本控制）
        skill_hash = self._calculate_skill_hash(manifest_path)
        
        # 5. 入库
        skill_id = self._insert_skill(manifest, skill_hash, audit_result)
        
        # 6. 更新依赖图谱
        self._update_dependency_graph(skill_name, manifest['dependencies'])
        
        # 7. 自动生成五图腾标签
        self._tag_totem_attributes(skill_id, manifest['totem_attributes'])
        
        # 8. 触发依赖该Skill的其他Skill检查（级联验证）
        self._cascade_validation(skill_name)
        
        return {
            "skill_id": skill_id,
            "status": "registered",
            "audit_passed": audit_result['passed'],
            "dependencies_resolved": len(manifest['dependencies']['required_skills']),
            "next_step": "部署到生产环境"
        }
    
    def resolve_dependencies(self, skill_name: str, target_version: Optional[str] = None) -> List[SkillMetadata]:
        """
        依赖解析与拓扑排序
        确保Skill按正确顺序加载，检查循环依赖
        """
        # 检查循环依赖
        if not nx.is_directed_acyclic_graph(self.dependency_graph):
            cycles = list(nx.simple_cycles(self.dependency_graph))
            raise CyclicDependencyError(f"发现循环依赖: {cycles}")
        
        # 拓扑排序：确保依赖项先加载
        load_order = list(nx.topological_sort(self.dependency_graph))
        
        # 筛选相关Skill
        relevant_skills = []
        for name in load_order:
            if name == skill_name or nx.has_path(self.dependency_graph, name, skill_name):
                metadata = self._get_skill_metadata(name)
                relevant_skills.append(metadata)
        
        return relevant_skills
    
    def get_skill_by_totem(self, totem_type: str, capability: Optional[str] = None) -> List[SkillMetadata]:
        """
        按五图腾分类发现Skill
        例如：找到所有"水-观自在"类型的直觉感知类Skill
        """
        query = """
            SELECT s.name, s.version, t.totem_type, s.health_score
            FROM skills s
            JOIN skill_totem_tags t ON s.id = t.skill_id
            WHERE t.totem_type = %s AND s.status = 'production'
        """
        params = [totem_type]
        
        if capability:
            query += " AND s.capability = %s"
            params.append(capability)
        
        cursor = self.db.cursor()
        cursor.execute(query, params)
        
        return [SkillMetadata(*row) for row in cursor.fetchall()]
    
    def health_check(self, skill_name: str) -> Dict:
        """
        Skill健康度检查（五维评估）
        """
        # 获取最近执行指标
        metrics = self._get_recent_metrics(skill_name, days=7)
        
        # 计算五维健康分
        health = {
            "availability": metrics['success_rate'] * 100,  # 可用性
            "performance": max(0, 100 - metrics['avg_latency']),  # 性能（延迟反向）
            "cost_efficiency": max(0, 100 - (metrics['avg_cost'] / 10)),  # 成本效率
            "quality": metrics['first_time_success'] * 100,  # 质量
            "totem_alignment": self._calculate_totem_alignment(skill_name)  # 图腾契合度
        }
        
        overall = sum(health.values()) / 5
        
        # 健康度影响依赖该Skill的其他Skill
        if overall < 60:
            self._alert_dependents(skill_name, health)
        
        return {
            "overall_score": round(overall, 1),
            "dimensions": health,
            "status": "healthy" if overall > 80 else "degraded" if overall > 60 else "critical",
            "recommendation": self._generate_health_recommendation(health)
        }
    
    def migrate_skill(self, skill_name: str, target_model: str) -> Dict:
        """
        模型迁移助手：当需要更换底层模型（如Kimi→其他）时
        实现"换模型迁移成本<1天"的关键功能
        """
        skill = self._get_skill_manifest(skill_name)
        
        # 1. 分析模型依赖
        model_deps = [dep for dep in skill['dependencies']['external_apis'] 
                     if 'kimi' in dep['name'] or 'llm' in dep['name']]
        
        # 2. 生成迁移清单
        migration_plan = {
            "api_changes": [],
            "prompt_adaptations": [],
            "test_revalidation": []
        }
        
        for dep in model_deps:
            if target_model == "gpt-4":
                migration_plan['api_changes'].append({
                    "old": dep['endpoint'],
                    "new": "https://api.openai.com/v1",
                    "auth_changes": "需替换为OpenAI Key"
                })
            elif target_model == "local-llm":
                migration_plan['api_changes'].append({
                    "old": dep['endpoint'],
                    "new": "http://localhost:8000/v1",
                    "note": "需确保本地模型支持相同context长度"
                })
        
        # 3. Prompt适配建议（不同模型的system prompt差异）
        if 'totem' in skill_name:
            migration_plan['prompt_adaptations'].append(
                "五图腾System Prompt需根据新模型的指令跟随能力调整权重"
            )
        
        # 4. 自动生成兼容性测试
        test_cases = self._generate_migration_tests(skill_name)
        
        return {
            "estimated_hours": len(migration_plan['api_changes']) * 2 + 4,
            "migration_plan": migration_plan,
            "test_cases": test_cases,
            "rollback_guide": "保留原模型配置，支持 instant rollback"
        }
    
    def _blue_team_pre_audit(self, manifest: Dict) -> Dict:
        """
        Skill注册前的蓝军审计
        """
        auditor = Skeptor7Auditor()
        
        findings = []
        
        # 检查R001: 敏感数据处理
        if 'privacy' not in manifest.get('security', {}):
            findings.append({
                "rule": "S001",
                "severity": "critical",
                "finding": "Skill未声明隐私数据处理策略"
            })
        
        # 检查R002: 资源限制
        if 'resource_limits' not in manifest:
            findings.append({
                "rule": "R004",
                "severity": "warning",
                "finding": "未设置资源限制，可能导致成本失控"
            })
        
        # 检查R003: 五图腾逻辑一致性
        if manifest['totem_attributes']['primary'] not in [
            "土-刘禹锡", "金-司马贺", "水-观自在", "木-孔子", "火-慧能"
        ]:
            findings.append({
                "rule": "T001",
                "severity": "critical",
                "finding": "无效的图腾分类"
            })
        
        critical_count = len([f for f in findings if f['severity'] == 'critical'])
        
        return {
            "passed": critical_count == 0,
            "severity": "critical" if critical_count > 0 else "warning" if findings else "info",
            "findings": findings
        }
    
    def visualize_dependency_graph(self, skill_name: Optional[str] = None) -> str:
        """
        生成Skill依赖图谱（用于架构治理）
        输出：Mermaid语法或SVG
        """
        if skill_name:
            # 子图：只显示相关依赖
            subgraph = nx.ego_graph(self.dependency_graph, skill_name, radius=2)
            return self._graph_to_mermaid(subgraph)
        else:
            # 全图
            return self._graph_to_mermaid(self.dependency_graph)
    
    def _graph_to_mermaid(self, graph: nx.DiGraph) -> str:
        """
        转换为Mermaid流程图语法
        """
        lines = ["graph TD"]
        
        for node in graph.nodes():
            # 根据图腾类型着色
            totem = graph.nodes[node].get('totem', 'unknown')
            color_map = {
                "土-刘禹锡": "fill:#d4c4b5",
                "金-司马贺": "fill:#f0e68c", 
                "水-观自在": "fill:#b0c4de",
                "木-孔子": "fill:#90ee90",
                "火-慧能": "fill:#ffb6c1"
            }
            style = color_map.get(totem, "fill:#ffffff")
            lines.append(f"    {node}[{node}]:::{totem.replace('-', '')}")
        
        for edge in graph.edges():
            lines.append(f"    {edge[0]} --> {edge[1]}")
        
        # 添加图例样式
        lines.append("    classDef tu fill:#d4c4b5")
        lines.append("    classDef jin fill:#f0e68c")
        lines.append("    classDef shui fill:#b0c4de")
        lines.append("    classDef mu fill:#90ee90")
        lines.append("    classDef huo fill:#ffb6c1")
        
        return "\n".join(lines)

# /opt/sri-agent-os/prompts/template_manager.py

class PromptTemplateManager:
    """
    Prompt模板库管理器
    支持：版本控制、效果追踪、A/B测试、智能推荐
    """
    
    def __init__(self):
        self.db = psycopg2.connect("dbname=sri_agent_os user=postgres")
        self.totem_library = self._load_totem_prompts()
    
    def create_template(self, name: str, totem_mode: str, base_content: str, 
                       variables: List[str]) -> str:
        """
        创建新模板（绑定五图腾模式）
        """
        template_id = str(uuid.uuid4())
        
        # 自动注入对应图腾的System Prompt
        system_prompt = self.totem_library[totem_mode]['system_prompt']
        
        full_template = {
            "id": template_id,
            "name": name,
            "totem_mode": totem_mode,
            "system_prompt": system_prompt,
            "user_template": base_content,
            "variables": variables,
            "version": "1.0.0",
            "created_at": datetime.now().isoformat(),
            "metrics": {
                "usage_count": 0,
                "avg_success_rate": 0.0,
                "avg_token_cost": 0.0
            }
        }
        
        self._save_template(full_template)
        return template_id
    
    def render(self, template_id: str, variables: Dict, context: Dict = None) -> str:
        """
        渲染模板（自动注入五图腾上下文）
        """
        template = self._get_template(template_id)
        
        # 变量替换
        user_content = template['user_template']
        for var, value in variables.items():
            user_content = user_content.replace(f"{{{{{var}}}}}", str(value))
        
        # 添加上下文增强（如果提供了历史记忆）
        if context:
            memory_enhancement = self._retrieve_relevant_memory(context, template['totem_mode'])
            user_content = f"{memory_enhancement}\n\n{user_content}"
        
        # 记录使用（用于复用率统计）
        self._record_usage(template_id)
        
        return {
            "system": template['system_prompt'],
            "user": user_content,
            "totem_mode": template['totem_mode'],
            "estimated_tokens": len(template['system_prompt']) + len(user_content)
        }
    
    def ab_test_create(self, base_template_id: str, variant_changes: Dict) -> str:
        """
        创建A/B测试变体（优化Prompt效果）
        """
        base = self._get_template(base_template_id)
        
        variant = base.copy()
        variant['id'] = str(uuid.uuid4())
        variant['parent_id'] = base_template_id
        variant['version'] = f"{base['version']}-variant"
        variant.update(variant_changes)
        
        # 分配流量：50/50
        self._setup_ab_test(base_template_id, variant['id'])
        
        return variant['id']
    
    def ab_test_evaluate(self, test_id: str) -> Dict:
        """
        评估A/B测试结果（基于质量指标）
        """
        results = self._query_ab_metrics(test_id)
        
        # 计算统计显著性
        base_rate = results['base']['success_rate']
        variant_rate = results['variant']['success_rate']
        cost_diff = results['base']['avg_cost'] - results['variant']['avg_cost']
        
        winner = None
        if variant_rate > base_rate * 1.1 and cost_diff >= 0:  # 成功率提升10%且成本不增
            winner = "variant"
        elif variant_rate >= base_rate and cost_diff > 0:  # 成功率持平但成本降低
            winner = "variant"
        else:
            winner = "base"
        
        return {
            "winner": winner,
            "base_metrics": results['base'],
            "variant_metrics": results['variant'],
            "improvement": f"{(variant_rate - base_rate) / base_rate * 100:.1f}%",
            "recommendation": f"采用{'变体' if winner == 'variant' else '原版本'}作为新版本"
        }
    
    def recommend_template(self, task_description: str, current_totem: Optional[str] = None) -> List[Dict]:
        """
        基于任务描述智能推荐模板（结合五图腾）
        """
        # 1. 语义匹配
        embedding = self._embed(task_description)
        semantic_matches = self._semantic_search(embedding, top_k=5)
        
        # 2. 图腾匹配（如果用户当前处于某决策模式）
        if current_totem:
            totem_matches = self._get_templates_by_totem(current_totem)
            # 加权融合
            combined = self._merge_rankings(semantic_matches, totem_matches, 
                                           weights=[0.6, 0.4])
        else:
            combined = semantic_matches
        
        # 3. 效果排序（优先推荐高成功率模板）
        scored = []
        for tmpl in combined:
            score = tmpl['similarity'] * 0.5 + tmpl['metrics']['avg_success_rate'] * 0.5
            scored.append({**tmpl, "recommend_score": score})
        
        return sorted(scored, key=lambda x: x['recommend_score'], reverse=True)[:3]

-- 物化视图：实时五维指标（每5分钟刷新）

CREATE MATERIALIZED VIEW mv_five_dimensions AS
WITH time_metrics AS (
    SELECT 
        DATE_TRUNC('day', created_at) as date,
        AVG(CASE WHEN task_type = 'knowledge_ingestion' THEN duration_seconds END) as avg_ingestion_time,
        AVG(CASE WHEN task_type LIKE 'workflow_%' THEN duration_seconds END) as avg_workflow_time,
        PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY duration_seconds) as p95_time
    FROM metrics_time
    WHERE created_at >= CURRENT_DATE - INTERVAL '30 days'
    GROUP BY DATE_TRUNC('day', created_at)
),
quality_metrics AS (
    SELECT 
        DATE_TRUNC('day', created_at) as date,
        AVG(CASE WHEN first_time_success THEN 1 ELSE 0 END) * 100 as first_time_success_rate,
        AVG(rework_count) as avg_rework,
        SUM(CASE WHEN verification_score >= 80 THEN 1 ELSE 0 END)::FLOAT / COUNT(*) * 100 as verification_pass_rate
    FROM metrics_quality
    GROUP BY DATE_TRUNC('day', created_at)
),
risk_metrics AS (
    SELECT 
        DATE_TRUNC('day', created_at) as date,
        COUNT(CASE WHEN severity = 'critical' THEN 1 END) as critical_events,
        COUNT(CASE WHEN severity = 'high' THEN 1 END) as high_risk_events,
        COUNT(*) as total_events
    FROM metrics_risk
    GROUP BY DATE_TRUNC('day', created_at)
),
cost_metrics AS (
    SELECT 
        DATE_TRUNC('day', timestamp) as date,
        SUM(total_cost_yuan) as daily_cost,
        AVG(total_cost_yuan) as avg_task_cost,
        SUM(prompt_tokens + completion_tokens) as daily_tokens
    FROM metrics_cost
    GROUP BY DATE_TRUNC('day', timestamp)
),
reuse_metrics AS (
    SELECT 
        DATE_TRUNC('day', access_time) as date,
        COUNT(DISTINCT asset_id) as unique_assets_used,
        COUNT(*) as total_accesses,
        COUNT(*)::FLOAT / COUNT(DISTINCT asset_id) as reuse_ratio
    FROM metrics_reuse
    GROUP BY DATE_TRUNC('day', access_time)
)
SELECT 
    t.date,
    -- 时间维度
    t.avg_ingestion_time,
    t.avg_workflow_time,
    t.p95_time,
    -- 质量维度
    q.first_time_success_rate,
    q.avg_rework,
    q.verification_pass_rate,
    -- 风险维度
    r.critical_events,
    r.high_risk_events,
    (r.critical_events + r.high_risk_events)::FLOAT / NULLIF(r.total_events, 0) * 100 as risk_rate,
    -- 成本维度
    c.daily_cost,
    c.avg_task_cost,
    c.daily_tokens,
    -- 复用维度
    u.unique_assets_used,
    u.total_accesses,
    u.reuse_ratio,
    -- 综合健康分
    (q.first_time_success_rate * 0.3 + (100 - r.risk_rate) * 0.3 + 
     LEAST(u.reuse_ratio * 20, 100) * 0.2 + (100 - LEAST(t.avg_workflow_time/60, 100)) * 0.2) as overall_health_score
FROM time_metrics t
LEFT JOIN quality_metrics q ON t.date = q.date
LEFT JOIN risk_metrics r ON t.date = r.date
LEFT JOIN cost_metrics c ON t.date = c.date
LEFT JOIN reuse_metrics u ON t.date = u.date
ORDER BY t.date DESC;

CREATE INDEX idx_mv_five_dimensions_date ON mv_five_dimensions(date);

class BackupManager:
    """
    3-2-1备份策略执行器
    3份副本，2种介质，1份异地
    """
    
    def __init__(self):
        self.db = psycopg2.connect("dbname=sri_agent_os user=postgres")
        self.s3_client = boto3.client('s3', endpoint_url=os.getenv('S3_ENDPOINT'))
        self.local_backup_path = "/backup"
        self.nas_backup_path = "/archive"
    
    def daily_backup(self):
        """
        每日备份任务
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        
        # 1. 数据库备份（副本1：本地）
        db_backup_file = f"{self.local_backup_path}/db_{timestamp}.sql"
        self._backup_database(db_backup_file)
        
        # 2. 对象存储备份（副本2：本地NAS）
        minio_backup_file = f"{self.nas_backup_path}/minio_{timestamp}.tar.gz"
        self._backup_minio(minio_backup_file)
        
        # 3. 云端异地备份（副本3：云存储）
        if self._verify_backup(db_backup_file):
            self._upload_to_cloud(db_backup_file, f"backups/db_{timestamp}.sql")
            self._upload_to_cloud(minio_backup_file, f"backups/minio_{timestamp}.tar.gz")
        
        # 4. 记录备份日志
        self._log_backup(db_backup_file, minio_backup_file, "success")
        
        # 5. 清理旧备份（保留30天）
        self._cleanup_old_backups(days=30)
    
    def verify_backup_integrity(self, backup_id: str) -> bool:
        """
        备份完整性验证（0错误容忍）
        """
        backup_info = self._get_backup_info(backup_id)
        
        # 检查3个副本是否都存在
        checks = [
            os.path.exists(backup_info['local_path']),
            os.path.exists(backup_info['nas_path']),
            self._check_s3_exists(backup_info['s3_key'])
        ]
        
        if not all(checks):
            # 触发告警
            self._alert_egbertie(f"备份完整性检查失败: {backup_id}")
            return False
        
        # 校验文件哈希
        local_hash = self._calculate_hash(backup_info['local_path'])
        s3_hash = self._get_s3_object_hash(backup_info['s3_key'])
        
        if local_hash != s3_hash:
            self._alert_egbertie(f"备份哈希不匹配: {backup_id}")
            return False
        
        return True

# /opt/sri-agent-os/core/token_optimizer/edge_processor.py

import hashlib
import json
from typing import Dict, List, Optional
import requests
import faiss  # 本地向量检索
import numpy as np

class EdgeTokenOptimizer:
    """
    边缘层Token优化器
    原则：能用本地模型解决的，绝不上云；能缓存复用的，绝不重复计算
    """
    
    def __init__(self):
        self.local_llm_url = "http://localhost:8000/v1/completions"
        self.local_embedding_model = self._load_local_embedding()  # BGE-M3等本地模型
        self.semantic_cache = self._load_cache_index()  # FAISS本地索引
        
    def process_document(self, file_path: str) -> Dict:
        """
        文档预处理全流程（目标：<50T等效云Token）
        """
        # 1. 文件指纹计算（缓存键）
        file_hash = self._compute_hash(file_path)
        
        # 2. 检查精确缓存（完全相同的文件）
        cached = self._check_exact_cache(file_hash)
        if cached:
            return {**cached, "cache_hit": True, "token_cost": 0}
        
        # 3. 本地解析（PDF/Word转文本，零Token）
        raw_text = self._local_parse(file_path)
        
        # 4. 语义指纹（前512字符的向量）
        semantic_key = self._compute_semantic_fingerprint(raw_text[:1000])
        
        # 5. 检查相似缓存（相似度>0.95视为重复）
        similar_doc = self._check_semantic_cache(semantic_key)
        if similar_doc and similar_doc['similarity'] > 0.95:
            return {
                "cache_hit": "similar",
                "reference_doc": similar_doc['doc_id'],
                "adaptation_needed": False,
                "token_cost": 0
            }
        
        # 6. 本地轻量摘要（Qwen-7B，本地计算=0云Token）
        local_summary = self._local_summarize(raw_text)
        
        # 7. 五图腾初分类（关键词规则+轻量模型）
        totem_hint = self._local_totem_classify(local_summary)
        
        # 8. 复杂度评分（决定后续流程）
        complexity = self._assess_complexity(raw_text, local_summary)
        
        # 9. 智能分块（避免超长上下文）
        chunks = self._smart_chunk(raw_text, max_chunk_tokens=1500)
        
        return {
            "file_hash": file_hash,
            "local_summary": local_summary,
            "totem_hint": totem_hint,
            "complexity": complexity,
            "chunks": chunks,
            "estimated_cloud_tokens": self._estimate_tokens(complexity),
            "cache_hit": False
        }
    
    def _local_parse(self, file_path: str) -> str:
        """
        本地文档解析（零Token）
        支持：PDF(pdfplumber), Word(python-docx), Markdown直接读取
        """
        import os
        
        ext = os.path.splitext(file_path)[1].lower()
        
        if ext == '.pdf':
            import pdfplumber
            with pdfplumber.open(file_path) as pdf:
                text = "\n".join(page.extract_text() or "" for page in pdf.pages)
        elif ext in ['.doc', '.docx']:
            import docx
            doc = docx.Document(file_path)
            text = "\n".join(para.text for para in doc.paragraphs)
        elif ext in ['.md', '.txt']:
            with open(file_path, 'r', encoding='utf-8') as f:
                text = f.read()
        else:
            text = ""
            
        return text
    
    def _local_summarize(self, text: str) -> str:
        """
        本地模型生成摘要（零云Token）
        使用4-bit量化的7B模型，本地GPU/CPU推理
        """
        # 截断至4k上下文（本地模型限制）
        truncated = text[:4000]
        
        prompt = f"""请用3句话概括以下文档的核心内容（50字以内）：
{truncated}

摘要："""
        
        response = requests.post(self.local_llm_url, json={
            "model": "local-preprocessor",
            "prompt": prompt,
            "max_tokens": 100,
            "temperature": 0.3
        })
        
        return response.json()['choices'][0]['text'].strip()
    
    def _local_totem_classify(self, summary: str) -> Dict:
        """
        五图腾轻量分类（规则+本地模型）
        先规则匹配，不确定再调本地模型
        """
        # 快速规则匹配（零Token）
        keywords = {
            "土-刘禹锡": ["长期", "根基", "价值", "可持续", "复利"],
            "金-司马贺": ["算法", "优化", "边界", "约束", "满意"],
            "水-观自在": ["直觉", "感知", "感受", "心", "觉"],
            "木-孔子": ["伦理", "仁义", "道德", "信用", "关系"],
            "火-慧能": ["创新", "突破", "顿悟", "革命", "本质"]
        }
        
        scores = {k: sum(1 for w in v if w in summary) for k, v in keywords.items()}
        best_match = max(scores, key=scores.get)
        
        # 如果最高分<2，调本地模型二次判断
        if scores[best_match] < 2:
            prompt = f"""分类以下文本到五图腾之一（仅输出分类名）：
文本：{summary}
选项：土-刘禹锡(长期), 金-司马贺(算法), 水-观自在(直觉), 木-孔子(伦理), 火-慧能(创新)
分类："""
            
            response = requests.post(self.local_llm_url, json={
                "model": "local-preprocessor", 
                "prompt": prompt,
                "max_tokens": 20,
                "temperature": 0.1
            })
            best_match = response.json()['choices'][0]['text'].strip()
        
        return {
            "primary_totem": best_match,
            "confidence": min(scores[best_match] / 3, 1.0),
            "method": "rule" if scores[best_match] >= 2 else "model"
        }
    
    def _assess_complexity(self, raw_text: str, summary: str) -> Dict:
        """
        复杂度评估（决定后续Token投入）
        """
        factors = {
            "length_tier": "short" if len(raw_text) < 2000 else "medium" if len(raw_text) < 8000 else "long",
            "technical_density": raw_text.count("。") / len(raw_text) if len(raw_text) > 0 else 0,  # 句号密度作为复杂度代理
            "has_structured_data": any(x in raw_text for x in ["表格", "数据", "指标", "百分比"]),
            "summary_depth": "deep" if any(w in summary for w in ["分析", "框架", "理论"]) else "surface"
        }
        
        # 复杂度评分 1-5
        score = 1
        if factors['length_tier'] == 'long': score += 1
        if factors['has_structured_data']: score += 1
        if factors['summary_depth'] == 'deep': score += 1
        if len(raw_text) > 10000: score += 1
            
        return {
            "score": score,
            "factors": factors,
            "recommended_pipeline": "light" if score <= 2 else "standard" if score <= 4 else "deep"
        }
    
    def _smart_chunk(self, text: str, max_chunk_tokens: int = 1500) -> List[Dict]:
        """
        智能分块：按语义边界分割，避免在句子中间截断
        目标：每块<1500Token，适合轻量处理
        """
        import re
        
        # 按段落分割
        paragraphs = text.split('\n\n')
        chunks = []
        current_chunk = ""
        current_tokens = 0
        
        for para in paragraphs:
            para_tokens = len(para) / 2  # 中文字符估算：1字≈0.5-1Token
            
            if current_tokens + para_tokens > max_chunk_tokens:
                if current_chunk:
                    chunks.append({
                        "content": current_chunk,
                        "tokens": current_tokens,
                        "type": "semantic_block"
                    })
                current_chunk = para
                current_tokens = para_tokens
            else:
                current_chunk += "\n\n" + para
                current_tokens += para_tokens
        
        if current_chunk:
            chunks.append({
                "content": current_chunk,
                "tokens": current_tokens,
                "type": "semantic_block"
            })
        
        return chunks
    
    def _estimate_tokens(self, complexity: Dict) -> int:
        """
        预估云Token消耗
        """
        pipeline = complexity['recommended_pipeline']
        estimates = {
            "light": 100,      # 轻量：摘要+简单分类
            "standard": 300,   # 标准：三遍阅读辅助
            "deep": 800        # 深度：完整五图腾分析
        }
        return estimates.get(pipeline, 300)

class TieredProcessingPipeline:
    """
    分级处理：简单文档走轻量流程，复杂文档才走深度内化
    目标：80%文档走轻量流程（<150T），20%走深度流程（<500T）
    """
    
    def __init__(self):
        self.kimi_api = KimiClient()  # 官方API
        self.edge_processor = EdgeTokenOptimizer()
        
    def process(self, file_path: str) -> Dict:
        # 边缘预处理（0云Token）
        edge_result = self.edge_processor.process_document(file_path)
        
        if edge_result.get('cache_hit'):
            return self._serve_from_cache(edge_result)
        
        # 根据复杂度分级处理
        pipeline = edge_result['complexity']['recommended_pipeline']
        
        if pipeline == 'light':
            return self._light_pipeline(edge_result)
        elif pipeline == 'standard':
            return self._standard_pipeline(edge_result)
        else:
            return self._deep_pipeline(edge_result)
    
    def _light_pipeline(self, edge_result: Dict) -> Dict:
        """
        轻量流程：适合新闻、短文、简单资料
        Token预算：<150T
        """
        chunks = edge_result['chunks']
        
        # 只处理第一个块（通常包含核心信息）
        main_chunk = chunks[0]['content']
        
        # 单次调用Kimi：摘要+标签+存储建议
        prompt = f"""请对以下文本进行处理（精简输出）：
文本：{main_chunk[:2000]}

任务：
1. 一句话核心观点（20字内）
2. 关键标签（3个）
3. 所属五图腾类别
4. 建议存储位置（Skill/Memory/Case/Workflow）

输出JSON格式，禁止解释性文字。
"""
        
        response = self.kimi_api.complete(
            prompt=prompt,
            max_tokens=200,  # 限制输出长度
            temperature=0.3
        )
        
        # 解析并存储
        result = json.loads(response['text'])
        
        return {
            "processing_tier": "light",
            "token_used": response['usage']['total_tokens'],
            "result": result,
            "full_content_stored": False,  # 只存摘要，原文存档
            "estimated_savings": 600  # 相比标准流程节省的Token
        }
    
    def _standard_pipeline(self, edge_result: Dict) -> Dict:
        """
        标准流程：适合中等复杂度文档
        Token预算：<300T
        """
        # 第一遍：结构提取（边缘层已完成，跳过）
        
        # 第二遍：要点提取（轻量级Prompt）
        chunks = edge_result['chunks'][:2]  # 只取前2块
        
        prompt = f"""基于以下文本片段，提取决策要点：
文本：{chunks[0]['content'][:1500]}

要求：
- 提取3个可执行要点
- 每个要点包含：核心观点+适用场景+风险提示
- 用Markdown表格输出

禁止长篇大论。
"""
        
        response = self.kimi_api.complete(
            prompt=prompt,
            max_tokens=400,
            temperature=0.4
        )
        
        points = response['text']
        
        # 第三遍：五图腾快速检查（非深度分析）
        totem_prompt = f"""用一句话从五图腾视角点评：
{points}

格式：
土(长期)：...
金(满意解)：...
水(直觉)：...
木(伦理)：...
火(创新)：...
"""
        
        totem_response = self.kimi_api.complete(
            prompt=totem_prompt,
            max_tokens=150,
            temperature=0.5
        )
        
        total_tokens = response['usage']['total_tokens'] + totem_response['usage']['total_tokens']
        
        return {
            "processing_tier": "standard",
            "token_used": total_tokens,
            "key_points": points,
            "totem_check": totem_response['text'],
            "full_analysis": False
        }
    
    def _deep_pipeline(self, edge_result: Dict) -> Dict:
        """
        深度流程：仅用于核心文档（<20%比例）
        Token预算：<800T
        """
        # 这里走完整的五重门流程，但仍有优化：
        # 1. 使用边缘层生成的摘要作为上下文，减少重复输入
        # 2. 分块并行处理（如果独立）
        # 3. 缓存中间结果（如第一遍的结构化结果）
        
        return self._full_internalization_workflow(edge_result)

import redis

class TokenCacheManager:
    """
    三级缓存策略：
    L1: 内存缓存（处理中重复调用）
    L2: Redis共享缓存（跨会话复用）
    L3: 向量语义缓存（相似内容复用）
    """
    
    def __init__(self):
        self.l1_cache = {}  # 进程内字典
        self.l2_cache = redis.Redis(host='localhost', port=6379, db=0)
        self.edge_processor = EdgeTokenOptimizer()  # L3语义缓存
        
    def get_cached_completion(self, prompt: str, context: str = "", 
                             use_semantic: bool = False) -> Optional[Dict]:
        """
        获取缓存的Completion结果
        """
        # L1: 精确匹配（当前进程）
        cache_key = self._hash_prompt(prompt, context)
        if cache_key in self.l1_cache:
            return {**self.l1_cache[cache_key], "cache_level": "L1"}
        
        # L2: Redis精确匹配（跨会话）
        cached = self.l2_cache.get(f"sri:completion:{cache_key}")
        if cached:
            result = json.loads(cached)
            self.l1_cache[cache_key] = result  # 回填L1
            return {**result, "cache_level": "L2"}
        
        # L3: 语义相似匹配（向量检索）
        if use_semantic:
            similar = self.edge_processor.find_semantic_similar(prompt + context)
            if similar and similar['similarity'] > 0.92:
                return {
                    "cache_level": "L3",
                    "similarity": similar['similarity'],
                    "adapted_result": self._adapt_result(similar['result'], context)
                }
        
        return None
    
    def cache_completion(self, prompt: str, context: str, result: Dict, 
                        ttl: int = 3600):
        """
        缓存Completion结果
        """
        cache_key = self._hash_prompt(prompt, context)
        
        # 写入L1
        self.l1_cache[cache_key] = result
        
        # 写入L2（带TTL）
        self.l2_cache.setex(
            f"sri:completion:{cache_key}",
            ttl,
            json.dumps(result)
        )
        
        # 如果是高质量结果，写入L3语义缓存
        if result.get('quality_score', 0) > 0.8:
            self.edge_processor.add_to_semantic_cache(prompt + context, result)

class PromptEvolutionEngine:
    """
    Prompt进化引擎
    基于Case Library和用户使用数据，自动优化Prompt
    """
    
    def __init__(self, case_library, memory_store):
        self.cases = case_library
        self.memory = memory_store
        
    async def evolve_prompt(self, prompt_id: str) -> PromptAsset:
        """
        Prompt自动进化：基于成功Case和失败Case的差异学习
        """
        # 1. 获取该Prompt的历史Case
        success_cases = self.cases.query(prompt_id=prompt_id, outcome="success", top_k=10)
        failure_cases = self.cases.query(prompt_id=prompt_id, outcome="failure", top_k=10)
        
        # 2. 分析差异（是什么让成功的成功？）
        diff_analysis = self._analyze_diff(success_cases, failure_cases)
        
        # 3. 生成优化建议（使用P8引擎L4模式）
        optimization = await self._generate_optimization(diff_analysis)
        
        # 4. 创建新版本Prompt
        current = self.memory.get_prompt(prompt_id)
        new_prompt = PromptAsset(
            template=optimization["improved_template"],
            version=self._bump_version(current.version),
            intent_tags=current.intent_tags,
            totem_affinity=optimization.get("totem_affinity", current.totem_affinity),
            parent_prompts=[current.prompt_id],
            evolved_from=success_cases[0].case_id if success_cases else None,
            created_by="ai_evolution"
        )
        
        # 5. 蓝军审计（必须人工确认或自动化测试通过）
        audit_result = await self._blue_army_audit(new_prompt)
        if audit_result["status"] == "approved":
            new_prompt.audit_status = "approved"
            self.memory.store_prompt(new_prompt)
            return new_prompt
        else:
            return None
    
    def _analyze_diff(self, success_cases, failure_cases) -> Dict:
        """分析成功与失败Case的差异"""
        return {
            "success_patterns": [c.execution_path for c in success_cases],
            "failure_patterns": [c.error_type for c in failure_cases],
            "key_differentiators": self._extract_differentiators(success_cases, failure_cases)
        }
    
    async def _blue_army_audit(self, prompt: PromptAsset) -> Dict:
        """蓝军审计Prompt质量"""
        # 模拟审计逻辑
        test_results = []
        for test_case in prompt.test_cases:
            result = await self._run_test(prompt, test_case)
            test_results.append(result)
        
        pass_rate = sum(1 for r in test_results if r["passed"]) / len(test_results)
        
        return {
            "status": "approved" if pass_rate > 0.8 else "rejected",
            "pass_rate": pass_rate,
            "issues": [r for r in test_results if not r["passed"]]
        }

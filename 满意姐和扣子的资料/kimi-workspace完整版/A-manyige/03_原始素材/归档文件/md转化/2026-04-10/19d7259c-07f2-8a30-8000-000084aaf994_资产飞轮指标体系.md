### 2.1 评估器注册表（app/core/registry.py）

关键设计：通过装饰器实现现有模块零改动挂载，支持 SKU 分级和指标自动采集。

"""
评估器注册表 - 实现现有 20+ 模块的无侵入式挂载
"""
import inspect
import time
import functools
from typing import Dict, List, Callable, Any, Optional
from dataclasses import dataclass, field
from enum import Enum
import asyncio

class SKUType(str, Enum):
    A = "A"  # 轻咨询：3-5 个核心评估器
    B = "B"  # 核心产品：全量 8+
    C = "C"  # 年度托管：全量 + diff 追踪

@dataclass
class EvaluatorMetadata:
    """评估器元数据"""
    name: str
    func: Callable
    category: str  # strategy/operation/risk/finance
    sku: List[SKUType]  # 支持的 SKU 类型
    description: str = ""
    version: str = "1.0.0"
    metrics_enabled: bool = True  # 是否采集指标
    
    # 运行时统计（指标层）
    total_calls: int = 0
    total_duration: float = 0.0
    success_count: int = 0
    failure_count: int = 0
    
    def avg_duration(self) -> float:
        if self.total_calls == 0:
            return 0.0
        return self.total_duration / self.total_calls
    
    def success_rate(self) -> float:
        if self.total_calls == 0:
            return 0.0
        return self.success_count / self.total_calls

class EvaluatorRegistry:
    """评估器注册表 - 单例模式"""
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._evaluators: Dict[str, EvaluatorMetadata] = {}
            cls._instance._categories: Dict[str, List[str]] = {}
        return cls._instance
    
    def register(
        self,
        name: str,
        category: str = "general",
        sku: List[SKUType] = None,
        description: str = "",
        version: str = "1.0.0",
        metrics_enabled: bool = True
    ):
        """
        装饰器：注册评估器，支持零改动挂载现有模块
        
        Usage:
            @registry.register(name="market_position", sku=[SKUType.B, SKUType.C])
            def evaluate_market(data: dict) -> dict:
                ...
        """
        if sku is None:
            sku = [SKUType.B, SKUType.C]
            
        def decorator(func: Callable) -> Callable:
            # 包装函数，自动注入指标采集
            @functools.wraps(func)
            async def async_wrapper(*args, **kwargs):
                return await self._execute_with_metrics(name, func, *args, **kwargs)
            
            @functools.wraps(func)
            def sync_wrapper(*args, **kwargs):
                return self._execute_with_metrics_sync(name, func, *args, **kwargs)
            
            # 判断是同步还是异步
            wrapped = async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
            
            metadata = EvaluatorMetadata(
                name=name,
                func=wrapped,
                original_func=func,  # 保留原始函数引用
                category=category,
                sku=sku,
                description=description or func.__doc__ or "",
                version=version,
                metrics_enabled=metrics_enabled
            )
            
            self._evaluators[name] = metadata
            if category not in self._categories:
                self._categories[category] = []
            self._categories[category].append(name)
            
            return wrapped
        return decorator
    
    async def _execute_with_metrics(self, name: str, func: Callable, *args, **kwargs):
        """异步执行并采集指标"""
        metadata = self._evaluators[name]
        start_time = time.time()
        
        try:
            result = await func(*args, **kwargs)
            metadata.success_count += 1
            return result
        except Exception as e:
            metadata.failure_count += 1
            raise
        finally:
            duration = time.time() - start_time
            metadata.total_calls += 1
            metadata.total_duration += duration
    
    def _execute_with_metrics_sync(self, name: str, func: Callable, *args, **kwargs):
        """同步执行并采集指标"""
        metadata = self._evaluators[name]
        start_time = time.time()
        
        try:
            result = func(*args, **kwargs)
            metadata.success_count += 1
            return result
        except Exception as e:
            metadata.failure_count += 1
            raise
        finally:
            duration = time.time() - start_time
            metadata.total_calls += 1
            metadata.total_duration += duration
    
    def get(self, name: str) -> Optional[EvaluatorMetadata]:
        return self._evaluators.get(name)
    
    def list_by_sku(self, sku: SKUType) -> List[EvaluatorMetadata]:
        """根据 SKU 类型获取评估器列表"""
        return [m for m in self._evaluators.values() if sku in m.sku]
    
    def list_all(self) -> List[EvaluatorMetadata]:
        return list(self._evaluators.values())
    
    def get_metrics(self) -> Dict[str, dict]:
        """获取所有评估器指标（用于经营层）"""
        return {
            name: {
                "total_calls": m.total_calls,
                "avg_duration": m.avg_duration(),
                "success_rate": m.success_rate(),
                "category": m.category,
                "version": m.version
            }
            for name, m in self._evaluators.items()
        }

# 全局注册表实例

### 2.2 现有模块加载器（app/evaluators/loader.py）

关键设计：自动发现并包装 existing_modules 目录下的离线模块。

"""
现有模块加载器 - 自动挂载 20+ 离线评估模块
"""
import importlib
import importlib.util
import sys
from pathlib import Path
from typing import List
import logging

from app.core.registry import registry, SKUType

logger = logging.getLogger(__name__)

class ExistingModuleLoader:
    """加载现有离线模块，通过装饰器自动注册"""
    
    def __init__(self, modules_path: str = "existing_modules"):
        self.modules_path = Path(modules_path)
        self.loaded_modules = []
    
    def load_all(self):
        """自动发现并加载所有模块"""
        if not self.modules_path.exists():
            logger.warning(f"Modules path {self.modules_path} not found, skipping")
            return
        
        for py_file in self.modules_path.glob("*.py"):
            if py_file.name.startswith("_"):
                continue
            self._load_module(py_file)
    
    def _load_module(self, file_path: Path):
        """加载单个模块并自动注册符合规范的函数"""
        module_name = f"existing_modules.{file_path.stem}"
        
        spec = importlib.util.spec_from_file_location(module_name, file_path)
        module = importlib.util.module_from_spec(spec)
        sys.modules[module_name] = module
        spec.loader.exec_module(module)
        
        # 自动注册带有 __evaluator__ 属性的函数
        for attr_name in dir(module):
            attr = getattr(module, attr_name)
            if callable(attr) and hasattr(attr, '__evaluator__'):
                config = attr.__evaluator__
                registry.register(
                    name=config.get("name", attr_name),
                    category=config.get("category", "general"),
                    sku=[SKUType(s) for s in config.get("sku", ["B", "C"])],
                    description=config.get("description", ""),
                    version=config.get("version", "1.0.0")
                )(attr)
                logger.info(f"Registered existing module: {config.get('name', attr_name)}")
                self.loaded_modules.append(attr_name)
        
        # 如果没有 __evaluator__ 标记，尝试智能推断（通过函数名）
        self._auto_register_by_naming(module, file_path.stem)
    
    def _auto_register_by_naming(self, module, module_name: str):
        """通过命名约定自动注册（兜底方案）"""
        for attr_name in dir(module):
            if attr_name.startswith("evaluate_"):
                attr = getattr(module, attr_name)
                if callable(attr):
                    eval_name = attr_name.replace("evaluate_", "")
                    registry.register(
                        name=f"{module_name}.{eval_name}",
                        category=self._infer_category(eval_name),
                        sku=[SKUType.B, SKUType.C]
                    )(attr)
                    logger.info(f"Auto-registered: {module_name}.{eval_name}")
    
    def _infer_category(self, name: str) -> str:
        """根据名称推断分类"""
        if any(k in name for k in ["market", "competition", "strategy"]):
            return "strategy"
        elif any(k in name for k in ["risk", "compliance", "legal"]):
            return "risk"
        elif any(k in name for k in ["finance", "cost", "revenue"]):
            return "finance"
        return "operation"

# 挂载装饰器（供现有模块使用，无需修改原函数逻辑）
def mount_evaluator(
    name: str = None,
    category: str = "general",
    sku: List[str] = None,
    description: str = "",
    version: str = "1.0.0"
):
    """
    供现有模块使用的挂载装饰器
    
    Usage（在 existing_modules/module_a.py 中）:
        from app.evaluators.loader import mount_evaluator
        
        @mount_evaluator(name="market_share", category="strategy")
        def calculate_market_share(data):
            ...
    """
    if sku is None:
        sku = ["B", "C"]
    
    def decorator(func):
        func.__evaluator__ = {
            "name": name or func.__name__,
            "category": category,
            "sku": sku,
            "description": description or func.__doc__ or "",
            "version": version
        }
        return func
    return decorator

"""
资产飞轮引擎 - 实现资产层的自增强循环
"""
import json
import hashlib
from datetime import datetime
from typing import List, Optional, Dict, Any
from sqlmodel import SQLModel, Field, Relationship, Session, select
from sqlalchemy import Column, JSON

from app.db.database import engine

class AssetBase(SQLModel):
    """资产基类"""
    name: str = Field(index=True)
    version: str = Field(default="1.0.0")
    content: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    tags: List[str] = Field(default_factory=list, sa_column=Column(JSON))
    reuse_count: int = Field(default=0)  # 复用次数（关键指标）
    last_used: Optional[datetime] = None
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)

class Prompt(AssetBase, table=True):
    """Prompt 资产"""
    __tablename__ = "prompts"
    id: Optional[int] = Field(default=None, primary_key=True)
    template: str = Field(default="")  # 提示词模板
    variables: List[str] = Field(default_factory=list, sa_column=Column(JSON))
    
    # 关联
    workflows: List["Workflow"] = Relationship(back_populates="prompt")

class Skill(AssetBase, table=True):
    """Skill 资产（可复用的评估能力）"""
    __tablename__ = "skills"
    id: Optional[int] = Field(default=None, primary_key=True)
    evaluator_name: str = Field(index=True)  # 关联的评估器
    config: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    
    # 关联
    workflows: List["Workflow"] = Relationship(back_populates="skill")

class Memory(AssetBase, table=True):
    """Memory 资产（案例库/经验沉淀）"""
    __tablename__ = "memories"
    id: Optional[int] = Field(default=None, primary_key=True)
    case_type: str = Field(index=True)  # 案例类型
    anonymized_data: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    evaluation_result: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    
    # 关联
    workflows: List["Workflow"] = Relationship(back_populates="memory")

class Workflow(SQLModel, table=True):
    """Workflow 资产（编排 Prompt + Skill + Memory）"""
    __tablename__ = "workflows"
    id: Optional[int] = Field(default=None, primary_key=True)
    name: str = Field(index=True)
    description: str = ""
    status: str = Field(default="active")  # active/archived
    
    # 外键关联
    prompt_id: Optional[int] = Field(default=None, foreign_key="prompts.id")
    skill_id: Optional[int] = Field(default=None, foreign_key="skills.id")
    memory_id: Optional[int] = Field(default=None, foreign_key="memories.id")
    
    # 关联关系
    prompt: Optional[Prompt] = Relationship(back_populates="workflows")
    skill: Optional[Skill] = Relationship(back_populates="workflows")
    memory: Optional[Memory] = Relationship(back_populates="workflows")
    
    # 执行统计（时间指标）
    execution_count: int = Field(default=0)
    avg_execution_time: float = Field(default=0.0)
    last_execution: Optional[datetime] = None
    
    created_at: datetime = Field(default_factory=datetime.utcnow)

class AssetFlywheel:
    """
    资产飞轮引擎
    
    核心逻辑：
    1. Workflow 执行 -> 沉淀为 Memory（案例库）
    2. Memory 分析 -> 优化 Prompt
    3. Prompt 升级 -> 提升 Skill 准确度
    4. Skill 增强 -> 加速 Workflow 执行
    """
    
    def __init__(self):
        self.engine = engine
    
    def execute_workflow(
        self,
        workflow_id: int,
        input_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        执行工作流，并自动沉淀资产
        """
        with Session(self.engine) as session:
            workflow = session.get(Workflow, workflow_id)
            if not workflow:
                raise ValueError(f"Workflow {workflow_id} not found")
            
            import time
            start_time = time.time()
            
            # 1. 加载 Prompt
            prompt_template = workflow.prompt.template if workflow.prompt else "{}"
            prompt_content = prompt_template.format(**input_data)
            
            # 2. 执行 Skill（调用注册的评估器）
            from app.core.registry import registry
            skill_config = workflow.skill.config if workflow.skill else {}
            evaluator = registry.get(workflow.skill.evaluator_name) if workflow.skill else None
            
            if evaluator:
                result = evaluator.original_func({
                    **input_data,
                    **skill_config,
                    "prompt": prompt_content
                })
            else:
                result = {"status": "noop", "data": input_data}
            
            execution_time = time.time() - start_time
            
            # 3. 更新 Workflow 统计（时间指标）
            workflow.execution_count += 1
            workflow.last_execution = datetime.utcnow()
            # 滑动平均更新
            workflow.avg_execution_time = (
                (workflow.avg_execution_time * (workflow.execution_count - 1) + execution_time)
                / workflow.execution_count
            )
            
            # 4. 复用率更新（Reuse 指标）
            if workflow.prompt:
                workflow.prompt.reuse_count += 1
                workflow.prompt.last_used = datetime.utcnow()
            if workflow.skill:
                workflow.skill.reuse_count += 1
                workflow.skill.last_used = datetime.utcnow()
            
            # 5. 沉淀 Memory（案例库）- 匿名化存储
            if result.get("worth_saving", True):
                memory = Memory(
                    name=f"Case_{workflow.name}_{datetime.utcnow().isoformat()}",
                    case_type=workflow.name,
                    anonymized_data=self._anonymize(input_data),
                    evaluation_result=result,
                    reuse_count=0
                )
                session.add(memory)
            
            session.commit()
            
            return {
                "workflow_id": workflow_id,
                "execution_time": execution_time,
                "result": result,
                "assets_reused": {
                    "prompt": workflow.prompt.reuse_count if workflow.prompt else 0,
                    "skill": workflow.skill.reuse_count if workflow.skill else 0
                }
            }
    
    def _anonymize(self, data: Dict) -> Dict:
        """数据匿名化处理"""
        # 简单实现：移除敏感字段
        sensitive_keys = ['name', 'email', 'phone', 'company_name']
        return {k: v for k, v in data.items() if k not in sensitive_keys}
    
    def get_reuse_metrics(self) -> Dict[str, float]:
        """
        获取复用率指标（经营层关键指标）
        """
        with Session(self.engine) as session:
            prompts = session.exec(select(Prompt)).all()
            skills = session.exec(select(Skill)).all()
            memories = session.exec(select(Memory)).all()
            
            total_reuse = sum(p.reuse_count for p in prompts) + \
                         sum(s.reuse_count for s in skills)
            total_assets = len(prompts) + len(skills)
            
            return {
                "overall_reuse_rate": total_reuse / max(total_assets, 1),
                "prompt_avg_reuse": sum(p.reuse_count for p in prompts) / max(len(prompts), 1),
                "skill_avg_reuse": sum(s.reuse_count for s in skills) / max(len(skills), 1),
                "memory_count": len(memories),
                "total_assets": total_assets
            }
    
    def optimize_prompt_from_memory(self, prompt_id: int) -> str:
        """
        基于 Memory 优化 Prompt（飞轮增强循环）
        """
        with Session(self.engine) as session:
            prompt = session.get(Prompt, prompt_id)
            if not prompt:
                raise ValueError(f"Prompt {prompt_id} not found")
            
            # 获取相关 Memory
            memories = session.exec(
                select(Memory).where(Memory.case_type == prompt.name)
            ).all()
            
            if len(memories) < 5:
                return "Insufficient memory for optimization"
            
            # 分析高频成功模式
            success_patterns = [
                m.evaluation_result for m in memories 
                if m.evaluation_result.get("score", 0) > 0.8
            ]
            
            # 生成优化建议（简化版）
            optimization = f"""
            # 基于 {len(success_patterns)} 个成功案例优化
            # 原模板: {prompt.template[:100]}...
            # 建议: 增加针对 {len(success_patterns)} 个成功模式的指导
            """
            
            return optimization

### 2.4 指标体系（app/metrics/collector.py）

实现五类经营指标（时间、质量、风险、成本、复用率）的自动采集。

"""
指标采集器 - 实现经营层的五类指标
"""
import time
from datetime import datetime, timedelta
from typing import Dict, List, Optional
from dataclasses import dataclass, asdict
from enum import Enum
import json

from sqlmodel import SQLModel, Field, Session, select
from sqlalchemy import Column, JSON

from app.db.database import engine

class MetricType(str, Enum):
    TIME = "time"           # 处理时长、响应时间
    QUALITY = "quality"     # 准确率、成功率
    RISK = "risk"           # 合规事件数、失败率
    COST = "cost"           # 单次任务成本、资源消耗
    REUSE = "reuse"         # 资产复用率

@dataclass
class MetricReading:
    """指标读数"""
    timestamp: datetime
    value: float
    unit: str
    context: Dict = None
    
    def to_dict(self):
        return {
            "timestamp": self.timestamp.isoformat(),
            "value": self.value,
            "unit": self.unit,
            "context": self.context or {}
        }

class MetricRecord(SQLModel, table=True):
    """指标记录表"""
    __tablename__ = "metrics"
    id: Optional[int] = Field(default=None, primary_key=True)
    metric_type: str = Field(index=True)  # MetricType
    name: str = Field(index=True)         # 具体指标名
    value: float
    unit: str
    context: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    created_at: datetime = Field(default_factory=datetime.utcnow)

class MetricsCollector:
    """
    五类指标采集器
    
    对应经营层关注的：
    - Time: 平均处理时长、工作流执行时间
    - Quality: 一次成功率、评估准确度
    - Risk: 合规事件数、失败率
    - Cost: 单位任务成本、资源消耗
    - Reuse: 资产复用率
    """
    
    def __init__(self):
        self.engine = engine
        self._buffer: List[MetricRecord] = []
        self._buffer_size = 100
    
    def record(
        self,
        metric_type: MetricType,
        name: str,
        value: float,
        unit: str,
        context: Dict = None
    ):
        """记录指标"""
        record = MetricRecord(
            metric_type=metric_type.value,
            name=name,
            value=value,
            unit=unit,
            context=context or {}
        )
        self._buffer.append(record)
        
        # 批量写入
        if len(self._buffer) >= self._buffer_size:
            self.flush()
    
    def flush(self):
        """批量刷盘"""
        if not self._buffer:
            return
        
        with Session(self.engine) as session:
            for record in self._buffer:
                session.add(record)
            session.commit()
        self._buffer.clear()
    
    def get_dashboard(
        self,
        start_time: Optional[datetime] = None,
        end_time: Optional[datetime] = None
    ) -> Dict[str, Any]:
        """
        生成指标看板（经营层视图）
        """
        if not end_time:
            end_time = datetime.utcnow()
        if not start_time:
            start_time = end_time - timedelta(days=30)
        
        with Session(self.engine) as session:
            # 获取各类指标最新值和趋势
            dashboard = {}
            
            for mtype in MetricType:
                records = session.exec(
                    select(MetricRecord)
                    .where(MetricRecord.metric_type == mtype.value)
                    .where(MetricRecord.created_at >= start_time)
                    .where(MetricRecord.created_at <= end_time)
                    .order_by(MetricRecord.created_at.desc())
                ).all()
                
                if not records:
                    continue
                
                latest = records[0]
                values = [r.value for r in records]
                
                dashboard[mtype.value] = {
                    "current": {
                        "value": latest.value,
                        "unit": latest.unit,
                        "updated_at": latest.created_at.isoformat()
                    },
                    "statistics": {
                        "avg": sum(values) / len(values),
                        "min": min(values),
                        "max": max(values),
                        "count": len(values)
                    },
                    "trend": [
                        {"time": r.created_at.isoformat(), "value": r.value}
                        for r in records[:30]  # 最近 30 个点
                    ][::-1]
                }
            
            # 计算复合指标
            dashboard["composite"] = self._calculate_composite_score(dashboard)
            
            return dashboard
    
    def _calculate_composite_score(self, dashboard: Dict) -> Dict:
        """计算综合经营评分"""
        scores = {}
        
        # 质量分（成功率）
        if "quality" in dashboard:
            scores["quality_score"] = dashboard["quality"]["current"]["value"] * 100
        
        # 效率分（基于时间，越短越好）
        if "time" in dashboard:
            avg_time = dashboard["time"]["statistics"]["avg"]
            scores["efficiency_score"] = max(0, 100 - avg_time / 60)  # 假设基准 60 秒
        
        # 风险分（合规事件越少越好）
        if "risk" in dashboard:
            risk_count = dashboard["risk"]["current"]["value"]
            scores["risk_score"] = max(0, 100 - risk_count * 10)
        
        # 成本分（单位成本越低越好）
        if "cost" in dashboard:
            cost = dashboard["cost"]["current"]["value"]
            scores["cost_score"] = max(0, 100 - cost / 10)
        
        # 复用分
        if "reuse" in dashboard:
            scores["reuse_score"] = dashboard["reuse"]["current"]["value"] * 100
        
        if scores:
            scores["overall"] = sum(scores.values()) / len(scores)
        
        return scores

# 装饰器：自动采集函数执行指标
def capture_metrics(
    metric_type: MetricType,
    name: str = None,
    unit: str = "seconds"
):
    """
    自动采集函数执行指标的装饰器
    
    Usage:
        @capture_metrics(MetricType.TIME, "workflow_execution")
        def execute_workflow(...):
            ...
    """
    def decorator(func):
        def wrapper(*args, **kwargs):
            collector = MetricsCollector()
            start_time = time.time()
            
            try:
                result = func(*args, **kwargs)
                
                # 记录成功指标
                if metric_type == MetricType.TIME:
                    duration = time.time() - start_time
                    collector.record(
                        metric_type=metric_type,
                        name=name or func.__name__,
                        value=duration,
                        unit=unit,
                        context={"status": "success"}
                    )
                elif metric_type == MetricType.QUALITY:
                    collector.record(
                        metric_type=metric_type,
                        name=name or func.__name__,
                        value=1.0,  # 成功
                        unit="success_rate",
                        context={}
                    )
                
                return result
                
            except Exception as e:
                # 记录失败指标
                if metric_type == MetricType.QUALITY:
                    collector.record(
                        metric_type=metric_type,
                        name=name or func.__name__,
                        value=0.0,  # 失败
                        unit="success_rate",
                        context={"error": str(e)}
                    )
                elif metric_type == MetricType.RISK:
                    collector.record(
                        metric_type=metric_type,
                        name=f"{name or func.__name__}_failure",
                        value=1.0,
                        unit="incident",
                        context={"error": str(e)}
                    )
                raise
            finally:
                collector.flush()
        
        return wrapper
    return decorator

async def run_evaluation(execution_id: str, request: EvaluationRequest):
    """后台执行评估"""
    store = execution_store[execution_id]
    store["status"] = "running"
    
    collector = MetricsCollector()
    flywheel = AssetFlywheel()
    report_service = ReportService()
    
    start_time = datetime.utcnow()
    
    try:
        # 1. 获取评估器列表（根据 SKU）
        evaluators = get_sku_evaluators(request.sku)
        if not evaluators:
            raise ValueError("No evaluators available for this SKU")
        
        store["logs"].append(f"Loaded {len(evaluators)} evaluators for SKU-{request.sku.value}")
        
        # 2. 执行评估（串行或并行）
        results = {}
        for evaluator in evaluators:
            try:
                # 检查是否使用工作流
                if request.workflow_id:
                    wf_result = flywheel.execute_workflow(
                        request.workflow_id,
                        request.input_data
                    )
                    results[evaluator.name] = wf_result
                else:
                    # 直接执行评估器
                    if asyncio.iscoroutinefunction(evaluator.original_func):
                        result = await evaluator.original_func(request.input_data)
                    else:
                        result = evaluator.original_func(request.input_data)
                    results[evaluator.name] = result
                
                # 采集质量指标
                collector.record(
                    metric_type=MetricType.QUALITY,
                    name=f"eval_{evaluator.name}",
                    value=1.0 if not result.get("error") else 0.0,
                    unit="success_rate"
                )
                
            except Exception as e:
                results[evaluator.name] = {"error": str(e)}
                collector.record(
                    metric_type=MetricType.RISK,
                    name=f"eval_{evaluator.name}_error",
                    value=1.0,
                    unit="incident"
                )
        
        # 3. 生成报告（根据 SKU 格式）
        report_format = "markdown" if request.sku == SKUType.A else "pdf"
        report_result = await report_service.generate(
            results=results,
            format=report_format,
            sku=request.sku,
            input_data=request.input_data
        )
        
        # 4. SKU-C: 添加历史对比
        if request.sku == SKUType.C:
            report_result["diff"] = await calculate_historical_diff(
                request.input_data.get("company_id"),
                results
            )
        
        # 5. 采集时间和成本指标
        duration = (datetime.utcnow() - start_time).total_seconds()
        collector.record(
            metric_type=MetricType.TIME,
            name="evaluation_total_time",
            value=duration,
            unit="seconds"
        )
        collector.record(
            metric_type=MetricType.COST,
            name="evaluation_cost",
            value=duration * 0.01,  # 假设成本模型：0.01 元/秒
            unit="yuan"
        )
        
        # 更新存储
        store["status"] = "completed"
        store["completed_at"] = datetime.utcnow()
        store["results"] = results
        store["report"] = report_result
        
    except Exception as e:
        store["status"] = "failed"
        store["error"] = str(e)
        collector.record(
            metric_type=MetricType.RISK,
            name="evaluation_fatal_error",
            value=1.0,
            unit="incident",
            context={"error": str(e)}
        )
    finally:
        collector.flush()

# app/intelligence/__init__.py
"""
情报采集层 - 基于现有工具链的封装
复用：kimi_search（联网搜索）+ agent-reach（14 平台抓取）
"""

# app/intelligence/collector.py
from typing import List, Dict, Any, Optional
from dataclasses import dataclass
from datetime import datetime
import asyncio
import hashlib
import json

from sqlmodel import SQLModel, Field, Session, select
from sqlalchemy import Column, JSON

from app.db.database import engine
from app.metrics.collector import MetricsCollector, MetricType

@dataclass
class IntelligenceSource:
    """情报来源配置"""
    name: str  # kimi_search, agent-reach-wechat, agent-reach-zhihu, etc.
    query_template: str
    weight: float = 1.0  # 数据可信度权重
    requires_auth: bool = False

class RawIntelligence(SQLModel, table=True):
    """原始情报表（合规存储）"""
    __tablename__ = "intelligence_raw"
    id: Optional[int] = Field(default=None, primary_key=True)
    source: str = Field(index=True)  # kimi_search / agent-reach-xxx
    query_hash: str = Field(index=True)  # 查询去重
    query_params: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    content: Dict = Field(default_factory=dict, sa_column=Column(JSON))
    collected_at: datetime = Field(default_factory=datetime.utcnow)
    expires_at: datetime  # 情报有效期
    is_processed: bool = Field(default=False)

class IntelligenceProcessor:
    """
    情报处理器
    - 缓存策略：相同查询 24h 内不复采
    - 合规检查：只存公开数据，自动敏感信息过滤
    - 质量评分：基于 source weight 和 content 完整性
    """
    
    def __init__(self):
        self.engine = engine
        self.metrics = MetricsCollector()
        self.sources = self._init_sources()
    
    def _init_sources(self) -> List[IntelligenceSource]:
        """初始化已授权的数据源（严格遵守合规红线）"""
        return [
            IntelligenceSource(
                name="kimi_search",
                query_template="site:{domain} {keyword}",
                weight=0.9,
                requires_auth=False  # 公开搜索
            ),
            IntelligenceSource(
                name="agent-reach-wechat",
                query_template="{keyword} site:mp.weixin.qq.com",
                weight=0.8,
                requires_auth=False  # 公众号公开文章
            ),
            IntelligenceSource(
                name="agent-reach-zhihu",
                query_template="{keyword} site:zhihu.com",
                weight=0.85,
                requires_auth=False
            ),
            # ... 其他 11 个平台
        ]
    
    async def collect(
        self,
        query: str,
        context: Dict[str, Any],
        force_refresh: bool = False
    ) -> Dict[str, Any]:
        """
        采集情报（异步，带缓存）
        
        Args:
            query: 查询关键词
            context: 上下文（如 company_name, industry）
            force_refresh: 是否强制刷新缓存
        """
        query_hash = self._hash_query(query, context)
        
        # 1. 检查缓存
        if not force_refresh:
            cached = self._get_cache(query_hash)
            if cached:
                self.metrics.record(
                    MetricType.REUSE, "intelligence_cache_hit", 1, "count"
                )
                return cached.content
        
        # 2. 并发采集多源数据
        tasks = []
        for source in self.sources:
            if source.name == "kimi_search":
                tasks.append(self._call_kimi_search(source, query, context))
            elif source.name.startswith("agent-reach"):
                tasks.append(self._call_agent_reach(source, query, context))
        
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 3. 融合与评分
        merged = self._merge_intelligence(results)
        
        # 4. 持久化（合规：只存公开摘要）
        self._save_intelligence(query_hash, query, context, merged)
        
        # 5. 采集指标
        self.metrics.record(
            MetricType.COST, "intelligence_collection", 
            len(self.sources) * 0.5, "yuan"  # 假设每次采集成本
        )
        
        return merged
    
    def _hash_query(self, query: str, context: Dict) -> str:
        """生成查询指纹"""
        content = f"{query}:{json.dumps(context, sort_keys=True)}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def _get_cache(self, query_hash: str) -> Optional[RawIntelligence]:
        """获取缓存（24h 内有效）"""
        with Session(self.engine) as session:
            result = session.exec(
                select(RawIntelligence)
                .where(RawIntelligence.query_hash == query_hash)
                .where(RawIntelligence.expires_at > datetime.utcnow())
            ).first()
            return result
    
    async def _call_kimi_search(
        self, 
        source: IntelligenceSource, 
        query: str, 
        context: Dict
    ) -> Dict:
        """调用现有 kimi_search 工具（模拟，实际为平台内部调用）"""
        try:
            # 实际实现：调用 OpenClaw 平台已有的 kimi_search Skill
            # from openclaw.skills import kimi_search
            # result = await kimi_search.search(query)
            
            # 模拟实现（零空气代码原则：实际应替换为真实调用）
            await asyncio.sleep(0.1)  # 模拟网络延迟
            return {
                "source": source.name,
                "data": [{"title": f"Result for {query}", "url": "https://example.com"}],
                "weight": source.weight,
                "status": "success"
            }
        except Exception as e:
            return {"source": source.name, "error": str(e), "weight": 0}
    
    async def _call_agent_reach(
        self,
        source: IntelligenceSource,
        query: str,
        context: Dict
    ) -> Dict:
        """调用现有 agent-reach 工具（14 平台抓取）"""
        try:
            platform = source.name.replace("agent-reach-", "")
            
            # 实际实现：调用 OpenClaw 平台已有的 agent-reach Skill
            # from openclaw.skills import agent_reach
            # result = await agent_reach.fetch(platform, query)
            
            await asyncio.sleep(0.15)
            return {
                "source": source.name,
                "platform": platform,
                "data": [{"content": f"Data from {platform}"}],
                "weight": source.weight,
                "status": "success"
            }
        except Exception as e:
            return {"source": source.name, "error": str(e), "weight": 0}
    
    def _merge_intelligence(self, results: List[Dict]) -> Dict:
        """多源数据融合（去重、加权评分）"""
        valid_results = [r for r in results if "error" not in r]
        
        if not valid_results:
            return {"status": "failed", "reason": "all_sources_failed"}
        
        # 加权平均计算置信度
        total_weight = sum(r.get("weight", 0) for r in valid_results)
        confidence = min(0.95, total_weight / len(valid_results))
        
        return {
            "status": "success",
            "confidence": confidence,
            "sources_count": len(valid_results),
            "data": valid_results,
            "collected_at": datetime.utcnow().isoformat()
        }
    
    def _save_intelligence(
        self, 
        query_hash: str, 
        query: str, 
        context: Dict, 
        merged: Dict
    ):
        """保存到数据库（自动匿名化）"""
        # 匿名化：移除可能的人员姓名、联系方式
        anonymized = self._anonymize_content(merged)
        
        with Session(self.engine) as session:
            intel = RawIntelligence(
                source="merged",
                query_hash=query_hash,
                query_params={"query": query, **context},
                content=anonymized,
                expires_at=datetime.utcnow() + timedelta(hours=24)
            )
            session.add(intel)
            session.commit()
    
    def _anonymize_content(self, content: Dict) -> Dict:
        """PIL 级别的数据匿名化（保护隐私）"""
        # 递归扫描并标记敏感字段
        sensitive_patterns = ['phone', 'email', 'name', 'contact', 'address']
        
        def scan_and_mask(obj):
            if isinstance(obj, dict):
                return {
                    k: "[REDACTED]" if any(p in k.lower() for p in sensitive_patterns) else scan_and_mask(v)
                    for k, v in obj.items()
                }
            elif isinstance(obj, list):
                return [scan_and_mask(item) for item in obj]
            elif isinstance(obj, str):
                # 简单的手机号/邮箱正则匹配
                import re
                text = obj
                text = re.sub(r'\b1[3-9]\d{9}\b', '[PHONE]', text)
                text = re.sub(r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b', '[EMAIL]', text)
                return text
            return obj
        
        return scan_and_mask(content)

# app/services/report_service.py
"""
报告生成服务 - 支持 Markdown/PDF，适配 SKU-A/B/C
"""

import tempfile
import os
from typing import Dict, Any, Optional
from datetime import datetime
from pathlib import Path

from jinja2 import Template
import markdown

# 根据约束优先使用 weasyprint，复杂排版用 reportlab
try:
    from weasyprint import HTML, CSS
    WEASYPRINT_AVAILABLE = True
except ImportError:
    WEASYPRINT_AVAILABLE = False

from app.core.registry import SKUType
from app.metrics.collector import MetricsCollector, MetricType

class ReportService:
    """报告生成器"""
    
    def __init__(self):
        self.metrics = MetricsCollector()
        self.templates_dir = Path(__file__).parent / "templates"
    
    async def generate(
        self,
        results: Dict[str, Any],
        format: str,  # markdown/pdf
        sku: SKUType,
        input_data: Dict[str, Any],
        template_name: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        生成报告
        
        - SKU-A: 轻量 Markdown，3-5 个核心模块，快速交付
        - SKU-B: PDF + Markdown，全量模块，综合排版
        - SKU-C: PDF + Markdown + diff 对比，历史追踪
        """
        start_time = datetime.utcnow()
        
        # 选择模板
        template = self._get_template(sku, template_name)
        
        # 准备上下文
        context = {
            "input_data": input_data,
            "results": results,
            "generated_at": datetime.utcnow().isoformat(),
            "sku": sku.value,
            "summary": self._generate_summary(results)
        }
        
        # SKU-C: 添加历史对比
        if sku == SKUType.C:
            context["historical_comparison"] = await self._load_historical_comparison(
                input_data.get("company_id")
            )
        
        # 生成 Markdown
        md_content = template.render(**context)
        
        # SKU-A 直接返回 Markdown
        if sku == SKUType.A:
            self._record_metrics(start_time, sku, "markdown")
            return {
                "format": "markdown",
                "content": md_content,
                "filename": f"report_{sku.value}_{datetime.utcnow().strftime('%Y%m%d')}.md"
            }
        
        # SKU-B/C 生成 PDF
        if format == "pdf" and WEASYPRINT_AVAILABLE:
            pdf_path = await self._generate_pdf(md_content, sku)
            self._record_metrics(start_time, sku, "pdf")
            return {
                "format": "pdf+markdown",
                "markdown_content": md_content,
                "pdf_url": pdf_path,
                "filename": f"report_{sku.value}_{datetime.utcnow().strftime('%Y%m%d')}.pdf"
            }
        
        # 降级到 Markdown
        return {
            "format": "markdown",
            "content": md_content,
            "warning": "PDF generation not available"
        }
    
    def _get_template(self, sku: SKUType, template_name: Optional[str]) -> Template:
        """获取 Jinja2 模板"""
        if template_name and (self.templates_dir / f"{template_name}.html").exists():
            template_path = self.templates_dir / f"{template_name}.html"
        else:
            # 默认模板
            template_path = self.templates_dir / f"report_{sku.value}.md"
            if not template_path.exists():
                template_path = self.templates_dir / "report_default.md"
        
        with open(template_path, 'r', encoding='utf-8') as f:
            return Template(f.read())
    
    def _generate_summary(self, results: Dict) -> Dict:
        """生成执行摘要"""
        total_modules = len(results)
        success_modules = sum(1 for r in results.values() if not r.get("error"))
        
        # 提取关键发现
        key_findings = []
        for module_name, result in results.items():
            if isinstance(result, dict) and "key_finding" in result:
                key_findings.append({
                    "module": module_name,
                    "finding": result["key_finding"],
                    "severity": result.get("severity", "info")
                })
        
        return {
            "total_modules": total_modules,
            "success_rate": success_modules / max(total_modules, 1),
            "key_findings": key_findings[:5],  # 前 5 个关键发现
            "risk_level": self._calculate_overall_risk(results)
        }
    
    def _calculate_overall_risk(self, results: Dict) -> str:
        """计算整体风险等级"""
        risk_scores = []
        for result in results.values():
            if isinstance(result, dict):
                if "risk_score" in result:
                    risk_scores.append(result["risk_score"])
                elif result.get("threat_level") == "high":
                    risk_scores.append(0.9)
        
        if not risk_scores:
            return "low"
        
        avg_risk = sum(risk_scores) / len(risk_scores)
        if avg_risk > 0.7:
            return "high"
        elif avg_risk > 0.4:
            return "medium"
        return "low"
    
    async def _generate_pdf(self, md_content: str, sku: SKUType) -> str:

    async def _load_historical_comparison(
        self, 
        company_id: Optional[str]
    ) -> Optional[Dict]:
        """加载历史对比数据（SKU-C 专属）"""
        if not company_id:
            return None
        
        # 查询历史记录
        from sqlmodel import Session, select
        from app.assets.models import Memory  # 假设的案例库模型
        
        with Session(engine) as session:
            # 获取最近两次评估结果
            memories = session.exec(
                select(Memory)
                .where(Memory.case_type == "evaluation")
                .where(Memory.anonymized_data["company_id"].as_string() == company_id)
                .order_by(Memory.created_at.desc())
                .limit(2)
            ).all()
            
            if len(memories) < 2:
                return {"available": False, "reason": "insufficient_history"}
            
            current = memories[0].evaluation_result
            previous = memories[1].evaluation_result
            
            # 计算 diff
            diff = self._calculate_diff(current, previous)
            
            return {
                "available": True,
                "previous_date": memories[1].created_at.isoformat(),
                "current_date": memories[0].created_at.isoformat(),
                "changes": diff,
                "trend": "improving" if diff.get("score_change", 0) > 0 else "declining"
            }
    
    def _calculate_diff(self, current: Dict, previous: Dict) -> Dict:
        """计算评估结果差异"""
        changes = {}
        
        # 对比关键指标
        for key in ["overall_score", "risk_score", "market_share"]:
            if key in current and key in previous:
                curr_val = current[key]
                prev_val = previous[key]
                if isinstance(curr_val, (int, float)) and isinstance(prev_val, (int, float)):
                    changes[key] = {
                        "current": curr_val,
                        "previous": prev_val,
                        "change": curr_val - prev_val,
                        "change_percent": ((curr_val - prev_val) / prev_val * 100) if prev_val != 0 else 0
                    }
        
        return changes
    
    def _record_metrics(self, start_time: datetime, sku: SKUType, format: str):
        """记录报告生成指标"""
        duration = (datetime.utcnow() - start_time).total_seconds()
        self.metrics.record(
            MetricType.TIME, 
            f"report_generation_{sku.value}_{format}", 
            duration, 
            "seconds"
        )

"""
OpenClaw 平台集成适配器

import os

class TriggerType(str, Enum):

    DATA_CHANGE = "data_change"       # 数据变更
    WEBHOOK = "webhook"               # 外部回调
    DAILY_ASSET = "daily_asset"       # 每日资产激活

class OpenClawAdapter:
    """
    OpenClaw 平台适配器
    
    核心职责：
    1. 将本系统注册为 OpenClaw 的 Skill
    2. 接收 Trigger Pipeline 的调用
    3. 复用平台已有的 195+ Skills 作为数据源
    """
    
    def __init__(self):
        self.platform_url = os.getenv("OPENCLAW_PLATFORM_URL")
        self.api_key = os.getenv("OPENCLAW_API_KEY")
        self.available_skills = self._load_platform_skills()
    
    def _load_platform_skills(self) -> List[str]:
        """加载平台可用 Skills（实际应通过 API 获取）"""
        # 模拟 195+ Skills 中的部分核心能力
        return [
            "kimi_search", "agent-reach-wechat", "agent-reach-zhihu",
            "data-analyzer", "pdf-parser", "image-recognition",
            # ... 其他 Skills
        ]
    
    async def handle_trigger(
        self,
        trigger_type: TriggerType,
        payload: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        处理来自 OpenClaw Trigger Pipeline 的调用
        
        使用场景：
        - daily_asset_runner.py 每日调用，激活资产复用检查
        - 数据变更时自动触发重新评估
        """
        if trigger_type == TriggerType.DAILY_ASSET:
            return await self._handle_daily_asset_runner(payload)
        elif trigger_type == TriggerType.DATA_CHANGE:
            return await self._handle_data_change_trigger(payload)
        elif trigger_type == TriggerType.MANUAL:
            return await self._handle_manual_trigger(payload)
        else:
            return {"status": "ignored", "reason": "unsupported_trigger"}
    
    async def _handle_daily_asset_runner(self, payload: Dict) -> Dict:
        """
        处理 daily_asset_runner.py 的每日激活
        
        逻辑：
        1. 扫描所有 Workflow 的复用率
        2. 对低复用率资产生成优化建议
        3. 触发 Case Library 的归档检查
        """
        from app.assets.flywheel import AssetFlywheel
        from app.metrics.collector import MetricsCollector
        
        flywheel = AssetFlywheel()
        metrics = MetricsCollector()
        
        # 获取复用率指标
        reuse_metrics = flywheel.get_reuse_metrics()
        
        actions = []
        
        # 策略：复用率 < 50% 的资产提示优化
        if reuse_metrics["overall_reuse_rate"] < 0.5:
            actions.append({
                "action": "alert_low_reuse",
                "value": reuse_metrics["overall_reuse_rate"],
                "suggestion": "Review assets with reuse rate < 50%"
            })
        
        # 策略：检查过期 Memory
        archived_count = await self._archive_expired_memories()
        if archived_count > 0:
            actions.append({
                "action": "archive_memories",
                "count": archived_count
            })
        
        # 记录每日运行指标
        metrics.record(
            MetricsCollector.MetricType.REUSE,
            "daily_reuse_rate",
            reuse_metrics["overall_reuse_rate"],
            "ratio"
        )
        
        return {
            "status": "completed",
            "trigger": "daily_asset_runner",
            "reuse_metrics": reuse_metrics,
            "actions_taken": actions,
            "timestamp": datetime.utcnow().isoformat()
        }
    
    async def _handle_data_change_trigger(self, payload: Dict) -> Dict:
        """数据变更时触发增量评估"""
        entity_type = payload.get("entity_type")  # company/industry/market
        entity_id = payload.get("entity_id")
        change_type = payload.get("change_type")  # update/insert/delete
        
        # 只处理更新场景
        if change_type != "update":
            return {"status": "skipped", "reason": "not_update"}
        
        # 查询相关活跃的 Workflow
        from app.assets.models import Workflow
        from sqlmodel import Session, select
        
        with Session(engine) as session:
            workflows = session.exec(
                select(Workflow)
                .where(Workflow.status == "active")
                .where(Workflow.description.contains(entity_type))
            ).all()
            
            triggered = []
            for wf in workflows:
                # 异步触发重新评估
                # 实际应接入 Celery/ARQ 等任务队列
                triggered.append({
                    "workflow_id": wf.id,
                    "workflow_name": wf.name
                })
            
            return {
                "status": "triggered",
                "entity": f"{entity_type}:{entity_id}",
                "workflows_affected": len(triggered),
                "details": triggered
            }
    
    async def _archive_expired_memories(self) -> int:
        """归档过期案例（超过 1 年未使用）"""
        from sqlmodel import Session, select, update
        from app.assets.models import Memory
        from datetime import timedelta
        
        with Session(engine) as session:
            # 查找 1 年未使用的 Memory
            expired = session.exec(
                select(Memory)
                .where(Memory.last_used < datetime.utcnow() - timedelta(days=365))
                .where(Memory.reuse_count < 5)  # 且复用次数少
            ).all()
            
            count = 0
            for mem in expired:
                # 软删除/归档
                mem.tags.append("archived")
                count += 1
            
            session.commit()
            return count
    
    def call_skill(self, skill_name: str, params: Dict) -> Dict:
        """
        调用 OpenClaw 平台已有的 Skill
        
        例如：在评估器中调用 kimi_search 进行情报补充

        if skill_name not in self.available_skills:

        # 模拟实现
        return {
            "skill": skill_name,
            "status": "called",
            "result": {"data": "from_platform_skill"}
        }

"""
生产环境中间件 - 可观测性、限流、审计
"""

import time
import uuid
from typing import Callable
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.types import ASGIApp

from app.metrics.collector import MetricsCollector, MetricType

class ObservabilityMiddleware(BaseHTTPMiddleware):
    """
    可观测性中间件
    - 结构化日志（时间、输入摘要、输出摘要、耗时）
    - 请求追踪 ID
    - 性能指标自动采集
    """
    
    def __init__(self, app: ASGIApp):
        super().__init__(app)
        self.metrics = MetricsCollector()
    
    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        request_id = str(uuid.uuid4())
        request.state.request_id = request_id
        
        start_time = time.time()
        
        # 记录请求摘要（脱敏）
        input_summary = {
            "method": request.method,
            "path": request.url.path,
            "query_params": str(request.query_params),
            "client_ip": request.client.host if request.client else None
        }
        
        try:
            response = await call_next(request)
            duration = time.time() - start_time
            
            # 记录成功指标
            self.metrics.record(
                MetricType.TIME,
                f"api_{request.url.path}",
                duration,
                "seconds",
                context={"method": request.method, "status": response.status_code}
            )
            
            # 添加追踪 ID 到响应头
            response.headers["X-Request-ID"] = request_id
            
            return response
            
        except Exception as e:
            duration = time.time() - start_time
            
            # 记录失败指标
            self.metrics.record(
                MetricType.RISK,
                f"api_error_{request.url.path}",
                1.0,
                "incident",
                context={"error": str(e), "duration": duration}
            )
            raise

class RateLimitMiddleware(BaseHTTPMiddleware):
    """
    简单限流中间件（基于内存，生产环境应使用 Redis）
    """
    
    def __init__(self, app: ASGIApp, requests_per_minute: int = 60):
        super().__init__(app)
        self.requests_per_minute = requests_per_minute
        self.requests = {}  # ip -> [(timestamp, count)]
    
    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        client_ip = request.client.host if request.client else "unknown"
        current_time = time.time()
        
        # 清理过期记录（1 分钟前）
        if client_ip in self.requests:
            self.requests[client_ip] = [
                (t, c) for t, c in self.requests[client_ip] 
                if current_time - t < 60
            ]
        else:
            self.requests[client_ip] = []
        
        # 计算当前请求数
        total_requests = sum(c for t, c in self.requests[client_ip])
        
        if total_requests >= self.requests_per_minute:
            from fastapi import HTTPException
            raise HTTPException(status_code=429, detail="Rate limit exceeded")
        
        # 记录本次请求
        self.requests[client_ip].append((current_time, 1))
        
        return await call_next(request)

    CURRENT_VERSION = "1.0.0"
    
    @staticmethod
    def migrate():
        """执行迁移"""
        # 创建基础表
        SQLModel.metadata.create_all(engine)
        
        # 检查/更新版本号
        with Session(engine) as session:
            # 创建版本表
            session.exec(text("""
                CREATE TABLE IF NOT EXISTS schema_version (
                    version VARCHAR(20) PRIMARY KEY,
                    applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            """))
            
            # 检查当前版本
            result = session.exec(text("SELECT version FROM schema_version ORDER BY applied_at DESC LIMIT 1"))
            current = result.first()
            
            if not current:
                # 初始化版本
                session.exec(text(
                    "INSERT INTO schema_version (version) VALUES (:version)"
                ), {"version": MigrationManager.CURRENT_VERSION})
                session.commit()
                print(f"Initialized database to version {MigrationManager.CURRENT_VERSION}")
            else:
                print(f"Current database version: {current[0]}")

class DebugLogger:
    """
    调试专用日志器
    支持：
    - 执行路径追踪
    - 变量状态快照
    - 性能热点标记
    """
    
    def __init__(self, name: str):
        self.logger = structlog.get_logger(name)
        self.tracer = trace.get_tracer(name)
    
    def log_execution(self, step: str, context: Dict = None):
        """记录执行步骤"""
        ctx = request_context.get()
        self.logger.info(
            f"execution_step_{step}",
            request_id=ctx.get("request_id"),
            step=step,
            context=context or {},
            timestamp=time.time()
        )
    
    def snapshot(self, variable_name: str, value: Any):
        """记录变量快照（用于调试复杂状态）"""
        # 敏感信息过滤
        safe_value = self._sanitize(value)
        self.logger.debug(
            "variable_snapshot",
            name=variable_name,
            value=safe_value,
            type=type(value).__name__
        )
    
    def profile(self, func):
        """性能剖析装饰器"""
        @functools.wraps(func)
        async def async_wrapper(*args, **kwargs):
            with self.tracer.start_as_current_span(func.__name__) as span:
                start = time.time()
                try:
                    result = await func(*args, **kwargs)
                    span.set_attribute("status", "success")
                    return result
                except Exception as e:
                    span.set_attribute("status", "error")
                    span.set_attribute("error.message", str(e))
                    raise
                finally:
                    duration = time.time() - start
                    span.set_attribute("duration_ms", duration * 1000)
                    self.logger.info(
                        "function_profiled",
                        function=func.__name__,
                        duration_ms=duration * 1000
                    )
        
        @functools.wraps(func)
        def sync_wrapper(*args, **kwargs):
            with self.tracer.start_as_current_span(func.__name__) as span:
                start = time.time()
                try:
                    result = func(*args, **kwargs)
                    span.set_attribute("status", "success")
                    return result
                except Exception as e:
                    span.set_attribute("status", "error")
                    span.record_exception(e)
                    raise
                finally:
                    duration = time.time() - start
                    span.set_attribute("duration_ms", duration * 1000)
        
        return async_wrapper if asyncio.iscoroutinefunction(func) else sync_wrapper
    
    def _sanitize(self, value: Any) -> Any:
        """脱敏处理"""
        if isinstance(value, dict):
            sensitive_keys = ['password', 'token', 'secret', 'api_key']
            return {
                k: "***" if any(s in k.lower() for s in sensitive_keys) else v
                for k, v in value.items()
            }
        return value

"""
沙箱执行器 - 隔离现有模块的执行环境
"""
import os
import sys
import time
import signal
import resource
import traceback
import multiprocessing
from typing import Callable, Any, Dict, Optional
from concurrent.futures import ProcessPoolExecutor, TimeoutError
import asyncio

class ExecutionTimeout(Exception):
    """执行超时"""
    pass

class ExecutionMemoryError(Exception):
    """内存超限"""
    pass

class SandboxExecutor:
    """
    进程级沙箱执行器
    
    特性：
    1. 超时控制（SIGALRM 或进程池）
    2. 内存限制（resource.setrlimit）
    3. 异常隔离（子进程异常不传播）
    4. 资源清理（强制垃圾回收）
    """
    
    def __init__(
        self,
        timeout: int = 60,
        memory_limit_mb: int = 512,
        cpu_limit_percent: int = 80
    ):
        self.timeout = timeout
        self.memory_limit = memory_limit_mb * 1024 * 1024
        self.cpu_limit = cpu_limit_percent
        self.executor = ProcessPoolExecutor(max_workers=4)
    
    def _limit_resources(self):
        """在子进程中设置资源限制"""
        # 内存限制
        resource.setrlimit(
            resource.RLIMIT_AS,
            (self.memory_limit, self.memory_limit)
        )
        # CPU 时间限制（软限制）
        resource.setrlimit(
            resource.RLIMIT_CPU,
            (self.timeout, self.timeout + 5)
        )
    
    def _execute_in_process(
        self,
        func: Callable,
        data: Dict
    ) -> Dict[str, Any]:
        """在独立进程中执行函数"""
        try:
            # 设置资源限制
            self._limit_resources()
            
            start_time = time.time()
            result = func(data)
            execution_time = time.time() - start_time
            
            return {
                "status": "success",
                "result": result,
                "_execution_time": execution_time,
                "_memory_used": resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
            }
        except MemoryError:
            return {
                "status": "error",
                "error": "Memory limit exceeded",
                "error_type": "ExecutionMemoryError"
            }
        except Exception as e:
            return {
                "status": "error",
                "error": str(e),
                "error_type": type(e).__name__,
                "stack_trace": traceback.format_exc()
            }
    
    async def execute_async(
        self,
        func: Callable,
        data: Dict
    ) -> Dict[str, Any]:
        """异步执行（用于现有同步模块）"""
        loop = asyncio.get_event_loop()
        
        try:
            # 在进程池中执行
            future = loop.run_in_executor(
                self.executor,
                self._execute_in_process,
                func,
                data
            )
            result = await asyncio.wait_for(future, timeout=self.timeout)
            return result
            
        except asyncio.TimeoutError:
            return {
                "status": "error",
                "error": f"Execution timeout after {self.timeout}s",
                "error_type": "ExecutionTimeout"
            }
    
    def execute(self, func: Callable, data: Dict) -> Dict[str, Any]:
        """同步执行包装"""
        return asyncio.run(self.execute_async(func, data))
    
    def __del__(self):
        """清理进程池"""
        self.executor.shutdown(wait=False)

"""
现有模块适配器 - 处理参数映射、类型转换、版本兼容
"""
import inspect
from typing import Callable, Dict, Any, Optional
from pydantic import BaseModel, ValidationError, create_model

class ModuleAdapter:
    """
    模块适配器
    
    解决现有模块的以下问题：
    1. 参数签名不一致 -> 自动映射
    2. 返回类型不统一 -> 包装标准化
    3. 缺少类型注解 -> 动态推断
    4. 异常处理不规范 -> 统一包装
    """
    
    def __init__(self, func: Callable):
        self.original_func = func
        self.signature = inspect.signature(func)
        self._build_input_model()
    
    def _build_input_model(self):
        """根据函数签名构建 Pydantic 模型（严格校验）"""
        params = {}
        for name, param in self.signature.parameters.items():
            if param.default != inspect.Parameter.empty:
                params[name] = (Optional[param.annotation], param.default)
            else:
                params[name] = (param.annotation, ...)
        
        self.InputModel = create_model(
            f"{self.original_func.__name__}_Input",
            **params
        )
    
    async def execute(self, data: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行适配后的函数
        
        流程：
        1. 参数校验（Pydantic）
        2. 必要字段补全（默认值）
        3. 沙箱执行（隔离）
        4. 结果标准化
        """
        # 1. 校验输入
        try:
            validated = self.InputModel(**data)
        except ValidationError as e:
            return {
                "status": "input_error",
                "error": "Input validation failed",
                "details": e.errors()
            }
        
        # 2. 转换为函数需要的格式
        kwargs = validated.dict()
        
        # 3. 沙箱执行
        from app.evaluators.sandbox import SandboxExecutor
        sandbox = SandboxExecutor(timeout=60, memory_limit_mb=512)
        
        result = await sandbox.execute_async(self.original_func, kwargs)
        
        # 4. 标准化输出
        if result["status"] == "success":
            return self._standardize_output(result["result"])
        else:
            return {
                "status": "execution_error",
                "error": result.get("error"),
                "error_type": result.get("error_type"),
                "original_function": self.original_func.__name__
            }
    
    def _standardize_output(self, raw_result: Any) -> Dict[str, Any]:
        """标准化输出格式"""
        if isinstance(raw_result, dict):
            return {
                "status": "success",
                "data": raw_result,
                "score": raw_result.get("score") or raw_result.get("confidence", 0.5),
                "key_finding": raw_result.get("key_finding") or raw_result.get("summary", ""),
                "recommendations": raw_result.get("recommendations", [])
            }
        else:
            # 非字典输出包装
            return {
                "status": "success",
                "data": {"value": raw_result},
                "score": 0.5,
                "key_finding": str(raw_result)[:200],
                "recommendations": []
            }

    sku: list = None,

            category=category,
            sku=[SKUType(s) for s in sku],

        )
        
        return wrapped

"""
Saga 模式 - 处理长事务（跨多个资产操作）
确保资产飞轮的原子性
"""
from typing import List, Callable, Any, Dict
from dataclasses import dataclass
from enum import Enum
import logging

logger = logging.getLogger(__name__)

class SagaStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    COMPLETED = "completed"
    COMPENSATING = "compensating"
    COMPENSATED = "compensated"
    FAILED = "failed"

@dataclass
class SagaStep:
    name: str
    action: Callable
    compensation: Callable  # 回滚操作
    params: Dict = None

class SagaOrchestrator:
    """
    Saga 编排器
    
    用于资产飞轮的复杂事务：
    - Workflow 执行
    - Memory 沉淀
    - Prompt 优化
    - Skill 更新
    
    任一步骤失败时触发补偿机制回滚
    """
    
    def __init__(self, saga_id: str):
        self.saga_id = saga_id
        self.steps: List[SagaStep] = []
        self.completed_steps: List[int] = []
        self.status = SagaStatus.PENDING
        self.context: Dict[str, Any] = {}
    
    def add_step(self, step: SagaStep):
        """添加步骤"""
        self.steps.append(step)
    
    async def execute(self) -> Dict[str, Any]:
        """执行 Saga"""
        self.status = SagaStatus.RUNNING
        
        for idx, step in enumerate(self.steps):
            try:
                logger.info(f"Saga {self.saga_id}: Executing step {step.name}")
                
                # 执行动作
                result = await self._execute_action(step)
                self.context[f"{step.name}_result"] = result
                self.completed_steps.append(idx)
                
            except Exception as e:
                logger.error(f"Saga {self.saga_id}: Step {step.name} failed: {e}")
                await self._compensate(idx)
                self.status = SagaStatus.FAILED
                return {
                    "status": "failed",
                    "failed_step": step.name,
                    "error": str(e),
                    "compensated_steps": len(self.completed_steps)
                }
        
        self.status = SagaStatus.COMPLETED
        return {
            "status": "success",
            "context": self.context,
            "steps_completed": len(self.steps)
        }
    
    async def _execute_action(self, step: SagaStep) -> Any:
        """执行单个动作"""
        if asyncio.iscoroutinefunction(step.action):
            return await step.action(step.params or {})
        else:
            return step.action(step.params or {})
    
    async def _compensate(self, failed_idx: int):
        """补偿（回滚）已完成的步骤"""
        self.status = SagaStatus.COMPENSATING
        
        # 逆向回滚
        for idx in reversed(self.completed_steps):
            step = self.steps[idx]
            try:
                logger.info(f"Saga {self.saga_id}: Compensating step {step.name}")
                
                if asyncio.iscoroutinefunction(step.compensation):
                    await step.compensation(self.context)
                else:
                    step.compensation(self.context)
                    
            except Exception as e:
                # 补偿失败，需要人工介入
                logger.critical(
                    f"Saga {self.saga_id}: Compensation failed for step {step.name}: {e}"
                )
                # 发送告警...
        
        self.status = SagaStatus.COMPENSATED

48 def should_escalate(self, task: Dict[str, Any]) -> Optional[str]: 

49 """ 

50 判定当前任务是否应优先寻求外援。 

51 返回 'knowledge' | 'technical' | None 

52 """ 

53 complexity = task.get("complexity", "P2") 

54 domain = task.get("domain", "") 

55 token_budget = task.get("token_budget", 50000) 

56 estimated_tokens = task.get("estimated_tokens", token_budget) 

57 internal_exhausted = task.get("internal_attempts", 0) >= 2 

58 novel_theory_required = task.get("novel_theory_required", False) 

59 tech_stack = task.get("tech_stack", "") 

60 

61 # 知识外援触发条件 

62 if novel_theory_required or domain in ("哲学", "经济学", "心理学", "组织行为", "决策科学"): 

63 return "knowledge" 

64 if estimated_tokens > token_budget * 0.8 and domain in ("研究", "框架设计", "方法论"): 

65 return "knowledge" 

66 

67 # 技术外援触发条件 

68 if tech_stack or domain in ("软件工程", "架构", "性能优化", "安全审计", "DevOps"): 

69 return "technical" 

70 if complexity == "P0" and internal_exhausted: 

71 return "technical" 

72 if task.get("needs_external_audit", False): 

73 return "technical" 

74 

75 return None

77 def create_request(self, req_type: str, task_id: str, background: str, specific_asks: List[str], deliverables: List[str]) -> Dict[str, Any]: 

78 """生成标准化的外援请求单""" 

79 template = self.CONSULTANT_TYPES.get(req_type, {}) 

80 request = { 

81 "task_id": task_id, 

82 "type": req_type, 

83 "type_name": template.get("name", req_type), 

84 "created_at": self.get_timestamp(), 

85 "background": background, 

86 "specific_asks": specific_asks, 

87 "expected_deliverables": deliverables, 

88 "status": "pending_dispatch", 

89 } 

90 self.active_requests.append(request) 

91 return request

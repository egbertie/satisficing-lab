30 def check(self, text: str) -> List[Dict[str, Any]]: 

31 """检查文本是否违反语言红线""" 

32 violations = [] 

33 for pattern, msg in self.RED_LINES: 

34 if pattern in text: 

35 violations.append({"pattern": pattern, "message": msg}) 

36 return violations

38 def suggest_fix(self, text: str) -> str: 

39 """尝试修复明显的语言违规""" 

40 fixes = { 

41 "好的！": "行，我来。", 

42 "没问题！": "已确认。", 

43 "这是一个好问题！": "这个问题我之前也记录过。", 

44 "处理完成": "验收通过", 

45 } 

46 result = text 

47 for bad, good in fixes.items(): 

48 result = result.replace(bad, good) 

49 return result

51 def delivery_format(self, conclusion: str, structure: Any, details: Optional[str] = None) -> str: 

52 """ 

53 11.1 默认交付结构 

54 1. 结论（1句话） 

55 2. 结构（表格/清单/分级） 

56 3. 细节（可选，按需展开） 

57 """ 

58 parts = [conclusion] 

59 if structure: 

60 if isinstance(structure, str): 

61 parts.append(structure) 

62 elif isinstance(structure, list): 

63 parts.append("\n".join(f"- {s}" for s in structure)) 

64 elif isinstance(structure, dict): 

65 for k, v in structure.items(): 

66 parts.append(f"{k}: {v}") 

67 if details: 

68 parts.append(f"\n细节:\n{details}") 

69 return "\n\n".join(parts)

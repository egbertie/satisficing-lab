34 def find_recent_memories(self, days: int = 7, limit: int = 5) -> List[Dict[str, Any]]: 

35 """检索最近n天的记忆文件""" 

36 results = [] 

37 if not self.memory_dir.exists(): 

38 return results 

39 

40 files = sorted(self.memory_dir.glob("*.md"), key=lambda f: f.stat().st_mtime, reverse=True) 

41 for f in files[:limit]: 

42 try: 

43 mtime = datetime.fromtimestamp(f.stat().st_mtime) 

44 results.append({ 

45 "path": str(f), 

46 "filename": f.name, 

47 "mtime": mtime.isoformat(), 

48 "size": f.stat().st_size, 

49 }) 

50 except Exception:

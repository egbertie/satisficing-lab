22 def audit(self, task_context: Dict[str, Any]) -> Dict[str, Any]: 

23 """ 

24 对任务进行蓝军审计 

25 禁止输出"一切正常"，必须发现至少1个问题 

26 """ 

27 accusations = [] 

28 suggestions = [] 

29 risk_level = "🟢可控" 

30 

31 # 进度虚报 

32 claimed = task_context.get("claimed_progress", 0) 

33 actual = task_context.get("actual_progress", 0) 

34 if claimed > 0 and abs(claimed - actual) / claimed > 0.10: 

35 accusations.append(f"[进度虚报] 声称{claimed}%，实际{actual}%") 

36 suggestions.append("修正进度声明，重测后再前进") 

37 risk_level = "🔴高危" if risk_level == "🟢可控" else risk_level 

38 

39 # 偏离核心目标 

40 if task_context.get("off_target", False): 

41 accusations.append("[目标偏离] 任务明显偏离核心目标") 

42 suggestions.append("回溯到TASK_MASTER.md确认核心目标") 

43 risk_level = "🟡中危" if risk_level == "🟢可控" else risk_level 

44 

45 # 资源透支 

46 if task_context.get("token_over_budget", False): 

47 accusations.append("[资源透支] Token已超过预算200%") 

48 suggestions.append("立即暂停，汇报消耗明细并申请压缩方案") 

49 risk_level = "🔴高危" 

50 

51 # /tmp滥用 

52 tmp_count = task_context.get("tmp_file_count", 0) 

53 if tmp_count > 10: 

54 accusations.append(f"[/tmp滥用] 发现{tmp_count}个/tmp文件，缺乏清理机制") 

55 suggestions.append("将临时文件迁移至workspace持久目录或添加自动清理") 

56 risk_level = "🟡中危" if risk_level in ("🟢可控",) else risk_level 

57 

58 # 韧性缺失 

59 if not task_context.get("has_timeout", True): 

60 accusations.append("[韧性缺失] 脚本缺少超时退出机制") 

61 suggestions.append("为所有循环和脚本添加timeout参数") 

62 risk_level = "🟡中危" if risk_level in ("🟢可控",) else risk_level 

63 

64 # 疲劳工作 

65 if task_context.get("user_fatigued", False): 

66 accusations.append("[健康风险] 用户过度疲劳但仍要求高强度任务") 

67 suggestions.append("建议暂停，休息后再继续") 

68 risk_level = "🔴高危" 

69 

70 # 蓝军铁律：必须发现至少1个问题 

71 if not accusations: 

72 accusations.append("[审计盲区] 本次审计未找到明显问题，这可能是更大的问题——意味着缺少足够的可观测性") 

73 suggestions.append("增加执行证据和中间检查点") 

74 risk_level = "🟡中危" 

75 

76 return { 

77 "risk_level": risk_level, 

78 "accusations": accusations, 

79 "suggestions": suggestions, 

80 "passed": len([a for a in accusations if "🔴" not in a and "[审计盲区]" not in a]) == 0, 

81 }

83 def format_audit(self, result: Dict[str, Any]) -> str: 

84 """格式化为审计文本""" 

85 lines = [ 

86 "[蓝军审计]", 

87 f"风险等级: {result['risk_level']}", 

88 "指控:", 

89 ] 

90 for i, acc in enumerate(result["accusations"], 1): 

91 lines.append(f"{i}. {acc}") 

92 if result["suggestions"]: 

93 lines.append("建议:") 

94 for s in result["suggestions"]: 

95 lines.append(f"- {s}") 

96 return "\n".join(lines)

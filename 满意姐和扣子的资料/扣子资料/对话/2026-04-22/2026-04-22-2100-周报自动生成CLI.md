### 2.2 数据收集脚本

#### 脚本1: `collect_data.sh`

```bash
#!/bin/bash
# ================================================
# 周报数据收集脚本
# 用法: ./collect_data.sh [--quick | --full]
# ================================================

set -e

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
DATA_DIR="$(dirname "$SCRIPT_DIR")/data"
LOG_DIR="$(dirname "$SCRIPT_DIR")/logs"
TEMPLATE_DIR="$(dirname "$SCRIPT_DIR")/templates"

# 日志函数
log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_DIR/workflow.log"
}

# 创建必要目录
mkdir -p "$DATA_DIR" "$LOG_DIR"

log "=== 开始收集周报数据 ==="

# 解析参数
MODE=${1:-"quick"}

if [ "$MODE" == "--quick" ]; then
    log "快速模式: 仅收集基础数据"
elif [ "$MODE" == "--full" ]; then
    log "完整模式: 收集全部数据"
fi

# 收集本周日期范围
WEEK_START=$(date -d "last monday" +%Y-%m-%d)
WEEK_END=$(date -d "last sunday" +%Y-%m-%d)

# 1. 收集已完成任务
collect_completed() {
    log "步骤1: 收集本周完成任务..."
    cat > "$DATA_DIR/completed_tasks.md" << 'TEMPLATE'
## 已完成任务收集

请在下方填写本周完成的任务（按格式填写）:

### 任务格式
- **[完成]** 任务名称 | 完成时间 | 产出说明

### 任务列表
TEMPLATE
    echo -e "\n请输入本周完成的任务（输入完成后按 Ctrl+D）:"
    cat >> "$DATA_DIR/completed_tasks.md"
    log "已完成任务收集完成"
}

# 2. 收集下周计划
collect_pending() {
    log "步骤2: 收集下周计划..."
    cat > "$DATA_DIR/pending_tasks.md" << 'TEMPLATE'
## 下周计划收集

请在下方填写下周计划任务:

### 任务格式
- **[计划]** 任务名称 | 优先级(P1/P2/P3) | 预计工时

### 计划列表
TEMPLATE
    echo -e "\n请输入下周计划（输入完成后按 Ctrl+D）:"
    cat >> "$DATA_DIR/pending_tasks.md"
    log "下周计划收集完成"
}

# 3. 收集数据指标
collect_metrics() {
    log "步骤3: 收集数据指标..."
    cat > "$DATA_DIR/metrics.json" << 'TEMPLATE'
{
    "completed_tasks_count": 0,
    "meetings_count": 0,
    "issues_resolved": 0,
    "code_commits": 0,
    "documents_created": 0,
    "collaboration_hours": 0,
    "learning_hours": 0,
    "custom_metrics": {}
}
TEMPLATE
    echo "请编辑 $DATA_DIR/metrics.json 填写具体数值"
    log "数据指标收集完成"
}

# 4. 生成配置文件
generate_config() {
    log "步骤4: 生成配置文件..."
    cat > "$DATA_DIR/config.yaml" << EOF
weekly_report:
  author: Egbertie
  week_start: "$WEEK_START"
  week_end: "$WEEK_END"
  generated_at: "$(date '+%Y-%m-%d %H:%M:%S')"
  
template:
  path: "$TEMPLATE_DIR/weekly_template.md"
  output_format: "md"
  
output:
  directory: "$(dirname "$SCRIPT_DIR")/output/weekly_reports"
  filename_pattern: "周报_{date}.md"
EOF
    log "配置文件生成完成"
}

# 执行收集
if [ "$MODE" == "--full" ]; then
    collect_completed
    collect_pending
fi
collect_metrics
generate_config

log "=== 数据收集完成 ==="
log "请编辑 $DATA_DIR 下的文件补充详细信息"
```

### 2.3 辅助数据读取脚本

#### 脚本2: `read_previous_reports.sh` - 读取历史周报

```bash
#!/bin/bash
# 读取最近N周的周报作为参考

WEEKS=${1:-4}
REPORT_DIR="$(dirname "$(dirname "${BASH_SOURCE[0]}")")/output/weekly_reports"

echo "=== 最近 $WEEKS 周周报摘要 ==="
echo ""

for i in $(seq 0 $((WEEKS-1))); do
    WEEK_DATE=$(date -d "$i weeks ago" +%Y-%m-%d)
    REPORT_FILE=$(ls "$REPORT_DIR"/*"$WEEK_DATE"* 2>/dev/null | head -1)
    
    if [ -f "$REPORT_FILE" ]; then
        echo "📄 $(basename "$REPORT_FILE")"
        echo "---"
        # 提取本周摘要
        grep -A 5 "本周重点摘要" "$REPORT_FILE" | head -10
        echo ""
    else
        echo "⚠️  未找到 $WEEK_DATE 的周报"
    fi
done
```

## 3. 一键生成脚本设计

### 3.1 主脚本: `weekly_workflow.sh`

```bash
#!/bin/bash
# ================================================
# 周报一键生成脚本
# 用法: ./weekly_workflow.sh [--collect] [--generate] [--all]
# ================================================

set -e

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
BASE_DIR="$(dirname "$SCRIPT_DIR")"
TEMPLATE_FILE="$BASE_DIR/templates/weekly_template.md"
DATA_DIR="$BASE_DIR/data"
OUTPUT_DIR="$BASE_DIR/output/weekly_reports"
LOG_FILE="$BASE_DIR/logs/workflow.log"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log() {
    echo -e "${GREEN}[$(date '+%H:%M:%S')]${NC} $1" | tee -a "$LOG_FILE"
}

error() {
    echo -e "${RED}[ERROR]${NC} $1" | tee -a "$LOG_FILE"
    exit 1
}

warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

# 创建目录结构
init_directories() {
    mkdir -p "$DATA_DIR" "$OUTPUT_DIR" "$BASE_DIR/logs" "$BASE_DIR/templates"
    log "目录结构初始化完成"
}

# 数据收集阶段
collect_data() {
    log "=== 阶段1: 数据收集 ==="
    
    # 检查必要文件
    if [ ! -f "$DATA_DIR/completed_tasks.md" ]; then
        warn "未找到已完成任务文件，正在创建模板..."
        cat > "$DATA_DIR/completed_tasks.md" << 'EOF'
## 本周完成任务

请在下方补充:
- 任务1 | 2026-04-16 | 产出: xxx
- 任务2 | 2026-04-18 | 产出: xxx
EOF
    fi
    
    if [ ! -f "$DATA_DIR/pending_tasks.md" ]; then
        warn "未找到下周计划文件，正在创建模板..."
        cat > "$DATA_DIR/pending_tasks.md" << 'EOF'
## 下周工作计划

请在下方补充:
- 计划任务1 | P1 | 4h
- 计划任务2 | P2 | 2h
EOF
    fi
    
    if [ ! -f "$DATA_DIR/metrics.json" ]; then
        warn "未找到数据指标文件，正在创建模板..."
        cat > "$DATA_DIR/metrics.json" << 'EOF'
{
    "completed_tasks_count": 0,
    "meetings_count": 0,
    "issues_resolved": 0,
    "code_commits": 0
}
EOF
    fi
    
    log "数据收集检查完成"
}

# 报告生成阶段
generate_report() {
    log "=== 阶段2: 生成周报 ==="
    
    # 生成日期
    REPORT_DATE=$(date '+%Y-%m-%d')
    WEEK_START=$(date -d "last monday" +%Y-%m-%d)
    WEEK_END=$(date -d "last sunday" +%Y-%m-%d)
    OUTPUT_FILE="$OUTPUT_DIR/周报_${REPORT_DATE}.md"
    
    # 读取数据文件
    COMPLETED=$(cat "$DATA_DIR/completed_tasks.md" 2>/dev/null || echo "")
    PENDING=$(cat "$DATA_DIR/pending_tasks.md" 2>/dev/null || echo "")
    
    # 解析指标
    if [ -f "$DATA_DIR/metrics.json" ]; then
        METRICS=$(cat "$DATA_DIR/metrics.json")
        TASKS_COUNT=$(echo "$METRICS" | grep -oP '"completed_tasks_count":\s*\K\d+' || echo "0")
        MEETINGS=$(echo "$METRICS" | grep -oP '"meetings_count":\s*\K\d+' || echo "0")
        ISSUES=$(echo "$METRICS" | grep -oP '"issues_resolved":\s*\K\d+' || echo "0")
        COMMITS=$(echo "$METRICS" | grep -oP '"code_commits":\s*\K\d+' || echo "0")
    else
        TASKS_COUNT=0
        MEETINGS=0
        ISSUES=0
        COMMITS=0
    fi
    
    # 生成周报
    cat > "$OUTPUT_FILE" << EOF
# 📊 Egbertie 工作周报

**报告周期**: $WEEK_START ~ $WEEK_END  
**生成时间**: $(date '+%Y-%m-%d %H:%M:%S')  
**汇报人**: Egbertie  

---

## 📌 本周重点摘要

请填写本周核心亮点...

---

## ✅ 本周完成工作

$COMPLETED

| 序号 | 任务名称 | 完成状态 | 产出说明 |
|:---:|:---|:---:|:---|
EOF

    # 生成任务表格行
    line_num=1
    while IFS= read -r line; do
        if [[ "$line" =~ \[.*\].*\| ]]; then
            task_name=$(echo "$line" | sed 's/.*\[\(.*\)\] *//' | cut -d'|' -f1 | xargs)
            task_date=$(echo "$line" | grep -oP '\d{4}-\d{2}-\d{2}' | head -1)
            task_output=$(echo "$line" | cut -d'|' -f3- | xargs)
            echo "| $line_num | $task_name | ✅ 完成 | $task_output |" >> "$OUTPUT_FILE"
            ((line_num++))
        fi
    done <<< "$COMPLETED"

    cat >> "$OUTPUT_FILE" << EOF

---

## 🎯 下周工作计划

$PENDING

| 序号 | 任务名称 | 优先级 | 预计工时 |
|:---:|:---|:---:|:---|
EOF

    # 生成计划表格行
    line_num=1
    while IFS= read -r line; do
        if [[ "$line" =~ \[.*计划.*\].*\| ]]; then
            plan_name=$(echo "$line" | sed 's/.*\[\(.*\)\] *//' | cut -d'|' -f1 | xargs)
            plan_priority=$(echo "$line" | grep -oP 'P[12]' | head -1 || echo "P2")
            plan_hours=$(echo "$line" | grep -oP '\d+h' | head -1 || echo "2h")
            echo "| $line_num | $plan_name | $plan_priority | $plan_hours |" >> "$OUTPUT_FILE"
            ((line_num++))
        fi
    done <<< "$PENDING"

    cat >> "$OUTPUT_FILE" << EOF

---

## 📈 数据统计

| 指标 | 数值 |
|:---|:---:|
| 完成任务数 | $TASKS_COUNT |
| 会议数量 | $MEETINGS |
| 解决问题数 | $ISSUES |
| 代码提交数 | $COMMITS |

---

## 💡 经验沉淀

请总结本周学到的方法论或可复用的经验...

---

## ⚠️ 风险与阻塞

| 问题 | 影响 | 状态 | 应对措施 |
|:---|:---:|:---:|:---|
| - | - | - | - |

---

## 📎 附件清单

- 无

---
*报告由 CLI 工具自动生成 | $(date '+%Y-%m-%d %H:%M:%S')*
EOF

    log "✅ 周报生成完成: $OUTPUT_FILE"
    echo ""
    echo "📄 生成文件: $OUTPUT_FILE"
}

# 完整工作流
run_full_workflow() {
    init_directories
    collect_data
    generate_report
    
    echo ""
    log "=== 周报生成完成 ==="
    echo ""
    echo "📁 产出文件: $OUTPUT_DIR/周报_$(date '+%Y-%m-%d').md"
    echo ""
    echo "💡 建议:"
    echo "   1. 查看生成的周报文件，补充详细信息"
    echo "   2. 编辑 $DATA_DIR 下的数据文件完善内容"
    echo "   3. 重新运行 --generate 更新周报"
}

# 主逻辑
case "${1:-"--all"}" in
    --collect)
        init_directories
        collect_data
        ;;
    --generate)
        generate_report
        ;;
    --all|--*)
        run_full_workflow
        ;;
    *)
        error "未知参数: $1"
        ;;
esac
```

### 3.2 快捷脚本: `weekly.sh`

```bash
#!/bin/bash
# 周报快捷命令 - 放置在 PATH 中

WORKFLOW_DIR="/path/to/周报工作流"  # 修改为实际路径

case "$1" in
    new)
        "$WORKFLOW_DIR/scripts/weekly_workflow.sh" --all
        ;;
    collect)
        "$WORKFLOW_DIR/scripts/weekly_workflow.sh" --collect
        ;;
    gen)
        "$WORKFLOW_DIR/scripts/weekly_workflow.sh" --generate
        ;;
    list)
        ls -la "$WORKFLOW_DIR/output/weekly_reports/" | tail -10
        ;;
    *)
        echo "用法: weekly [new|collect|gen|list]"
        echo "  new     - 一键生成周报"
        echo "  collect - 仅收集数据"
        echo "  gen     - 仅生成周报"
        echo "  list    - 查看历史周报"
        ;;
esac
```

### 4.5 常见问题

| 问题 | 解决方案 |
|:---|:---|
| 脚本无执行权限 | `chmod +x scripts/*.sh` |
